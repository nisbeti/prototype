بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

الٓرۚ كِتَٰبٌ أُحۡكِمَتۡ ءَايَٰتُهُۥ ثُمَّ فُصِّلَتۡ مِن لَّدُنۡ حَكِيمٍ خَبِيرٍ ۝١ الۤرۚ كِتَٰبٌ أُحۡكِمَتۡ ءَایَٰتُهُۥ ثُمَّ فُصِّلَتۡ مِن لَّدُنۡ حَكِیمٍ خَبِیرٍ ۝١ الٓرٰ ۫ كِتٰبٌ اُحْكِمَتْ اٰیٰتُهٗ ثُمَّ فُصِّلَتْ مِنْ لَّدُنْ حَكِیْمٍ خَبِیْرٍ ۟ۙ أَلَّا تَعۡبُدُوٓاْ إِلَّا ٱللَّهَۚ إِنَّنِي لَكُم مِّنۡهُ نَذِيرٞ وَبَشِيرٞ ۝٢ أَلَّا تَعۡبُدُوۤا۟ إِلَّا ٱللَّهَۚ إِنَّنِی لَكُم مِّنۡهُ نَذِیرࣱ وَبَشِیرࣱ ۝٢ اَلَّا تَعْبُدُوْۤا اِلَّا اللّٰهَ ؕ اِنَّنِیْ لَكُمْ مِّنْهُ نَذِیْرٌ وَّبَشِیْرٌ ۟ۙ وَأَنِ ٱسۡتَغۡفِرُواْ رَبَّكُمۡ ثُمَّ تُوبُوٓاْ إِلَيۡهِ يُمَتِّعۡكُم مَّتَٰعًا حَسَنًا إِلَىٰٓ أَجَلٖ مُّسَمّٗى وَيُؤۡتِ كُلَّ ذِي فَضۡلٖ فَضۡلَهُۥۖ وَإِن تَوَلَّوۡاْ فَإِنِّيٓ أَخَافُ عَلَيۡكُمۡ عَذَابَ يَوۡمٖ كَبِيرٍ ۝٣ وَأَنِ ٱسۡتَغۡفِرُوا۟ رَبَّكُمۡ ثُمَّ تُوبُوۤا۟ إِلَیۡهِ یُمَتِّعۡكُم مَّتَٰعًا حَسَنًا إِلَىٰۤ أَجَلࣲ مُّسَمࣰّى وَیُؤۡتِ كُلَّ ذِی فَضۡلࣲ فَضۡلَهُۥۖ وَإِن تَوَلَّوۡا۟ فَإِنِّیۤ أَخَافُ عَلَیۡكُمۡ عَذَابَ یَوۡمࣲ كَبِیرٍ ۝٣ وَّاَنِ اسْتَغْفِرُوْا رَبَّكُمْ ثُمَّ تُوْبُوْۤا اِلَیْهِ یُمَتِّعْكُمْ مَّتَاعًا حَسَنًا اِلٰۤی اَجَلٍ مُّسَمًّی وَّیُؤْتِ كُلَّ ذِیْ فَضْلٍ فَضْلَهٗ ؕ وَاِنْ تَوَلَّوْا فَاِنِّیْۤ اَخَافُ عَلَیْكُمْ عَذَابَ یَوْمٍ كَبِیْرٍ ۟ إِلَى ٱللَّهِ مَرۡجِعُكُمۡۖ وَهُوَ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرٌ ۝٤ إِلَى ٱللَّهِ مَرۡجِعُكُمۡۖ وَهُوَ عَلَىٰ كُلِّ شَیۡءࣲ قَدِیرٌ ۝٤ اِلَی اللّٰهِ مَرْجِعُكُمْ ۚ وَهُوَ عَلٰی كُلِّ شَیْءٍ قَدِیْرٌ ۟ أَلَآ إِنَّهُمۡ يَثۡنُونَ صُدُورَهُمۡ لِيَسۡتَخۡفُواْ مِنۡهُۚ أَلَا حِينَ يَسۡتَغۡشُونَ ثِيَابَهُمۡ يَعۡلَمُ مَا يُسِرُّونَ وَمَا يُعۡلِنُونَۚ إِنَّهُۥ عَلِيمُۢ بِذَاتِ ٱلصُّدُورِ ۝٥ أَلَاۤ إِنَّهُمۡ یَثۡنُونَ صُدُورَهُمۡ لِیَسۡتَخۡفُوا۟ مِنۡهُۚ أَلَا حِینَ یَسۡتَغۡشُونَ ثِیَابَهُمۡ یَعۡلَمُ مَا یُسِرُّونَ وَمَا یُعۡلِنُونَۚ إِنَّهُۥ عَلِیمُۢ بِذَاتِ ٱلصُّدُورِ ۝٥ اَلَاۤ اِنَّهُمْ یَثْنُوْنَ صُدُوْرَهُمْ لِیَسْتَخْفُوْا مِنْهُ ؕ اَلَا حِیْنَ یَسْتَغْشُوْنَ ثِیَابَهُمْ ۙ یَعْلَمُ مَا یُسِرُّوْنَ وَمَا یُعْلِنُوْنَ ۚ اِنَّهٗ عَلِیْمٌۢ بِذَاتِ الصُّدُوْرِ ۟ وَمَا مِن دَآبَّةٖ فِي ٱلۡأَرۡضِ إِلَّا عَلَى ٱللَّهِ رِزۡقُهَا وَيَعۡلَمُ مُسۡتَقَرَّهَا وَمُسۡتَوۡدَعَهَاۚ كُلّٞ فِي كِتَٰبٖ مُّبِينٖ ۝٦ وَمَا مِن دَاۤبَّةࣲ فِی ٱلۡأَرۡضِ إِلَّا عَلَى ٱللَّهِ رِزۡقُهَا وَیَعۡلَمُ مُسۡتَقَرَّهَا وَمُسۡتَوۡدَعَهَاۚ كُلࣱّ فِی كِتَٰبࣲ مُّبِینࣲ ۝٦ وَمَا مِنْ دَآبَّةٍ فِی الْاَرْضِ اِلَّا عَلَی اللّٰهِ رِزْقُهَا وَیَعْلَمُ مُسْتَقَرَّهَا وَمُسْتَوْدَعَهَا ؕ كُلٌّ فِیْ كِتٰبٍ مُّبِیْنٍ ۟ وَهُوَ ٱلَّذِي خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ فِي سِتَّةِ أَيَّامٖ وَكَانَ عَرۡشُهُۥ عَلَى ٱلۡمَآءِ لِيَبۡلُوَكُمۡ أَيُّكُمۡ أَحۡسَنُ عَمَلٗاۗ وَلَئِن قُلۡتَ إِنَّكُم مَّبۡعُوثُونَ مِنۢ بَعۡدِ ٱلۡمَوۡتِ لَيَقُولَنَّ ٱلَّذِينَ كَفَرُوٓاْ إِنۡ هَٰذَآ إِلَّا سِحۡرٞ مُّبِينٞ ۝٧ وَهُوَ ٱلَّذِی خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ فِی سِتَّةِ أَیَّامࣲ وَكَانَ عَرۡشُهُۥ عَلَى ٱلۡمَاۤءِ لِیَبۡلُوَكُمۡ أَیُّكُمۡ أَحۡسَنُ عَمَلࣰاۗ وَلَئِن قُلۡتَ إِنَّكُم مَّبۡعُوثُونَ مِنۢ بَعۡدِ ٱلۡمَوۡتِ لَیَقُولَنَّ ٱلَّذِینَ كَفَرُوۤا۟ إِنۡ هَٰذَاۤ إِلَّا سِحۡرࣱ مُّبِینࣱ ۝٧ وَهُوَ الَّذِیْ خَلَقَ السَّمٰوٰتِ وَالْاَرْضَ فِیْ سِتَّةِ اَیَّامٍ وَّكَانَ عَرْشُهٗ عَلَی الْمَآءِ لِیَبْلُوَكُمْ اَیُّكُمْ اَحْسَنُ عَمَلًا ؕ وَلَىِٕنْ قُلْتَ اِنَّكُمْ مَّبْعُوْثُوْنَ مِنْ بَعْدِ الْمَوْتِ لَیَقُوْلَنَّ الَّذِیْنَ كَفَرُوْۤا اِنْ هٰذَاۤ اِلَّا سِحْرٌ مُّبِیْنٌ ۟ وَلَئِنۡ أَخَّرۡنَا عَنۡهُمُ ٱلۡعَذَابَ إِلَىٰٓ أُمَّةٖ مَّعۡدُودَةٖ لَّيَقُولُنَّ مَا يَحۡبِسُهُۥٓۗ أَلَا يَوۡمَ يَأۡتِيهِمۡ لَيۡسَ مَصۡرُوفًا عَنۡهُمۡ وَحَاقَ بِهِم مَّا كَانُواْ بِهِۦ يَسۡتَهۡزِءُونَ ۝٨ وَلَئِنۡ أَخَّرۡنَا عَنۡهُمُ ٱلۡعَذَابَ إِلَىٰۤ أُمَّةࣲ مَّعۡدُودَةࣲ لَّیَقُولُنَّ مَا یَحۡبِسُهُۥۤۗ أَلَا یَوۡمَ یَأۡتِیهِمۡ لَیۡسَ مَصۡرُوفًا عَنۡهُمۡ وَحَاقَ بِهِم مَّا كَانُوا۟ بِهِۦ یَسۡتَهۡزِءُونَ ۝٨ وَلَىِٕنْ اَخَّرْنَا عَنْهُمُ الْعَذَابَ اِلٰۤی اُمَّةٍ مَّعْدُوْدَةٍ لَّیَقُوْلُنَّ مَا یَحْبِسُهٗ ؕ اَلَا یَوْمَ یَاْتِیْهِمْ لَیْسَ مَصْرُوْفًا عَنْهُمْ وَحَاقَ بِهِمْ مَّا كَانُوْا بِهٖ یَسْتَهْزِءُوْنَ ۟۠ وَلَئِنۡ أَذَقۡنَا ٱلۡإِنسَٰنَ مِنَّا رَحۡمَةٗ ثُمَّ نَزَعۡنَٰهَا مِنۡهُ إِنَّهُۥ لَيَـُٔوسٞ كَفُورٞ ۝٩ وَلَئِنۡ أَذَقۡنَا ٱلۡإِنسَٰنَ مِنَّا رَحۡمَةࣰ ثُمَّ نَزَعۡنَٰهَا مِنۡهُ إِنَّهُۥ لَیَءُوسࣱ كَفُورࣱ ۝٩ وَلَىِٕنْ اَذَقْنَا الْاِنْسَانَ مِنَّا رَحْمَةً ثُمَّ نَزَعْنٰهَا مِنْهُ ۚ اِنَّهٗ لَیَـُٔوْسٌ كَفُوْرٌ ۟ وَلَئِنۡ أَذَقۡنَٰهُ نَعۡمَآءَ بَعۡدَ ضَرَّآءَ مَسَّتۡهُ لَيَقُولَنَّ ذَهَبَ ٱلسَّيِّـَٔاتُ عَنِّيٓۚ إِنَّهُۥ لَفَرِحٞ فَخُورٌ ۝١٠ وَلَئِنۡ أَذَقۡنَٰهُ نَعۡمَاۤءَ بَعۡدَ ضَرَّاۤءَ مَسَّتۡهُ لَیَقُولَنَّ ذَهَبَ ٱلسَّیِّءَاتُ عَنِّیۤۚ إِنَّهُۥ لَفَرِحࣱ فَخُورٌ ۝١٠ وَلَىِٕنْ اَذَقْنٰهُ نَعْمَآءَ بَعْدَ ضَرَّآءَ مَسَّتْهُ لَیَقُوْلَنَّ ذَهَبَ السَّیِّاٰتُ عَنِّیْ ؕ اِنَّهٗ لَفَرِحٌ فَخُوْرٌ ۟ۙ إِلَّا ٱلَّذِينَ صَبَرُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ أُوْلَٰٓئِكَ لَهُم مَّغۡفِرَةٞ وَأَجۡرٞ كَبِيرٞ ۝١١ إِلَّا ٱلَّذِینَ صَبَرُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ أُو۟لَٰۤئِكَ لَهُم مَّغۡفِرَةࣱ وَأَجۡرࣱ كَبِیرࣱ ۝١١ اِلَّا الَّذِیْنَ صَبَرُوْا وَعَمِلُوا الصّٰلِحٰتِ ؕ اُولٰٓىِٕكَ لَهُمْ مَّغْفِرَةٌ وَّاَجْرٌ كَبِیْرٌ ۟ فَلَعَلَّكَ تَارِكُۢ بَعۡضَ مَا يُوحَىٰٓ إِلَيۡكَ وَضَآئِقُۢ بِهِۦ صَدۡرُكَ أَن يَقُولُواْ لَوۡلَآ أُنزِلَ عَلَيۡهِ كَنزٌ أَوۡ جَآءَ مَعَهُۥ مَلَكٌۚ إِنَّمَآ أَنتَ نَذِيرٞۚ وَٱللَّهُ عَلَىٰ كُلِّ شَيۡءٖ وَكِيلٌ ۝١٢ فَلَعَلَّكَ تَارِكُۢ بَعۡضَ مَا یُوحَىٰۤ إِلَیۡكَ وَضَاۤئِقُۢ بِهِۦ صَدۡرُكَ أَن یَقُولُوا۟ لَوۡلَاۤ أُنزِلَ عَلَیۡهِ كَنزٌ أَوۡ جَاۤءَ مَعَهُۥ مَلَكٌۚ إِنَّمَاۤ أَنتَ نَذِیرࣱۚ وَٱللَّهُ عَلَىٰ كُلِّ شَیۡءࣲ وَكِیلٌ ۝١٢ فَلَعَلَّكَ تَارِكٌ بَعْضَ مَا یُوْحٰۤی اِلَیْكَ وَضَآىِٕقٌ بِهٖ صَدْرُكَ اَنْ یَّقُوْلُوْا لَوْلَاۤ اُنْزِلَ عَلَیْهِ كَنْزٌ اَوْ جَآءَ مَعَهٗ مَلَكٌ ؕ اِنَّمَاۤ اَنْتَ نَذِیْرٌ ؕ وَاللّٰهُ عَلٰی كُلِّ شَیْءٍ وَّكِیْلٌ ۟ؕ أَمۡ يَقُولُونَ ٱفۡتَرَىٰهُۖ قُلۡ فَأۡتُواْ بِعَشۡرِ سُوَرٖ مِّثۡلِهِۦ مُفۡتَرَيَٰتٖ وَٱدۡعُواْ مَنِ ٱسۡتَطَعۡتُم مِّن دُونِ ٱللَّهِ إِن كُنتُمۡ صَٰدِقِينَ ۝١٣ أَمۡ یَقُولُونَ ٱفۡتَرَىٰهُۖ قُلۡ فَأۡتُوا۟ بِعَشۡرِ سُوَرࣲ مِّثۡلِهِۦ مُفۡتَرَیَٰتࣲ وَٱدۡعُوا۟ مَنِ ٱسۡتَطَعۡتُم مِّن دُونِ ٱللَّهِ إِن كُنتُمۡ صَٰدِقِینَ ۝١٣ اَمْ یَقُوْلُوْنَ افْتَرٰىهُ ؕ قُلْ فَاْتُوْا بِعَشْرِ سُوَرٍ مِّثْلِهٖ مُفْتَرَیٰتٍ وَّادْعُوْا مَنِ اسْتَطَعْتُمْ مِّنْ دُوْنِ اللّٰهِ اِنْ كُنْتُمْ صٰدِقِیْنَ ۟ فَإِلَّمۡ يَسۡتَجِيبُواْ لَكُمۡ فَٱعۡلَمُوٓاْ أَنَّمَآ أُنزِلَ بِعِلۡمِ ٱللَّهِ وَأَن لَّآ إِلَٰهَ إِلَّا هُوَۖ فَهَلۡ أَنتُم مُّسۡلِمُونَ ۝١٤ فَإِلَّمۡ یَسۡتَجِیبُوا۟ لَكُمۡ فَٱعۡلَمُوۤا۟ أَنَّمَاۤ أُنزِلَ بِعِلۡمِ ٱللَّهِ وَأَن لَّاۤ إِلَٰهَ إِلَّا هُوَۖ فَهَلۡ أَنتُم مُّسۡلِمُونَ ۝١٤ فَاِلَّمْ یَسْتَجِیْبُوْا لَكُمْ فَاعْلَمُوْۤا اَنَّمَاۤ اُنْزِلَ بِعِلْمِ اللّٰهِ وَاَنْ لَّاۤ اِلٰهَ اِلَّا هُوَ ۚ فَهَلْ اَنْتُمْ مُّسْلِمُوْنَ ۟ مَن كَانَ يُرِيدُ ٱلۡحَيَوٰةَ ٱلدُّنۡيَا وَزِينَتَهَا نُوَفِّ إِلَيۡهِمۡ أَعۡمَٰلَهُمۡ فِيهَا وَهُمۡ فِيهَا لَا يُبۡخَسُونَ ۝١٥ مَن كَانَ یُرِیدُ ٱلۡحَیَوٰةَ ٱلدُّنۡیَا وَزِینَتَهَا نُوَفِّ إِلَیۡهِمۡ أَعۡمَٰلَهُمۡ فِیهَا وَهُمۡ فِیهَا لَا یُبۡخَسُونَ ۝١٥ مَنْ كَانَ یُرِیْدُ الْحَیٰوةَ الدُّنْیَا وَزِیْنَتَهَا نُوَفِّ اِلَیْهِمْ اَعْمَالَهُمْ فِیْهَا وَهُمْ فِیْهَا لَا یُبْخَسُوْنَ ۟ أُوْلَٰٓئِكَ ٱلَّذِينَ لَيۡسَ لَهُمۡ فِي ٱلۡأٓخِرَةِ إِلَّا ٱلنَّارُۖ وَحَبِطَ مَا صَنَعُواْ فِيهَا وَبَٰطِلٞ مَّا كَانُواْ يَعۡمَلُونَ ۝١٦ أُو۟لَٰۤئِكَ ٱلَّذِینَ لَیۡسَ لَهُمۡ فِی ٱلۡءَاخِرَةِ إِلَّا ٱلنَّارُۖ وَحَبِطَ مَا صَنَعُوا۟ فِیهَا وَبَٰطِلࣱ مَّا كَانُوا۟ یَعۡمَلُونَ ۝١٦ اُولٰٓىِٕكَ الَّذِیْنَ لَیْسَ لَهُمْ فِی الْاٰخِرَةِ اِلَّا النَّارُ ۖؗ وَحَبِطَ مَا صَنَعُوْا فِیْهَا وَبٰطِلٌ مَّا كَانُوْا یَعْمَلُوْنَ ۟ أَفَمَن كَانَ عَلَىٰ بَيِّنَةٖ مِّن رَّبِّهِۦ وَيَتۡلُوهُ شَاهِدٞ مِّنۡهُ وَمِن قَبۡلِهِۦ كِتَٰبُ مُوسَىٰٓ إِمَامٗا وَرَحۡمَةًۚ أُوْلَٰٓئِكَ يُؤۡمِنُونَ بِهِۦۚ وَمَن يَكۡفُرۡ بِهِۦ مِنَ ٱلۡأَحۡزَابِ فَٱلنَّارُ مَوۡعِدُهُۥۚ فَلَا تَكُ فِي مِرۡيَةٖ مِّنۡهُۚ إِنَّهُ ٱلۡحَقُّ مِن رَّبِّكَ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يُؤۡمِنُونَ ۝١٧ أَفَمَن كَانَ عَلَىٰ بَیِّنَةࣲ مِّن رَّبِّهِۦ وَیَتۡلُوهُ شَاهِدࣱ مِّنۡهُ وَمِن قَبۡلِهِۦ كِتَٰبُ مُوسَىٰۤ إِمَامࣰا وَرَحۡمَةًۚ أُو۟لَٰۤئِكَ یُؤۡمِنُونَ بِهِۦۚ وَمَن یَكۡفُرۡ بِهِۦ مِنَ ٱلۡأَحۡزَابِ فَٱلنَّارُ مَوۡعِدُهُۥۚ فَلَا تَكُ فِی مِرۡیَةࣲ مِّنۡهُۚ إِنَّهُ ٱلۡحَقُّ مِن رَّبِّكَ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا یُؤۡمِنُونَ ۝١٧ اَفَمَنْ كَانَ عَلٰی بَیِّنَةٍ مِّنْ رَّبِّهٖ وَیَتْلُوْهُ شَاهِدٌ مِّنْهُ وَمِنْ قَبْلِهٖ كِتٰبُ مُوْسٰۤی اِمَامًا وَّرَحْمَةً ؕ اُولٰٓىِٕكَ یُؤْمِنُوْنَ بِهٖ ؕ وَمَنْ یَّكْفُرْ بِهٖ مِنَ الْاَحْزَابِ فَالنَّارُ مَوْعِدُهٗ ۚ فَلَا تَكُ فِیْ مِرْیَةٍ مِّنْهُ ۗ اِنَّهُ الْحَقُّ مِنْ رَّبِّكَ وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا یُؤْمِنُوْنَ ۟ وَمَنۡ أَظۡلَمُ مِمَّنِ ٱفۡتَرَىٰ عَلَى ٱللَّهِ كَذِبًاۚ أُوْلَٰٓئِكَ يُعۡرَضُونَ عَلَىٰ رَبِّهِمۡ وَيَقُولُ ٱلۡأَشۡهَٰدُ هَٰٓؤُلَآءِ ٱلَّذِينَ كَذَبُواْ عَلَىٰ رَبِّهِمۡۚ أَلَا لَعۡنَةُ ٱللَّهِ عَلَى ٱلظَّٰلِمِينَ ۝١٨ وَمَنۡ أَظۡلَمُ مِمَّنِ ٱفۡتَرَىٰ عَلَى ٱللَّهِ كَذِبًاۚ أُو۟لَٰۤئِكَ یُعۡرَضُونَ عَلَىٰ رَبِّهِمۡ وَیَقُولُ ٱلۡأَشۡهَٰدُ هَٰۤؤُلَاۤءِ ٱلَّذِینَ كَذَبُوا۟ عَلَىٰ رَبِّهِمۡۚ أَلَا لَعۡنَةُ ٱللَّهِ عَلَى ٱلظَّٰلِمِینَ ۝١٨ وَمَنْ اَظْلَمُ مِمَّنِ افْتَرٰی عَلَی اللّٰهِ كَذِبًا ؕ اُولٰٓىِٕكَ یُعْرَضُوْنَ عَلٰی رَبِّهِمْ وَیَقُوْلُ الْاَشْهَادُ هٰۤؤُلَآءِ الَّذِیْنَ كَذَبُوْا عَلٰی رَبِّهِمْ ۚ اَلَا لَعْنَةُ اللّٰهِ عَلَی الظّٰلِمِیْنَ ۟ۙ ٱلَّذِينَ يَصُدُّونَ عَن سَبِيلِ ٱللَّهِ وَيَبۡغُونَهَا عِوَجٗا وَهُم بِٱلۡأٓخِرَةِ هُمۡ كَٰفِرُونَ ۝١٩ ٱلَّذِینَ یَصُدُّونَ عَن سَبِیلِ ٱللَّهِ وَیَبۡغُونَهَا عِوَجࣰا وَهُم بِٱلۡءَاخِرَةِ هُمۡ كَٰفِرُونَ ۝١٩ الَّذِیْنَ یَصُدُّوْنَ عَنْ سَبِیْلِ اللّٰهِ وَیَبْغُوْنَهَا عِوَجًا ؕ وَهُمْ بِالْاٰخِرَةِ هُمْ كٰفِرُوْنَ ۟ أُوْلَٰٓئِكَ لَمۡ يَكُونُواْ مُعۡجِزِينَ فِي ٱلۡأَرۡضِ وَمَا كَانَ لَهُم مِّن دُونِ ٱللَّهِ مِنۡ أَوۡلِيَآءَۘ يُضَٰعَفُ لَهُمُ ٱلۡعَذَابُۚ مَا كَانُواْ يَسۡتَطِيعُونَ ٱلسَّمۡعَ وَمَا ٱلسَّمۡعَ وَمَا كَانُوا۟ یُبۡصِرُونَ ۝٢٠ اُولٰٓىِٕكَ لَمْ یَكُوْنُوْا مُعْجِزِیْنَ فِی الْاَرْضِ وَمَا كَانَ لَهُمْ مِّنْ دُوْنِ اللّٰهِ مِنْ اَوْلِیَآءَ ۘ یُضٰعَفُ لَهُمُ الْعَذَابُ ؕ مَا كَانُوْا كٰرِهُوْنَ ۟ وَيَٰقَوۡمِ لَآ أَسۡـَٔلُكُمۡ عَلَيۡهِ مَالًاۖ إِنۡ أَجۡرِيَ إِلَّا عَلَى ٱللَّهِۚ وَمَآ أَنَا۠ بِطَارِدِ ٱلَّذِينَ ءَامَنُوٓاْۚ إِنَّهُم مُّلَٰقُواْ رَبِّهِمۡ وَلَٰكِنِّيٓ أَرَىٰكُمۡ قَوۡمٗا تَجۡهَلُونَ ۝٢٩ وَیَٰقَوۡمِ لَاۤ أَسۡءَلُكُمۡ عَلَیۡهِ مَالًاۖ إِنۡ أَجۡرِیَ إِلَّا عَلَى ٱللَّهِۚ وَمَاۤ أَنَا۠ بِطَارِدِ ٱلَّذِینَ ءَامَنُوۤا۟ۚ إِنَّهُم مُّلَٰقُوا۟ رَبِّهِمۡ وَلَٰكِنِّیۤ أَرَىٰكُمۡ قَوۡمࣰا تَجۡهَلُونَ ۝٢٩ وَیٰقَوْمِ لَاۤ اَسْـَٔلُكُمْ عَلَیْهِ مَالًا ؕ اِنْ اَجْرِیَ اِلَّا عَلَی اللّٰهِ وَمَاۤ اَنَا بِطَارِدِ الَّذِیْنَ اٰمَنُوْا ؕ اِنَّهُمْ مُّلٰقُوْا رَبِّهِمْ وَلٰكِنِّیْۤ اَرٰىكُمْ قَوْمًا تَجْهَلُوْنَ ۟ وَيَٰقَوۡمِ مَن يَنصُرُنِي مِنَ ٱللَّهِ إِن طَرَدتُّهُمۡۚ أَفَلَا تَذَكَّرُونَ ۝٣٠ وَیَٰقَوۡمِ مَن یَنصُرُنِی مِنَ ٱللَّهِ إِن طَرَدتُّهُمۡۚ أَفَلَا تَذَكَّرُونَ ۝٣٠ وَیٰقَوْمِ مَنْ یَّنْصُرُنِیْ مِنَ اللّٰهِ اِنْ طَرَدْتُّهُمْ ؕ اَفَلَا تَذَكَّرُوْنَ ۟ وَلَآ أَقُولُ لَكُمۡ عِندِي خَزَآئِنُ ٱللَّهِ وَلَآ أَعۡلَمُ ٱلۡغَيۡبَ وَلَآ أَقُولُ إِنِّي مَلَكٞ وَلَآ أَقُولُ لِلَّذِينَ تَزۡدَرِيٓ أَعۡيُنُكُمۡ لَن يُؤۡتِيَهُمُ ٱللَّهُ خَيۡرًاۖ ٱللَّهُ أَعۡلَمُ بِمَا فِيٓ أَنفُسِهِمۡ إِنِّيٓ إِذٗا لَّمِنَ ٱلظَّٰلِمِينَ ۝٣١ وَلَاۤ أَقُولُ لَكُمۡ عِندِی خَزَاۤئِنُ ٱللَّهِ وَلَاۤ أَعۡلَمُ ٱلۡغَیۡبَ وَلَاۤ أَقُولُ إِنِّی مَلَكࣱ وَلَاۤ أَقُولُ لِلَّذِینَ تَزۡدَرِیۤ أَعۡیُنُكُمۡ لَن یُؤۡتِیَهُمُ ٱللَّهُ خَیۡرًاۖ ٱللَّهُ أَعۡلَمُ بِمَا فِیۤ أَنفُسِهِمۡ إِنِّیۤ إِذࣰا لَّمِنَ ٱلظَّٰلِمِینَ ۝٣١ وَلَاۤ اَقُوْلُ لَكُمْ عِنْدِیْ خَزَآىِٕنُ اللّٰهِ وَلَاۤ اَعْلَمُ الْغَیْبَ وَلَاۤ اَقُوْلُ اِنِّیْ مَلَكٌ وَّلَاۤ اَقُوْلُ لِلَّذِیْنَ تَزْدَرِیْۤ اَعْیُنُكُمْ لَنْ یُّؤْتِیَهُمُ اللّٰهُ خَیْرًا ؕ اَللّٰهُ اَعْلَمُ بِمَا فِیْۤ اَنْفُسِهِمْ ۖۚ اِنِّیْۤ اِذًا لَّمِنَ الظّٰلِمِیْنَ ۟ قَالُواْ يَٰنُوحُ قَدۡ جَٰدَلۡتَنَا فَأَكۡثَرۡتَ جِدَٰلَنَا فَأۡتِنَا بِمَا تَعِدُنَآ إِن كُنتَ مِنَ ٱلصَّٰدِقِينَ ۝٣٢ قَالُوا۟ یَٰنُوحُ قَدۡ جَٰدَلۡتَنَا فَأَكۡثَرۡتَ جِدَٰلَنَا فَأۡتِنَا بِمَا تَعِدُنَاۤ إِن كُنتَ مِنَ ٱلصَّٰدِقِینَ ۝٣٢ قَالُوْا یٰنُوْحُ قَدْ جٰدَلْتَنَا فَاَكْثَرْتَ جِدَالَنَا فَاْتِنَا بِمَا تَعِدُنَاۤ اِنْ كُنْتَ مِنَ الصّٰدِقِیْنَ ۟ قَالَ إِنَّمَا يَأۡتِيكُم بِهِ ٱللَّهُ إِن شَآءَ وَمَآ أَنتُم بِمُعۡجِزِينَ ۝٣٣ قَالَ إِنَّمَا یَأۡتِیكُم بِهِ ٱللَّهُ إِن شَاۤءَ وَمَاۤ أَنتُم بِمُعۡجِزِینَ ۝٣٣ قَالَ اِنَّمَا یَاْتِیْكُمْ بِهِ اللّٰهُ اِنْ شَآءَ وَمَاۤ اَنْتُمْ بِمُعْجِزِیْنَ ۟ وَلَا يَنفَعُكُمۡ نُصۡحِيٓ إِنۡ أَرَدتُّ أَنۡ أَنصَحَ لَكُمۡ إِن كَانَ ٱللَّهُ يُرِيدُ أَن يُغۡوِيَكُمۡۚ هُوَ رَبُّكُمۡ وَإِلَيۡهِ تُرۡجَعُونَ ۝٣٤ وَلَا یَنفَعُكُمۡ نُصۡحِیۤ إِنۡ أَرَدتُّ أَنۡ أَنصَحَ لَكُمۡ إِن كَانَ ٱللَّهُ یُرِیدُ أَن یُغۡوِیَكُمۡۚ هُوَ رَبُّكُمۡ وَإِلَیۡهِ تُرۡجَعُونَ ۝٣٤ وَلَا یَنْفَعُكُمْ نُصْحِیْۤ اِنْ اَرَدْتُّ اَنْ اَنْصَحَ لَكُمْ اِنْ كَانَ اللّٰهُ یُرِیْدُ اَنْ یُّغْوِیَكُمْ ؕ هُوَ رَبُّكُمْ ۫ وَاِلَیْهِ تُرْجَعُوْنَ ۟ؕ أَمۡ يَقُولُونَ ٱفۡتَرَىٰهُۖ قُلۡ إِنِ ٱفۡتَرَيۡتُهُۥ فَعَلَيَّ إِجۡرَامِي وَأَنَا۠ بَرِيٓءٞ مِّمَّا تُجۡرِمُونَ ۝٣٥ أَمۡ یَقُولُونَ ٱفۡتَرَىٰهُۖ قُلۡ إِنِ ٱفۡتَرَیۡتُهُۥ فَعَلَیَّ إِجۡرَامِی وَأَنَا۠ بَرِیۤءࣱ مِّمَّا تُجۡرِمُونَ ۝٣٥ اَمْ یَقُوْلُوْنَ افْتَرٰىهُ ؕ قُلْ اِنِ افْتَرَیْتُهٗ فَعَلَیَّ اِجْرَامِیْ وَاَنَا بَرِیْٓءٌ مِّمَّا تُجْرِمُوْنَ ۟۠ وَأُوحِيَ إِلَىٰ یَّعْصِمُنِیْ مِنَ الْمَآءِ ؕ قَالَ لَا عَاصِمَ الْیَوْمَ مِنْ اَمْرِ اللّٰهِ اِلَّا مَنْ رَّحِمَ ۚ وَحَالَ بَیْنَهُمَا الْمَوْجُ فَكَانَ مِنَ الْمُغْرَقِیْنَ ۟ وَقِيلَ يَٰٓأَرۡضُ ٱبۡلَعِي مَآءَكِ وَيَٰسَمَآءُ أَقۡلِعِي وَغِيضَ ٱلۡمَآءُ وَقُضِيَ ٱلۡأَمۡرُ وَٱسۡتَوَتۡ عَلَى ٱلۡجُودِيِّۖ وَقِيلَ بُعۡدٗا لِّلۡقَوۡمِ ٱلظَّٰلِمِينَ ۝٤٤ وَقِیلَ یَٰۤأَرۡضُ ٱبۡلَعِی مَاۤءَكِ وَیَٰسَمَاۤءُ أَقۡلِعِی وَغِیضَ ٱلۡمَاۤءُ وَقُضِیَ ٱلۡأَمۡرُ وَٱسۡتَوَتۡ عَلَى ٱلۡجُودِیِّۖ وَقِیلَ بُعۡدࣰا لِّلۡقَوۡمِ ٱلظَّٰلِمِینَ ۝٤٤ وَقِیْلَ یٰۤاَرْضُ ابْلَعِیْ مَآءَكِ وَیٰسَمَآءُ اَقْلِعِیْ وَغِیْضَ الْمَآءُ وَقُضِیَ الْاَمْرُ وَاسْتَوَتْ عَلَی الْجُوْدِیِّ وَقِیْلَ بُعْدًا لِّلْقَوْمِ الظّٰلِمِیْنَ ۟ وَنَادَىٰ نُوحٞ رَّبَّهُۥ فَقَالَ رَبِّ إِنَّ ٱبۡنِي مِنۡ أَهۡلِي وَإِنَّ وَعۡدَكَ ٱلۡحَقُّ وَأَنتَ أَحۡكَمُ ٱلۡحَٰكِمِينَ ۝٤٥ وَنَادَىٰ نُوحࣱ رَّبَّهُۥ فَقَالَ رَبِّ إِنَّ ٱبۡنِی مِنۡ أَهۡلِی وَإِنَّ وَعۡدَكَ ٱلۡحَقُّ وَأَنتَ أَحۡكَمُ ٱلۡحَٰكِمِینَ ۝٤٥ وَنَادٰی نُوْحٌ رَّبَّهٗ فَقَالَ رَبِّ اِنَّ ابْنِیْ مِنْ اَهْلِیْ وَاِنَّ وَعْدَكَ الْحَقُّ وَاَنْتَ اَحْكَمُ الْحٰكِمِیْنَ ۟ قَالَ يَٰنُوحُ إِنَّهُۥ لَيۡسَ مِنۡ أَهۡلِكَۖ إِنَّهُۥ عَمَلٌ غَيۡرُ صَٰلِحٖۖ فَلَا تَسۡـَٔلۡنِ مَا لَيۡسَ لَكَ بِهِۦ عِلۡمٌۖ إِنِّيٓ أَعِظُكَ أَن تَكُونَ مِنَ ٱلۡجَٰهِلِينَ ۝٤٦ قَالَ یَٰنُوحُ إِنَّهُۥ لَیۡسَ مِنۡ أَهۡلِكَۖ إِنَّهُۥ عَمَلٌ غَیۡرُ صَٰلِحࣲۖ فَلَا تَسۡءَلۡنِ مَا لَیۡسَ لَكَ بِهِۦ عِلۡمٌۖ إِنِّیۤ أَعِظُكَ أَن تَكُونَ مِنَ ٱلۡجَٰهِلِینَ ۝٤٦ قَالَ یٰنُوْحُ اِنَّهٗ لَیْسَ مِنْ اَهْلِكَ ۚ اِنَّهٗ عَمَلٌ غَیْرُ صَالِحٍ ۖؗۗ فَلَا تَسْـَٔلْنِ مَا لَیْسَ لَكَ بِهٖ عِلْمٌ ؕ اِنِّیْۤ اَعِظُكَ اَنْ تَكُوْنَ مِنَ الْجٰهِلِیْنَ ۟ قَالَ رَبِّ إِنِّيٓ أَعُوذُ بِكَ أَنۡ أَسۡـَٔلَكَ مَا لَيۡسَ لِي بِهِۦ عِلۡمٞۖ وَإِلَّا تَغۡفِرۡ لِي وَتَرۡحَمۡنِيٓ أَكُن مِّنَ ٱلۡخَٰسِرِينَ ۝٤٧ قَالَ رَبِّ إِنِّیۤ أَعُوذُ بِكَ أَنۡ أَسۡءَلَكَ مَا لَیۡسَ لِی بِهِۦ عِلۡمࣱۖ وَإِلَّا تَغۡفِرۡ لِی وَتَرۡحَمۡنِیۤ أَكُن مِّنَ ٱلۡخَٰسِرِینَ ۝٤٧ قَالَ رَبِّ اِنِّیْۤ اَعُوْذُ بِكَ اَنْ اَسْـَٔلَكَ مَا لَیْسَ لِیْ بِهٖ عِلْمٌ ؕ وَاِلَّا تَغْفِرْ لِیْ وَتَرْحَمْنِیْۤ اَكُنْ مِّنَ الْخٰسِرِیْنَ ۟ قِيلَ يَٰنُوحُ ٱهۡبِطۡ بِسَلَٰمٖ مِّنَّا وَبَرَكَٰتٍ عَلَيۡكَ وَعَلَىٰٓ أُمَمٖ مِّمَّن مَّعَكَۚ وَأُمَمٞ سَنُمَتِّعُهُمۡ ثُمَّ يَمَسُّهُم مِّنَّا عَذَابٌ أَلِيمٞ ۝٤٨ قِیلَ یَٰنُوحُ ٱهۡبِطۡ بِسَلَٰمࣲ مِّنَّا وَبَرَكَٰتٍ عَلَیۡكَ وَعَلَىٰۤ أُمَمࣲ مِّمَّن مَّعَكَۚ وَأُمَمࣱ سَنُمَتِّعُهُمۡ ثُمَّ یَمَسُّهُم مِّنَّا عَذَابٌ أَلِیمࣱ ۝٤٨ قِیْلَ یٰنُوْحُ اهْبِطْ بِسَلٰمٍ مِّنَّا وَبَرَكٰتٍ عَلَیْكَ وَعَلٰۤی اُمَمٍ مِّمَّنْ مَّعَكَ ؕ وَاُمَمٌ سَنُمَتِّعُهُمْ ثُمَّ یَمَسُّهُمْ مِّنَّا عَذَابٌ اَلِیْمٌ ۟ تِلۡكَ مِنۡ أَنۢبَآءِ ٱلۡغَيۡبِ نُوحِيهَآ إِلَيۡكَۖ مَا كُنتَ تَعۡلَمُهَآ أَنتَ وَلَا قَوۡمُكَ مِن قَبۡلِ هَٰذَاۖ فَٱصۡبِرۡۖ إِنَّ ٱلۡعَٰقِبَةَ لِلۡمُتَّقِينَ ۝٤٩ تِلۡكَ مِنۡ أَنۢبَاۤءِ ٱلۡغَیۡبِ نُوحِیهَاۤ إِلَیۡكَۖ مَا كُنتَ تَعۡلَمُهَاۤ أَنتَ وَلَا قَوۡمُكَ مِن قَبۡلِ هَٰذَاۖ فَٱصۡبِرۡۖ إِنَّ ٱلۡعَٰقِبَةَ لِلۡمُتَّقِینَ ۝٤٩ تِلْكَ مِنْ اَنْۢبَآءِ الْغَیْبِ نُوْحِیْهَاۤ اِلَیْكَ ۚ مَا كُنْتَ تَعْلَمُهَاۤ اَنْتَ وَلَا قَوْمُكَ مِنْ قَبْلِ هٰذَا ۛؕ فَاصْبِرْ ۛؕ اِنَّ الْعَاقِبَةَ لِلْمُتَّقِیْنَ ۟۠ وَإِلَىٰ عَادٍ أَخَاهُمۡ هُودٗاۚ قَالَ يَٰقَوۡمِ ٱعۡبُدُواْ ٱللَّهَ مَا لَكُم مِّنۡ إِلَٰهٍ غَيۡرُهُۥٓۖ إِنۡ أَنتُمۡ إِلَّا مُفۡتَرُونَ ۝٥٠ وَإِلَىٰ عَادٍ أَخَاهُمۡ هُودࣰاۚ قَالَ یَٰقَوۡمِ ٱعۡبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنۡ إِلَٰهٍ غَیۡرُهُۥۤۖ إِنۡ أَنتُمۡ إِلَّا مُفۡتَرُونَ ۝٥٠ وَاِلٰی عَادٍ اَخَاهُمْ هُوْدًا ؕ قَالَ یٰقَوْمِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَیْرُهٗ ؕ اِنْ اَنْتُمْ اِلَّا مُفْتَرُوْنَ ۟ يَٰقَوۡمِ لَآ أَسۡـَٔلُكُمۡ عَلَيۡهِ أَجۡرًاۖ إِنۡ أَجۡرِيَ إِلَّا عَلَى ٱلَّذِي فَطَرَنِيٓۚ أَفَلَا تَعۡقِلُونَ ۝٥١ یَٰقَوۡمِ لَاۤ أَسۡءَلُكُمۡ عَلَیۡهِ أَجۡرًاۖ إِنۡ أَجۡرِیَ إِلَّا عَلَى ٱلَّذِی فَطَرَنِیۤۚ أَفَلَا تَعۡقِلُونَ ۝٥١ یٰقَوْمِ لَاۤ اَسْـَٔلُكُمْ عَلَیْهِ اَجْرًا ؕ اِنْ اَجْرِیَ اِلَّا عَلَی الَّذِیْ فَطَرَنِیْ ؕ اَفَلَا تَعْقِلُوْنَ ۟ وَيَٰقَوۡمِ ٱسۡتَغۡفِرُواْ رَبَّكُمۡ ثُمَّ تُوبُوٓاْ إِلَيۡهِ يُرۡسِلِ ٱلسَّمَآءَ عَلَيۡكُم مِّدۡرَارٗا وَيَزِدۡكُمۡ قُوَّةً إِلَىٰ قُوَّتِكُمۡ وَلَا تَتَوَلَّوۡاْ مُجۡرِمِينَ ۝٥٢ وَیَٰقَوۡمِ ٱسۡتَغۡفِرُوا۟ رَبَّكُمۡ ثُمَّ تُوبُوۤا۟ إِلَیۡهِ یُرۡسِلِ ٱلسَّمَاۤءَ عَلَیۡكُم مِّدۡرَارࣰا وَیَزِدۡكُمۡ قُوَّةً إِلَىٰ قُوَّتِكُمۡ وَلَا تَتَوَلَّوۡا۟ مُجۡرِمِینَ ۝٥٢ وَیٰقَوْمِ اسْتَغْفِرُوْا رَبَّكُمْ ثُمَّ تُوْبُوْۤا اِلَیْهِ یُرْسِلِ السَّمَآءَ عَلَیْكُمْ مِّدْرَارًا وَّیَزِدْكُمْ قُوَّةً اِلٰی قُوَّتِكُمْ وَلَا تَتَوَلَّوْا مُجْرِمِیْنَ ۟ قَالُواْ يَٰهُودُ مَا جِئۡتَنَا بِبَيِّنَةٖ وَمَا نَحۡنُ بِتَارِكِيٓ ءَالِهَتِنَا عَن قَوۡلِكَ وَمَا نَحۡنُ لَكَ بِمُؤۡمِنِينَ ۝٥٣ قَالُوا۟ یَٰهُودُ مَا جِئۡتَنَا بِبَیِّنَةࣲ وَمَا نَحۡنُ بِتَارِكِیۤ ءَالِهَتِنَا عَن قَوۡلِكَ وَمَا نَحۡنُ لَكَ بِمُؤۡمِنِینَ ۝٥٣ قَالُوْا یٰهُوْدُ مَا جِئْتَنَا بِبَیِّنَةٍ وَّمَا نَحْنُ بِتَارِكِیْۤ اٰلِهَتِنَا عَنْ قَوْلِكَ وَمَا نَحْنُ لَكَ بِمُؤْمِنِیْنَ ۟ إِن نَّقُولُ إِلَّا ٱعۡتَرَىٰكَ بَعۡضُ ءَالِهَتِنَا بِسُوٓءٖۗ قَالَ إِنِّيٓ أُشۡهِدُ ٱللَّهَ وَٱشۡهَدُوٓاْ أَنِّي بَرِيٓءٞ مِّمَّا تُشۡرِكُونَ ۝٥٤ إِن نَّقُولُ إِلَّا ٱعۡتَرَىٰكَ بَعۡضُ ءَالِهَتِنَا بِسُوۤءࣲۗ قَالَ إِنِّیۤ أُشۡهِدُ ٱللَّهَ وَٱشۡهَدُوۤا۟ أَنِّی بَرِیۤءࣱ مِّمَّا تُشۡرِكُونَ ۝٥٤ اِنْ نَّقُوْلُ اِلَّا اعْتَرٰىكَ بَعْضُ اٰلِهَتِنَا بِسُوْٓءٍ ؕ قَالَ اِنِّیْۤ اُشْهِدُ اللّٰهَ وَاشْهَدُوْۤا اَنِّیْ بَرِیْٓءٌ مِّمَّا تُشْرِكُوْنَ ۟ۙ مِن دُونِهِۦۖ فَكِيدُونِي جَمِيعٗا ثُمَّ لَا تُنظِرُونِ ۝٥٥ مِن دُونِهِۦۖ فَكِیدُونِی جَمِیعࣰا ثُمَّ لَا تُنظِرُونِ ۝٥٥ مِنْ دُوْنِهٖ فَكِیْدُوْنِیْ جَمِیْعًا ثُمَّ لَا تُنْظِرُوْنِ ۟ إِنِّي تَوَكَّلۡتُ عَلَى ٱللَّهِ رَبِّي وَرَبِّكُمۚ مَّا مِن دَآبَّةٍ إِلَّا هُوَ ءَاخِذُۢ بِنَاصِيَتِهَآۚ إِنَّ رَبِّي عَلَىٰ صِرَٰطٖ مُّسۡتَقِيمٖ ۝٥٦ إِنِّی تَوَكَّلۡتُ عَلَى ٱللَّهِ رَبِّی وَرَبِّكُمۚ مَّا مِن دَاۤبَّةٍ إِلَّا هُوَ ءَاخِذُۢ بِنَاصِیَتِهَاۤۚ إِنَّ رَبِّی عَلَىٰ صِرَٰطࣲ مُّسۡتَقِیمࣲ ۝٥٦ اِنِّیْ تَوَكَّلْتُ عَلَی اللّٰهِ رَبِّیْ وَرَبِّكُمْ ؕ مَا مِنْ دَآبَّةٍ اِلَّا هُوَ اٰخِذٌ بِنَاصِیَتِهَا ؕ اِنَّ رَبِّیْ عَلٰی صِرَاطٍ مُّسْتَقِیْمٍ ۟ فَإِن تَوَلَّوۡاْ فَقَدۡ أَبۡلَغۡتُكُم مَّآ أُرۡسِلۡتُ بِهِۦٓ إِلَيۡكُمۡۚ وَيَسۡتَخۡلِفُ رَبِّي قَوۡمًا غَيۡرَكُمۡ وَلَا تَضُرُّونَهُۥ شَيۡـًٔاۚ إِنَّ رَبِّي عَلَىٰ كُلِّ شَيۡءٍ حَفِيظٞ ۝٥٧ فَإِن تَوَلَّوۡا۟ فَقَدۡ أَبۡلَغۡتُكُم مَّاۤ أُرۡسِلۡتُ بِهِۦۤ إِلَیۡكُمۡۚ وَیَسۡتَخۡلِفُ رَبِّی قَوۡمًا غَیۡرَكُمۡ وَلَا تَضُرُّونَهُۥ شَیۡءًاۚ إِنَّ رَبِّی عَلَىٰ كُلِّ شَیۡءٍ حَفِیظࣱ ۝٥٧ فَاِنْ تَوَلَّوْا فَقَدْ اَبْلَغْتُكُمْ مَّاۤ اُرْسِلْتُ بِهٖۤ اِلَیْكُمْ ؕ وَیَسْتَخْلِفُ رَبِّیْ قَوْمًا غَیْرَكُمْ ۚ وَلَا تَضُرُّوْنَهٗ شَیْـًٔا ؕ اِنَّ رَبِّیْ عَلٰی كُلِّ شَیْءٍ حَفِیْظٌ ۟ وَلَمَّا جَآءَ أَمۡرُنَا نَجَّيۡنَا هُودٗا وَٱلَّذِينَ ءَامَنُواْ مَعَهُۥ بِرَحۡمَةٖ مِّنَّا وَنَجَّيۡنَٰهُم مِّنۡ عَذَابٍ غَلِيظٖ ۝٥٨ وَلَمَّا جَاۤءَ أَمۡرُنَا نَجَّیۡنَا هُودࣰا وَٱلَّذِینَ ءَامَنُوا۟ مَعَهُۥ بِرَحۡمَةࣲ مِّنَّا وَنَجَّیۡنَٰهُم مِّنۡ عَذَابٍ غَلِیظࣲ ۝٥٨ وَلَمَّا جَآءَ اَمْرُنَا نَجَّیْنَا هُوْدًا وَّالَّذِیْنَ اٰمَنُوْا مَعَهٗ بِرَحْمَةٍ مِّنَّا ۚ وَنَجَّیْنٰهُمْ مِّنْ عَذَابٍ غَلِیْظٍ ۟ وَتِلۡكَ عَادٞۖ جَحَدُواْ بِـَٔايَٰتِ رَبِّهِمۡ وَعَصَوۡاْ رُسُلَهُۥ وَٱتَّبَعُوٓاْ أَمۡرَ كُلِّ جَبَّارٍ عَنِيدٖ ۝٥٩ وَتِلۡكَ عَادࣱۖ جَحَدُوا۟ بِءَایَٰتِ رَبِّهِمۡ وَعَصَوۡا۟ رُسُلَهُۥ وَٱتَّبَعُوۤا۟ أَمۡرَ كُلِّ جَبَّارٍ عَنِیدࣲ ۝٥٩ وَتِلْكَ عَادٌ ۙ۫ جَحَدُوْا بِاٰیٰتِ رَبِّهِمْ وَعَصَوْا رُسُلَهٗ وَاتَّبَعُوْۤا اَمْرَ كُلِّ جَبَّارٍ عَنِیْدٍ ۟ وَأُتۡبِعُواْ فِي هَٰذِهِ ٱلدُّنۡيَا لَعۡنَةٗ وَيَوۡمَ ٱلۡقِيَٰمَةِۗ أَلَآ إِنَّ عَادٗا كَفَرُواْ رَبَّهُمۡۗ أَلَا بُعۡدٗا لِّعَادٖ قَوۡمِ هُودٖ ۝٦٠ وَأُتۡبِعُوا۟ فِی هَٰذِهِ ٱلدُّنۡیَا لَعۡنَةࣰ وَیَوۡمَ ٱلۡقِیَٰمَةِۗ أَلَاۤ إِنَّ عَادࣰا كَفَرُوا۟ رَبَّهُمۡۗ أَلَا بُعۡدࣰا لِّعَادࣲ قَوۡمِ هُودࣲ ۝٦٠ وَاُتْبِعُوْا فِیْ هٰذِهِ الدُّنْیَا لَعْنَةً وَّیَوْمَ الْقِیٰمَةِ ؕ اَلَاۤ اِنَّ عَادًا كَفَرُوْا رَبَّهُمْ ؕ اَلَا بُعْدًا لِّعَادٍ قَوْمِ هُوْدٍ ۟۠ وَإِلَىٰ ثَمُودَ أَخَاهُمۡ صَٰلِحٗاۚ قَالَ يَٰقَوۡمِ ٱعۡبُدُواْ ٱللَّهَ مَا لَكُم مِّنۡ إِلَٰهٍ غَيۡرُهُۥۖ هُوَ أَنشَأَكُم مِّنَ ٱلۡأَرۡضِ وَٱسۡتَعۡمَرَكُمۡ فِيهَا فَٱسۡتَغۡفِرُوهُ ثُمَّ تُوبُوٓاْ إِلَيۡهِۚ إِنَّ رَبِّي قَرِيبٞ مُّجِيبٞ ۝٦١ وَإِلَىٰ ثَمُودَ أَخَاهُمۡ صَٰلِحࣰاۚ قَالَ یَٰقَوۡمِ ٱعۡبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنۡ إِلَٰهٍ غَیۡرُهُۥۖ هُوَ أَنشَأَكُم مِّنَ ٱلۡأَرۡضِ وَٱسۡتَعۡمَرَكُمۡ فِیهَا فَٱسۡتَغۡفِرُوهُ ثُمَّ تُوبُوۤا۟ إِلَیۡهِۚ إِنَّ رَبِّی قَرِیبࣱ مُّجِیبࣱ ۝٦١ وَاِلٰی ثَمُوْدَ اَخَاهُمْ صٰلِحًا ۘ قَالَ یٰقَوْمِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَیْرُهٗ ؕ هُوَ اَنْشَاَكُمْ مِّنَ الْاَرْضِ وَاسْتَعْمَرَكُمْ فِیْهَا فَاسْتَغْفِرُوْهُ ثُمَّ تُوْبُوْۤا اِلَیْهِ ؕ اِنَّ رَبِّیْ قَرِیْبٌ مُّجِیْبٌ ۟ قَالُواْ يَٰصَٰلِحُ قَدۡ كُنتَ فِينَا مَرۡجُوّٗا قَبۡلَ هَٰذَآۖ أَتَنۡهَىٰنَآ أَن نَّعۡبُدَ مَا يَعۡبُدُ ءَابَآؤُنَا وَإِنَّنَا لَفِي شَكّٖ مِّمَّا تَدۡعُونَآ إِلَيۡهِ مُرِيبٖ ۝٦٢ قَالُوا۟ یَٰصَٰلِحُ قَدۡ كُنتَ فِینَا مَرۡجُوࣰّا قَبۡلَ هَٰذَاۤۖ أَتَنۡهَىٰنَاۤ أَن نَّعۡبُدَ مَا یَعۡبُدُ ءَابَاۤؤُنَا وَإِنَّنَا لَفِی شَكࣲّ مِّمَّا تَدۡعُونَاۤ إِلَیۡهِ مُرِیبࣲ ۝٦٢ قَالُوْا یٰصٰلِحُ قَدْ كُنْتَ فِیْنَا مَرْجُوًّا قَبْلَ هٰذَاۤ اَتَنْهٰىنَاۤ اَنْ نَّعْبُدَ مَا یَعْبُدُ اٰبَآؤُنَا وَاِنَّنَا لَفِیْ شَكٍّ مِّمَّا تَدْعُوْنَاۤ اِلَیْهِ مُرِیْبٍ ۟ قَالَ يَٰقَوۡمِ أَرَءَيۡتُمۡ إِن كُنتُ عَلَىٰ بَيِّنَةٖ مِّن رَّبِّي وَءَاتَىٰنِي مِنۡهُ رَحۡمَةٗ فَمَن يَنصُرُنِي مِنَ ٱللَّهِ إِنۡ عَصَيۡتُهُۥۖ فَمَا تَزِيدُونَنِي غَيۡرَ تَخۡسِيرٖ ۝٦٣ قَالَ یَٰقَوۡمِ أَرَءَیۡتُمۡ إِن كُنتُ عَلَىٰ بَیِّنَةࣲ مِّن رَّبِّی وَءَاتَىٰنِی مِنۡهُ رَحۡمَةࣰ فَمَن یَنصُرُنِی مِنَ ٱللَّهِ إِنۡ عَصَیۡتُهُۥۖ فَمَا تَزِیدُونَنِی غَیۡرَ تَخۡسِیرࣲ ۝٦٣ قَالَ یٰقَوْمِ اَرَءَیْتُمْ اِنْ كُنْتُ عَلٰی بَیِّنَةٍ مِّنْ رَّبِّیْ وَاٰتٰىنِیْ مِنْهُ رَحْمَةً فَمَنْ یَّنْصُرُنِیْ مِنَ اللّٰهِ اِنْ عَصَیْتُهٗ ۫ فَمَا تَزِیْدُوْنَنِیْ غَیْرَ تَخْسِیْرٍ ۟ وَيَٰقَوۡمِ هَٰذِهِۦ نَاقَةُ ٱللَّهِ لَكُمۡ ءَايَةٗۖ فَذَرُوهَا تَأۡكُلۡ فِيٓ أَرۡضِ ٱللَّهِۖ وَلَا تَمَسُّوهَا بِسُوٓءٖ فَيَأۡخُذَكُمۡ عَذَابٞ قَرِيبٞ ۝٦٤ وَیَٰقَوۡمِ هَٰذِهِۦ نَاقَةُ ٱللَّهِ لَكُمۡ ءَایَةࣰۖ فَذَرُوهَا تَأۡكُلۡ فِیۤ أَرۡضِ ٱللَّهِۖ وَلَا تَمَسُّوهَا بِسُوۤءࣲ فَیَأۡخُذَكُمۡ عَذَابࣱ قَرِیبࣱ ۝٦٤ وَیٰقَوْمِ هٰذِهٖ نَاقَةُ اللّٰهِ لَكُمْ اٰیَةً فَذَرُوْهَا تَاْكُلْ فِیْۤ اَرْضِ اللّٰهِ وَلَا تَمَسُّوْهَا بِسُوْٓءٍ فَیَاْخُذَكُمْ عَذَابٌ قَرِیْبٌ ۟ فَعَقَرُوهَا فَقَالَ تَمَتَّعُواْ فِي دَارِكُمۡ ثَلَٰثَةَ أَيَّامٖۖ ذَٰلِكَ وَعۡدٌ غَيۡرُ مَكۡذُوبٖ ۝٦٥ فَعَقَرُوهَا فَقَالَ تَمَتَّعُوا۟ فِی دَارِكُمۡ ثَلَٰثَةَ أَیَّامࣲۖ ذَٰلِكَ وَعۡدٌ غَیۡرُ مَكۡذُوبࣲ ۝٦٥ فَعَقَرُوْهَا فَقَالَ تَمَتَّعُوْا فِیْ دَارِكُمْ ثَلٰثَةَ اَیَّامٍ ؕ ذٰلِكَ وَعْدٌ غَیْرُ مَكْذُوْبٍ ۟ فَلَمَّا جَآءَ أَمۡرُنَا نَجَّيۡنَا صَٰلِحٗا وَٱلَّذِينَ ءَامَنُواْ مَعَهُۥ بِرَحۡمَةٖ مِّنَّا وَمِنۡ خِزۡيِ يَوۡمِئِذٍۚ إِنَّ رَبَّكَ هُوَ ٱلۡقَوِيُّ ٱلۡعَزِيزُ ۝٦٦ فَلَمَّا جَاۤءَ أَمۡرُنَا نَجَّیۡنَا صَٰلِحࣰا وَٱلَّذِینَ ءَامَنُوا۟ مَعَهُۥ بِرَحۡمَةࣲ مِّنَّا وَمِنۡ خِزۡیِ یَوۡمِئِذٍۚ إِنَّ رَبَّكَ هُوَ ٱلۡقَوِیُّ ٱلۡعَزِیزُ ۝٦٦ فَلَمَّا جَآءَ اَمْرُنَا نَجَّیْنَا صٰلِحًا وَّالَّذِیْنَ اٰمَنُوْا مَعَهٗ بِرَحْمَةٍ مِّنَّا وَمِنْ خِزْیِ یَوْمِىِٕذٍ ؕ اِنَّ رَبَّكَ هُوَ الْقَوِیُّ الْعَزِیْزُ ۟ وَأَخَذَ ٱلَّذِينَ ظَلَمُواْ ٱلصَّيۡحَةُ فَأَصۡبَحُواْ فِي دِيَٰرِهِمۡ جَٰثِمِينَ ۝٦٧ وَأَخَذَ ٱلَّذِینَ ظَلَمُوا۟ ٱلصَّیۡحَةُ فَأَصۡبَحُوا۟ فِی دِیَٰرِهِمۡ جَٰثِمِینَ ۝٦٧ وَاَخَذَ الَّذِیْنَ ظَلَمُوا أَلَاۤ إِنَّ ثَمُودَا۟ كَفَرُوا۟ رَبَّهُمۡۗ أَلَا بُعۡدࣰا لِّثَمُودَ ۝٦٨ كَاَنْ لَّمْ یَغْنَوْا فِیْهَا ؕ اَلَاۤ اِنَّ ثَمُوْدَاۡ كَفَرُوْا رَبَّهُمْ ؕ اَلَا بُعْدًا لِّثَمُوْدَ ۟۠ وَلَقَدۡ جَآءَتۡ رَءَآ أَيۡدِيَهُمۡ لَا تَصِلُ إِلَيۡهِ نَكِرَهُمۡ وَأَوۡجَسَ مِنۡهُمۡ خِيفَةٗۚ قَالُواْ لَا تَخَفۡ إِنَّآ أُرۡسِلۡنَآ إِلَىٰ قَوۡمِ لُوطٖ ۝٧٠ فَلَمَّا رَءَاۤ أَیۡدِیَهُمۡ لَا تَصِلُ إِلَیۡهِ عَجِیبࣱ ۝٧٢ قَالَتْ یٰوَیْلَتٰۤی ءَاَلِدُ وَاَنَا عَجُوْزٌ وَّهٰذَا بَعْلِیْ شَیْخًا ؕ اِنَّ هٰذَا لَشَیْءٌ عَجِیْبٌ ۟ قَالُوٓاْ أَتَعۡجَبِينَ مِنۡ أَمۡرِ ٱللَّهِۖ رَحۡمَتُ ٱللَّهِ وَبَرَكَٰتُهُۥ عَلَيۡكُمۡ أَهۡلَ ٱلۡبَيۡتِۚ إِنَّهُۥ حَمِيدٞ مَّجِيدٞ ۝٧٣ قَالُوۤا۟ أَتَعۡجَبِینَ مِنۡ أَمۡرِ ٱللَّهِۖ رَحۡمَتُ ٱللَّهِ وَبَرَكَٰتُهُۥ عَلَیۡكُمۡ أَهۡلَ ٱلۡبَیۡتِۚ إِنَّهُۥ حَمِیدࣱ مَّجِیدࣱ ۝٧٣ قَالُوْۤا اَتَعْجَبِیْنَ مِنْ اَمْرِ اللّٰهِ رَحْمَتُ اللّٰهِ وَبَرَكٰتُهٗ عَلَیْكُمْ اَهْلَ الْبَیْتِ ؕ اِنَّهٗ حَمِیْدٌ مَّجِیْدٌ ۟ فَلَمَّا ذَهَبَ عَنۡ إِبۡرَٰهِيمَ ٱلرَّوۡعُ وَجَآءَتۡهُ ٱلۡبُشۡرَىٰ يُجَٰدِلُنَا فِي قَوۡمِ لُوطٍ ۝٧٤ فَلَمَّا ذَهَبَ عَنۡ إِبۡرَٰهِیمَ ٱلرَّوۡعُ وَجَاۤءَتۡهُ ٱلۡبُشۡرَىٰ یُجَٰدِلُنَا فِی قَوۡمِ لُوطٍ ۝٧٤ فَلَمَّا ذَهَبَ عَنْ اِبْرٰهِیْمَ الرَّوْعُ وَجَآءَتْهُ الْبُشْرٰی یُجَادِلُنَا فِیْ قَوْمِ لُوْطٍ ۟ؕ إِنَّ إِبۡرَٰهِيمَ لَحَلِيمٌ أَوَّٰهٞ مُّنِيبٞ ۝٧٥ إِنَّ إِبۡرَٰهِیمَ لَحَلِیمٌ أَوَّٰهࣱ مُّنِیبࣱ ۝٧٥ اِنَّ اِبْرٰهِیْمَ لَحَلِیْمٌ اَوَّاهٌ مُّنِیْبٌ ۟ يَٰٓإِبۡرَٰهِيمُ أَعۡرِضۡ عَنۡ هَٰذَآۖ إِنَّهُۥ قَدۡ جَآءَ أَمۡرُ رَبِّكَۖ وَإِنَّهُمۡ ءَاتِيهِمۡ عَذَابٌ غَيۡرُ مَرۡدُودٖ ۝٧٦ یَٰۤإِبۡرَٰهِیمُ أَعۡرِضۡ عَنۡ هَٰذَاۤۖ إِنَّهُۥ قَدۡ جَاۤءَ أَمۡرُ رَبِّكَۖ وَإِنَّهُمۡ ءَاتِیهِمۡ عَذَابٌ غَیۡرُ مَرۡدُودࣲ ۝٧٦ یٰۤاِبْرٰهِیْمُ اَعْرِضْ عَنْ هٰذَا ۚ اِنَّهٗ قَدْ جَآءَ اَمْرُ رَبِّكَ ۚ وَاِنَّهُمْ اٰتِیْهِمْ عَذَابٌ غَیْرُ مَرْدُوْدٍ ۟ وَلَمَّا جَآءَتۡ رُسُلُنَا لُوطٗا سِيٓءَ بِهِمۡ وَضَاقَ بِهِمۡ ذَرۡعٗا وَقَالَ هَٰذَا يَوۡمٌ عَصِيبٞ ۝٧٧ وَلَمَّا جَاۤءَتۡ رُسُلُنَا لُوطࣰا سِیۤءَ بِهِمۡ وَضَاقَ بِهِمۡ ذَرۡعࣰا وَقَالَ هَٰذَا یَوۡمٌ عَصِیبࣱ ۝٧٧ وَلَمَّا جَآءَتْ رُسُلُنَا لُوْطًا سِیْٓءَ بِهِمْ وَضَاقَ بِهِمْ ذَرْعًا وَّقَالَ هٰذَا یَوْمٌ عَصِیْبٌ ۟ وَجَآءَهُۥ قَوۡمُهُۥ يُهۡرَعُونَ إِلَيۡهِ وَمِن قَبۡلُ كَانُواْ يَعۡمَلُونَ ٱلسَّيِّـَٔاتِۚ قَالَ يَٰقَوۡمِ هَٰٓؤُلَآءِ بَنَاتِي هُنَّ أَطۡهَرُ لَكُمۡۖ فَٱتَّقُواْ ٱللَّهَ وَلَا تُخۡزُونِ فِي ضَيۡفِيٓۖ أَلَيۡسَ مِنكُمۡ رَجُلٞ رَّشِيدٞ ۝٧٨ وَجَاۤءَهُۥ قَوۡمُهُۥ یُهۡرَعُونَ إِلَیۡهِ وَمِن قَبۡلُ كَانُوا۟ یَعۡمَلُونَ ٱلسَّیِّءَاتِۚ قَالَ یَٰقَوۡمِ هَٰۤؤُلَاۤءِ بَنَاتِی هُنَّ أَطۡهَرُ لَكُمۡۖ فَٱتَّقُوا۟ ٱللَّهَ وَلَا تُخۡزُونِ فِی ضَیۡفِیۤۖ أَلَیۡسَ مِنكُمۡ رَجُلࣱ رَّشِیدࣱ ۝٧٨ وَجَآءَهٗ قَوْمُهٗ یُهْرَعُوْنَ اِلَیْهِ ؕ وَمِنْ قَبْلُ كَانُوْا یَعْمَلُوْنَ السَّیِّاٰتِ ؕ قَالَ یٰقَوْمِ هٰۤؤُلَآءِ بَنَاتِیْ هُنَّ اَطْهَرُ لَكُمْ فَاتَّقُوا اللّٰهَ وَلَا تُخْزُوْنِ فِیْ ضَیْفِیْ ؕ اَلَیْسَ مِنْكُمْ رَجُلٌ رَّشِیْدٌ ۟ قَالُواْ لَقَدۡ عَلِمۡتَ مَا لَنَا فِي بَنَاتِكَ مِنۡ حَقّٖ وَإِنَّكَ لَتَعۡلَمُ مَا نُرِيدُ ۝٧٩ قَالُوا۟ لَقَدۡ عَلِمۡتَ مَا لَنَا فِی بَنَاتِكَ مِنۡ حَقࣲّ وَإِنَّكَ لَتَعۡلَمُ مَا نُرِیدُ ۝٧٩ قَالُوْا لَقَدْ عَلِمْتَ مَا لَنَا فِیْ بَنٰتِكَ مِنْ حَقٍّ ۚ وَاِنَّكَ لَتَعْلَمُ مَا نُرِیْدُ ۟ قَالَ لَوۡ أَنَّ لِي بِكُمۡ قُوَّةً أَوۡ ءَاوِيٓ إِلَىٰ رُكۡنٖ شَدِيدٖ ۝٨٠ قَالَ لَوۡ أَنَّ لِی بِكُمۡ قُوَّةً أَوۡ ءَاوِیۤ إِلَىٰ رُكۡنࣲ شَدِیدࣲ ۝٨٠ قَالَ لَوْ اَنَّ لِیْ بِكُمْ قُوَّةً اَوْ اٰوِیْۤ اِلٰی رُكْنٍ شَدِیْدٍ ۟ قَالُواْ يَٰلُوطُ إِنَّا رُسُلُ رَبِّكَ لَن يَصِلُوٓاْ إِلَيۡكَۖ فَأَسۡرِ بِأَهۡلِكَ بِقِطۡعٖ مِّنَ ٱلَّيۡلِ وَلَا يَلۡتَفِتۡ مِنكُمۡ أَحَدٌ إِلَّا ٱمۡرَأَتَكَۖ إِنَّهُۥ مُصِيبُهَا مَآ أَصَابَهُمۡۚ إِنَّ مَوۡعِدَهُمُ ٱلصُّبۡحُۚ أَلَيۡسَ ٱلصُّبۡحُ بِقَرِيبٖ ۝٨١ قَالُوا۟ یَٰلُوطُ إِنَّا رُسُلُ رَبِّكَ لَن یَصِلُوۤا۟ إِلَیۡكَۖ فَأَسۡرِ بِأَهۡلِكَ بِقِطۡعࣲ مِّنَ ٱلَّیۡلِ وَلَا یَلۡتَفِتۡ مِنكُمۡ أَحَدٌ إِلَّا ٱمۡرَأَتَكَۖ إِنَّهُۥ مُصِیبُهَا مَاۤ أَصَابَهُمۡۚ إِنَّ مَوۡعِدَهُمُ ٱلصُّبۡحُۚ أَلَیۡسَ ٱلصُّبۡحُ بِقَرِیبࣲ ۝٨١ قَالُوْا یٰلُوْطُ اِنَّا رُسُلُ رَبِّكَ لَنْ یَّصِلُوْۤا اِلَیْكَ فَاَسْرِ بِاَهْلِكَ بِقِطْعٍ مِّنَ الَّیْلِ وَلَا یَلْتَفِتْ مِنْكُمْ اَحَدٌ اِلَّا امْرَاَتَكَ ؕ اِنَّهٗ مُصِیْبُهَا مَاۤ اَصَابَهُمْ ؕ اِنَّ مَوْعِدَهُمُ الصُّبْحُ ؕ اَلَیْسَ الصُّبْحُ بِقَرِیْبٍ ۟ فَلَمَّا جَآءَ أَمۡرُنَا جَعَلۡنَا عَٰلِيَهَا سَافِلَهَا وَأَمۡطَرۡنَا عَلَيۡهَا حِجَارَةٗ مِّن سِجِّيلٖ مَّنضُودٖ ۝٨٢ فَلَمَّا جَاۤءَ أَمۡرُنَا جَعَلۡنَا عَٰلِیَهَا سَافِلَهَا وَأَمۡطَرۡنَا عَلَیۡهَا حِجَارَةࣰ مِّن سِجِّیلࣲ مَّنضُودࣲ ۝٨٢ فَلَمَّا جَآءَ اَمْرُنَا جَعَلْنَا عَالِیَهَا سَافِلَهَا وَاَمْطَرْنَا عَلَیْهَا حِجَارَةً مِّنْ سِجِّیْلٍ ۙ۬ مَّنْضُوْدٍ ۟ۙ مُّسَوَّمَةً عِندَ رَبِّكَۖ وَمَا هِيَ مِنَ ٱلظَّٰلِمِينَ بِبَعِيدٖ ۝٨٣ مُّسَوَّمَةً عِندَ رَبِّكَۖ وَمَا هِیَ مِنَ ٱلظَّٰلِمِینَ بِبَعِیدࣲ ۝٨٣ مُّسَوَّمَةً عِنْدَ رَبِّكَ ؕ وَمَا هِیَ مِنَ الظّٰلِمِیْنَ بِبَعِیْدٍ ۟۠ وَإِلَىٰ مَدۡيَنَ أَخَاهُمۡ شُعَيۡبٗاۚ قَالَ يَٰقَوۡمِ ٱعۡبُدُواْ ٱللَّهَ مَا لَكُم مِّنۡ إِلَٰهٍ غَيۡرُهُۥۖ وَلَا تَنقُصُواْ ٱلۡمِكۡيَالَ وَٱلۡمِيزَانَۖ إِنِّيٓ أَرَىٰكُم بِخَيۡرٖ وَإِنِّيٓ أَخَافُ عَلَيۡكُمۡ عَذَابَ يَوۡمٖ مُّحِيطٖ ۝٨٤ وَإِلَىٰ مَدۡیَنَ أَخَاهُمۡ شُعَیۡبࣰاۚ قَالَ یَٰقَوۡمِ ٱعۡبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنۡ إِلَٰهٍ غَیۡرُهُۥۖ وَلَا تَنقُصُوا۟ ٱلۡمِكۡیَالَ وَٱلۡمِیزَانَۖ إِنِّیۤ أَرَىٰكُم بِخَیۡرࣲ وَإِنِّیۤ أَخَافُ عَلَیۡكُمۡ عَذَابَ یَوۡمࣲ مُّحِیطࣲ ۝٨٤ وَاِلٰی مَدْیَنَ اَخَاهُمْ شُعَیْبًا ؕ قَالَ یٰقَوْمِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَیْرُهٗ ؕ وَلَا تَنْقُصُوا الْمِكْیَالَ وَالْمِیْزَانَ اِنِّیْۤ اَرٰىكُمْ بِخَیْرٍ وَّاِنِّیْۤ اَخَافُ عَلَیْكُمْ عَذَابَ یَوْمٍ مُّحِیْطٍ ۟ وَيَٰقَوۡمِ أَوۡفُواْ ٱلۡمِكۡيَالَ وَٱلۡمِيزَانَ بِٱلۡقِسۡطِۖ وَلَا تَبۡخَسُواْ لَأَنتَ ٱلۡحَلِیمُ ٱلرَّشِیدُ ۝٨٧ قَالُوْا یٰشُعَیْبُ اَصَلٰوتُكَ تَاْمُرُكَ اَنْ نَّتْرُكَ مَا یَعْبُدُ اٰبَآؤُنَاۤ اَوْ اَنْ نَّفْعَلَ فِیْۤ اَمْوَالِنَا مَا نَشٰٓؤُا ؕ اِنَّكَ لَاَنْتَ الْحَلِیْمُ الرَّشِیْدُ ۟ قَالَ يَٰقَوۡمِ أَرَءَيۡتُمۡ إِن كُنتُ عَلَىٰ بَيِّنَةٖ مِّن رَّبِّي وَرَزَقَنِي مِنۡهُ رِزۡقًا حَسَنٗاۚ وَمَآ أُرِيدُ أَنۡ أُخَالِفَكُمۡ إِلَىٰ مَآ أَنۡهَىٰكُمۡ عَنۡهُۚ إِنۡ أُرِيدُ إِلَّا ٱلۡإِصۡلَٰحَ مَا ٱسۡتَطَعۡتُۚ وَمَا تَوۡفِيقِيٓ إِلَّا بِٱللَّهِۚ عَلَيۡهِ تَوَكَّلۡتُ وَإِلَيۡهِ أُنِيبُ ۝٨٨ قَالَ یَٰقَوۡمِ أَرَءَیۡتُمۡ إِن كُنتُ عَلَىٰ بَیِّنَةࣲ مِّن رَّبِّی وَرَزَقَنِی مِنۡهُ رِزۡقًا حَسَنࣰاۚ وَمَاۤ أُرِیدُ أَنۡ أُخَالِفَكُمۡ إِلَىٰ مَاۤ أَنۡهَىٰكُمۡ عَنۡهُۚ إِنۡ أُرِیدُ إِلَّا ٱلۡإِصۡلَٰحَ مَا ٱسۡتَطَعۡتُۚ وَمَا تَوۡفِیقِیۤ إِلَّا بِٱللَّهِۚ عَلَیۡهِ تَوَكَّلۡتُ وَإِلَیۡهِ أُنِیبُ ۝٨٨ قَالَ یٰقَوْمِ اَرَءَیْتُمْ اِنْ كُنْتُ عَلٰی بَیِّنَةٍ مِّنْ رَّبِّیْ وَرَزَقَنِیْ مِنْهُ رِزْقًا حَسَنًا ؕ وَمَاۤ اُرِیْدُ اَنْ اُخَالِفَكُمْ اِلٰی مَاۤ اَنْهٰىكُمْ عَنْهُ ؕ اِنْ اُرِیْدُ اِلَّا الْاِصْلَاحَ مَا اسْتَطَعْتُ ؕ وَمَا تَوْفِیْقِیْۤ اِلَّا بِاللّٰهِ ؕ عَلَیْهِ تَوَكَّلْتُ وَاِلَیْهِ اُنِیْبُ ۟ وَيَٰقَوۡمِ لَا يَجۡرِمَنَّكُمۡ شِقَاقِيٓ أَن يُصِيبَكُم مِّثۡلُ مَآ أَصَابَ قَوۡمَ نُوحٍ أَوۡ قَوۡمَ هُودٍ أَوۡ قَوۡمَ صَٰلِحٖۚ وَمَا قَوۡمُ لُوطٖ مِّنكُم بِبَعِيدٖ ۝٨٩ وَیَٰقَوۡمِ لَا یَجۡرِمَنَّكُمۡ شِقَاقِیۤ أَن یُصِیبَكُم مِّثۡلُ مَاۤ أَصَابَ قَوۡمَ نُوحٍ أَوۡ قَوۡمَ هُودٍ أَوۡ قَوۡمَ صَٰلِحࣲۚ وَمَا قَوۡمُ لُوطࣲ مِّنكُم بِبَعِیدࣲ ۝٨٩ وَیٰقَوْمِ لَا یَجْرِمَنَّكُمْ شِقَاقِیْۤ اَنْ یُّصِیْبَكُمْ مِّثْلُ مَاۤ اَصَابَ قَوْمَ نُوْحٍ اَوْ قَوْمَ هُوْدٍ اَوْ قَوْمَ صٰلِحٍ ؕ وَمَا قَوْمُ لُوْطٍ مِّنْكُمْ بِبَعِیْدٍ ۟ وَٱسۡتَغۡفِرُواْ رَبَّكُمۡ ثُمَّ تُوبُوٓاْ إِلَيۡهِۚ إِنَّ رَبِّي رَحِيمٞ وَدُودٞ ۝٩٠ وَٱسۡتَغۡفِرُوا۟ رَبَّكُمۡ ثُمَّ تُوبُوۤا۟ إِلَیۡهِۚ إِنَّ رَبِّی رَحِیمࣱ وَدُودࣱ ۝٩٠ وَاسْتَغْفِرُوْا رَبَّكُمْ ثُمَّ تُوْبُوْۤا اِلَیْهِ ؕ اِنَّ رَبِّیْ رَحِیْمٌ وَّدُوْدٌ ۟ قَالُواْ يَٰشُعَيۡبُ مَا نَفۡقَهُ كَثِيرٗا مِّمَّا تَقُولُ وَإِنَّا لَنَرَىٰكَ فِينَا ضَعِيفٗاۖ وَلَوۡلَا رَهۡطُكَ لَرَجَمۡنَٰكَۖ وَمَآ أَنتَ عَلَيۡنَا بِعَزِيزٖ ۝٩١ قَالُوا۟ یَٰشُعَیۡبُ مَا نَفۡقَهُ كَثِیرࣰا مِّمَّا تَقُولُ وَإِنَّا لَنَرَىٰكَ فِینَا ضَعِیفࣰاۖ وَلَوۡلَا رَهۡطُكَ لَرَجَمۡنَٰكَۖ وَمَاۤ أَنتَ عَلَیۡنَا بِعَزِیزࣲ ۝٩١ قَالُوْا یٰشُعَیْبُ مَا نَفْقَهُ كَثِیْرًا مِّمَّا تَقُوْلُ وَاِنَّا لَنَرٰىكَ فِیْنَا ضَعِیْفًا ۚ وَلَوْلَا رَهْطُكَ لَرَجَمْنٰكَ ؗ وَمَاۤ اَنْتَ عَلَیْنَا بِعَزِیْزٍ ۟ قَالَ يَٰقَوۡمِ أَرَهۡطِيٓ أَعَزُّ عَلَيۡكُم مِّنَ ٱللَّهِ وَٱتَّخَذۡتُمُوهُ وَرَآءَكُمۡ ظِهۡرِيًّاۖ إِنَّ رَبِّي بِمَا تَعۡمَلُونَ مُحِيطٞ ۝٩٢ قَالَ یَٰقَوۡمِ أَرَهۡطِیۤ أَعَزُّ عَلَیۡكُم مِّنَ ٱللَّهِ وَٱتَّخَذۡتُمُوهُ وَرَاۤءَكُمۡ ظِهۡرِیًّاۖ إِنَّ رَبِّی بِمَا تَعۡمَلُونَ مُحِیطࣱ ۝٩٢ قَالَ یٰقَوْمِ اَرَهْطِیْۤ اَعَزُّ عَلَیْكُمْ مِّنَ اللّٰهِ ؕ وَاتَّخَذْتُمُوْهُ وَرَآءَكُمْ ظِهْرِیًّا ؕ اِنَّ رَبِّیْ بِمَا تَعْمَلُوْنَ مُحِیْطٌ ۟ وَيَٰقَوۡمِ ٱعۡمَلُواْ عَلَىٰ مَكَانَتِكُمۡ إِنِّي عَٰمِلٞۖ سَوۡفَ تَعۡلَمُونَ مَن يَأۡتِيهِ عَذَابٞ يُخۡزِيهِ وَمَنۡ هُوَ كَٰذِبٞۖ وَٱرۡتَقِبُوٓاْ إِنِّي مَعَكُمۡ رَقِيبٞ ۝٩٣ وَیَٰقَوۡمِ ٱعۡمَلُوا۟ عَلَىٰ مَكَانَتِكُمۡ إِنِّی عَٰمِلࣱۖ سَوۡفَ تَعۡلَمُونَ مَن یَأۡتِیهِ عَذَابࣱ یُخۡزِیهِ وَمَنۡ هُوَ كَٰذِبࣱۖ وَٱرۡتَقِبُوۤا۟ إِنِّی مَعَكُمۡ رَقِیبࣱ ۝٩٣ وَیٰقَوْمِ اعْمَلُوْا عَلٰی مَكَانَتِكُمْ اِنِّیْ عَامِلٌ ؕ سَوْفَ تَعْلَمُوْنَ ۙ مَنْ یَّاْتِیْهِ عَذَابٌ یُّخْزِیْهِ وَمَنْ هُوَ كَاذِبٌ ؕ وَارْتَقِبُوْۤا اِنِّیْ مَعَكُمْ رَقِیْبٌ ۟ وَلَمَّا جَآءَ أَمۡرُنَا نَجَّيۡنَا شُعَيۡبٗا وَٱلَّذِينَ ءَامَنُواْ مَعَهُۥ بِرَحۡمَةٖ مِّنَّا وَأَخَذَتِ ٱلَّذِينَ ظَلَمُواْ ٱلصَّيۡحَةُ فَأَصۡبَحُواْ فِي دِيَٰرِهِمۡ جَٰثِمِينَ ۝٩٤ وَلَمَّا جَاۤءَ أَمۡرُنَا نَجَّیۡنَا شُعَیۡبࣰا وَٱلَّذِینَ ءَامَنُوا۟ مَعَهُۥ بِرَحۡمَةࣲ مِّنَّا وَأَخَذَتِ ٱلَّذِینَ ظَلَمُوا۟ ٱلصَّیۡحَةُ فَأَصۡبَحُوا۟ فِی دِیَٰرِهِمۡ جَٰثِمِینَ ۝٩٤ وَلَمَّا جَآءَ اَمْرُنَا نَجَّیْنَا شُعَیْبًا وَّالَّذِیْنَ اٰمَنُوْا مَعَهٗ بِرَحْمَةٍ مِّنَّا وَاَخَذَتِ الَّذِیْنَ ظَلَمُوا الصَّیْحَةُ فَاَصْبَحُوْا فِیْ دِیَارِهِمْ جٰثِمِیْنَ ۟ۙ كَأَن لَّمۡ يَغۡنَوۡاْ فِيهَآۗ أَلَا بُعۡدٗا لِّمَدۡيَنَ كَمَا بَعِدَتۡ ثَمُودُ ۝٩٥ كَأَن لَّمۡ یَغۡنَوۡا۟ فِیهَاۤۗ أَلَا بُعۡدࣰا لِّمَدۡیَنَ كَمَا بَعِدَتۡ ثَمُودُ ۝٩٥ كَاَنْ لَّمْ یَغْنَوْا فِیْهَا ؕ اَلَا بُعْدًا لِّمَدْیَنَ كَمَا بَعِدَتْ ثَمُوْدُ ۟۠ وَلَقَدۡ أَرۡسَلۡنَا مُوسَىٰ بِـَٔايَٰتِنَا وَسُلۡطَٰنٖ مُّبِينٍ ۝٩٦ وَلَقَدۡ أَرۡسَلۡنَا مُوسَىٰ بِءَایَٰتِنَا وَسُلۡطَٰنࣲ مُّبِینٍ ۝٩٦ وَلَقَدْ اَرْسَلْنَا مُوْسٰی بِاٰیٰتِنَا وَسُلْطٰنٍ مُّبِیْنٍ ۟ۙ إِلَىٰ فِرۡعَوۡنَ وَمَلَإِيْهِۦ فَٱتَّبَعُوٓاْ أَمۡرَ فِرۡعَوۡنَۖ وَمَآ أَمۡرُ فِرۡعَوۡنَ بِرَشِيدٖ ۝٩٧ إِلَىٰ فِرۡعَوۡنَ وَمَلَإِی۟هِۦ فَٱتَّبَعُوۤا۟ أَمۡرَ فِرۡعَوۡنَۖ وَمَاۤ أَمۡرُ فِرۡعَوۡنَ بِرَشِیدࣲ ۝٩٧ اِلٰی فِرْعَوْنَ وَمَلَاۡىِٕهٖ فَاتَّبَعُوْۤا اَمْرَ فِرْعَوْنَ ۚ وَمَاۤ اَمْرُ فِرْعَوْنَ بِرَشِیْدٍ ۟ يَقۡدُمُ قَوۡمَهُۥ يَوۡمَ ٱلۡقِيَٰمَةِ فَأَوۡرَدَهُمُ ٱلنَّارَۖ وَبِئۡسَ ٱلۡوِرۡدُ ٱلۡمَوۡرُودُ ۝٩٨ یَقۡدُمُ قَوۡمَهُۥ یَوۡمَ ٱلۡقِیَٰمَةِ فَأَوۡرَدَهُمُ ٱلنَّارَۖ وَبِئۡسَ ٱلۡوِرۡدُ ٱلۡمَوۡرُودُ ۝٩٨ یَقْدُمُ قَوْمَهٗ یَوْمَ الْقِیٰمَةِ فَاَوْرَدَهُمُ النَّارَ ؕ وَبِئْسَ الْوِرْدُ الْمَوْرُوْدُ ۟ وَأُتۡبِعُواْ فِي هَٰذِهِۦ لَعۡنَةٗ وَيَوۡمَ ٱلۡقِيَٰمَةِۚ بِئۡسَ ٱلرِّفۡدُ ٱلۡمَرۡفُودُ ۝٩٩ وَأُتۡبِعُوا۟ فِی هَٰذِهِۦ لَعۡنَةࣰ وَیَوۡمَ ٱلۡقِیَٰمَةِۚ بِئۡسَ ٱلرِّفۡدُ ٱلۡمَرۡفُودُ ۝٩٩ وَاُتْبِعُوْا فِیْ هٰذِهٖ لَعْنَةً وَّیَوْمَ الْقِیٰمَةِ ؕ بِئْسَ الرِّفْدُ الْمَرْفُوْدُ ۟ ذَٰلِكَ مِنۡ وَذٰلِكَ یَوْمٌ مَّشْهُوْدٌ ۟ وَمَا نُؤَخِّرُهُۥٓ إِلَّا لِأَجَلٖ مَّعۡدُودٖ ۝١٠٤ وَمَا نُؤَخِّرُهُۥۤ إِلَّا لِأَجَلࣲ مَّعۡدُودࣲ ۝١٠٤ وَمَا نُؤَخِّرُهٗۤ اِلَّا لِاَجَلٍ مَّعْدُوْدٍ ۟ؕ يَوۡمَ يَأۡتِ اِلَّا بِاِذْنِهٖ ۚ فَمِنْهُمْ شَقِیٌّ وَّسَعِیْدٌ ۟ فَأَمَّا ٱلَّذِينَ شَقُواْ فَفِي ٱلنَّارِ لَهُمۡ فِيهَا زَفِيرٞ وَشَهِيقٌ ۝١٠٦ فَأَمَّا ٱلَّذِینَ شَقُوا۟ فَفِی ٱلنَّارِ لَهُمۡ فِیهَا زَفِیرࣱ وَشَهِیقٌ ۝١٠٦ فَاَمَّا الَّذِیْنَ شَقُوْا فَفِی النَّارِ لَهُمْ فِیْهَا زَفِیْرٌ وَّشَهِیْقٌ ۟ۙ خَٰلِدِينَ فِيهَا مَا دَامَتِ ٱلسَّمَٰوَٰتُ وَٱلۡأَرۡضُ إِلَّا مَا شَآءَ رَبُّكَۚ إِنَّ رَبَّكَ وَالْاَرْضُ اِلَّا مَا شَآءَ رَبُّكَ ؕ اِنَّ رَبَّكَ فَعَّالٌ لِّمَا یُرِیْدُ ۟ وَأَمَّا ٱلَّذِينَ سُعِدُواْ فَفِي ٱلۡجَنَّةِ خَٰلِدِينَ فِيهَا مَا دَامَتِ ٱلسَّمَٰوَٰتُ وَٱلۡأَرۡضُ إِلَّا مَا شَآءَ رَبُّكَۖ عَطَآءً غَيۡرَ مَجۡذُوذٖ ۝١٠٨ وَأَمَّا ٱلَّذِینَ سُعِدُوا۟ فَفِی ٱلۡجَنَّةِ خَٰلِدِینَ فِیهَا مَا دَامَتِ ٱلسَّمَٰوَٰتُ وَٱلۡأَرۡضُ إِلَّا مَا شَاۤءَ رَبُّكَۖ عَطَاۤءً غَیۡرَ مَجۡذُوذࣲ ۝١٠٨ وَاَمَّا الَّذِیْنَ سُعِدُوْا فَفِی الْجَنَّةِ خٰلِدِیْنَ فِیْهَا مَا دَامَتِ السَّمٰوٰتُ وَالْاَرْضُ اِلَّا مَا شَآءَ رَبُّكَ ؕ عَطَآءً غَیْرَ مَجْذُوْذٍ ۟ فَلَا تَكُ فِي مِرۡيَةٖ مِّمَّا يَعۡبُدُ هَٰٓؤُلَآءِۚ مَا يَعۡبُدُونَ إِلَّا كَمَا يَعۡبُدُ ءَابَآؤُهُم مِّن قَبۡلُۚ وَإِنَّا لَمُوَفُّوهُمۡ نَصِيبَهُمۡ غَيۡرَ مَنقُوصٖ ۝١٠٩ فَلَا تَكُ فِی مِرۡیَةࣲ مِّمَّا یَعۡبُدُ هَٰۤؤُلَاۤءِۚ مَا یَعۡبُدُونَ إِلَّا كَمَا یَعۡبُدُ ءَابَاۤؤُهُم مِّن قَبۡلُۚ وَإِنَّا لَمُوَفُّوهُمۡ نَصِیبَهُمۡ غَیۡرَ مَنقُوصࣲ ۝١٠٩ فَلَا تَكُ فِیْ مِرْیَةٍ مِّمَّا یَعْبُدُ هٰۤؤُلَآءِ ؕ مَا یَعْبُدُوْنَ اِلَّا كَمَا یَعْبُدُ اٰبَآؤُهُمْ مِّنْ قَبْلُ ؕ وَاِنَّا لَمُوَفُّوْهُمْ نَصِیْبَهُمْ غَیْرَ مَنْقُوْصٍ ۟۠ وَلَقَدۡ ءَاتَيۡنَا مُوسَى ٱلۡكِتَٰبَ فَٱخۡتُلِفَ فِيهِۚ وَلَوۡلَا كَلِمَةٞ سَبَقَتۡ مِن رَّبِّكَ لَقُضِيَ بَيۡنَهُمۡۚ وَإِنَّهُمۡ لَفِي شَكّٖ مِّنۡهُ مُرِيبٖ ۝١١٠ وَلَقَدۡ ءَاتَیۡنَا مُوسَى ٱلۡكِتَٰبَ فَٱخۡتُلِفَ فِیهِۚ وَلَوۡلَا كَلِمَةࣱ سَبَقَتۡ مِن رَّبِّكَ لَقُضِیَ بَیۡنَهُمۡۚ وَإِنَّهُمۡ لَفِی شَكࣲّ مِّنۡهُ مُرِیبࣲ ۝١١٠ وَلَقَدْ اٰتَیْنَا مُوْسَی الْكِتٰبَ فَاخْتُلِفَ فِیْهِ ؕ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِنْ رَّبِّكَ لَقُضِیَ بَیْنَهُمْ ؕ وَاِنَّهُمْ لَفِیْ شَكٍّ مِّنْهُ مُرِیْبٍ ۟ وَإِنَّ كُلّٗا لَّمَّا لَيُوَفِّيَنَّهُمۡ رَبُّكَ أَعۡمَٰلَهُمۡۚ إِنَّهُۥ بِمَا يَعۡمَلُونَ خَبِيرٞ ۝١١١ وَإِنَّ كُلࣰّا لَّمَّا لَیُوَفِّیَنَّهُمۡ رَبُّكَ أَعۡمَٰلَهُمۡۚ إِنَّهُۥ بِمَا یَعۡمَلُونَ خَبِیرࣱ ۝١١١ وَاِنَّ كُلًّا لَّمَّا لَیُوَفِّیَنَّهُمْ رَبُّكَ اَعْمَالَهُمْ ؕ اِنَّهٗ بِمَا یَعْمَلُوْنَ خَبِیْرٌ ۟ فَٱسۡتَقِمۡ كَمَآ أُمِرۡتَ وَمَن تَابَ مَعَكَ وَلَا تَطۡغَوۡاْۚ إِنَّهُۥ بِمَا تَعۡمَلُونَ بَصِيرٞ ۝١١٢ فَٱسۡتَقِمۡ كَمَاۤ أُمِرۡتَ وَمَن تَابَ مَعَكَ وَلَا تَطۡغَوۡا۟ۚ إِنَّهُۥ بِمَا تَعۡمَلُونَ بَصِیرࣱ ۝١١٢ فَاسْتَقِمْ كَمَاۤ اُمِرْتَ وَمَنْ تَابَ مَعَكَ وَلَا تَطْغَوْا ؕ اِنَّهٗ بِمَا تَعْمَلُوْنَ بَصِیْرٌ ۟ وَلَا تَرۡكَنُوٓاْ إِلَى ٱلَّذِينَ ظَلَمُواْ فَتَمَسَّكُمُ ٱلنَّارُ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِنۡ أَوۡلِيَآءَ ثُمَّ لَا یَزَالُوْنَ مُخْتَلِفِیْنَ ۟ۙ إِلَّا مَن رَّحِمَ رَبُّكَۚ وَلِذَٰلِكَ خَلَقَهُمۡۗ وَتَمَّتۡ كَلِمَةُ رَبِّكَ لَأَمۡلَأَنَّ جَهَنَّمَ مِنَ ٱلۡجِنَّةِ وَٱلنَّاسِ أَجۡمَعِينَ ۝١١٩ إِلَّا مَن رَّحِمَ رَبُّكَۚ وَلِذَٰلِكَ خَلَقَهُمۡۗ وَتَمَّتۡ كَلِمَةُ رَبِّكَ لَأَمۡلَأَنَّ جَهَنَّمَ مِنَ ٱلۡجِنَّةِ وَٱلنَّاسِ أَجۡمَعِینَ ۝١١٩ اِلَّا مَنْ رَّحِمَ رَبُّكَ ؕ وَلِذٰلِكَ خَلَقَهُمْ ؕ وَتَمَّتْ كَلِمَةُ رَبِّكَ لَاَمْلَـَٔنَّ جَهَنَّمَ مِنَ الْجِنَّةِ وَالنَّاسِ اَجْمَعِیْنَ ۟ وَكُلّٗا نَّقُصُّ عَلَيۡكَ مِنۡ أَنۢبَآءِ ٱلرُّسُلِ مَا نُثَبِّتُ بِهِۦ فُؤَادَكَۚ وَجَآءَكَ فِي هَٰذِهِ ٱلۡحَقُّ وَمَوۡعِظَةٞ وَذِكۡرَىٰ لِلۡمُؤۡمِنِينَ ۝١٢٠ وَكُلࣰّا نَّقُصُّ عَلَیۡكَ مِنۡ أَنۢبَاۤءِ ٱلرُّسُلِ مَا نُثَبِّتُ بِهِۦ فُؤَادَكَۚ وَجَاۤءَكَ فِی هَٰذِهِ ٱلۡحَقُّ وَمَوۡعِظَةࣱ وَذِكۡرَىٰ لِلۡمُؤۡمِنِینَ ۝١٢٠ وَكُلًّا نَّقُصُّ عَلَیْكَ مِنْ اَنْۢبَآءِ الرُّسُلِ مَا نُثَبِّتُ بِهٖ فُؤَادَكَ ۚ وَجَآءَكَ فِیْ هٰذِهِ الْحَقُّ وَمَوْعِظَةٌ وَّذِكْرٰی لِلْمُؤْمِنِیْنَ ۟ وَقُل لِّلَّذِينَ لَا يُؤۡمِنُونَ ٱعۡمَلُواْ عَلَىٰ مَكَانَتِكُمۡ إِنَّا عَٰمِلُونَ ۝١٢١ وَقُل لِّلَّذِینَ لَا یُؤۡمِنُونَ ٱعۡمَلُوا۟ عَلَىٰ مَكَانَتِكُمۡ إِنَّا عَٰمِلُونَ ۝١٢١ وَقُلْ لِّلَّذِیْنَ لَا یُؤْمِنُوْنَ اعْمَلُوْا عَلٰی مَكَانَتِكُمْ ؕ اِنَّا عٰمِلُوْنَ ۟ۙ وَٱنتَظِرُوٓاْ إِنَّا مُنتَظِرُونَ ۝١٢٢ وَٱنتَظِرُوۤا۟ إِنَّا مُنتَظِرُونَ ۝١٢٢ وَانْتَظِرُوْا ۚ اِنَّا مُنْتَظِرُوْنَ ۟ وَلِلَّهِ غَيۡبُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَإِلَيۡهِ يُرۡجَعُ ٱلۡأَمۡرُ كُلُّهُۥ فَٱعۡبُدۡهُ وَتَوَكَّلۡ عَلَيۡهِۚ وَمَا رَبُّكَ بِغَٰفِلٍ عَمَّا تَعۡمَلُونَ ۝١٢٣ وَلِلَّهِ غَیۡبُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَإِلَیۡهِ یُرۡجَعُ ٱلۡأَمۡرُ كُلُّهُۥ فَٱعۡبُدۡهُ وَتَوَكَّلۡ عَلَیۡهِۚ وَمَا رَبُّكَ بِغَٰفِلٍ عَمَّا تَعۡمَلُونَ ۝١٢٣ وَلِلّٰهِ غَیْبُ السَّمٰوٰتِ وَالْاَرْضِ وَاِلَیْهِ یُرْجَعُ الْاَمْرُ كُلُّهٗ فَاعْبُدْهُ وَتَوَكَّلْ عَلَیْهِ ؕ وَمَا رَبُّكَ بِغَافِلٍ عَمَّا تَعْمَلُوْنَ ۟۠
