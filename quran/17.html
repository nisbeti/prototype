بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

سُبۡحَٰنَ ٱلَّذِيٓ أَسۡرَىٰ بِعَبۡدِهِۦ لَيۡلٗا مِّنَ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ إِلَى ٱلۡمَسۡجِدِ ٱلۡأَقۡصَا ٱلَّذِي بَٰرَكۡنَا حَوۡلَهُۥ لِنُرِيَهُۥ مِنۡ ءَايَٰتِنَآۚ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلۡبَصِيرُ ۝١ سُبۡحَٰنَ ٱلَّذِیۤ أَسۡرَىٰ بِعَبۡدِهِۦ لَیۡلࣰا مِّنَ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ إِلَى ٱلۡمَسۡجِدِ ٱلۡأَقۡصَا ٱلَّذِی بَٰرَكۡنَا حَوۡلَهُۥ لِنُرِیَهُۥ مِنۡ ءَایَٰتِنَاۤۚ إِنَّهُۥ هُوَ ٱلسَّمِیعُ ٱلۡبَصِیرُ ۝١ سُبْحٰنَ الَّذِیْۤ اَسْرٰی بِعَبْدِهٖ لَیْلًا مِّنَ الْمَسْجِدِ الْحَرَامِ اِلَی الْمَسْجِدِ الْاَقْصَا الَّذِیْ بٰرَكْنَا حَوْلَهٗ لِنُرِیَهٗ مِنْ اٰیٰتِنَا ؕ اِنَّهٗ هُوَ السَّمِیْعُ الْبَصِیْرُ ۟ وَءَاتَيۡنَا مُوسَى ٱلۡكِتَٰبَ وَجَعَلۡنَٰهُ هُدٗى لِّبَنِيٓ إِسۡرَٰٓءِيلَ أَلَّا تَتَّخِذُواْ مِن دُونِي وَكِيلٗا ۝٢ وَءَاتَیۡنَا مُوسَى ٱلۡكِتَٰبَ وَجَعَلۡنَٰهُ هُدࣰى لِّبَنِیۤ إِسۡرَٰۤءِیلَ أَلَّا تَتَّخِذُوا۟ مِن دُونِی وَكِیلࣰا ۝٢ وَاٰتَیْنَا مُوْسَی الْكِتٰبَ وَجَعَلْنٰهُ هُدًی لِّبَنِیْۤ اِسْرَآءِیْلَ اَلَّا تَتَّخِذُوْا مِنْ دُوْنِیْ وَكِیْلًا ۟ؕ ذُرِّيَّةَ مَنۡ حَمَلۡنَا مَعَ نُوحٍۚ إِنَّهُۥ كَانَ عَبۡدٗا شَكُورٗا ۝٣ ذُرِّیَّةَ مَنۡ حَمَلۡنَا مَعَ نُوحٍۚ إِنَّهُۥ كَانَ عَبۡدࣰا شَكُورࣰا ۝٣ ذُرِّیَّةَ مَنْ حَمَلْنَا مَعَ نُوْحٍ ؕ اِنَّهٗ كَانَ عَبْدًا شَكُوْرًا ۟ وَقَضَيۡنَآ إِلَىٰ بَنِيٓ إِسۡرَٰٓءِيلَ فِي ٱلۡكِتَٰبِ لَتُفۡسِدُنَّ فِي ٱلۡأَرۡضِ مَرَّتَيۡنِ وَلَتَعۡلُنَّ عُلُوّٗا كَبِيرٗا ۝٤ وَقَضَیۡنَاۤ إِلَىٰ بَنِیۤ إِسۡرَٰۤءِیلَ فِی ٱلۡكِتَٰبِ لَتُفۡسِدُنَّ فِی ٱلۡأَرۡضِ مَرَّتَیۡنِ وَلَتَعۡلُنَّ عُلُوࣰّا كَبِیرࣰا ۝٤ وَقَضَیْنَاۤ اِلٰی بَنِیْۤ اِسْرَآءِیْلَ فِی الْكِتٰبِ لَتُفْسِدُنَّ فِی الْاَرْضِ مَرَّتَیْنِ وَلَتَعْلُنَّ عُلُوًّا كَبِیْرًا ۟ فَإِذَا جَآءَ وَعۡدُ أُولَىٰهُمَا بَعَثۡنَا عَلَيۡكُمۡ عِبَادٗا لَّنَآ أُوْلِي بَأۡسٖ شَدِيدٖ فَجَاسُواْ خِلَٰلَ ٱلدِّيَارِۚ وَكَانَ وَعۡدٗا مَّفۡعُولٗا ۝٥ فَإِذَا جَاۤءَ وَعۡدُ أُولَىٰهُمَا بَعَثۡنَا عَلَیۡكُمۡ عِبَادࣰا لَّنَاۤ أُو۟لِی بَأۡسࣲ شَدِیدࣲ فَجَاسُوا۟ خِلَٰلَ ٱلدِّیَارِۚ وَكَانَ وَعۡدࣰا مَّفۡعُولࣰا ۝٥ فَاِذَا جَآءَ وَعْدُ اُوْلٰىهُمَا بَعَثْنَا عَلَیْكُمْ عِبَادًا لَّنَاۤ اُولِیْ بَاْسٍ شَدِیْدٍ فَجَاسُوْا خِلٰلَ الدِّیَارِ وَكَانَ وَعْدًا مَّفْعُوْلًا ۟ ثُمَّ رَدَدۡنَا لَكُمُ ٱلۡكَرَّةَ عَلَيۡهِمۡ وَأَمۡدَدۡنَٰكُم بِأَمۡوَٰلٖ وَبَنِينَ وَجَعَلۡنَٰكُمۡ أَكۡثَرَ نَفِيرًا ۝٦ ثُمَّ رَدَدۡنَا لَكُمُ ٱلۡكَرَّةَ عَلَیۡهِمۡ وَأَمۡدَدۡنَٰكُم بِأَمۡوَٰلࣲ وَبَنِینَ وَجَعَلۡنَٰكُمۡ أَكۡثَرَ نَفِیرًا ۝٦ ثُمَّ رَدَدْنَا لَكُمُ الْكَرَّةَ عَلَیْهِمْ وَاَمْدَدْنٰكُمْ بِاَمْوَالٍ وَّبَنِیْنَ وَجَعَلْنٰكُمْ اَكْثَرَ نَفِیْرًا ۟ إِنۡ أَحۡسَنتُمۡ أَحۡسَنتُمۡ لِأَنفُسِكُمۡۖ وَإِنۡ أَسَأۡتُمۡ فَلَهَاۚ فَإِذَا جَآءَ وَعۡدُ ٱلۡأٓخِرَةِ لِيَسُـُٔواْ وُجُوهَكُمۡ وَلِيَدۡخُلُواْ ٱلۡمَسۡجِدَ كَمَا دَخَلُوهُ أَوَّلَ مَرَّةٖ وَلِيُتَبِّرُواْ مَا عَلَوۡاْ تَتۡبِيرًا ۝٧ إِنۡ أَحۡسَنتُمۡ أَحۡسَنتُمۡ لِأَنفُسِكُمۡۖ وَإِنۡ أَسَأۡتُمۡ فَلَهَاۚ فَإِذَا جَاۤءَ وَعۡدُ ٱلۡءَاخِرَةِ لِیَسُۥۤءُوا۟ وُجُوهَكُمۡ وَلِیَدۡخُلُوا۟ ٱلۡمَسۡجِدَ كَمَا دَخَلُوهُ أَوَّلَ مَرَّةࣲ وَلِیُتَبِّرُوا۟ مَا عَلَوۡا۟ تَتۡبِیرًا ۝٧ اِنْ اَحْسَنْتُمْ اَحْسَنْتُمْ لِاَنْفُسِكُمْ ۫ وَاِنْ اَسَاْتُمْ فَلَهَا ؕ فَاِذَا جَآءَ وَعْدُ الْاٰخِرَةِ لِیَسُوْٓءٗا وُجُوْهَكُمْ وَلِیَدْخُلُوا الْمَسْجِدَ كَمَا دَخَلُوْهُ اَوَّلَ مَرَّةٍ وَّلِیُتَبِّرُوْا مَا عَلَوْا تَتْبِیْرًا ۟ عَسَىٰ رَبُّكُمۡ أَن يَرۡحَمَكُمۡۚ وَإِنۡ عُدتُّمۡ عُدۡنَاۚ وَجَعَلۡنَا جَهَنَّمَ لِلۡكَٰفِرِينَ حَصِيرًا ۝٨ عَسَىٰ رَبُّكُمۡ أَن یَرۡحَمَكُمۡۚ وَإِنۡ عُدتُّمۡ عُدۡنَاۚ وَجَعَلۡنَا جَهَنَّمَ لِلۡكَٰفِرِینَ حَصِیرًا ۝٨ عَسٰی رَبُّكُمْ اَنْ یَّرْحَمَكُمْ ۚ وَاِنْ عُدْتُّمْ عُدْنَا ۘ وَجَعَلْنَا جَهَنَّمَ لِلْكٰفِرِیْنَ حَصِیْرًا ۟ إِنَّ هَٰذَا ٱلۡقُرۡءَانَ يَهۡدِي لِلَّتِي هِيَ أَقۡوَمُ وَيُبَشِّرُ ٱلۡمُؤۡمِنِينَ ٱلَّذِينَ يَعۡمَلُونَ ٱلصَّٰلِحَٰتِ أَنَّ لَهُمۡ أَجۡرٗا كَبِيرٗا ۝٩ إِنَّ هَٰذَا ٱلۡقُرۡءَانَ یَهۡدِی لِلَّتِی هِیَ أَقۡوَمُ وَیُبَشِّرُ ٱلۡمُؤۡمِنِینَ ٱلَّذِینَ یَعۡمَلُونَ ٱلصَّٰلِحَٰتِ أَنَّ لَهُمۡ أَجۡرࣰا كَبِیرࣰا ۝٩ اِنَّ هٰذَا الْقُرْاٰنَ یَهْدِیْ لِلَّتِیْ هِیَ اَقْوَمُ وَیُبَشِّرُ الْمُؤْمِنِیْنَ الَّذِیْنَ یَعْمَلُوْنَ الصّٰلِحٰتِ اَنَّ لَهُمْ اَجْرًا كَبِیْرًا ۟ۙ وَأَنَّ ٱلَّذِينَ لَا يُؤۡمِنُونَ بِٱلۡأٓخِرَةِ أَعۡتَدۡنَا لَهُمۡ عَذَابًا أَلِيمٗا ۝١٠ وَأَنَّ ٱلَّذِینَ لَا یُؤۡمِنُونَ بِٱلۡءَاخِرَةِ أَعۡتَدۡنَا لَهُمۡ عَذَابًا أَلِیمࣰا ۝١٠ وَّاَنَّ الَّذِیْنَ لَا یُؤْمِنُوْنَ بِالْاٰخِرَةِ اَعْتَدْنَا لَهُمْ عَذَابًا اَلِیْمًا ۟۠ وَيَدۡعُ ٱلۡإِنسَٰنُ بِٱلشَّرِّ دُعَآءَهُۥ بِٱلۡخَيۡرِۖ وَكَانَ ٱلۡإِنسَٰنُ عَجُولٗا ۝١١ وَیَدۡعُ ٱلۡإِنسَٰنُ بِٱلشَّرِّ دُعَاۤءَهُۥ بِٱلۡخَیۡرِۖ وَكَانَ ٱلۡإِنسَٰنُ عَجُولࣰا ۝١١ وَیَدْعُ الْاِنْسَانُ بِالشَّرِّ دُعَآءَهٗ بِالْخَیْرِ ؕ وَكَانَ الْاِنْسَانُ عَجُوْلًا ۟ وَجَعَلۡنَا ٱلَّيۡلَ وَٱلنَّهَارَ ءَايَتَيۡنِۖ فَمَحَوۡنَآ ءَايَةَ ٱلَّيۡلِ وَجَعَلۡنَآ ءَايَةَ ٱلنَّهَارِ مُبۡصِرَةٗ لِّتَبۡتَغُواْ فَضۡلٗا مِّن رَّبِّكُمۡ وَلِتَعۡلَمُواْ عَدَدَ ٱلسِّنِينَ وَٱلۡحِسَابَۚ وَكُلَّ شَيۡءٖ فَصَّلۡنَٰهُ تَفۡصِيلٗا ۝١٢ وَجَعَلۡنَا ٱلَّیۡلَ وَٱلنَّهَارَ ءَایَتَیۡنِۖ فَمَحَوۡنَاۤ ءَایَةَ ٱلَّیۡلِ وَجَعَلۡنَاۤ ءَایَةَ ٱلنَّهَارِ مُبۡصِرَةࣰ لِّتَبۡتَغُوا۟ فَضۡلࣰا مِّن رَّبِّكُمۡ وَلِتَعۡلَمُوا۟ عَدَدَ ٱلسِّنِینَ وَٱلۡحِسَابَۚ وَكُلَّ شَیۡءࣲ فَصَّلۡنَٰهُ تَفۡصِیلࣰا ۝١٢ وَجَعَلْنَا الَّیْلَ وَالنَّهَارَ اٰیَتَیْنِ فَمَحَوْنَاۤ اٰیَةَ الَّیْلِ وَجَعَلْنَاۤ اٰیَةَ النَّهَارِ مُبْصِرَةً لِّتَبْتَغُوْا فَضْلًا مِّنْ رَّبِّكُمْ وَلِتَعْلَمُوْا عَدَدَ السِّنِیْنَ وَالْحِسَابَ ؕ وَكُلَّ شَیْءٍ فَصَّلْنٰهُ تَفْصِیْلًا ۟ وَكُلَّ إِنسَٰنٍ أَلۡزَمۡنَٰهُ طَٰٓئِرَهُۥ فِي عُنُقِهِۦۖ وَنُخۡرِجُ لَهُۥ يَوۡمَ ٱلۡقِيَٰمَةِ كِتَٰبٗا يَلۡقَىٰهُ مَنشُورًا ۝١٣ وَكُلَّ إِنسَٰنٍ أَلۡزَمۡنَٰهُ طَٰۤئِرَهُۥ فِی عُنُقِهِۦۖ وَنُخۡرِجُ لَهُۥ یَوۡمَ ٱلۡقِیَٰمَةِ كِتَٰبࣰا یَلۡقَىٰهُ مَنشُورًا ۝١٣ وَكُلَّ اِنْسَانٍ اَلْزَمْنٰهُ طٰٓىِٕرَهٗ فِیْ عُنُقِهٖ ؕ وَنُخْرِجُ لَهٗ یَوْمَ الْقِیٰمَةِ كِتٰبًا یَّلْقٰىهُ مَنْشُوْرًا ۟ ٱقۡرَأۡ كِتَٰبَكَ كَفَىٰ بِنَفۡسِكَ ٱلۡيَوۡمَ عَلَيۡكَ حَسِيبٗا ۝١٤ ٱقۡرَأۡ كِتَٰبَكَ كَفَىٰ بِنَفۡسِكَ ٱلۡیَوۡمَ عَلَیۡكَ حَسِیبࣰا ۝١٤ اِقْرَاْ كِتٰبَكَ ؕ كَفٰی بِنَفْسِكَ الْیَوْمَ عَلَیْكَ حَسِیْبًا ۟ؕ مَّنِ ٱهۡتَدَىٰ فَإِنَّمَا يَهۡتَدِي لِنَفۡسِهِۦۖ وَمَن ضَلَّ فَإِنَّمَا يَضِلُّ عَلَيۡهَاۚ وَلَا تَزِرُ وَازِرَةٞ وِزۡرَ أُخۡرَىٰۗ وَمَا كُنَّا مُعَذِّبِينَ حَتَّىٰ نَبۡعَثَ رَسُولٗا ۝١٥ مَّنِ ٱهۡتَدَىٰ فَإِنَّمَا یَهۡتَدِی لِنَفۡسِهِۦۖ وَمَن ضَلَّ فَإِنَّمَا یَضِلُّ عَلَیۡهَاۚ وَلَا تَزِرُ وَازِرَةࣱ وِزۡرَ أُخۡرَىٰۗ وَمَا كُنَّا مُعَذِّبِینَ حَتَّىٰ نَبۡعَثَ رَسُولࣰا ۝١٥ مَنِ اهْتَدٰی فَاِنَّمَا یَهْتَدِیْ لِنَفْسِهٖ ۚ وَمَنْ ضَلَّ فَاِنَّمَا یَضِلُّ عَلَیْهَا ؕ وَلَا تَزِرُ وَازِرَةٌ وِّزْرَ اُخْرٰی ؕ وَمَا كُنَّا مُعَذِّبِیْنَ حَتّٰی نَبْعَثَ رَسُوْلًا ۟ وَإِذَآ أَرَدۡنَآ أَن نُّهۡلِكَ قَرۡيَةً أَمَرۡنَا مُتۡرَفِيهَا فَفَسَقُواْ فِيهَا فَحَقَّ عَلَيۡهَا ٱلۡقَوۡلُ فَدَمَّرۡنَٰهَا تَدۡمِيرٗا ۝١٦ وَإِذَاۤ أَرَدۡنَاۤ أَن نُّهۡلِكَ قَرۡیَةً أَمَرۡنَا مُتۡرَفِیهَا فَفَسَقُوا۟ فِیهَا فَحَقَّ عَلَیۡهَا ٱلۡقَوۡلُ فَدَمَّرۡنَٰهَا تَدۡمِیرࣰا تَقۡرَبُواْ مَالَ ٱلۡيَتِيمِ إِلَّا بِٱلَّتِي هِيَ أَحۡسَنُ حَتَّىٰ يَبۡلُغَ أَشُدَّهُۥۚ وَأَوۡفُواْ بِٱلۡعَهۡدِۖ إِنَّ ٱلۡعَهۡدَ كَانَ مَسۡـُٔولٗا ۝٣٤ وَلَا تَقۡرَبُوا۟ مَالَ ٱلۡیَتِیمِ إِلَّا نَجَّىٰكُمۡ إِلَى ٱلۡبَرِّ أَعۡرَضۡتُمۡۚ وَكَانَ ٱلۡإِنسَٰنُ كَفُورًا ۝٦٧ وَاِذَا مَسَّكُمُ الضُّرُّ فِی الْبَحْرِ ضَلَّ مَنْ تَدْعُوْنَ اِلَّاۤ اِیَّاهُ ۚ فَلَمَّا نَجّٰىكُمْ اِلَی الْبَرِّ اَعْرَضْتُمْ ؕ وَكَانَ الْاِنْسَانُ كَفُوْرًا ۟ أَفَأَمِنتُمۡ أَن يَخۡسِفَ بِكُمۡ جَانِبَ ٱلۡبَرِّ أَوۡ يُرۡسِلَ عَلَيۡكُمۡ حَاصِبٗا ثُمَّ لَا تَجِدُواْ لَكُمۡ وَكِيلًا ۝٦٨ أَفَأَمِنتُمۡ أَن یَخۡسِفَ بِكُمۡ جَانِبَ ٱلۡبَرِّ أَوۡ یُرۡسِلَ عَلَیۡكُمۡ حَاصِبࣰا ثُمَّ لَا تَجِدُوا۟ لَكُمۡ وَكِیلًا ۝٦٨ اَفَاَمِنْتُمْ اَنْ یَّخْسِفَ بِكُمْ جَانِبَ الْبَرِّ اَوْ یُرْسِلَ عَلَیْكُمْ حَاصِبًا ثُمَّ لَا تَجِدُوْا لَكُمْ وَكِیْلًا ۟ۙ أَمۡ أَمِنتُمۡ أَن يُعِيدَكُمۡ فِيهِ تَارَةً أُخۡرَىٰ فَيُرۡسِلَ عَلَيۡكُمۡ قَاصِفٗا مِّنَ ٱلرِّيحِ فَيُغۡرِقَكُم بِمَا كَفَرۡتُمۡ ثُمَّ لَا تَجِدُواْ لَكُمۡ عَلَيۡنَا بِهِۦ تَبِيعٗا ۝٦٩ أَمۡ أَمِنتُمۡ أَن یُعِیدَكُمۡ فِیهِ تَارَةً أُخۡرَىٰ فَیُرۡسِلَ عَلَیۡكُمۡ قَاصِفࣰا مِّنَ ٱلرِّیحِ فَیُغۡرِقَكُم بِمَا كَفَرۡتُمۡ ثُمَّ لَا تَجِدُوا۟ لَكُمۡ عَلَیۡنَا بِهِۦ لَنَذْهَبَنَّ بِالَّذِیْۤ اَوْحَیْنَاۤ اِلَیْكَ ثُمَّ لَا تَجِدُ لَكَ بِهٖ عَلَیْنَا وَكِیْلًا ۟ۙ إِلَّا رَحۡمَةٗ مِّن رَّبِّكَۚ إِنَّ فَضۡلَهُۥ كَانَ عَلَيۡكَ كَبِيرٗا ۝٨٧ إِلَّا رَحۡمَةࣰ مِّن رَّبِّكَۚ إِنَّ فَضۡلَهُۥ كَانَ عَلَیۡكَ كَبِیرࣰا ۝٨٧ اِلَّا رَحْمَةً مِّنْ رَّبِّكَ ؕ اِنَّ فَضْلَهٗ كَانَ عَلَیْكَ كَبِیْرًا ۟ قُل لَّئِنِ ٱجۡتَمَعَتِ ٱلۡإِنسُ وَٱلۡجِنُّ عَلَىٰٓ أَن يَأۡتُواْ یَأۡتُونَ بِمِثۡلِهِۦ وَلَوۡ كَانَ بَعۡضُهُمۡ لِبَعۡضࣲ ظَهِیرࣰا ۝٨٨ قُلْ لَّىِٕنِ اجْتَمَعَتِ الْاِنْسُ وَالْجِنُّ عَلٰۤی اَنْ یَّاْتُوْا بِمِثْلِ هٰذَا الْقُرْاٰنِ لَا یَاْتُوْنَ بِمِثْلِهٖ وَلَوْ كَانَ بَعْضُهُمْ لِبَعْضٍ ظَهِیْرًا ۟ وَلَقَدۡ صَرَّفۡنَا لِلنَّاسِ فِي هَٰذَا ٱلۡقُرۡءَانِ مِن كُلِّ مَثَلٖ فَأَبَىٰٓ أَكۡثَرُ ٱلنَّاسِ إِلَّا كُفُورٗا ۝٨٩ وَلَقَدۡ صَرَّفۡنَا لِلنَّاسِ فِی هَٰذَا زَعَمۡتَ عَلَيۡنَا كِسَفًا أَوۡ تَأۡتِيَ بِٱللَّهِ وَٱلۡمَلَٰٓئِكَةِ قَبِيلًا ۝٩٢ أَوۡ تُسۡقِطَ ٱلسَّمَاۤءَ كَمَا زَعَمۡتَ عَلَیۡنَا كِسَفًا أَوۡ تَأۡتِیَ بِٱللَّهِ وَٱلۡمَلَٰۤئِكَةِ قَبِیلًا ۝٩٢ اَوْ ۝٩٧ وَمَنْ یَّهْدِ اللّٰهُ فَهُوَ الْمُهْتَدِ ۚ وَمَنْ یُّضْلِلْ فَلَنْ تَجِدَ لَهُمْ اَوْلِیَآءَ مِنْ دُوْنِهٖ ؕ وَنَحْشُرُهُمْ یَوْمَ الْقِیٰمَةِ عَلٰی وُجُوْهِهِمْ عُمْیًا وَّبُكْمًا وَّصُمًّا ؕ مَاْوٰىهُمْ جَهَنَّمُ ؕ كُلَّمَا خَبَتْ زِدْنٰهُمْ سَعِیْرًا ۟ ذَٰلِكَ جَزَآؤُهُم بِأَنَّهُمۡ كَفَرُواْ بِـَٔايَٰتِنَا وَقَالُوٓاْ أَءِذَا كُنَّا عِظَٰمٗا وَرُفَٰتًا أَءِنَّا لَمَبۡعُوثُونَ خَلۡقٗا جَدِيدًا ۝٩٨ ذَٰلِكَ جَزَاۤؤُهُم بِأَنَّهُمۡ كَفَرُوا۟ بِءَایَٰتِنَا وَقَالُوۤا۟ أَءِذَا كُنَّا عِظَٰمࣰا وَرُفَٰتًا أَءِنَّا لَمَبۡعُوثُونَ خَلۡقࣰا جَدِیدًا ۝٩٨ ذٰلِكَ جَزَآؤُهُمْ بِاَنَّهُمْ كَفَرُوْا بِاٰیٰتِنَا وَقَالُوْۤا ءَاِذَا كُنَّا عِظَامًا وَّرُفَاتًا ءَاِنَّا لَمَبْعُوْثُوْنَ خَلْقًا جَدِیْدًا ۟ أَوَلَمۡ يَرَوۡاْ أَنَّ ٱللَّهَ ٱلَّذِي خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ قَادِرٌ عَلَىٰٓ أَن يَخۡلُقَ مِثۡلَهُمۡ وَجَعَلَ لَهُمۡ أَجَلٗا لَّا رَيۡبَ فِيهِ فَأَبَى ٱلظَّٰلِمُونَ إِلَّا كُفُورٗا ۝٩٩ أَوَلَمۡ یَرَوۡا۟ أَنَّ ٱللَّهَ ٱلَّذِی خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ قَادِرٌ عَلَىٰۤ أَن یَخۡلُقَ مِثۡلَهُمۡ وَجَعَلَ لَهُمۡ أَجَلࣰا لَّا رَیۡبَ فِیهِ فَأَبَى ٱلظَّٰلِمُونَ إِلَّا كُفُورࣰا ۝٩٩ اَوَلَمْ یَرَوْا اَنَّ اللّٰهَ الَّذِیْ خَلَقَ السَّمٰوٰتِ وَالْاَرْضَ قَادِرٌ عَلٰۤی اَنْ یَّخْلُقَ مِثْلَهُمْ وَجَعَلَ لَهُمْ اَجَلًا لَّا رَیْبَ فِیْهِ ؕ فَاَبَی الظّٰلِمُوْنَ اِلَّا كُفُوْرًا ۟ قُل لَّوۡ أَنتُمۡ تَمۡلِكُونَ خَزَآئِنَ رَحۡمَةِ رَبِّيٓ إِذٗا لَّأَمۡسَكۡتُمۡ خَشۡيَةَ ٱلۡإِنفَاقِۚ وَكَانَ ٱلۡإِنسَٰنُ قَتُورٗا ۝١٠٠ قُل لَّوۡ أَنتُمۡ تَمۡلِكُونَ خَزَاۤئِنَ رَحۡمَةِ رَبِّیۤ إِذࣰا لَّأَمۡسَكۡتُمۡ خَشۡیَةَ ٱلۡإِنفَاقِۚ وَكَانَ ٱلۡإِنسَٰنُ قَتُورࣰا ۝١٠٠ قُلْ لَّوْ اَنْتُمْ تَمْلِكُوْنَ خَزَآىِٕنَ رَحْمَةِ رَبِّیْۤ اِذًا لَّاَمْسَكْتُمْ خَشْیَةَ الْاِنْفَاقِ ؕ وَكَانَ الْاِنْسَانُ قَتُوْرًا ۟۠ وَلَقَدۡ ءَاتَيۡنَا مُوسَىٰ تِسۡعَ ءَايَٰتِۭ بَيِّنَٰتٖۖ فَسۡـَٔلۡ بَنِيٓ إِسۡرَٰٓءِيلَ إِذۡ جَآءَهُمۡ فَقَالَ لَهُۥ فِرۡعَوۡنُ إِنِّي لَأَظُنُّكَ يَٰمُوسَىٰ مَسۡحُورٗا ۝١٠١ وَلَقَدۡ ءَاتَیۡنَا مُوسَىٰ تِسۡعَ ءَایَٰتِۭ بَیِّنَٰتࣲۖ فَسۡءَلۡ بَنِیۤ إِسۡرَٰۤءِیلَ إِذۡ جَاۤءَهُمۡ فَقَالَ لَهُۥ فِرۡعَوۡنُ إِنِّی لَأَظُنُّكَ یَٰمُوسَىٰ مَسۡحُورࣰا ۝١٠١ وَلَقَدْ اٰتَیْنَا مُوْسٰی تِسْعَ اٰیٰتٍ بَیِّنٰتٍ فَسْـَٔلْ بَنِیْۤ اِسْرَآءِیْلَ اِذْ جَآءَهُمْ فَقَالَ لَهٗ فِرْعَوْنُ اِنِّیْ لَاَظُنُّكَ یٰمُوْسٰی مَسْحُوْرًا ۟ قَالَ لَقَدۡ عَلِمۡتَ مَآ أَنزَلَ هَٰٓؤُلَآءِ إِلَّا رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ بَصَآئِرَ وَإِنِّي لَأَظُنُّكَ يَٰفِرۡعَوۡنُ مَثۡبُورٗا ۝١٠٢ قَالَ لَقَدۡ عَلِمۡتَ مَاۤ أَنزَلَ هَٰۤؤُلَاۤءِ إِلَّا رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ بَصَاۤئِرَ وَإِنِّی لَأَظُنُّكَ یَٰفِرۡعَوۡنُ مَثۡبُورࣰا ۝١٠٢ قَالَ لَقَدْ عَلِمْتَ مَاۤ اَنْزَلَ هٰۤؤُلَآءِ اِلَّا رَبُّ السَّمٰوٰتِ وَالْاَرْضِ بَصَآىِٕرَ ۚ وَاِنِّیْ لَاَظُنُّكَ یٰفِرْعَوْنُ مَثْبُوْرًا ۟ فَأَرَادَ أَن يَسۡتَفِزَّهُم مِّنَ ٱلۡأَرۡضِ فَأَغۡرَقۡنَٰهُ وَمَن مَّعَهُۥ جَمِيعٗا ۝١٠٣ فَأَرَادَ أَن یَسۡتَفِزَّهُم مِّنَ ٱلۡأَرۡضِ فَأَغۡرَقۡنَٰهُ وَمَن مَّعَهُۥ جَمِیعࣰا ۝١٠٣ فَاَرَادَ اَنْ یَّسْتَفِزَّهُمْ مِّنَ الْاَرْضِ فَاَغْرَقْنٰهُ وَمَنْ مَّعَهٗ جَمِیْعًا ۟ۙ وَقُلۡنَا مِنۢ بَعۡدِهِۦ لِبَنِيٓ إِسۡرَٰٓءِيلَ ٱسۡكُنُواْ ٱلۡأَرۡضَ فَإِذَا جَآءَ وَعۡدُ ٱلۡأٓخِرَةِ جِئۡنَا بِكُمۡ لَفِيفٗا ۝١٠٤ وَقُلۡنَا مِنۢ بَعۡدِهِۦ لِبَنِیۤ إِسۡرَٰۤءِیلَ ٱسۡكُنُوا۟ ٱلۡأَرۡضَ فَإِذَا جَاۤءَ وَعۡدُ ٱلۡءَاخِرَةِ جِئۡنَا بِكُمۡ لَفِیفࣰا ۝١٠٤ وَّقُلْنَا مِنْ بَعْدِهٖ لِبَنِیْۤ اِسْرَآءِیْلَ اسْكُنُوا الْاَرْضَ فَاِذَا جَآءَ وَعْدُ الْاٰخِرَةِ جِئْنَا بِكُمْ لَفِیْفًا ۟ؕ وَبِٱلۡحَقِّ أَنزَلۡنَٰهُ وَبِٱلۡحَقِّ نَزَلَۗ وَمَآ أَرۡسَلۡنَٰكَ إِلَّا مُبَشِّرٗا وَنَذِيرٗا ۝١٠٥ وَبِٱلۡحَقِّ أَنزَلۡنَٰهُ وَبِٱلۡحَقِّ نَزَلَۗ وَمَاۤ أَرۡسَلۡنَٰكَ إِلَّا مُبَشِّرࣰا وَنَذِیرࣰا ۝١٠٥ وَبِالْحَقِّ اَنْزَلْنٰهُ وَبِالْحَقِّ نَزَلَ ؕ وَمَاۤ اَرْسَلْنٰكَ اِلَّا مُبَشِّرًا وَّنَذِیْرًا ۟ۘ وَقُرۡءَانٗا فَرَقۡنَٰهُ لِتَقۡرَأَهُۥ عَلَى ٱلنَّاسِ عَلَىٰ مُكۡثٖ وَنَزَّلۡنَٰهُ تَنزِيلٗا ۝١٠٦ وَقُرۡءَانࣰا فَرَقۡنَٰهُ لِتَقۡرَأَهُۥ عَلَى ٱلنَّاسِ عَلَىٰ مُكۡثࣲ وَنَزَّلۡنَٰهُ تَنزِیلࣰا ۝١٠٦ وَقُرْاٰنًا فَرَقْنٰهُ لِتَقْرَاَهٗ عَلَی النَّاسِ عَلٰی مُكْثٍ وَّنَزَّلْنٰهُ تَنْزِیْلًا ۟ قُلۡ ءَامِنُواْ بِهِۦٓ أَوۡ لَا تُؤۡمِنُوٓاْۚ إِنَّ ٱلَّذِينَ أُوتُواْ ٱلۡعِلۡمَ مِن قَبۡلِهِۦٓ إِذَا يُتۡلَىٰ عَلَيۡهِمۡ يَخِرُّونَۤ لِلۡأَذۡقَانِۤ سُجَّدٗاۤ ۝١٠٧ قُلۡ ءَامِنُوا۟ بِهِۦۤ أَوۡ لَا تُؤۡمِنُوۤا۟ۚ إِنَّ ٱلَّذِینَ أُوتُوا۟ ٱلۡعِلۡمَ مِن قَبۡلِهِۦۤ إِذَا یُتۡلَىٰ عَلَیۡهِمۡ یَ̅خِ̅رُّ̅و̅نَ̅ ̅لِ̅لۡ̅أَ̅ذۡ̅قَ̅ا̅نِ̅ ̅سُ̅جَّ̅دࣰ̅ا̅ ۝١٠٧ قُلْ اٰمِنُوْا بِهٖۤ اَوْ لَا تُؤْمِنُوْا ؕ اِنَّ الَّذِیْنَ اُوْتُوا الْعِلْمَ مِنْ قَبْلِهٖۤ اِذَا یُتْلٰی عَلَیْهِمْ یَخِرُّوْنَ لِلْاَذْقَانِ سُجَّدًا ۟ۙ وَيَقُولُونَ سُبۡحَٰنَ رَبِّنَآ إِن كَانَ وَعۡدُ رَبِّنَا لَمَفۡعُولٗا ۝١٠٨ وَیَقُولُونَ سُبۡحَٰنَ رَبِّنَاۤ إِن كَانَ وَعۡدُ رَبِّنَا لَمَفۡعُولࣰا ۝١٠٨ وَّیَقُوْلُوْنَ سُبْحٰنَ رَبِّنَاۤ اِنْ كَانَ وَعْدُ رَبِّنَا لَمَفْعُوْلًا ۟ وَيَخِرُّونَ لِلۡأَذۡقَانِ يَبۡكُونَ وَيَزِيدُهُمۡ خُشُوعٗا ۝١٠٩ وَیَخِرُّونَ لِلۡأَذۡقَانِ یَبۡكُونَ وَیَزِیدُهُمۡ خُشُوعࣰا ۝١٠٩ وَیَخِرُّوْنَ لِلْاَذْقَانِ یَبْكُوْنَ وَیَزِیْدُهُمْ خُشُوْعًا ۟ قُلِ ٱدۡعُواْ ٱللَّهَ أَوِ ٱدۡعُواْ ٱلرَّحۡمَٰنَۖ أَيّٗا مَّا تَدۡعُواْ فَلَهُ ٱلۡأَسۡمَآءُ ٱلۡحُسۡنَىٰۚ وَلَا تَجۡهَرۡ بِصَلَاتِكَ وَلَا تُخَافِتۡ بِهَا وَٱبۡتَغِ بَيۡنَ ذَٰلِكَ سَبِيلٗا ۝١١٠ قُلِ ٱدۡعُوا۟ ٱللَّهَ أَوِ ٱدۡعُوا۟ ٱلرَّحۡمَٰنَۖ أَیࣰّا مَّا تَدۡعُوا۟ فَلَهُ ٱلۡأَسۡمَاۤءُ ٱلۡحُسۡنَىٰۚ وَلَا تَجۡهَرۡ بِصَلَاتِكَ وَلَا تُخَافِتۡ بِهَا وَٱبۡتَغِ بَیۡنَ ذَٰلِكَ سَبِیلࣰا ۝١١٠ قُلِ ادْعُوا اللّٰهَ اَوِ ادْعُوا الرَّحْمٰنَ ؕ اَیًّا مَّا تَدْعُوْا فَلَهُ الْاَسْمَآءُ الْحُسْنٰی ۚ وَلَا تَجْهَرْ بِصَلَاتِكَ وَلَا تُخَافِتْ بِهَا وَابْتَغِ بَیْنَ ذٰلِكَ سَبِیْلًا ۟ وَقُلِ ٱلۡحَمۡدُ لِلَّهِ ٱلَّذِي لَمۡ يَتَّخِذۡ وَلَدٗا وَلَمۡ يَكُن لَّهُۥ شَرِيكٞ فِي ٱلۡمُلۡكِ وَلَمۡ يَكُن لَّهُۥ وَلِيّٞ مِّنَ ٱلذُّلِّۖ وَكَبِّرۡهُ تَكۡبِيرَۢا ۝١١١ وَقُلِ ٱلۡحَمۡدُ لِلَّهِ ٱلَّذِی لَمۡ یَتَّخِذۡ وَلَدࣰا وَلَمۡ یَكُن لَّهُۥ شَرِیكࣱ فِی ٱلۡمُلۡكِ وَلَمۡ یَكُن لَّهُۥ وَلِیࣱّ مِّنَ ٱلذُّلِّۖ وَكَبِّرۡهُ تَكۡبِیرَۢا ۝١١١ وَقُلِ الْحَمْدُ لِلّٰهِ الَّذِیْ لَمْ یَتَّخِذْ وَلَدًا وَّلَمْ یَكُنْ لَّهٗ شَرِیْكٌ فِی الْمُلْكِ وَلَمْ یَكُنْ لَّهٗ وَلِیٌّ مِّنَ الذُّلِّ وَكَبِّرْهُ تَكْبِیْرًا ۟۠
