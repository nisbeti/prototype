بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

ٱقۡتَرَبَ لِلنَّاسِ حِسَابُهُمۡ وَهُمۡ فِي غَفۡلَةٖ مُّعۡرِضُونَ ۝١ ٱقۡتَرَبَ لِلنَّاسِ حِسَابُهُمۡ وَهُمۡ فِی غَفۡلَةࣲ مُّعۡرِضُونَ ۝١ اِقْتَرَبَ لِلنَّاسِ حِسَابُهُمْ وَهُمْ فِیْ غَفْلَةٍ مُّعْرِضُوْنَ ۟ۚ مَا يَأۡتِيهِم مِّن ذِكۡرٖ مِّن رَّبِّهِم مُّحۡدَثٍ إِلَّا ٱسۡتَمَعُوهُ وَهُمۡ يَلۡعَبُونَ ۝٢ مَا یَأۡتِیهِم مِّن ذِكۡرࣲ مِّن رَّبِّهِم مُّحۡدَثٍ إِلَّا ٱسۡتَمَعُوهُ وَهُمۡ یَلۡعَبُونَ ۝٢ مَا یَاْتِیْهِمْ مِّنْ ذِكْرٍ مِّنْ رَّبِّهِمْ مُّحْدَثٍ اِلَّا اسْتَمَعُوْهُ وَهُمْ یَلْعَبُوْنَ ۟ۙ لَاهِيَةٗ قُلُوبُهُمۡۗ وَأَسَرُّواْ ٱلنَّجۡوَى ٱلَّذِينَ ظَلَمُواْ هَلۡ هَٰذَآ إِلَّا بَشَرٞ مِّثۡلُكُمۡۖ أَفَتَأۡتُونَ ٱلسِّحۡرَ وَأَنتُمۡ تُبۡصِرُونَ ۝٣ لَاهِیَةࣰ قُلُوبُهُمۡۗ وَأَسَرُّوا۟ ٱلنَّجۡوَى ٱلَّذِینَ ظَلَمُوا۟ هَلۡ هَٰذَاۤ إِلَّا بَشَرࣱ مِّثۡلُكُمۡۖ أَفَتَأۡتُونَ ٱلسِّحۡرَ وَأَنتُمۡ تُبۡصِرُونَ ۝٣ لَاهِیَةً قُلُوْبُهُمْ ؕ وَاَسَرُّوا النَّجْوَی ۖۗ الَّذِیْنَ ظَلَمُوْا ۖۗ هَلْ هٰذَاۤ اِلَّا بَشَرٌ مِّثْلُكُمْ ۚ اَفَتَاْتُوْنَ السِّحْرَ وَاَنْتُمْ تُبْصِرُوْنَ ۟ قَالَ رَبِّي يَعۡلَمُ ٱلۡقَوۡلَ فِي ٱلسَّمَآءِ وَٱلۡأَرۡضِۖ وَهُوَ ٱلسَّمِيعُ ٱلۡعَلِيمُ ۝٤ قَالَ رَبِّی یَعۡلَمُ ٱلۡقَوۡلَ فِی ٱلسَّمَاۤءِ وَٱلۡأَرۡضِۖ وَهُوَ ٱلسَّمِیعُ ٱلۡعَلِیمُ ۝٤ قٰلَ رَبِّیْ یَعْلَمُ الْقَوْلَ فِی السَّمَآءِ وَالْاَرْضِ ؗ وَهُوَ السَّمِیْعُ الْعَلِیْمُ ۟ بَلۡ قَالُوٓاْ أَضۡغَٰثُ أَحۡلَٰمِۭ بَلِ ٱفۡتَرَىٰهُ بَلۡ هُوَ شَاعِرٞ فَلۡيَأۡتِنَا بِـَٔايَةٖ كَمَآ أُرۡسِلَ ٱلۡأَوَّلُونَ ۝٥ بَلۡ قَالُوۤا۟ أَضۡغَٰثُ أَحۡلَٰمِۭ بَلِ ٱفۡتَرَىٰهُ بَلۡ هُوَ شَاعِرࣱ فَلۡیَأۡتِنَا بِءَایَةࣲ كَمَاۤ أُرۡسِلَ ٱلۡأَوَّلُونَ ۝٥ بَلْ قَالُوْۤا اَضْغَاثُ اَحْلَامٍ بَلِ افْتَرٰىهُ بَلْ هُوَ شَاعِرٌ ۖۚ فَلْیَاْتِنَا بِاٰیَةٍ كَمَاۤ اُرْسِلَ الْاَوَّلُوْنَ ۟ مَآ ءَامَنَتۡ قَبۡلَهُم مِّن قَرۡيَةٍ أَهۡلَكۡنَٰهَآۖ أَفَهُمۡ يُؤۡمِنُونَ ۝٦ مَاۤ ءَامَنَتۡ قَبۡلَهُم مِّن قَرۡیَةٍ أَهۡلَكۡنَٰهَاۤۖ أَفَهُمۡ یُؤۡمِنُونَ ۝٦ مَاۤ اٰمَنَتْ قَبْلَهُمْ مِّنْ قَرْیَةٍ اَهْلَكْنٰهَا ۚ اَفَهُمْ یُؤْمِنُوْنَ ۟ وَمَآ أَرۡسَلۡنَا قَبۡلَكَ إِلَّا رِجَالٗا نُّوحِيٓ إِلَيۡهِمۡۖ فَسۡـَٔلُوٓاْ أَهۡلَ ٱلذِّكۡرِ إِن كُنتُمۡ لَا تَعۡلَمُونَ ۝٧ وَمَاۤ أَرۡسَلۡنَا قَبۡلَكَ إِلَّا رِجَالࣰا نُّوحِیۤ إِلَیۡهِمۡۖ فَسۡءَلُوۤا۟ أَهۡلَ ٱلذِّكۡرِ إِن كُنتُمۡ لَا تَعۡلَمُونَ ۝٧ وَمَاۤ اَرْسَلْنَا قَبْلَكَ اِلَّا رِجَالًا نُّوْحِیْۤ اِلَیْهِمْ فَسْـَٔلُوْۤا اَهْلَ الذِّكْرِ اِنْ كُنْتُمْ لَا تَعْلَمُوْنَ ۟ وَمَا جَعَلۡنَٰهُمۡ جَسَدٗا لَّا يَأۡكُلُونَ ٱلطَّعَامَ وَمَا كَانُواْ خَٰلِدِينَ ۝٨ وَمَا جَعَلۡنَٰهُمۡ جَسَدࣰا لَّا یَأۡكُلُونَ ٱلطَّعَامَ وَمَا كَانُوا۟ خَٰلِدِینَ ۝٨ وَمَا جَعَلْنٰهُمْ جَسَدًا لَّا یَاْكُلُوْنَ الطَّعَامَ وَمَا كَانُوْا خٰلِدِیْنَ ۟ ثُمَّ صَدَقۡنَٰهُمُ ٱلۡوَعۡدَ فَأَنجَيۡنَٰهُمۡ وَمَن نَّشَآءُ وَأَهۡلَكۡنَا ٱلۡمُسۡرِفِينَ ۝٩ ثُمَّ صَدَقۡنَٰهُمُ ٱلۡوَعۡدَ فَأَنجَیۡنَٰهُمۡ وَمَن نَّشَاۤءُ وَأَهۡلَكۡنَا ٱلۡمُسۡرِفِینَ ۝٩ ثُمَّ صَدَقْنٰهُمُ الْوَعْدَ فَاَنْجَیْنٰهُمْ وَمَنْ نَّشَآءُ وَاَهْلَكْنَا الْمُسْرِفِیْنَ ۟ لَقَدۡ أَنزَلۡنَآ إِلَيۡكُمۡ كِتَٰبٗا فِيهِ ذِكۡرُكُمۡۚ أَفَلَا تَعۡقِلُونَ ۝١٠ لَقَدۡ أَنزَلۡنَاۤ إِلَیۡكُمۡ كِتَٰبࣰا فِیهِ ذِكۡرُكُمۡۚ أَفَلَا تَعۡقِلُونَ ۝١٠ لَقَدْ اَنْزَلْنَاۤ اِلَیْكُمْ كِتٰبًا فِیْهِ ذِكْرُكُمْ ؕ اَفَلَا تَعْقِلُوْنَ ۟۠ وَكَمۡ قَصَمۡنَا مِن قَرۡيَةٖ كَانَتۡ ظَالِمَةٗ وَأَنشَأۡنَا بَعۡدَهَا قَوۡمًا ءَاخَرِينَ ۝١١ وَكَمۡ قَصَمۡنَا مِن قَرۡیَةࣲ كَانَتۡ ظَالِمَةࣰ وَأَنشَأۡنَا بَعۡدَهَا قَوۡمًا ءَاخَرِینَ ۝١١ وَكَمْ قَصَمْنَا مِنْ قَرْیَةٍ كَانَتْ ظَالِمَةً وَّاَنْشَاْنَا بَعْدَهَا قَوْمًا اٰخَرِیْنَ ۟ فَلَمَّآ أَحَسُّواْ بَأۡسَنَآ إِذَا هُم مِّنۡهَا يَرۡكُضُونَ ۝١٢ فَلَمَّاۤ أَحَسُّوا۟ بَأۡسَنَاۤ إِذَا هُم مِّنۡهَا یَرۡكُضُونَ ۝١٢ فَلَمَّاۤ اَحَسُّوْا بَاْسَنَاۤ اِذَا هُمْ مِّنْهَا یَرْكُضُوْنَ ۟ؕ لَا تَرۡكُضُواْ وَٱرۡجِعُوٓاْ إِلَىٰ مَآ أُتۡرِفۡتُمۡ فِيهِ وَمَسَٰكِنِكُمۡ لَعَلَّكُمۡ تُسۡـَٔلُونَ ۝١٣ لَا تَرۡكُضُوا۟ وَٱرۡجِعُوۤا۟ إِلَىٰ مَاۤ أُتۡرِفۡتُمۡ فِیهِ وَمَسَٰكِنِكُمۡ لَعَلَّكُمۡ تُسۡءَلُونَ ۝١٣ لَا تَرْكُضُوْا وَارْجِعُوْۤا اِلٰی مَاۤ اُتْرِفْتُمْ فِیْهِ وَمَسٰكِنِكُمْ لَعَلَّكُمْ تُسْـَٔلُوْنَ ۟ قَالُواْ يَٰوَيۡلَنَآ إِنَّا كُنَّا ظَٰلِمِينَ ۝١٤ قَالُوا۟ یَٰوَیۡلَنَاۤ إِنَّا كُنَّا ظَٰلِمِینَ ۝١٤ قَالُوْا یٰوَیْلَنَاۤ اِنَّا كُنَّا ظٰلِمِیْنَ ۟ فَمَا زَالَت تِّلۡكَ دَعۡوَىٰهُمۡ حَتَّىٰ جَعَلۡنَٰهُمۡ حَصِيدًا خَٰمِدِينَ ۝١٥ فَمَا زَالَت تِّلۡكَ دَعۡوَىٰهُمۡ حَتَّىٰ جَعَلۡنَٰهُمۡ حَصِیدًا خَٰمِدِینَ ۝١٥ فَمَا زَالَتْ تِّلْكَ دَعْوٰىهُمْ حَتّٰی جَعَلْنٰهُمْ حَصِیْدًا خٰمِدِیْنَ ۟ وَمَا خَلَقۡنَا ٱلسَّمَآءَ وَٱلۡأَرۡضَ وَمَا بَيۡنَهُمَا لَٰعِبِينَ ۝١٦ وَمَا خَلَقۡنَا ٱلسَّمَاۤءَ وَٱلۡأَرۡضَ وَمَا بَیۡنَهُمَا لَٰعِبِینَ ۝١٦ وَمَا خَلَقْنَا السَّمَآءَ وَالْاَرْضَ وَمَا بَیْنَهُمَا لٰعِبِیْنَ ۟ لَوۡ أَرَدۡنَآ أَن نَّتَّخِذَ لَهۡوٗا لَّٱتَّخَذۡنَٰهُ مِن لَّدُنَّآ إِن كُنَّا فَٰعِلِينَ ۝١٧ لَوۡ أَرَدۡنَاۤ أَن نَّتَّخِذَ لَهۡوࣰا لَّٱتَّخَذۡنَٰهُ مِن لَّدُنَّاۤ إِن كُنَّا فَٰعِلِینَ ۝١٧ لَوْ اَرَدْنَاۤ اَنْ نَّتَّخِذَ لَهْوًا لَّاتَّخَذْنٰهُ مِنْ لَّدُنَّاۤ ۖۗ اِنْ كُنَّا فٰعِلِیْنَ ۟ بَلۡ نَقۡذِفُ بِٱلۡحَقِّ عَلَى ٱلۡبَٰطِلِ فَيَدۡمَغُهُۥ فَإِذَا هُوَ زَاهِقٞۚ وَلَكُمُ ٱلۡوَيۡلُ مِمَّا تَصِفُونَ ۝١٨ بَلۡ نَقۡذِفُ بِٱلۡحَقِّ عَلَى ٱلۡبَٰطِلِ فَیَدۡمَغُهُۥ فَإِذَا هُوَ زَاهِقࣱۚ وَلَكُمُ ٱلۡوَیۡلُ مِمَّا تَصِفُونَ ۝١٨ بَلْ نَقْذِفُ بِالْحَقِّ عَلَی الْبَاطِلِ فَیَدْمَغُهٗ فَاِذَا هُوَ زَاهِقٌ ؕ وَلَكُمُ الْوَیْلُ مِمَّا تَصِفُوْنَ ۟ وَلَهُۥ مَن فِي ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۚ وَمَنۡ عِندَهُۥ لَا يَسۡتَكۡبِرُونَ عَنۡ عِبَادَتِهِۦ وَلَا يَسۡتَحۡسِرُونَ ۝١٩ وَلَهُۥ مَن فِی ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۚ وَمَنۡ عِندَهُۥ لَا یَسۡتَكۡبِرُونَ عَنۡ عِبَادَتِهِۦ وَلَا یَسۡتَحۡسِرُونَ ۝١٩ وَلَهٗ مَنْ فِی السَّمٰوٰتِ وَالْاَرْضِ ؕ وَمَنْ عِنْدَهٗ لَا یَسْتَكْبِرُوْنَ عَنْ عِبَادَتِهٖ وَلَا یَسْتَحْسِرُوْنَ ۟ۚ يُسَبِّحُونَ ٱلَّيۡلَ وَٱلنَّهَارَ لَا يَفۡتُرُونَ ۝٢٠ یُسَبِّحُونَ ٱلَّیۡلَ وَٱلنَّهَارَ لَا یَفۡتُرُونَ ۝٢٠ یُسَبِّحُوْنَ الَّیْلَ وَالنَّهَارَ لَا یَفْتُرُوْنَ ۟ أَمِ ٱتَّخَذُوٓاْ ءَالِهَةٗ مِّنَ ٱلۡأَرۡضِ هُمۡ يُنشِرُونَ ۝٢١ أَمِ ٱتَّخَذُوۤا۟ ءَالِهَةࣰ مِّنَ ٱلۡأَرۡضِ هُمۡ یُنشِرُونَ ۝٢١ اَمِ اتَّخَذُوْۤا اٰلِهَةً مِّنَ الْاَرْضِ هُمْ یُنْشِرُوْنَ ۟ لَوۡ كَانَ فِيهِمَآ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَاۚ فَسُبۡحَٰنَ ٱللَّهِ رَبِّ ٱلۡعَرۡشِ عَمَّا يَصِفُونَ ۝٢٢ لَوۡ كَانَ فِیهِمَاۤ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَاۚ فَسُبۡحَٰنَ ٱللَّهِ رَبِّ ٱلۡعَرۡشِ عَمَّا یَصِفُونَ ۝٢٢ لَوْ كَانَ فِیْهِمَاۤ اٰلِهَةٌ اِلَّا اللّٰهُ لَفَسَدَتَا ۚ فَسُبْحٰنَ اللّٰهِ رَبِّ الْعَرْشِ عَمَّا یَصِفُوْنَ ۟ لَا يُسۡـَٔلُ عَمَّا يَفۡعَلُ وَهُمۡ يُسۡـَٔلُونَ ۝٢٣ لَا یُسۡءَلُ عَمَّا یَفۡعَلُ وَهُمۡ یُسۡءَلُونَ ۝٢٣ لَا یُسْـَٔلُ عَمَّا یَفْعَلُ وَهُمْ یُسْـَٔلُوْنَ ۟ أَمِ ٱتَّخَذُواْ مِن دُونِهِۦٓ ءَالِهَةٗۖ قُلۡ هَاتُواْ بُرۡهَٰنَكُمۡۖ هَٰذَا ذِكۡرُ مَن مَّعِيَ وَذِكۡرُ مَن قَبۡلِيۚ بَلۡ أَكۡثَرُهُمۡ لَا يَعۡلَمُونَ ٱلۡحَقَّۖ فَهُم مُّعۡرِضُونَ ۝٢٤ أَمِ ٱتَّخَذُوا۟ مِن دُونِهِۦۤ ءَالِهَةࣰۖ قُلۡ هَاتُوا۟ بُرۡهَٰنَكُمۡۖ هَٰذَا ذِكۡرُ مَن مَّعِیَ وَذِكۡرُ مَن قَبۡلِیۚ بَلۡ أَكۡثَرُهُمۡ لَا یَعۡلَمُونَ ٱلۡحَقَّۖ فَهُم مُّعۡرِضُونَ ۝٢٤ اَمِ اتَّخَذُوْا مِنْ دُوْنِهٖۤ اٰلِهَةً ؕ قُلْ هَاتُوْا بُرْهَانَكُمْ ۚ هٰذَا ذِكْرُ مَنْ مَّعِیَ وَذِكْرُ مَنْ قَبْلِیْ ؕ بَلْ اَكْثَرُهُمْ لَا یَعْلَمُوْنَ ۙ الْحَقَّ فَهُمْ مُّعْرِضُوْنَ ۟ وَمَآ أَرۡسَلۡنَا مِن قَبۡلِكَ مِن رَّسُولٍ إِلَّا نُوحِيٓ إِلَيۡهِ أَنَّهُۥ لَآ إِلَٰهَ إِلَّآ أَنَا۠ فَٱعۡبُدُونِ ۝٢٥ وَمَاۤ أَرۡسَلۡنَا مِن قَبۡلِكَ مِن رَّسُولٍ إِلَّا نُوحِیۤ إِلَیۡهِ أَنَّهُۥ لَاۤ إِلَٰهَ إِلَّاۤ أَنَا۠ فَٱعۡبُدُونِ ۝٢٥ وَمَاۤ اَرْسَلْنَا مِنْ قَبْلِكَ مِنْ رَّسُوْلٍ اِلَّا نُوْحِیْۤ اِلَیْهِ اَنَّهٗ لَاۤ اِلٰهَ اِلَّاۤ اَنَا فَاعْبُدُوْنِ ۟ وَقَالُواْ ٱتَّخَذَ ٱلرَّحۡمَٰنُ وَلَدٗاۗ سُبۡحَٰنَهُۥۚ بَلۡ عِبَادٞ مُّكۡرَمُونَ ۝٢٦ وَقَالُوا۟ ٱتَّخَذَ ٱلرَّحۡمَٰنُ وَلَدࣰاۗ سُبۡحَٰنَهُۥۚ بَلۡ عِبَادࣱ مُّكۡرَمُونَ ۝٢٦ وَقَالُوا اتَّخَذَ الرَّحْمٰنُ وَلَدًا سُبْحٰنَهٗ ؕ بَلْ عِبَادٌ مُّكْرَمُوْنَ ۟ۙ لَا يَسۡبِقُونَهُۥ بِٱلۡقَوۡلِ وَهُم بِأَمۡرِهِۦ يَعۡمَلُونَ ۝٢٧ لَا یَسۡبِقُونَهُۥ بِٱلۡقَوۡلِ وَهُم بِأَمۡرِهِۦ یَعۡمَلُونَ ۝٢٧ لَا یَسْبِقُوْنَهٗ بِالْقَوْلِ وَهُمْ بِاَمْرِهٖ یَعْمَلُوْنَ ۟ يَعۡلَمُ مَا بَيۡنَ أَيۡدِيهِمۡ وَمَا خَلۡفَهُمۡ وَلَا يَشۡفَعُونَ إِلَّا لِمَنِ ٱرۡتَضَىٰ وَهُم مِّنۡ خَشۡيَتِهِۦ مُشۡفِقُونَ ۝٢٨ یَعۡلَمُ مَا بَیۡنَ أَیۡدِیهِمۡ وَمَا خَلۡفَهُمۡ وَلَا یَشۡفَعُونَ إِلَّا لِمَنِ ٱرۡتَضَىٰ وَهُم مِّنۡ خَشۡیَتِهِۦ مُشۡفِقُونَ ۝٢٨ یَعْلَمُ مَا بَیْنَ اَیْدِیْهِمْ وَمَا خَلْفَهُمْ وَلَا یَشْفَعُوْنَ ۙ اِلَّا لِمَنِ ارْتَضٰی وَهُمْ مِّنْ خَشْیَتِهٖ مُشْفِقُوْنَ ۟ وَمَن يَقُلۡ مِنۡهُمۡ إِنِّيٓ إِلَٰهٞ مِّن دُونِهِۦ فَذَٰلِكَ نَجۡزِيهِ جَهَنَّمَۚ كَذَٰلِكَ نَجۡزِي ٱلظَّٰلِمِينَ ۝٢٩ وَمَن یَقُلۡ مِنۡهُمۡ إِنِّیۤ إِلَٰهࣱ مِّن دُونِهِۦ فَذَٰلِكَ نَجۡزِیهِ جَهَنَّمَۚ كَذَٰلِكَ نَجۡزِی ٱلظَّٰلِمِینَ ۝٢٩ وَمَنْ یَّقُلْ مِنْهُمْ اِنِّیْۤ اِلٰهٌ مِّنْ دُوْنِهٖ فَذٰلِكَ نَجْزِیْهِ جَهَنَّمَ ؕ كَذٰلِكَ نَجْزِی الظّٰلِمِیْنَ ۟۠ أَوَلَمۡ يَرَ ٱلَّذِينَ كَفَرُوٓاْ أَنَّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ كَانَتَا رَتۡقٗا فَفَتَقۡنَٰهُمَاۖ وَجَعَلۡنَا مِنَ ٱلۡمَآءِ كُلَّ شَيۡءٍ حَيٍّۚ أَفَلَا يُؤۡمِنُونَ ۝٣٠ أَوَلَمۡ یَرَ ٱلَّذِینَ كَفَرُوۤا۟ أَنَّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ كَانَتَا رَتۡقࣰا فَفَتَقۡنَٰهُمَاۖ وَجَعَلۡنَا مِنَ ٱلۡمَاۤءِ كُلَّ شَیۡءٍ حَیٍّۚ أَفَلَا یُؤۡمِنُونَ ۝٣٠ اَوَلَمْ یَرَ الَّذِیْنَ كَفَرُوْۤا اَنَّ السَّمٰوٰتِ وَالْاَرْضَ كَانَتَا رَتْقًا فَفَتَقْنٰهُمَا ؕ وَجَعَلْنَا مِنَ الْمَآءِ كُلَّ شَیْءٍ حَیٍّ ؕ اَفَلَا یُؤْمِنُوْنَ ۟ وَجَعَلۡنَا فِي ٱلۡأَرۡضِ رَوَٰسِيَ أَن تَمِيدَ بِهِمۡ وَجَعَلۡنَا فِيهَا فِجَاجٗا سُبُلٗا لَّعَلَّهُمۡ يَهۡتَدُونَ ۝٣١ وَجَعَلۡنَا فِی ٱلۡأَرۡضِ رَوَٰسِیَ أَن تَمِیدَ بِهِمۡ وَجَعَلۡنَا فِیهَا فِجَاجࣰا سُبُلࣰا لَّعَلَّهُمۡ یَهۡتَدُونَ ۝٣١ وَجَعَلْنَا فِی الْاَرْضِ رَوَاسِیَ اَنْ تَمِیْدَ بِهِمْ ۪ وَجَعَلْنَا فِیْهَا فِجَاجًا سُبُلًا لَّعَلَّهُمْ یَهْتَدُوْنَ ۟ وَجَعَلۡنَا ٱلسَّمَآءَ سَقۡفٗا مَّحۡفُوظٗاۖ وَهُمۡ عَنۡ ءَايَٰتِهَا مُعۡرِضُونَ ۝٣٢ وَجَعَلۡنَا ٱلسَّمَاۤءَ سَقۡفࣰا مَّحۡفُوظࣰاۖ وَهُمۡ عَنۡ ءَایَٰتِهَا مُعۡرِضُونَ ۝٣٢ وَجَعَلْنَا السَّمَآءَ سَقْفًا مَّحْفُوْظًا ۖۚ وَّهُمْ عَنْ اٰیٰتِهَا مُعْرِضُوْنَ ۟ وَهُوَ ٱلَّذِي خَلَقَ ٱلَّيۡلَ وَٱلنَّهَارَ وَٱلشَّمۡسَ وَٱلۡقَمَرَۖ كُلّٞ فِي فَلَكٖ يَسۡبَحُونَ ۝٣٣ وَهُوَ ٱلَّذِی خَلَقَ ٱلَّیۡلَ وَٱلنَّهَارَ وَٱلشَّمۡسَ وَٱلۡقَمَرَۖ كُلࣱّ فِی فَلَكࣲ یَسۡبَحُونَ ۝٣٣ وَهُوَ الَّذِیْ خَلَقَ الَّیْلَ وَالنَّهَارَ وَالشَّمْسَ وَالْقَمَرَ ؕ كُلٌّ فِیْ فَلَكٍ یَّسْبَحُوْنَ ۟ وَمَا جَعَلۡنَا لِبَشَرٖ مِّن قَبۡلِكَ ٱلۡخُلۡدَۖ أَفَإِيْن مِّتَّ فَهُمُ ٱلۡخَٰلِدُونَ ۝٣٤ وَمَا جَعَلۡنَا لِبَشَرࣲ مِّن قَبۡلِكَ ٱلۡخُلۡدَۖ أَفَإِی۟ن مِّتَّ فَهُمُ ٱلۡخَٰلِدُونَ ۝٣٤ وَمَا جَعَلْنَا لِبَشَرٍ مِّنْ قَبْلِكَ الْخُلْدَ ؕ اَفَاۡىِٕنْ مِّتَّ فَهُمُ الْخٰلِدُوْنَ ۟ كُلُّ نَفۡسٖ ذَآئِقَةُ ٱلۡمَوۡتِۗ وَنَبۡلُوكُم بِٱلشَّرِّ وَٱلۡخَيۡرِ فِتۡنَةٗۖ وَإِلَيۡنَا تُرۡجَعُونَ ۝٣٥ كُلُّ نَفۡسࣲ ذَاۤئِقَةُ ٱلۡمَوۡتِۗ وَنَبۡلُوكُم بِٱلشَّرِّ وَٱلۡخَیۡرِ فِتۡنَةࣰۖ وَإِلَیۡنَا تُرۡجَعُونَ ۝٣٥ كُلُّ نَفْسٍ ذَآىِٕقَةُ الْمَوْتِ ؕ وَنَبْلُوْكُمْ بِالشَّرِّ وَالْخَیْرِ فِتْنَةً ؕ وَاِلَیْنَا تُرْجَعُوْنَ ۟ وَإِذَا رَءَاكَ ٱلَّذِينَ كَفَرُوٓاْ إِن أَنفُسِهِمۡ وَلَا هُم مِّنَّا يُصۡحَبُونَ ۝٤٣ أَمۡ لَهُمۡ ءَالِهَةࣱ تَمۡنَعُهُم مِّن دُونِنَاۚ لَا یَسۡتَطِیعُونَ نَصۡرَ أَنفُسِهِمۡ وَلَا هُم مِّنَّا یُصۡحَبُونَ ۝٤٣ اَمْ لَهُمْ اٰلِهَةٌ تَمْنَعُهُمْ مِّنْ دُوْنِنَا ؕ لَا یَسْتَطِیْعُوْنَ نَصْرَ اَنْفُسِهِمْ وَلَا هُمْ مِّنَّا یُصْحَبُوْنَ ۟ بَلۡ مَتَّعۡنَا هَٰٓؤُلَآءِ وَءَابَآءَهُمۡ حَتَّىٰ طَالَ عَلَيۡهِمُ ٱلۡعُمُرُۗ أَفَلَا يَرَوۡنَ أَنَّا نَأۡتِي ٱلۡأَرۡضَ نَنقُصُهَا مِنۡ أَطۡرَافِهَآۚ أَفَهُمُ ٱلۡغَٰلِبُونَ ۝٤٤ بَلۡ مَتَّعۡنَا هَٰۤؤُلَاۤءِ وَءَابَاۤءَهُمۡ حَتَّىٰ طَالَ عَلَیۡهِمُ ٱلۡعُمُرُۗ أَفَلَا یَرَوۡنَ أَنَّا نَأۡتِی ٱلۡأَرۡضَ نَنقُصُهَا مِنۡ أَطۡرَافِهَاۤۚ أَفَهُمُ ٱلۡغَٰلِبُونَ ۝٤٤ بَلْ مَتَّعْنَا هٰۤؤُلَآءِ وَاٰبَآءَهُمْ حَتّٰی طَالَ عَلَیْهِمُ الْعُمُرُ ؕ اَفَلَا یَرَوْنَ اَنَّا نَاْتِی الْاَرْضَ نَنْقُصُهَا مِنْ اَطْرَافِهَا ؕ اَفَهُمُ الْغٰلِبُوْنَ ۟ قُلۡ إِنَّمَآ أُنذِرُكُم بِٱلۡوَحۡيِۚ وَلَا يَسۡمَعُ ٱلصُّمُّ ٱلدُّعَآءَ إِذَا مَا يُنذَرُونَ ۝٤٥ قُلۡ إِنَّمَاۤ أُنذِرُكُم بِٱلۡوَحۡیِۚ وَلَا یَسۡمَعُ ٱلصُّمُّ ٱلدُّعَاۤءَ إِذَا مَا یُنذَرُونَ ۝٤٥ قُلْ اِنَّمَاۤ اُنْذِرُكُمْ بِالْوَحْیِ ۖؗ وَلَا یَسْمَعُ الصُّمُّ الدُّعَآءَ اِذَا مَا یُنْذَرُوْنَ ۟ وَلَئِن مَّسَّتۡهُمۡ نَفۡحَةٞ مِّنۡ عَذَابِ رَبِّكَ لَيَقُولُنَّ يَٰوَيۡلَنَآ إِنَّا كُنَّا ظَٰلِمِينَ ۝٤٦ وَلَئِن مَّسَّتۡهُمۡ نَفۡحَةࣱ مِّنۡ عَذَابِ رَبِّكَ لَیَقُولُنَّ یَٰوَیۡلَنَاۤ إِنَّا كُنَّا ظَٰلِمِینَ ۝٤٦ وَلَىِٕنْ مَّسَّتْهُمْ نَفْحَةٌ مِّنْ عَذَابِ رَبِّكَ لَیَقُوْلُنَّ یٰوَیْلَنَاۤ اِنَّا كُنَّا ظٰلِمِیْنَ ۟ وَنَضَعُ ٱلۡمَوَٰزِينَ ٱلۡقِسۡطَ لِيَوۡمِ ٱلۡقِيَٰمَةِ فَلَا تُظۡلَمُ نَفۡسٞ شَيۡـٔٗاۖ وَإِن كَانَ مِثۡقَالَ حَبَّةٖ مِّنۡ خَرۡدَلٍ أَتَيۡنَا بِهَاۗ وَكَفَىٰ بِنَا حَٰسِبِينَ ۝٤٧ وَنَضَعُ ٱلۡمَوَٰزِینَ ٱلۡقِسۡطَ لِیَوۡمِ ٱلۡقِیَٰمَةِ فَلَا تُظۡلَمُ نَفۡسࣱ شَیۡءࣰاۖ وَإِن كَانَ مِثۡقَالَ حَبَّةࣲ مِّنۡ خَرۡدَلٍ أَتَیۡنَا بِهَاۗ وَكَفَىٰ بِنَا حَٰسِبِینَ ۝٤٧ وَنَضَعُ الْمَوَازِیْنَ الْقِسْطَ لِیَوْمِ الْقِیٰمَةِ فَلَا تُظْلَمُ نَفْسٌ شَیْـًٔا ؕ وَاِنْ كَانَ مِثْقَالَ حَبَّةٍ مِّنْ خَرْدَلٍ اَتَیْنَا بِهَا ؕ وَكَفٰی بِنَا حٰسِبِیْنَ ۟ وَلَقَدۡ ءَاتَيۡنَا مُوسَىٰ وَهَٰرُونَ ٱلۡفُرۡقَانَ وَضِيَآءٗ وَذِكۡرٗا لِّلۡمُتَّقِينَ ۝٤٨ وَلَقَدۡ ءَاتَیۡنَا مُوسَىٰ وَهَٰرُونَ ٱلۡفُرۡقَانَ وَضِیَاۤءࣰ وَذِكۡرࣰا لِّلۡمُتَّقِینَ ۝٤٨ وَلَقَدْ اٰتَیْنَا مُوْسٰی وَهٰرُوْنَ الْفُرْقَانَ وَضِیَآءً وَّذِكْرًا لِّلْمُتَّقِیْنَ ۟ۙ ٱلَّذِينَ يَخۡشَوۡنَ رَبَّهُم بِٱلۡغَيۡبِ وَهُم مِّنَ ٱلسَّاعَةِ مُشۡفِقُونَ ۝٤٩ ٱلَّذِینَ یَخۡشَوۡنَ رَبَّهُم بِٱلۡغَیۡبِ وَهُم مِّنَ ٱلسَّاعَةِ مُشۡفِقُونَ ۝٤٩ الَّذِیْنَ یَخْشَوْنَ رَبَّهُمْ بِالْغَیْبِ وَهُمْ مِّنَ السَّاعَةِ مُشْفِقُوْنَ ۟ وَهَٰذَا ذِكۡرٞ مُّبَارَكٌ أَنزَلۡنَٰهُۚ أَفَأَنتُمۡ لَهُۥ مُنكِرُونَ ۝٥٠ وَهَٰذَا ذِكۡرࣱ مُّبَارَكٌ أَنزَلۡنَٰهُۚ أَفَأَنتُمۡ لَهُۥ مُنكِرُونَ ۝٥٠ وَهٰذَا ذِكْرٌ مُّبٰرَكٌ اَنْزَلْنٰهُ ؕ اَفَاَنْتُمْ لَهٗ مُنْكِرُوْنَ ۟۠ وَلَقَدۡ ءَاتَيۡنَآ إِبۡرَٰهِيمَ رُشۡدَهُۥ مِن قَبۡلُ وَكُنَّا بِهِۦ عَٰلِمِينَ ۝٥١ وَلَقَدۡ ءَاتَیۡنَاۤ إِبۡرَٰهِیمَ رُشۡدَهُۥ مِن قَبۡلُ وَكُنَّا بِهِۦ عَٰلِمِینَ ۝٥١ وَلَقَدْ اٰتَیْنَاۤ اِبْرٰهِیْمَ رُشْدَهٗ مِنْ قَبْلُ وَكُنَّا بِهٖ عٰلِمِیْنَ ۟ۚ إِذۡ قَالَ لِأَبِيهِ وَقَوۡمِهِۦ مَا هَٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِيٓ أَنتُمۡ لَهَا عَٰكِفُونَ ۝٥٢ إِذۡ قَالَ لِأَبِیهِ وَقَوۡمِهِۦ مَا هَٰذِهِ ٱلتَّمَاثِیلُ ٱلَّتِیۤ أَنتُمۡ لَهَا عَٰكِفُونَ ۝٥٢ اِذْ قَالَ لِاَبِیْهِ وَقَوْمِهٖ مَا هٰذِهِ التَّمَاثِیْلُ الَّتِیْۤ اَنْتُمْ لَهَا عٰكِفُوْنَ ۟ قَالُواْ وَجَدۡنَآ ءَابَآءَنَا لَهَا عَٰبِدِينَ ۝٥٣ قَالُوا۟ وَجَدۡنَاۤ ءَابَاۤءَنَا لَهَا عَٰبِدِینَ ۝٥٣ قَالُوْا وَجَدْنَاۤ اٰبَآءَنَا لَهَا عٰبِدِیْنَ ۟ قَالَ لَقَدۡ كُنتُمۡ أَنتُمۡ وَءَابَآؤُكُمۡ فِي ضَلَٰلٖ مُّبِينٖ ۝٥٤ قَالَ لَقَدۡ كُنتُمۡ أَنتُمۡ وَءَابَاۤؤُكُمۡ فِی ضَلَٰلࣲ مُّبِینࣲ ۝٥٤ قَالَ لَقَدْ كُنْتُمْ اَنْتُمْ وَاٰبَآؤُكُمْ فِیْ ضَلٰلٍ مُّبِیْنٍ ۟ قَالُوٓاْ أَجِئۡتَنَا بِٱلۡحَقِّ أَمۡ أَنتَ مِنَ ٱللَّٰعِبِينَ ۝٥٥ قَالُوۤا۟ أَجِئۡتَنَا بِٱلۡحَقِّ أَمۡ أَنتَ مِنَ ٱللَّٰعِبِینَ ۝٥٥ قَالُوْۤا اَجِئْتَنَا بِالْحَقِّ اَمْ اَنْتَ مِنَ اللّٰعِبِیْنَ ۟ قَالَ بَل رَّبُّكُمۡ رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ ٱلَّذِي فَطَرَهُنَّ وَأَنَا۠ عَلَىٰ ذَٰلِكُم مِّنَ ٱلشَّٰهِدِينَ ۝٥٦ قَالَ بَل رَّبُّكُمۡ رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ ٱلَّذِی فَطَرَهُنَّ وَأَنَا۠ عَلَىٰ ذَٰلِكُم مِّنَ ٱلشَّٰهِدِینَ ۝٥٦ قَالَ بَلْ رَّبُّكُمْ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ الَّذِیْ فَطَرَهُنَّ ۖؗ وَاَنَا عَلٰی ذٰلِكُمْ مِّنَ الشّٰهِدِیْنَ ۟ وَتَٱللَّهِ لَأَكِيدَنَّ أَصۡنَٰمَكُم بَعۡدَ أَن تُوَلُّواْ مُدۡبِرِينَ ۝٥٧ وَتَٱللَّهِ لَأَكِیدَنَّ أَصۡنَٰمَكُم بَعۡدَ أَن تُوَلُّوا۟ مُدۡبِرِینَ ۝٥٧ وَتَاللّٰهِ لَاَكِیْدَنَّ اَصْنَامَكُمْ بَعْدَ اَنْ تُوَلُّوْا مُدْبِرِیْنَ ۟ فَجَعَلَهُمۡ جُذَٰذًا إِلَّا كَبِيرٗا لَّهُمۡ لَعَلَّهُمۡ إِلَيۡهِ يَرۡجِعُونَ ۝٥٨ فَجَعَلَهُمۡ جُذَٰذًا إِلَّا كَبِیرࣰا لَّهُمۡ لَعَلَّهُمۡ إِلَیۡهِ یَرۡجِعُونَ ۝٥٨ فَجَعَلَهُمْ جُذٰذًا اِلَّا كَبِیْرًا لَّهُمْ لَعَلَّهُمْ اِلَیْهِ یَرْجِعُوْنَ ۟ قَالُواْ مَن فَعَلَ هَٰذَا بِـَٔالِهَتِنَآ إِنَّهُۥ لَمِنَ ٱلظَّٰلِمِينَ ۝٥٩ قَالُوا۟ مَن فَعَلَ هَٰذَا بِءَالِهَتِنَاۤ إِنَّهُۥ لَمِنَ ٱلظَّٰلِمِینَ ۝٥٩ قَالُوْا مَنْ فَعَلَ هٰذَا بِاٰلِهَتِنَاۤ اِنَّهٗ لَمِنَ الظّٰلِمِیْنَ ۟ قَالُواْ سَمِعۡنَا فَتٗى يَذۡكُرُهُمۡ يُقَالُ لَهُۥٓ إِبۡرَٰهِيمُ ۝٦٠ قَالُوا۟ سَمِعۡنَا فَتࣰى یَذۡكُرُهُمۡ یُقَالُ لَهُۥۤ إِبۡرَٰهِیمُ ۝٦٠ قَالُوْا سَمِعْنَا فَتًی یَّذْكُرُهُمْ یُقَالُ لَهٗۤ اِبْرٰهِیْمُ ۟ؕ قَالُواْ فَأۡتُواْ بِهِۦ عَلَىٰٓ أَعۡيُنِ ٱلنَّاسِ لَعَلَّهُمۡ يَشۡهَدُونَ ۝٦١ قَالُوا۟ فَأۡتُوا۟ بِهِۦ عَلَىٰۤ أَعۡیُنِ ٱلنَّاسِ لَعَلَّهُمۡ یَشۡهَدُونَ ۝٦١ قَالُوْا فَاْتُوْا بِهٖ عَلٰۤی اَعْیُنِ النَّاسِ لَعَلَّهُمْ یَشْهَدُوْنَ ۟ قَالُوٓاْ ءَأَنتَ فَعَلۡتَ هَٰذَا بِـَٔالِهَتِنَا يَٰٓإِبۡرَٰهِيمُ ۝٦٢ قَالُوۤا۟ ءَأَنتَ فَعَلۡتَ هَٰذَا بِءَالِهَتِنَا یَٰۤإِبۡرَٰهِیمُ ۝٦٢ قَالُوْۤا ءَاَنْتَ فَعَلْتَ هٰذَا بِاٰلِهَتِنَا یٰۤاِبْرٰهِیْمُ ۟ؕ قَالَ بَلۡ فَعَلَهُۥ كَبِيرُهُمۡ هَٰذَا فَسۡـَٔلُوهُمۡ إِن كَانُواْ يَنطِقُونَ ۝٦٣ قَالَ بَلۡ فَعَلَهُۥ كَبِیرُهُمۡ هَٰذَا فَسۡءَلُوهُمۡ إِن كَانُوا۟ یَنطِقُونَ ۝٦٣ قَالَ بَلْ فَعَلَهٗ ۖۗ كَبِیْرُهُمْ هٰذَا فَسْـَٔلُوْهُمْ اِنْ كَانُوْا یَنْطِقُوْنَ ۟ فَرَجَعُوٓاْ إِلَىٰٓ أَنفُسِهِمۡ فَقَالُوٓاْ إِنَّكُمۡ أَنتُمُ ٱلظَّٰلِمُونَ ۝٦٤ فَرَجَعُوۤا۟ إِلَىٰۤ أَنفُسِهِمۡ فَقَالُوۤا۟ إِنَّكُمۡ أَنتُمُ ٱلظَّٰلِمُونَ ۝٦٤ فَرَجَعُوْۤا اِلٰۤی اَنْفُسِهِمْ فَقَالُوْۤا اِنَّكُمْ اَنْتُمُ الظّٰلِمُوْنَ ۟ۙ ثُمَّ نُكِسُواْ عَلَىٰ رُءُوسِهِمۡ لَقَدۡ عَلِمۡتَ مَا هَٰٓؤُلَآءِ يَنطِقُونَ ۝٦٥ ثُمَّ نُكِسُوا۟ عَلَىٰ رُءُوسِهِمۡ لَقَدۡ عَلِمۡتَ مَا هَٰۤؤُلَاۤءِ یَنطِقُونَ ۝٦٥ ثُمَّ نُكِسُوْا عَلٰی رُءُوْسِهِمْ ۚ لَقَدْ عَلِمْتَ مَا هٰۤؤُلَآءِ یَنْطِقُوْنَ ۟ قَالَ أَفَتَعۡبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَنفَعُكُمۡ شَيۡـٔٗا وَلَا يَضُرُّكُمۡ ۝٦٦ قَالَ أَفَتَعۡبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا یَنفَعُكُمۡ شَیۡءࣰا وَلَا یَضُرُّكُمۡ ۝٦٦ قَالَ اَفَتَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ مَا لَا یَنْفَعُكُمْ شَیْـًٔا وَّلَا یَضُرُّكُمْ ۟ؕ أُفّٖ لَّكُمۡ وَلِمَا تَعۡبُدُونَ مِن دُونِ ٱللَّهِۚ أَفَلَا تَعۡقِلُونَ ۝٦٧ أُفࣲّ لَّكُمۡ وَلِمَا تَعۡبُدُونَ مِن دُونِ ٱللَّهِۚ أَفَلَا تَعۡقِلُونَ ۝٦٧ اُفٍّ لَّكُمْ وَلِمَا تَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ ؕ اَفَلَا تَعْقِلُوْنَ ۟ قَالُواْ حَرِّقُوهُ وَٱنصُرُوٓاْ ءَالِهَتَكُمۡ إِن كُنتُمۡ فَٰعِلِينَ ۝٦٨ قَالُوا۟ حَرِّقُوهُ وَٱنصُرُوۤا۟ ءَالِهَتَكُمۡ إِن كُنتُمۡ فَٰعِلِینَ ۝٦٨ قَالُوْا حَرِّقُوْهُ وَانْصُرُوْۤا اٰلِهَتَكُمْ اِنْ كُنْتُمْ فٰعِلِیْنَ ۟ قُلۡنَا يَٰنَارُ كُونِي بَرۡدٗا وَسَلَٰمًا عَلَىٰٓ إِبۡرَٰهِيمَ ۝٦٩ قُلۡنَا یَٰنَارُ كُونِی بَرۡدࣰا وَسَلَٰمًا عَلَىٰۤ إِبۡرَٰهِیمَ ۝٦٩ قُلْنَا یٰنَارُ كُوْنِیْ بَرْدًا وَّسَلٰمًا عَلٰۤی اِبْرٰهِیْمَ ۟ۙ وَأَرَادُواْ بِهِۦ كَيۡدٗا فَجَعَلۡنَٰهُمُ ٱلۡأَخۡسَرِينَ ۝٧٠ وَأَرَادُوا۟ بِهِۦ كَیۡدࣰا فَجَعَلۡنَٰهُمُ ٱلۡأَخۡسَرِینَ ۝٧٠ وَاَرَادُوْا بِهٖ كَیْدًا فَجَعَلْنٰهُمُ الْاَخْسَرِیْنَ ۟ۚ وَنَجَّيۡنَٰهُ وَلُوطًا إِلَى ٱلۡأَرۡضِ ٱلَّتِي بَٰرَكۡنَا فِيهَا لِلۡعَٰلَمِينَ ۝٧١ وَنَجَّیۡنَٰهُ وَلُوطًا إِلَى ٱلۡأَرۡضِ ٱلَّتِی بَٰرَكۡنَا فِیهَا لِلۡعَٰلَمِینَ ۝٧١ وَنَجَّیْنٰهُ وَلُوْطًا اِلَی الْاَرْضِ الَّتِیْ بٰرَكْنَا فِیْهَا لِلْعٰلَمِیْنَ ۟ وَوَهَبۡنَا لَهُۥٓ إِسۡحَٰقَ وَيَعۡقُوبَ نَافِلَةٗۖ وَكُلّٗا جَعَلۡنَا صَٰلِحِينَ ۝٧٢ وَوَهَبۡنَا لَهُۥۤ إِسۡحَٰقَ وَیَعۡقُوبَ نَافِلَةࣰۖ وَكُلࣰّا جَعَلۡنَا صَٰلِحِینَ ۝٧٢ وَوَهَبْنَا لَهٗۤ اِسْحٰقَ ؕ وَیَعْقُوْبَ نَافِلَةً ؕ وَكُلًّا جَعَلْنَا صٰلِحِیْنَ ۟ وَجَعَلۡنَٰهُمۡ أَئِمَّةٗ يَهۡدُونَ بِأَمۡرِنَا وَأَوۡحَيۡنَآ إِلَيۡهِمۡ فِعۡلَ ٱلۡخَيۡرَٰتِ وَإِقَامَ ٱلصَّلَوٰةِ وَإِيتَآءَ ٱلزَّكَوٰةِۖ وَكَانُواْ لَنَا عَٰبِدِينَ ۝٧٣ وَجَعَلۡنَٰهُمۡ أَئِمَّةࣰ یَهۡدُونَ بِأَمۡرِنَا وَأَوۡحَیۡنَاۤ إِلَیۡهِمۡ فِعۡلَ ٱلۡخَیۡرَٰتِ وَإِقَامَ ٱلصَّلَوٰةِ وَإِیتَاۤءَ ٱلزَّكَوٰةِۖ وَكَانُوا۟ لَنَا عَٰبِدِینَ ۝٧٣ وَجَعَلْنٰهُمْ اَىِٕمَّةً یَّهْدُوْنَ بِاَمْرِنَا وَاَوْحَیْنَاۤ اِلَیْهِمْ فِعْلَ الْخَیْرٰتِ وَاِقَامَ الصَّلٰوةِ وَاِیْتَآءَ الزَّكٰوةِ ۚ وَكَانُوْا لَنَا عٰبِدِیْنَ ۟ۚۙ وَلُوطًا ءَاتَيۡنَٰهُ حُكۡمٗا وَعِلۡمٗا وَنَجَّيۡنَٰهُ مِنَ ٱلۡقَرۡيَةِ ٱلَّتِي كَانَت تَّعۡمَلُ ٱلۡخَبَٰٓئِثَۚ إِنَّهُمۡ كَانُواْ قَوۡمَ سَوۡءٖ فَٰسِقِينَ ۝٧٤ وَلُوطًا ءَاتَیۡنَٰهُ حُكۡمࣰا وَعِلۡمࣰا وَنَجَّیۡنَٰهُ مِنَ ٱلۡقَرۡیَةِ ٱلَّتِی كَانَت تَّعۡمَلُ ٱلۡخَبَٰۤئِثَۚ إِنَّهُمۡ كَانُوا۟ قَوۡمَ سَوۡءࣲ فَٰسِقِینَ ۝٧٤ وَلُوْطًا اٰتَیْنٰهُ حُكْمًا وَّعِلْمًا وَّنَجَّیْنٰهُ مِنَ الْقَرْیَةِ الَّتِیْ كَانَتْ تَّعْمَلُ الْخَبٰٓىِٕثَ ؕ اِنَّهُمْ كَانُوْا قَوْمَ سَوْءٍ فٰسِقِیْنَ ۟ۙ وَأَدۡخَلۡنَٰهُ فِي رَحۡمَتِنَآۖ إِنَّهُۥ مِنَ ٱلصَّٰلِحِينَ ۝٧٥ وَأَدۡخَلۡنَٰهُ فِی رَحۡمَتِنَاۤۖ إِنَّهُۥ مِنَ ٱلصَّٰلِحِینَ ۝٧٥ وَاَدْخَلْنٰهُ فِیْ رَحْمَتِنَا ؕ اِنَّهٗ مِنَ الصّٰلِحِیْنَ ۟۠ وَنُوحًا إِذۡ نَادَىٰ مِن قَبۡلُ فَٱسۡتَجَبۡنَا لَهُۥ فَنَجَّيۡنَٰهُ وَأَهۡلَهُۥ مِنَ ٱلۡكَرۡبِ ٱلۡعَظِيمِ ۝٧٦ وَنُوحًا إِذۡ نَادَىٰ مِن قَبۡلُ فَٱسۡتَجَبۡنَا لَهُۥ فَنَجَّیۡنَٰهُ وَأَهۡلَهُۥ مِنَ ٱلۡكَرۡبِ ٱلۡعَظِیمِ ۝٧٦ وَنُوْحًا اِذْ نَادٰی مِنْ قَبْلُ فَاسْتَجَبْنَا لَهٗ فَنَجَّیْنٰهُ وَاَهْلَهٗ مِنَ الْكَرْبِ الْعَظِیْمِ ۟ۚ وَنَصَرۡنَٰهُ مِنَ ٱلۡقَوۡمِ ٱلَّذِينَ كَذَّبُواْ بِـَٔايَٰتِنَآۚ إِنَّهُمۡ كَانُواْ قَوۡمَ سَوۡءٖ فَأَغۡرَقۡنَٰهُمۡ أَجۡمَعِينَ ۝٧٧ وَنَصَرۡنَٰهُ مِنَ ٱلۡقَوۡمِ ٱلَّذِینَ كَذَّبُوا۟ بِءَایَٰتِنَاۤۚ إِنَّهُمۡ كَانُوا۟ قَوۡمَ سَوۡءࣲ فَأَغۡرَقۡنَٰهُمۡ أَجۡمَعِینَ ۝٧٧ وَنَصَرْنٰهُ مِنَ الْقَوْمِ الَّذِیْنَ كَذَّبُوْا بِاٰیٰتِنَا ؕ اِنَّهُمْ كَانُوْا قَوْمَ سَوْءٍ فَاَغْرَقْنٰهُمْ اَجْمَعِیْنَ ۟ وَدَاوُۥدَ وَسُلَيۡمَٰنَ إِذۡ يَحۡكُمَانِ فِي ٱلۡحَرۡثِ إِذۡ نَفَشَتۡ فِيهِ غَنَمُ ٱلۡقَوۡمِ وَكُنَّا لِحُكۡمِهِمۡ شَٰهِدِينَ ۝٧٨ وَدَاوُۥدَ وَسُلَیۡمَٰنَ إِذۡ یَحۡكُمَانِ فِی ٱلۡحَرۡثِ إِذۡ نَفَشَتۡ فِیهِ غَنَمُ ٱلۡقَوۡمِ وَكُنَّا لِحُكۡمِهِمۡ شَٰهِدِینَ ۝٧٨ وَدَاوٗدَ وَسُلَیْمٰنَ اِذْ یَحْكُمٰنِ فِی الْحَرْثِ اِذْ نَفَشَتْ فِیْهِ غَنَمُ الْقَوْمِ ۚ وَكُنَّا لِحُكْمِهِمْ شٰهِدِیْنَ ۟ۗۙ فَفَهَّمۡنَٰهَا سُلَيۡمَٰنَۚ وَكُلًّا ءَاتَيۡنَا حُكۡمٗا وَعِلۡمٗاۚ وَسَخَّرۡنَا مَعَ دَاوُۥدَ ٱلۡجِبَالَ يُسَبِّحۡنَ وَٱلطَّيۡرَۚ وَكُنَّا فَٰعِلِينَ ۝٧٩ فَفَهَّمۡنَٰهَا سُلَیۡمَٰنَۚ وَكُلًّا ءَاتَیۡنَا حُكۡمࣰا وَعِلۡمࣰاۚ وَسَخَّرۡنَا مَعَ دَاوُۥدَ ٱلۡجِبَالَ یُسَبِّحۡنَ وَٱلطَّیۡرَۚ وَكُنَّا فَٰعِلِینَ ۝٧٩ فَفَهَّمْنٰهَا سُلَیْمٰنَ ۚ وَكُلًّا اٰتَیْنَا حُكْمًا وَّعِلْمًا ؗ وَّسَخَّرْنَا مَعَ دَاوٗدَ الْجِبَالَ یُسَبِّحْنَ وَالطَّیْرَ ؕ وَكُنَّا فٰعِلِیْنَ ۟ وَعَلَّمۡنَٰهُ صَنۡعَةَ لَبُوسٖ لَّكُمۡ لِتُحۡصِنَكُم مِّنۢ بَأۡسِكُمۡۖ فَهَلۡ أَنتُمۡ شَٰكِرُونَ ۝٨٠ وَعَلَّمۡنَٰهُ صَنۡعَةَ لَبُوسࣲ لَّكُمۡ لِتُحۡصِنَكُم مِّنۢ بَأۡسِكُمۡۖ فَهَلۡ أَنتُمۡ شَٰكِرُونَ ۝٨٠ وَعَلَّمْنٰهُ صَنْعَةَ لَبُوْسٍ لَّكُمْ لِتُحْصِنَكُمْ مِّنْ بَاْسِكُمْ ۚ فَهَلْ اَنْتُمْ شٰكِرُوْنَ ۟ وَلِسُلَيۡمَٰنَ ٱلرِّيحَ عَاصِفَةٗ تَجۡرِي بِأَمۡرِهِۦٓ إِلَى ٱلۡأَرۡضِ ٱلَّتِي بَٰرَكۡنَا فِيهَاۚ وَكُنَّا بِكُلِّ شَيۡءٍ عَٰلِمِينَ ۝٨١ وَلِسُلَیۡمَٰنَ ٱلرِّیحَ عَاصِفَةࣰ تَجۡرِی بِأَمۡرِهِۦۤ إِلَى ٱلۡأَرۡضِ ٱلَّتِی بَٰرَكۡنَا فِیهَاۚ وَكُنَّا بِكُلِّ شَیۡءٍ عَٰلِمِینَ ۝٨١ وَلِسُلَیْمٰنَ الرِّیْحَ عَاصِفَةً تَجْرِیْ بِاَمْرِهٖۤ اِلَی الْاَرْضِ الَّتِیْ بٰرَكْنَا فِیْهَا ؕ وَكُنَّا بِكُلِّ شَیْءٍ عٰلِمِیْنَ ۟ وَمِنَ ٱلشَّيَٰطِينِ مَن يَغُوصُونَ لَهُۥ وَيَعۡمَلُونَ عَمَلٗا دُونَ ذَٰلِكَۖ وَكُنَّا لَهُمۡ حَٰفِظِينَ ۝٨٢ وَمِنَ ٱلشَّیَٰطِینِ مَن یَغُوصُونَ لَهُۥ وَیَعۡمَلُونَ عَمَلࣰا دُونَ ذَٰلِكَۖ وَكُنَّا لَهُمۡ حَٰفِظِینَ ۝٨٢ وَمِنَ الشَّیٰطِیْنِ مَنْ یَّغُوْصُوْنَ لَهٗ وَیَعْمَلُوْنَ عَمَلًا دُوْنَ ذٰلِكَ ۚ وَكُنَّا لَهُمْ حٰفِظِیْنَ ۟ۙ وَأَيُّوبَ إِذۡ نَادَىٰ رَبَّهُۥٓ أَنِّي مَسَّنِيَ ٱلضُّرُّ وَأَنتَ أَرۡحَمُ ٱلرَّٰحِمِينَ ۝٨٣ وَأَیُّوبَ إِذۡ نَادَىٰ رَبَّهُۥۤ أَنِّی مَسَّنِیَ ٱلضُّرُّ وَأَنتَ أَرۡحَمُ ٱلرَّٰحِمِینَ ۝٨٣ وَاَیُّوْبَ اِذْ نَادٰی رَبَّهٗۤ اَنِّیْ مَسَّنِیَ الضُّرُّ وَاَنْتَ اَرْحَمُ الرّٰحِمِیْنَ ۟ۚۖ فَٱسۡتَجَبۡنَا لَهُۥ فَكَشَفۡنَا مَا بِهِۦ مِن ضُرّٖۖ وَءَاتَيۡنَٰهُ أَهۡلَهُۥ وَمِثۡلَهُم مَّعَهُمۡ رَحۡمَةٗ مِّنۡ عِندِنَا وَذِكۡرَىٰ لِلۡعَٰبِدِينَ ۝٨٤ فَٱسۡتَجَبۡنَا لَهُۥ فَكَشَفۡنَا مَا بِهِۦ مِن ضُرࣲّۖ وَءَاتَیۡنَٰهُ أَهۡلَهُۥ وَمِثۡلَهُم مَّعَهُمۡ رَحۡمَةࣰ مِّنۡ عِندِنَا وَذِكۡرَىٰ لِلۡعَٰبِدِینَ ۝٨٤ فَاسْتَجَبْنَا لَهٗ فَكَشَفْنَا مَا بِهٖ مِنْ ضُرٍّ وَّاٰتَیْنٰهُ اَهْلَهٗ وَمِثْلَهُمْ مَّعَهُمْ رَحْمَةً مِّنْ عِنْدِنَا وَذِكْرٰی لِلْعٰبِدِیْنَ ۟ وَإِسۡمَٰعِيلَ وَإِدۡرِيسَ وَذَا ٱلۡكِفۡلِۖ كُلّٞ مِّنَ ٱلصَّٰبِرِينَ ۝٨٥ وَإِسۡمَٰعِیلَ وَإِدۡرِیسَ وَذَا ٱلۡكِفۡلِۖ كُلࣱّ مِّنَ ٱلصَّٰبِرِینَ ۝٨٥ وَاِسْمٰعِیْلَ وَاِدْرِیْسَ وَذَا الْكِفْلِ ؕ كُلٌّ مِّنَ الصّٰبِرِیْنَ ۟ۚۖ وَأَدۡخَلۡنَٰهُمۡ فِي رَحۡمَتِنَآۖ إِنَّهُم مِّنَ ٱلصَّٰلِحِينَ ۝٨٦ وَأَدۡخَلۡنَٰهُمۡ فِی رَحۡمَتِنَاۤۖ إِنَّهُم مِّنَ ٱلصَّٰلِحِینَ ۝٨٦ وَاَدْخَلْنٰهُمْ فِیْ رَحْمَتِنَا ؕ اِنَّهُمْ مِّنَ الصّٰلِحِیْنَ ۟ وَذَا ٱلنُّونِ إِذ ذَّهَبَ مُغَٰضِبٗا فَظَنَّ أَن لَّن نَّقۡدِرَ عَلَيۡهِ فَنَادَىٰ فِي ٱلظُّلُمَٰتِ أَن لَّآ إِلَٰهَ إِلَّآ أَنتَ سُبۡحَٰنَكَ إِنِّي كُنتُ مِنَ ٱلظَّٰلِمِينَ ۝٨٧ وَذَا ٱلنُّونِ إِذ ذَّهَبَ مُغَٰضِبࣰا فَظَنَّ أَن لَّن نَّقۡدِرَ عَلَیۡهِ فَنَادَىٰ فِی ٱلظُّلُمَٰتِ أَن لَّاۤ إِلَٰهَ إِلَّاۤ أَنتَ سُبۡحَٰنَكَ إِنِّی كُنتُ مِنَ ٱلظَّٰلِمِینَ ۝٨٧ وَذَا النُّوْنِ اِذْ ذَّهَبَ مُغَاضِبًا فَظَنَّ اَنْ لَّنْ نَّقْدِرَ عَلَیْهِ فَنَادٰی فِی الظُّلُمٰتِ اَنْ لَّاۤ اِلٰهَ اِلَّاۤ اَنْتَ سُبْحٰنَكَ ۖۗ اِنِّیْ كُنْتُ مِنَ الظّٰلِمِیْنَ ۟ۚۖ فَٱسۡتَجَبۡنَا لَهُۥ وَنَجَّيۡنَٰهُ مِنَ ٱلۡغَمِّۚ وَكَذَٰلِكَ نُـۨجِي ٱلۡمُؤۡمِنِينَ ۝٨٨ فَٱسۡتَجَبۡنَا لَهُۥ وَنَجَّیۡنَٰهُ مِنَ ٱلۡغَمِّۚ وَكَذَٰلِكَ نُۨجِی ٱلۡمُؤۡمِنِینَ ۝٨٨ فَاسْتَجَبْنَا لَهٗ ۙ وَنَجَّیْنٰهُ مِنَ الْغَمِّ ؕ وَكَذٰلِكَ نُـجِی الْمُؤْمِنِیْنَ ۟ وَزَكَرِيَّآ إِذۡ نَادَىٰ رَبَّهُۥ رَبِّ لَا تَذَرۡنِي فَرۡدٗا وَأَنتَ خَيۡرُ ٱلۡوَٰرِثِينَ ۝٨٩ وَزَكَرِیَّاۤ إِذۡ نَادَىٰ رَبَّهُۥ رَبِّ لَا تَذَرۡنِی فَرۡدࣰا وَأَنتَ خَیۡرُ ٱلۡوَٰرِثِینَ ۝٨٩ وَزَكَرِیَّاۤ اِذْ نَادٰی رَبَّهٗ رَبِّ لَا تَذَرْنِیْ فَرْدًا وَّاَنْتَ خَیْرُ الْوٰرِثِیْنَ ۟ۚۖ فَٱسۡتَجَبۡنَا لَهُۥ وَوَهَبۡنَا لَهُۥ يَحۡيَىٰ وَأَصۡلَحۡنَا لَهُۥ زَوۡجَهُۥٓۚ إِنَّهُمۡ كَانُواْ يُسَٰرِعُونَ فِي ٱلۡخَيۡرَٰتِ وَيَدۡعُونَنَا رَغَبٗا وَرَهَبٗاۖ وَكَانُواْ لَنَا خَٰشِعِينَ ۝٩٠ فَٱسۡتَجَبۡنَا لَهُۥ وَوَهَبۡنَا لَهُۥ یَحۡیَىٰ وَأَصۡلَحۡنَا لَهُۥ زَوۡجَهُۥۤۚ إِنَّهُمۡ كَانُوا۟ یُسَٰرِعُونَ فِی ٱلۡخَیۡرَٰتِ وَیَدۡعُونَنَا رَغَبࣰا وَرَهَبࣰاۖ وَكَانُوا۟ لَنَا خَٰشِعِینَ ۝٩٠ فَاسْتَجَبْنَا لَهٗ ؗ وَوَهَبْنَا لَهٗ یَحْیٰی وَاَصْلَحْنَا لَهٗ زَوْجَهٗ ؕ اِنَّهُمْ كَانُوْا یُسٰرِعُوْنَ فِی الْخَیْرٰتِ وَیَدْعُوْنَنَا رَغَبًا وَّرَهَبًا ؕ وَكَانُوْا لَنَا خٰشِعِیْنَ ۟ وَٱلَّتِيٓ أَحۡصَنَتۡ فَرۡجَهَا فَنَفَخۡنَا فِيهَا مِن رُّوحِنَا وَجَعَلۡنَٰهَا وَٱبۡنَهَآ ءَايَةٗ لِّلۡعَٰلَمِينَ ۝٩١ وَٱلَّتِیۤ أَحۡصَنَتۡ فَرۡجَهَا فَنَفَخۡنَا فِیهَا مِن رُّوحِنَا وَجَعَلۡنَٰهَا وَٱبۡنَهَاۤ ءَایَةࣰ لِّلۡعَٰلَمِینَ ۝٩١ وَالَّتِیْۤ اَحْصَنَتْ فَرْجَهَا فَنَفَخْنَا فِیْهَا مِنْ رُّوْحِنَا وَجَعَلْنٰهَا وَابْنَهَاۤ اٰیَةً لِّلْعٰلَمِیْنَ ۟ إِنَّ هَٰذِهِۦٓ أُمَّتُكُمۡ أُمَّةٗ وَٰحِدَةٗ وَأَنَا۠ رَبُّكُمۡ فَٱعۡبُدُونِ ۝٩٢ إِنَّ هَٰذِهِۦۤ أُمَّتُكُمۡ أُمَّةࣰ وَٰحِدَةࣰ وَأَنَا۠ رَبُّكُمۡ فَٱعۡبُدُونِ ۝٩٢ اِنَّ هٰذِهٖۤ اُمَّتُكُمْ اُمَّةً وَّاحِدَةً ۖؗ وَّاَنَا رَبُّكُمْ فَاعْبُدُوْنِ ۟ وَتَقَطَّعُوٓاْ أَمۡرَهُم بَيۡنَهُمۡۖ كُلٌّ إِلَيۡنَا رَٰجِعُونَ ۝٩٣ وَتَقَطَّعُوۤا۟ أَمۡرَهُم بَیۡنَهُمۡۖ كُلٌّ إِلَیۡنَا رَٰجِعُونَ ۝٩٣ وَتَقَطَّعُوْۤا اَمْرَهُمْ بَیْنَهُمْ ؕ كُلٌّ اِلَیْنَا رٰجِعُوْنَ ۟۠ فَمَن يَعۡمَلۡ مِنَ ٱلصَّٰلِحَٰتِ وَهُوَ مُؤۡمِنٞ فَلَا كُفۡرَانَ لِسَعۡيِهِۦ وَإِنَّا لَهُۥ كَٰتِبُونَ ۝٩٤ فَمَن یَعۡمَلۡ مِنَ ٱلصَّٰلِحَٰتِ وَهُوَ مُؤۡمِنࣱ فَلَا كُفۡرَانَ لِسَعۡیِهِۦ وَإِنَّا لَهُۥ كَٰتِبُونَ ۝٩٤ فَمَنْ یَّعْمَلْ مِنَ الصّٰلِحٰتِ وَهُوَ مُؤْمِنٌ فَلَا كُفْرَانَ لِسَعْیِهٖ ۚ وَاِنَّا لَهٗ كٰتِبُوْنَ ۟ وَحَرَٰمٌ عَلَىٰ قَرۡيَةٍ أَهۡلَكۡنَٰهَآ أَنَّهُمۡ لَا يَرۡجِعُونَ ۝٩٥ وَحَرَٰمٌ عَلَىٰ قَرۡیَةٍ أَهۡلَكۡنَٰهَاۤ أَنَّهُمۡ لَا یَرۡجِعُونَ ۝٩٥ وَحَرٰمٌ عَلٰی قَرْیَةٍ اَهْلَكْنٰهَاۤ اَنَّهُمْ لَا یَرْجِعُوْنَ ۟ حَتَّىٰٓ إِذَا فُتِحَتۡ يَأۡجُوجُ وَمَأۡجُوجُ وَهُم مِّن كُلِّ حَدَبٖ يَنسِلُونَ ۝٩٦ حَتَّىٰۤ إِذَا فُتِحَتۡ یَأۡجُوجُ وَمَأۡجُوجُ وَهُم مِّن كُلِّ حَدَبࣲ یَنسِلُونَ ۝٩٦ حَتّٰۤی اِذَا فُتِحَتْ یَاْجُوْجُ وَمَاْجُوْجُ وَهُمْ مِّنْ كُلِّ حَدَبٍ یَّنْسِلُوْنَ ۟ وَٱقۡتَرَبَ ٱلۡوَعۡدُ ٱلۡحَقُّ فَإِذَا هِيَ شَٰخِصَةٌ أَبۡصَٰرُ ٱلَّذِينَ كَفَرُواْ يَٰوَيۡلَنَا قَدۡ كُنَّا فِي غَفۡلَةٖ مِّنۡ هَٰذَا بَلۡ كُنَّا ظَٰلِمِينَ ۝٩٧ وَٱقۡتَرَبَ ٱلۡوَعۡدُ ٱلۡحَقُّ فَإِذَا هِیَ شَٰخِصَةٌ أَبۡصَٰرُ ٱلَّذِینَ كَفَرُوا۟ یَٰوَیۡلَنَا قَدۡ كُنَّا فِی غَفۡلَةࣲ مِّنۡ هَٰذَا بَلۡ كُنَّا ظَٰلِمِینَ ۝٩٧ وَاقْتَرَبَ الْوَعْدُ الْحَقُّ فَاِذَا هِیَ شَاخِصَةٌ اَبْصَارُ الَّذِیْنَ كَفَرُوْا ؕ یٰوَیْلَنَا قَدْ كُنَّا فِیْ غَفْلَةٍ مِّنْ هٰذَا بَلْ كُنَّا ظٰلِمِیْنَ ۟ إِنَّكُمۡ وَمَا تَعۡبُدُونَ مِن دُونِ ٱللَّهِ حَصَبُ جَهَنَّمَ أَنتُمۡ لَهَا وَٰرِدُونَ ۝٩٨ إِنَّكُمۡ وَمَا تَعۡبُدُونَ مِن دُونِ ٱللَّهِ حَصَبُ جَهَنَّمَ أَنتُمۡ لَهَا وَٰرِدُونَ ۝٩٨ اِنَّكُمْ وَمَا تَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ حَصَبُ جَهَنَّمَ ؕ اَنْتُمْ لَهَا وٰرِدُوْنَ ۟ لَوۡ كَانَ هَٰٓؤُلَآءِ ءَالِهَةٗ مَّا وَرَدُوهَاۖ وَكُلّٞ فِيهَا خَٰلِدُونَ ۝٩٩ لَوۡ كَانَ هَٰۤؤُلَاۤءِ ءَالِهَةࣰ مَّا وَرَدُوهَاۖ وَكُلࣱّ فِیهَا خَٰلِدُونَ ۝٩٩ لَوْ كَانَ هٰۤؤُلَآءِ اٰلِهَةً مَّا وَرَدُوْهَا ؕ وَكُلٌّ فِیْهَا خٰلِدُوْنَ ۟ لَهُمۡ فِيهَا زَفِيرٞ وَهُمۡ فِيهَا لَا يَسۡمَعُونَ ۝١٠٠ لَهُمۡ فِیهَا زَفِیرࣱ وَهُمۡ فِیهَا لَا یَسۡمَعُونَ ۝١٠٠ لَهُمْ فِیْهَا زَفِیْرٌ وَّهُمْ فِیْهَا لَا یَسْمَعُوْنَ ۟ إِنَّ ٱلَّذِينَ سَبَقَتۡ لَهُم مِّنَّا ٱلۡحُسۡنَىٰٓ أُوْلَٰٓئِكَ عَنۡهَا مُبۡعَدُونَ ۝١٠١ إِنَّ ٱلَّذِینَ سَبَقَتۡ لَهُم مِّنَّا ٱلۡحُسۡنَىٰۤ أُو۟لَٰۤئِكَ عَنۡهَا مُبۡعَدُونَ ۝١٠١ اِنَّ الَّذِیْنَ سَبَقَتْ لَهُمْ مِّنَّا الْحُسْنٰۤی ۙ اُولٰٓىِٕكَ عَنْهَا مُبْعَدُوْنَ ۟ۙ لَا يَسۡمَعُونَ حَسِيسَهَاۖ وَهُمۡ فِي مَا ٱشۡتَهَتۡ أَنفُسُهُمۡ خَٰلِدُونَ ۝١٠٢ لَا یَسۡمَعُونَ حَسِیسَهَاۖ وَهُمۡ فِی مَا ٱشۡتَهَتۡ أَنفُسُهُمۡ خَٰلِدُونَ ۝١٠٢ لَا یَسْمَعُوْنَ حَسِیْسَهَا ۚ وَهُمْ فِیْ مَا اشْتَهَتْ اَنْفُسُهُمْ خٰلِدُوْنَ ۟ۚ لَا يَحۡزُنُهُمُ ٱلۡفَزَعُ ٱلۡأَكۡبَرُ وَتَتَلَقَّىٰهُمُ ٱلۡمَلَٰٓئِكَةُ هَٰذَا يَوۡمُكُمُ ٱلَّذِي كُنتُمۡ تُوعَدُونَ ۝١٠٣ لَا یَحۡزُنُهُمُ ٱلۡفَزَعُ ٱلۡأَكۡبَرُ وَتَتَلَقَّىٰهُمُ ٱلۡمَلَٰۤئِكَةُ هَٰذَا یَوۡمُكُمُ ٱلَّذِی كُنتُمۡ تُوعَدُونَ ۝١٠٣ لَا یَحْزُنُهُمُ الْفَزَعُ الْاَكْبَرُ وَتَتَلَقّٰىهُمُ الْمَلٰٓىِٕكَةُ ؕ هٰذَا یَوْمُكُمُ الَّذِیْ كُنْتُمْ تُوْعَدُوْنَ ۟ يَوۡمَ نَطۡوِي ٱلسَّمَآءَ كَطَيِّ ٱلسِّجِلِّ لِلۡكُتُبِۚ كَمَا بَدَأۡنَآ أَوَّلَ خَلۡقٖ نُّعِيدُهُۥۚ وَعۡدًا عَلَيۡنَآۚ إِنَّا كُنَّا فَٰعِلِينَ ۝١٠٤ یَوۡمَ نَطۡوِی ٱلسَّمَاۤءَ كَطَیِّ ٱلسِّجِلِّ لِلۡكُتُبِۚ كَمَا بَدَأۡنَاۤ أَوَّلَ خَلۡقࣲ نُّعِیدُهُۥۚ وَعۡدًا عَلَیۡنَاۤۚ إِنَّا كُنَّا فَٰعِلِینَ ۝١٠٤ یَوْمَ نَطْوِی السَّمَآءَ كَطَیِّ السِّجِلِّ لِلْكُتُبِ ؕ كَمَا بَدَاْنَاۤ اَوَّلَ خَلْقٍ نُّعِیْدُهٗ ؕ وَعْدًا عَلَیْنَا ؕ اِنَّا كُنَّا فٰعِلِیْنَ ۟ وَلَقَدۡ كَتَبۡنَا فِي ٱلزَّبُورِ مِنۢ بَعۡدِ ٱلذِّكۡرِ أَنَّ ٱلۡأَرۡضَ يَرِثُهَا عِبَادِيَ ٱلصَّٰلِحُونَ ۝١٠٥ وَلَقَدۡ كَتَبۡنَا فِی ٱلزَّبُورِ مِنۢ بَعۡدِ ٱلذِّكۡرِ أَنَّ ٱلۡأَرۡضَ یَرِثُهَا عِبَادِیَ ٱلصَّٰلِحُونَ ۝١٠٥ وَلَقَدْ كَتَبْنَا فِی الزَّبُوْرِ مِنْ بَعْدِ الذِّكْرِ اَنَّ الْاَرْضَ یَرِثُهَا عِبَادِیَ الصّٰلِحُوْنَ ۟ إِنَّ فِي هَٰذَا لَبَلَٰغٗا لِّقَوۡمٍ عَٰبِدِينَ ۝١٠٦ إِنَّ فِی هَٰذَا لَبَلَٰغࣰا لِّقَوۡمٍ عَٰبِدِینَ ۝١٠٦ اِنَّ فِیْ هٰذَا لَبَلٰغًا لِّقَوْمٍ عٰبِدِیْنَ ۟ؕ وَمَآ أَرۡسَلۡنَٰكَ إِلَّا رَحۡمَةٗ لِّلۡعَٰلَمِينَ ۝١٠٧ وَمَاۤ أَرۡسَلۡنَٰكَ إِلَّا رَحۡمَةࣰ لِّلۡعَٰلَمِینَ ۝١٠٧ وَمَاۤ اَرْسَلْنٰكَ اِلَّا رَحْمَةً لِّلْعٰلَمِیْنَ ۟ قُلۡ إِنَّمَا يُوحَىٰٓ إِلَيَّ أَنَّمَآ إِلَٰهُكُمۡ إِلَٰهٞ وَٰحِدٞۖ فَهَلۡ أَنتُم مُّسۡلِمُونَ ۝١٠٨ قُلۡ إِنَّمَا یُوحَىٰۤ إِلَیَّ أَنَّمَاۤ إِلَٰهُكُمۡ إِلَٰهࣱ وَٰحِدࣱۖ فَهَلۡ أَنتُم مُّسۡلِمُونَ ۝١٠٨ قُلْ اِنَّمَا یُوْحٰۤی اِلَیَّ اَنَّمَاۤ اِلٰهُكُمْ اِلٰهٌ وَّاحِدٌ ۚ فَهَلْ اَنْتُمْ مُّسْلِمُوْنَ ۟ فَإِن تَوَلَّوۡاْ فَقُلۡ ءَاذَنتُكُمۡ عَلَىٰ سَوَآءٖۖ وَإِنۡ أَدۡرِيٓ أَقَرِيبٌ أَم بَعِيدٞ مَّا تُوعَدُونَ ۝١٠٩ فَإِن تَوَلَّوۡا۟ فَقُلۡ ءَاذَنتُكُمۡ عَلَىٰ سَوَاۤءࣲۖ وَإِنۡ أَدۡرِیۤ أَقَرِیبٌ أَم بَعِیدࣱ مَّا تُوعَدُونَ ۝١٠٩ فَاِنْ تَوَلَّوْا فَقُلْ اٰذَنْتُكُمْ عَلٰی سَوَآءٍ ؕ وَاِنْ اَدْرِیْۤ اَقَرِیْبٌ اَمْ بَعِیْدٌ مَّا تُوْعَدُوْنَ ۟ إِنَّهُۥ يَعۡلَمُ ٱلۡجَهۡرَ مِنَ ٱلۡقَوۡلِ وَيَعۡلَمُ مَا تَكۡتُمُونَ ۝١١٠ إِنَّهُۥ یَعۡلَمُ ٱلۡجَهۡرَ مِنَ ٱلۡقَوۡلِ وَیَعۡلَمُ مَا تَكۡتُمُونَ ۝١١٠ اِنَّهٗ یَعْلَمُ الْجَهْرَ مِنَ الْقَوْلِ وَیَعْلَمُ مَا تَكْتُمُوْنَ ۟ وَإِنۡ أَدۡرِي لَعَلَّهُۥ فِتۡنَةٞ لَّكُمۡ وَمَتَٰعٌ إِلَىٰ حِينٖ ۝١١١ وَإِنۡ أَدۡرِی لَعَلَّهُۥ فِتۡنَةࣱ لَّكُمۡ وَمَتَٰعٌ إِلَىٰ حِینࣲ ۝١١١ وَاِنْ اَدْرِیْ لَعَلَّهٗ فِتْنَةٌ لَّكُمْ وَمَتَاعٌ اِلٰی حِیْنٍ ۟ قَٰلَ رَبِّ ٱحۡكُم بِٱلۡحَقِّۗ وَرَبُّنَا ٱلرَّحۡمَٰنُ ٱلۡمُسۡتَعَانُ عَلَىٰ مَا تَصِفُونَ ۝١١٢ قَٰلَ رَبِّ ٱحۡكُم بِٱلۡحَقِّۗ وَرَبُّنَا ٱلرَّحۡمَٰنُ ٱلۡمُسۡتَعَانُ عَلَىٰ مَا تَصِفُونَ ۝١١٢ قٰلَ رَبِّ احْكُمْ بِالْحَقِّ ؕ وَرَبُّنَا الرَّحْمٰنُ الْمُسْتَعَانُ عَلٰی مَا تَصِفُوْنَ ۟۠
