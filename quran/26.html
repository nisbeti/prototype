بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

طسٓمٓ ۝١ طسۤمۤ ۝١ طٰسٓمّٓ ۟ تِلۡكَ ءَايَٰتُ ٱلۡكِتَٰبِ ٱلۡمُبِينِ ۝٢ تِلۡكَ ءَایَٰتُ ٱلۡكِتَٰبِ ٱلۡمُبِینِ ۝٢ تِلْكَ اٰیٰتُ الْكِتٰبِ الْمُبِیْنِ ۟ لَعَلَّكَ بَٰخِعٞ نَّفۡسَكَ أَلَّا يَكُونُواْ مُؤۡمِنِينَ ۝٣ لَعَلَّكَ بَٰخِعࣱ نَّفۡسَكَ أَلَّا یَكُونُوا۟ مُؤۡمِنِینَ ۝٣ لَعَلَّكَ بَاخِعٌ نَّفْسَكَ اَلَّا یَكُوْنُوْا مُؤْمِنِیْنَ ۟ إِن نَّشَأۡ نُنَزِّلۡ عَلَيۡهِم مِّنَ ٱلسَّمَآءِ ءَايَةٗ فَظَلَّتۡ أَعۡنَٰقُهُمۡ لَهَا خَٰضِعِينَ ۝٤ إِن نَّشَأۡ نُنَزِّلۡ عَلَیۡهِم مِّنَ ٱلسَّمَاۤءِ ءَایَةࣰ فَظَلَّتۡ أَعۡنَٰقُهُمۡ لَهَا خَٰضِعِینَ ۝٤ اِنْ نَّشَاْ نُنَزِّلْ عَلَیْهِمْ مِّنَ السَّمَآءِ اٰیَةً فَظَلَّتْ اَعْنَاقُهُمْ لَهَا خٰضِعِیْنَ ۟ وَمَا يَأۡتِيهِم مِّن ذِكۡرٖ مِّنَ ٱلرَّحۡمَٰنِ مُحۡدَثٍ إِلَّا كَانُواْ عَنۡهُ مُعۡرِضِينَ ۝٥ وَمَا یَأۡتِیهِم مِّن ذِكۡرࣲ مِّنَ ٱلرَّحۡمَٰنِ مُحۡدَثٍ إِلَّا كَانُوا۟ عَنۡهُ مُعۡرِضِینَ ۝٥ وَمَا یَاْتِیْهِمْ مِّنْ ذِكْرٍ مِّنَ الرَّحْمٰنِ مُحْدَثٍ اِلَّا كَانُوْا عَنْهُ مُعْرِضِیْنَ ۟ فَقَدۡ كَذَّبُواْ فَسَيَأۡتِيهِمۡ أَنۢبَٰٓؤُاْ مَا كَانُواْ بِهِۦ يَسۡتَهۡزِءُونَ ۝٦ فَقَدۡ كَذَّبُوا۟ فَسَیَأۡتِیهِمۡ أَنۢبَٰۤؤُا۟ مَا كَانُوا۟ بِهِۦ یَسۡتَهۡزِءُونَ ۝٦ فَقَدْ كَذَّبُوْا فَسَیَاْتِیْهِمْ اَنْۢبٰٓؤُا مَا كَانُوْا بِهٖ یَسْتَهْزِءُوْنَ ۟ أَوَلَمۡ يَرَوۡاْ إِلَى ٱلۡأَرۡضِ كَمۡ أَنۢبَتۡنَا فِيهَا مِن كُلِّ زَوۡجٖ كَرِيمٍ ۝٧ أَوَلَمۡ یَرَوۡا۟ إِلَى ٱلۡأَرۡضِ كَمۡ أَنۢبَتۡنَا فِیهَا مِن كُلِّ زَوۡجࣲ كَرِیمٍ ۝٧ اَوَلَمْ یَرَوْا اِلَی الْاَرْضِ كَمْ اَنْۢبَتْنَا فِیْهَا مِنْ كُلِّ زَوْجٍ كَرِیْمٍ ۟ إِنَّ فِي ذَٰلِكَ لَأٓيَةٗۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِينَ ۝٨ إِنَّ فِی ذَٰلِكَ لَءَایَةࣰۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِینَ ۝٨ اِنَّ فِیْ ذٰلِكَ لَاٰیَةً ؕ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِیْنَ ۟ وَإِنَّ رَبَّكَ لَهُوَ ٱلۡعَزِيزُ ٱلرَّحِيمُ ۝٩ وَإِنَّ رَبَّكَ لَهُوَ ٱلۡعَزِیزُ ٱلرَّحِیمُ ۝٩ وَاِنَّ رَبَّكَ لَهُوَ الْعَزِیْزُ الرَّحِیْمُ ۟۠ وَإِذۡ نَادَىٰ رَبُّكَ مُوسَىٰٓ أَنِ ٱئۡتِ ٱلۡقَوۡمَ ٱلظَّٰلِمِينَ ۝١٠ وَإِذۡ نَادَىٰ رَبُّكَ مُوسَىٰۤ أَنِ ٱئۡتِ ٱلۡقَوۡمَ ٱلظَّٰلِمِینَ ۝١٠ وَاِذْ نَادٰی رَبُّكَ مُوْسٰۤی اَنِ ائْتِ الْقَوْمَ الظّٰلِمِیْنَ ۟ۙ قَوۡمَ فِرۡعَوۡنَۚ أَلَا يَتَّقُونَ ۝١١ قَوۡمَ فِرۡعَوۡنَۚ أَلَا یَتَّقُونَ ۝١١ قَوْمَ فِرْعَوْنَ ؕ اَلَا یَتَّقُوْنَ ۟ قَالَ رَبِّ إِنِّيٓ أَخَافُ أَن يُكَذِّبُونِ ۝١٢ قَالَ رَبِّ إِنِّیۤ أَخَافُ أَن یُكَذِّبُونِ ۝١٢ قَالَ رَبِّ اِنِّیْۤ اَخَافُ اَنْ یُّكَذِّبُوْنِ ۟ؕ وَيَضِيقُ صَدۡرِي وَلَا يَنطَلِقُ لِسَانِي فَأَرۡسِلۡ إِلَىٰ هَٰرُونَ ۝١٣ وَیَضِیقُ صَدۡرِی وَلَا یَنطَلِقُ لِسَانِی فَأَرۡسِلۡ إِلَىٰ هَٰرُونَ ۝١٣ وَیَضِیْقُ صَدْرِیْ وَلَا یَنْطَلِقُ لِسَانِیْ فَاَرْسِلْ اِلٰی هٰرُوْنَ ۟ وَلَهُمۡ عَلَيَّ ذَنۢبٞ فَأَخَافُ أَن يَقۡتُلُونِ ۝١٤ وَلَهُمۡ عَلَیَّ ذَنۢبࣱ فَأَخَافُ أَن یَقۡتُلُونِ ۝١٤ وَلَهُمْ عَلَیَّ ذَنْۢبٌ فَاَخَافُ اَنْ یَّقْتُلُوْنِ ۟ۚۖ قَالَ كَلَّاۖ فَٱذۡهَبَا بِـَٔايَٰتِنَآۖ إِنَّا مَعَكُم مُّسۡتَمِعُونَ ۝١٥ قَالَ كَلَّاۖ فَٱذۡهَبَا بِءَایَٰتِنَاۤۖ إِنَّا مَعَكُم مُّسۡتَمِعُونَ ۝١٥ قَالَ كَلَّا ۚ فَاذْهَبَا بِاٰیٰتِنَاۤ اِنَّا مَعَكُمْ مُّسْتَمِعُوْنَ ۟ فَأۡتِيَا فِرۡعَوۡنَ فَقُولَآ إِنَّا رَسُولُ رَبِّ ٱلۡعَٰلَمِينَ ۝١٦ فَأۡتِیَا فِرۡعَوۡنَ فَقُولَاۤ إِنَّا رَسُولُ رَبِّ ٱلۡعَٰلَمِینَ ۝١٦ فَاْتِیَا فِرْعَوْنَ فَقُوْلَاۤ اِنَّا رَسُوْلُ رَبِّ الْعٰلَمِیْنَ ۟ۙ أَنۡ أَرۡسِلۡ مَعَنَا بَنِيٓ إِسۡرَٰٓءِيلَ ۝١٧ أَنۡ أَرۡسِلۡ مَعَنَا بَنِیۤ إِسۡرَٰۤءِیلَ ۝١٧ اَنْ اَرْسِلْ مَعَنَا بَنِیْۤ اِسْرَآءِیْلَ ۟ؕ قَالَ أَلَمۡ نُرَبِّكَ فِينَا وَلِيدٗا وَلَبِثۡتَ فِينَا مِنۡ عُمُرِكَ سِنِينَ ۝١٨ قَالَ أَلَمۡ نُرَبِّكَ فِینَا وَلِیدࣰا وَلَبِثۡتَ فِینَا مِنۡ عُمُرِكَ سِنِینَ ۝١٨ قَالَ اَلَمْ نُرَبِّكَ فِیْنَا وَلِیْدًا وَّلَبِثْتَ فِیْنَا مِنْ عُمُرِكَ سِنِیْنَ ۟ۙ وَفَعَلۡتَ فَعۡلَتَكَ ٱلَّتِي فَعَلۡتَ وَأَنتَ مِنَ ٱلۡكَٰفِرِينَ ۝١٩ وَفَعَلۡتَ فَعۡلَتَكَ ٱلَّتِی فَعَلۡتَ وَأَنتَ مِنَ ٱلۡكَٰفِرِینَ ۝١٩ وَفَعَلْتَ فَعْلَتَكَ الَّتِیْ فَعَلْتَ وَاَنْتَ مِنَ الْكٰفِرِیْنَ ۟ قَالَ فَعَلۡتُهَآ إِذٗا وَأَنَا۠ مِنَ ٱلضَّآلِّينَ ۝٢٠ قَالَ فَعَلۡتُهَاۤ إِذࣰا وَأَنَا۠ مِنَ ٱلضَّاۤلِّینَ ۝٢٠ قَالَ فَعَلْتُهَاۤ اِذًا وَّاَنَا مِنَ الضَّآلِّیْنَ ۟ؕ فَفَرَرۡتُ مِنكُمۡ لَمَّا خِفۡتُكُمۡ فَوَهَبَ لِي رَبِّي حُكۡمٗا وَجَعَلَنِي مِنَ ٱلۡمُرۡسَلِينَ ۝٢١ فَفَرَرۡتُ مِنكُمۡ لَمَّا خِفۡتُكُمۡ فَوَهَبَ لِی رَبِّی حُكۡمࣰا وَجَعَلَنِی مِنَ ٱلۡمُرۡسَلِینَ ۝٢١ فَفَرَرْتُ مِنْكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِیْ رَبِّیْ حُكْمًا وَّجَعَلَنِیْ مِنَ الْمُرْسَلِیْنَ ۟ وَتِلۡكَ نِعۡمَةٞ تَمُنُّهَا عَلَيَّ أَنۡ عَبَّدتَّ بَنِيٓ إِسۡرَٰٓءِيلَ ۝٢٢ وَتِلۡكَ نِعۡمَةࣱ تَمُنُّهَا عَلَیَّ أَنۡ عَبَّدتَّ بَنِیۤ إِسۡرَٰۤءِیلَ ۝٢٢ وَتِلْكَ نِعْمَةٌ تَمُنُّهَا عَلَیَّ اَنْ عَبَّدْتَّ بَنِیْۤ اِسْرَآءِیْلَ ۟ؕ قَالَ فِرۡعَوۡنُ وَمَا رَبُّ ٱلۡعَٰلَمِينَ ۝٢٣ قَالَ فِرۡعَوۡنُ وَمَا رَبُّ ٱلۡعَٰلَمِینَ ۝٢٣ قَالَ فِرْعَوْنُ وَمَا رَبُّ الْعٰلَمِیْنَ ۟ؕ قَالَ رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَآۖ إِن كُنتُم مُّوقِنِينَ ۝٢٤ قَالَ رَبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَیۡنَهُمَاۤۖ إِن كُنتُم مُّوقِنِینَ ۝٢٤ قَالَ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ وَمَا بَیْنَهُمَا ؕ اِنْ كُنْتُمْ مُّوْقِنِیْنَ ۟ قَالَ لِمَنۡ حَوۡلَهُۥٓ أَلَا تَسۡتَمِعُونَ ۝٢٥ قَالَ لِمَنۡ حَوۡلَهُۥۤ أَلَا تَسۡتَمِعُونَ ۝٢٥ قَالَ لِمَنْ حَوْلَهٗۤ اَلَا تَسْتَمِعُوْنَ ۟ قَالَ رَبُّكُمۡ وَرَبُّ ءَابَآئِكُمُ ٱلۡأَوَّلِينَ ۝٢٦ قَالَ رَبُّكُمۡ وَرَبُّ ءَابَاۤئِكُمُ ٱلۡأَوَّلِینَ ۝٢٦ قَالَ رَبُّكُمْ وَرَبُّ اٰبَآىِٕكُمُ الْاَوَّلِیْنَ ۟ قَالَ إِنَّ رَسُولَكُمُ ٱلَّذِيٓ أُرۡسِلَ إِلَيۡكُمۡ لَمَجۡنُونٞ ۝٢٧ قَالَ إِنَّ رَسُولَكُمُ ٱلَّذِیۤ أُرۡسِلَ إِلَیۡكُمۡ لَمَجۡنُونࣱ ۝٢٧ قَالَ اِنَّ رَسُوْلَكُمُ الَّذِیْۤ اُرْسِلَ اِلَیْكُمْ لَمَجْنُوْنٌ ۟ قَالَ رَبُّ ٱلۡمَشۡرِقِ وَٱلۡمَغۡرِبِ وَمَا بَيۡنَهُمَآۖ إِن كُنتُمۡ تَعۡقِلُونَ ۝٢٨ قَالَ رَبُّ ٱلۡمَشۡرِقِ وَٱلۡمَغۡرِبِ وَمَا بَیۡنَهُمَاۤۖ إِن كُنتُمۡ تَعۡقِلُونَ ۝٢٨ قَالَ رَبُّ الْمَشْرِقِ وَالْمَغْرِبِ وَمَا بَیْنَهُمَا ؕ اِنْ كُنْتُمْ تَعْقِلُوْنَ ۟ قَالَ لَئِنِ ٱتَّخَذۡتَ إِلَٰهًا غَيۡرِي لَأَجۡعَلَنَّكَ مِنَ ٱلۡمَسۡجُونِينَ ۝٢٩ قَالَ لَئِنِ ٱتَّخَذۡتَ إِلَٰهًا غَیۡرِی لَأَجۡعَلَنَّكَ مِنَ ٱلۡمَسۡجُونِینَ ۝٢٩ قَالَ لَىِٕنِ اتَّخَذْتَ اِلٰهًا غَیْرِیْ لَاَجْعَلَنَّكَ مِنَ الْمَسْجُوْنِیْنَ ۟ قَالَ أَوَلَوۡ جِئۡتُكَ بِشَيۡءٖ مُّبِينٖ ۝٣٠ قَالَ أَوَلَوۡ جِئۡتُكَ بِشَیۡءࣲ مُّبِینࣲ ۝٣٠ قَالَ اَوَلَوْ جِئْتُكَ بِشَیْءٍ مُّبِیْنٍ ۟ۚ قَالَ فَأۡتِ بِهِۦٓ إِن كُنتَ مِنَ ٱلصَّٰدِقِينَ ۝٣١ قَالَ فَأۡتِ بِهِۦۤ إِن كُنتَ مِنَ ٱلصَّٰدِقِینَ ۝٣١ قَالَ فَاْتِ بِهٖۤ اِنْ كُنْتَ مِنَ الصّٰدِقِیْنَ ۟ فَأَلۡقَىٰ عَصَاهُ فَإِذَا هِيَ ثُعۡبَانٞ مُّبِينٞ ۝٣٢ فَأَلۡقَىٰ عَصَاهُ فَإِذَا هِیَ ثُعۡبَانࣱ مُّبِینࣱ ۝٣٢ فَاَلْقٰی عَصَاهُ فَاِذَا هِیَ ثُعْبَانٌ مُّبِیْنٌ ۟ۚۖ وَنَزَعَ يَدَهُۥ فَإِذَا هِيَ بَيۡضَآءُ لِلنَّٰظِرِينَ ۝٣٣ وَنَزَعَ یَدَهُۥ فَإِذَا هِیَ بَیۡضَاۤءُ لِلنَّٰظِرِینَ ۝٣٣ وَّنَزَعَ یَدَهٗ فَاِذَا هِیَ بَیْضَآءُ لِلنّٰظِرِیْنَ ۟۠ قَالَ لِلۡمَلَإِ حَوۡلَهُۥٓ إِنَّ هَٰذَا لَسَٰحِرٌ عَلِيمٞ ۝٣٤ قَالَ لِلۡمَلَإِ حَوۡلَهُۥۤ إِنَّ هَٰذَا لَسَٰحِرٌ عَلِیمࣱ ۝٣٤ قَالَ لِلْمَلَاِ حَوْلَهٗۤ اِنَّ هٰذَا لَسٰحِرٌ عَلِیْمٌ ۟ۙ يُرِيدُ أَن يُخۡرِجَكُم مِّنۡ أَرۡضِكُم بِسِحۡرِهِۦ فَمَاذَا تَأۡمُرُونَ ۝٣٥ یُرِیدُ أَن یُخۡرِجَكُم مِّنۡ أَرۡضِكُم بِسِحۡرِهِۦ فَمَاذَا تَأۡمُرُونَ ۝٣٥ یُّرِیْدُ اَنْ یُّخْرِجَكُمْ مِّنْ اَرْضِكُمْ بِسِحْرِهٖ ۖۗ فَمَاذَا تَاْمُرُوْنَ ۟ قَالُوٓاْ أَرۡجِهۡ وَأَخَاهُ وَٱبۡعَثۡ فِي ٱلۡمَدَآئِنِ حَٰشِرِينَ ۝٣٦ قَالُوۤا۟ أَرۡجِهۡ وَأَخَاهُ وَٱبۡعَثۡ فِی ٱلۡمَدَاۤئِنِ حَٰشِرِینَ ۝٣٦ قَالُوْۤا اَرْجِهْ وَاَخَاهُ وَابْعَثْ فِی الْمَدَآىِٕنِ حٰشِرِیْنَ ۟ۙ يَأۡتُوكَ بِكُلِّ سَحَّارٍ عَلِيمٖ ۝٣٧ یَأۡتُوكَ بِكُلِّ سَحَّارٍ عَلِیمࣲ ۝٣٧ یَاْتُوْكَ بِكُلِّ سَحَّارٍ عَلِیْمٍ ۟ فَجُمِعَ ٱلسَّحَرَةُ لِمِيقَٰتِ يَوۡمٖ مَّعۡلُومٖ ۝٣٨ فَجُمِعَ ٱلسَّحَرَةُ لِمِیقَٰتِ یَوۡمࣲ مَّعۡلُومࣲ ۝٣٨ فَجُمِعَ السَّحَرَةُ لِمِیْقَاتِ یَوْمٍ مَّعْلُوْمٍ ۟ۙ وَقِيلَ لِلنَّاسِ هَلۡ أَنتُم مُّجۡتَمِعُونَ ۝٣٩ وَقِیلَ لِلنَّاسِ هَلۡ أَنتُم مُّجۡتَمِعُونَ ۝٣٩ وَّقِیْلَ لِلنَّاسِ هَلْ اَنْتُمْ مُّجْتَمِعُوْنَ ۟ۙ لَعَلَّنَا نَتَّبِعُ ٱلسَّحَرَةَ إِن كَانُواْ هُمُ ٱلۡغَٰلِبِينَ ۝٤٠ لَعَلَّنَا نَتَّبِعُ ٱلسَّحَرَةَ إِن كَانُوا۟ هُمُ ٱلۡغَٰلِبِینَ ۝٤٠ لَعَلَّنَا نَتَّبِعُ السَّحَرَةَ اِنْ كَانُوْا هُمُ الْغٰلِبِیْنَ ۟ فَلَمَّا جَآءَ ٱلسَّحَرَةُ قَالُواْ لِفِرۡعَوۡنَ أَئِنَّ لَنَا لَأَجۡرًا إِن كُنَّا نَحۡنُ ٱلۡغَٰلِبِينَ ۝٤١ فَلَمَّا جَاۤءَ ٱلسَّحَرَةُ قَالُوا۟ لِفِرۡعَوۡنَ أَئِنَّ لَنَا لَأَجۡرًا إِن كُنَّا نَحۡنُ ٱلۡغَٰلِبِینَ ۝٤١ فَلَمَّا جَآءَ السَّحَرَةُ قَالُوْا لِفِرْعَوْنَ اَىِٕنَّ لَنَا لَاَجْرًا اِنْ كُنَّا نَحْنُ الْغٰلِبِیْنَ ۟ قَالَ نَعَمۡ وَإِنَّكُمۡ إِذٗا لَّمِنَ ٱلۡمُقَرَّبِينَ ۝٤٢ قَالَ نَعَمۡ وَإِنَّكُمۡ إِذࣰا لَّمِنَ ٱلۡمُقَرَّبِینَ ۝٤٢ قَالَ نَعَمْ وَاِنَّكُمْ اِذًا لَّمِنَ الْمُقَرَّبِیْنَ ۟ قَالَ لَهُم مُّوسَىٰٓ أَلۡقُواْ مَآ أَنتُم مُّلۡقُونَ ۝٤٣ قَالَ لَهُم مُّوسَىٰۤ أَلۡقُوا۟ مَاۤ أَنتُم مُّلۡقُونَ ۝٤٣ قَالَ لَهُمْ مُّوْسٰۤی اَلْقُوْا مَاۤ اَنْتُمْ مُّلْقُوْنَ ۟ فَأَلۡقَوۡاْ حِبَالَهُمۡ وَعِصِيَّهُمۡ وَقَالُواْ بِعِزَّةِ فِرۡعَوۡنَ إِنَّا لَنَحۡنُ ٱلۡغَٰلِبُونَ ۝٤٤ فَأَلۡقَوۡا۟ حِبَالَهُمۡ وَعِصِیَّهُمۡ وَقَالُوا۟ بِعِزَّةِ فِرۡعَوۡنَ إِنَّا لَنَحۡنُ ٱلۡغَٰلِبُونَ ۝٤٤ فَاَلْقَوْا حِبَالَهُمْ وَعِصِیَّهُمْ وَقَالُوْا بِعِزَّةِ فِرْعَوْنَ اِنَّا لَنَحْنُ الْغٰلِبُوْنَ ۟ فَأَلۡقَىٰ مُوسَىٰ عَصَاهُ فَإِذَا هِيَ تَلۡقَفُ مَا يَأۡفِكُونَ ۝٤٥ فَأَلۡقَىٰ مُوسَىٰ عَصَاهُ فَإِذَا هِیَ تَلۡقَفُ مَا یَأۡفِكُونَ ۝٤٥ فَاَلْقٰی مُوْسٰی عَصَاهُ فَاِذَا هِیَ تَلْقَفُ مَا یَاْفِكُوْنَ ۟ۚۖ فَأُلۡقِيَ ٱلسَّحَرَةُ سَٰجِدِينَ ۝٤٦ فَأُلۡقِیَ ٱلسَّحَرَةُ سَٰجِدِینَ ۝٤٦ فَاُلْقِیَ السَّحَرَةُ سٰجِدِیْنَ ۟ۙ قَالُوٓاْ ءَامَنَّا بِرَبِّ ٱلۡعَٰلَمِينَ ۝٤٧ قَالُوۤا۟ ءَامَنَّا بِرَبِّ ٱلۡعَٰلَمِینَ ۝٤٧ قَالُوْۤا اٰمَنَّا بِرَبِّ الْعٰلَمِیْنَ ۟ۙ رَبِّ مُوسَىٰ وَهَٰرُونَ ۝٤٨ رَبِّ مُوسَىٰ وَهَٰرُونَ ۝٤٨ رَبِّ مُوْسٰی وَهٰرُوْنَ ۟ قَالَ ءَامَنتُمۡ لَهُۥ قَبۡلَ أَنۡ ءَاذَنَ لَكُمۡۖ إِنَّهُۥ لَكَبِيرُكُمُ ٱلَّذِي عَلَّمَكُمُ ٱلسِّحۡرَ فَلَسَوۡفَ تَعۡلَمُونَۚ لَأُقَطِّعَنَّ أَيۡدِيَكُمۡ وَأَرۡجُلَكُم مِّنۡ خِلَٰفٖ وَلَأُصَلِّبَنَّكُمۡ أَجۡمَعِينَ ۝٤٩ قَالَ ءَامَنتُمۡ لَهُۥ قَبۡلَ أَنۡ ءَاذَنَ لَكُمۡۖ إِنَّهُۥ لَكَبِیرُكُمُ ٱلَّذِی عَلَّمَكُمُ ٱلسِّحۡرَ فَلَسَوۡفَ تَعۡلَمُونَۚ لَأُقَطِّعَنَّ أَیۡدِیَكُمۡ وَأَرۡجُلَكُم مِّنۡ خِلَٰفࣲ وَلَأُصَلِّبَنَّكُمۡ أَجۡمَعِینَ ۝٤٩ قَالَ اٰمَنْتُمْ لَهٗ قَبْلَ اَنْ اٰذَنَ لَكُمْ ۚ اِنَّهٗ لَكَبِیْرُكُمُ الَّذِیْ عَلَّمَكُمُ السِّحْرَ ۚ فَلَسَوْفَ تَعْلَمُوْنَ ؕ۬ لَاُقَطِّعَنَّ اَیْدِیَكُمْ وَاَرْجُلَكُمْ مِّنْ خِلَافٍ وَّلَاُوصَلِّبَنَّكُمْ اَجْمَعِیْنَ ۟ۚ قَالُواْ لَا ضَيۡرَۖ إِنَّآ إِلَىٰ رَبِّنَا مُنقَلِبُونَ ۝٥٠ قَالُوا۟ لَا ضَیۡرَۖ إِنَّاۤ إِلَىٰ رَبِّنَا مُنقَلِبُونَ ۝٥٠ قَالُوْا لَا ضَیْرَ ؗ اِنَّاۤ اِلٰی رَبِّنَا مُنْقَلِبُوْنَ ۟ۚ إِنَّا نَطۡمَعُ أَن يَغۡفِرَ لَنَا رَبُّنَا خَطَٰيَٰنَآ أَن كُنَّآ أَوَّلَ ٱلۡمُؤۡمِنِينَ ۝٥١ إِنَّا نَطۡمَعُ أَن یَغۡفِرَ لَنَا رَبُّنَا خَطَٰیَٰنَاۤ أَن كُنَّاۤ أَوَّلَ ٱلۡمُؤۡمِنِینَ ۝٥١ اِنَّا نَطْمَعُ اَنْ یَّغْفِرَ لَنَا رَبُّنَا خَطٰیٰنَاۤ اَنْ كُنَّاۤ اَوَّلَ الْمُؤْمِنِیْنَ ۟ؕ۠ وَأَوۡحَيۡنَآ إِلَىٰ مُوسَىٰٓ أَنۡ أَسۡرِ بِعِبَادِيٓ إِنَّكُم مُّتَّبَعُونَ ۝٥٢ وَأَوۡحَیۡنَاۤ إِلَىٰ مُوسَىٰۤ أَنۡ أَسۡرِ بِعِبَادِیۤ إِنَّكُم مُّتَّبَعُونَ ۝٥٢ وَاَوْحَیْنَاۤ اِلٰی مُوْسٰۤی اَنْ اَسْرِ بِعِبَادِیْۤ اِنَّكُمْ مُّتَّبَعُوْنَ ۟ فَأَرۡسَلَ فِرۡعَوۡنُ فِي ٱلۡمَدَآئِنِ حَٰشِرِينَ ۝٥٣ فَأَرۡسَلَ فِرۡعَوۡنُ فِی ٱلۡمَدَاۤئِنِ حَٰشِرِینَ ۝٥٣ فَاَرْسَلَ فِرْعَوْنُ فِی الْمَدَآىِٕنِ حٰشِرِیْنَ ۟ۚ إِنَّ هَٰٓؤُلَآءِ لَشِرۡذِمَةٞ قَلِيلُونَ ۝٥٤ إِنَّ هَٰۤؤُلَاۤءِ لَشِرۡذِمَةࣱ قَلِیلُونَ ۝٥٤ اِنَّ هٰۤؤُلَآءِ لَشِرْذِمَةٌ قَلِیْلُوْنَ ۟ۙ وَإِنَّهُمۡ لَنَا لَغَآئِظُونَ ۝٥٥ وَإِنَّهُمۡ لَنَا لَغَاۤئِظُونَ ۝٥٥ وَاِنَّهُمْ لَنَا لَغَآىِٕظُوْنَ ۟ۙ وَإِنَّا لَجَمِيعٌ حَٰذِرُونَ ۝٥٦ وَإِنَّا لَجَمِیعٌ حَٰذِرُونَ ۝٥٦ وَاِنَّا لَجَمِیْعٌ حٰذِرُوْنَ ۟ؕ فَأَخۡرَجۡنَٰهُم مِّن جَنَّٰتٖ وَعُيُونٖ ۝٥٧ فَأَخۡرَجۡنَٰهُم مِّن جَنَّٰتࣲ وَعُیُونࣲ ۝٥٧ فَاَخْرَجْنٰهُمْ مِّنْ جَنّٰتٍ وَّعُیُوْنٍ ۟ۙ وَكُنُوزٖ وَمَقَامٖ كَرِيمٖ ۝٥٨ وَكُنُوزࣲ وَمَقَامࣲ كَرِیمࣲ ۝٥٨ وَّكُنُوْزٍ وَّمَقَامٍ كَرِیْمٍ ۟ۙ كَذَٰلِكَۖ وَأَوۡرَثۡنَٰهَا بَنِيٓ إِسۡرَٰٓءِيلَ ۝٥٩ كَذَٰلِكَۖ وَأَوۡرَثۡنَٰهَا بَنِیۤ إِسۡرَٰۤءِیلَ ۝٥٩ كَذٰلِكَ ؕ وَاَوْرَثْنٰهَا بَنِیْۤ اِسْرَآءِیْلَ ۟ؕ فَأَتۡبَعُوهُم مُّشۡرِقِينَ ۝٦٠ فَأَتۡبَعُوهُم مُّشۡرِقِینَ ۝٦٠ فَاَتْبَعُوْهُمْ مُّشْرِقِیْنَ ۟ فَلَمَّا تَرَٰٓءَا ٱلۡجَمۡعَانِ قَالَ أَصۡحَٰبُ مُوسَىٰٓ إِنَّا لَمُدۡرَكُونَ ۝٦١ فَلَمَّا تَرَٰۤءَا ٱلۡجَمۡعَانِ قَالَ أَصۡحَٰبُ مُوسَىٰۤ إِنَّا لَمُدۡرَكُونَ ۝٦١ فَلَمَّا تَرَآءَ الْجَمْعٰنِ قَالَ اَصْحٰبُ مُوْسٰۤی اِنَّا لَمُدْرَكُوْنَ ۟ۚ قَالَ كَلَّآۖ إِنَّ مَعِيَ رَبِّي سَيَهۡدِينِ ۝٦٢ قَالَ كَلَّاۤۖ إِنَّ مَعِیَ رَبِّی سَیَهۡدِینِ ۝٦٢ قَالَ كَلَّا ۚ اِنَّ مَعِیَ رَبِّیْ سَیَهْدِیْنِ ۟ فَأَوۡحَيۡنَآ إِلَىٰ مُوسَىٰٓ أَنِ ٱضۡرِب بِّعَصَاكَ ٱلۡبَحۡرَۖ فَٱنفَلَقَ فَكَانَ كُلُّ فِرۡقٖ كَٱلطَّوۡدِ ٱلۡعَظِيمِ ۝٦٣ فَأَوۡحَیۡنَاۤ إِلَىٰ مُوسَىٰۤ أَنِ ٱضۡرِب بِّعَصَاكَ ٱلۡبَحۡرَۖ فَٱنفَلَقَ فَكَانَ كُلُّ فِرۡقࣲ كَٱلطَّوۡدِ ٱلۡعَظِیمِ ٱلۡأَرۡذَلُونَ ۝١١١ قَالُوۤا۟ أَنُؤۡمِنُ لَكَ وَٱتَّبَعَكَ ٱلۡأَرۡذَلُونَ ۝١١١ قَالُوْۤا اَنُؤْمِنُ لَكَ وَاتَّبَعَكَ الْاَرْذَلُوْنَ ۟ؕ قَالَ وَمَا عِلۡمِي بِمَا كَانُواْ يَعۡمَلُونَ ۝١١٢ قَالَ وَمَا عِلۡمِی بِمَا كَانُوا۟ یَعۡمَلُونَ ۝١١٢ قَالَ وَمَا عِلْمِیْ بِمَا كَانُوْا یَعْمَلُوْنَ ۟ۚ إِنۡ حِسَابُهُمۡ إِلَّا عَلَىٰ رَبِّيۖ لَوۡ تَشۡعُرُونَ ۝١١٣ إِنۡ حِسَابُهُمۡ إِلَّا عَلَىٰ رَبِّیۖ لَوۡ تَشۡعُرُونَ ۝١١٣ اِنْ حِسَابُهُمْ اِلَّا عَلٰی رَبِّیْ لَوْ تَشْعُرُوْنَ ۟ۚ وَمَآ أَنَا۠ بِطَارِدِ ٱلۡمُؤۡمِنِينَ ۝١١٤ وَمَاۤ أَنَا۠ بِطَارِدِ ٱلۡمُؤۡمِنِینَ ۝١١٤ وَمَاۤ اَنَا بِطَارِدِ الْمُؤْمِنِیْنَ ۟ۚ إِنۡ أَنَا۠ إِلَّا نَذِيرٞ مُّبِينٞ ۝١١٥ إِنۡ أَنَا۠ إِلَّا نَذِیرࣱ مُّبِینࣱ ۝١١٥ اِنْ اَنَا اِلَّا نَذِیْرٌ مُّبِیْنٌ ۟ؕ قَالُواْ لَئِن لَّمۡ تَنتَهِ يَٰنُوحُ لَتَكُونَنَّ مِنَ ٱلۡمَرۡجُومِينَ ۝١١٦ قَالُوا۟ لَئِن لَّمۡ تَنتَهِ یَٰنُوحُ لَتَكُونَنَّ مِنَ ٱلۡمَرۡجُومِینَ ۝١١٦ قَالُوْا لَىِٕنْ لَّمْ تَنْتَهِ یٰنُوْحُ لَتَكُوْنَنَّ مِنَ الْمَرْجُوْمِیْنَ ۟ؕ قَالَ رَبِّ إِنَّ قَوۡمِي كَذَّبُونِ ۝١١٧ قَالَ رَبِّ إِنَّ قَوۡمِی كَذَّبُونِ ۝١١٧ قَالَ رَبِّ اِنَّ قَوْمِیْ كَذَّبُوْنِ ۟ۚۖ فَٱفۡتَحۡ بَيۡنِي وَبَيۡنَهُمۡ فَتۡحٗا وَنَجِّنِي وَمَن مَّعِيَ مِنَ ٱلۡمُؤۡمِنِينَ ۝١١٨ فَٱفۡتَحۡ بَیۡنِی وَبَیۡنَهُمۡ فَتۡحࣰا وَنَجِّنِی وَمَن مَّعِیَ مِنَ ٱلۡمُؤۡمِنِینَ ۝١١٨ فَافْتَحْ بَیْنِیْ وَبَیْنَهُمْ فَتْحًا وَّنَجِّنِیْ وَمَنْ مَّعِیَ مِنَ الْمُؤْمِنِیْنَ ۟ فَأَنجَيۡنَٰهُ وَمَن مَّعَهُۥ فِي ٱلۡفُلۡكِ ٱلۡمَشۡحُونِ ۝١١٩ فَأَنجَیۡنَٰهُ وَمَن مَّعَهُۥ فِی ٱلۡفُلۡكِ ٱلۡمَشۡحُونِ ۝١١٩ فَاَنْجَیْنٰهُ وَمَنْ مَّعَهٗ فِی الْفُلْكِ الْمَشْحُوْنِ ۟ۚ ثُمَّ أَغۡرَقۡنَا بَعۡدُ ٱلۡبَاقِينَ ۝١٢٠ ثُمَّ أَغۡرَقۡنَا بَعۡدُ ٱلۡبَاقِینَ ۝١٢٠ ثُمَّ اَغْرَقْنَا بَعْدُ الْبٰقِیْنَ ۟ؕ إِنَّ فِي ذَٰلِكَ لَأٓيَةٗۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِينَ ۝١٢١ إِنَّ فِی ذَٰلِكَ لَءَایَةࣰۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِینَ ۝١٢١ اِنَّ فِیْ ذٰلِكَ لَاٰیَةً ؕ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِیْنَ ۟ وَإِنَّ رَبَّكَ لَهُوَ ٱلۡعَزِيزُ ٱلرَّحِيمُ ۝١٢٢ وَإِنَّ رَبَّكَ لَهُوَ ٱلۡعَزِیزُ ٱلرَّحِیمُ ۝١٢٢ وَاِنَّ رَبَّكَ لَهُوَ الْعَزِیْزُ الرَّحِیْمُ ۟۠ كَذَّبَتۡ عَادٌ ٱلۡمُرۡسَلِينَ ۝١٢٣ كَذَّبَتۡ عَادٌ ٱلۡمُرۡسَلِینَ ۝١٢٣ كَذَّبَتْ عَادُ لْمُرْسَلِیْنَ ۟ۚۖ إِذۡ قَالَ لَهُمۡ أَخُوهُمۡ هُودٌ أَلَا تَتَّقُونَ ۝١٢٤ إِذۡ قَالَ لَهُمۡ أَخُوهُمۡ هُودٌ أَلَا تَتَّقُونَ ۝١٢٤ اِذْ قَالَ لَهُمْ اَخُوْهُمْ هُوْدٌ اَلَا تَتَّقُوْنَ ۟ۚ إِنِّي لَكُمۡ رَسُولٌ أَمِينٞ ۝١٢٥ إِنِّی لَكُمۡ رَسُولٌ أَمِینࣱ ۝١٢٥ اِنِّیْ لَكُمْ رَسُوْلٌ اَمِیْنٌ ۟ۙ فَٱتَّقُواْ ٱللَّهَ وَأَطِيعُونِ ۝١٢٦ فَٱتَّقُوا۟ ٱللَّهَ وَأَطِیعُونِ ۝١٢٦ فَاتَّقُوا اللّٰهَ وَاَطِیْعُوْنِ ۟ۚ وَمَآ أَسۡـَٔلُكُمۡ عَلَيۡهِ مِنۡ أَجۡرٍۖ إِنۡ أَجۡرِيَ إِلَّا عَلَىٰ رَبِّ ٱلۡعَٰلَمِينَ ۝١٢٧ وَمَاۤ أَسۡءَلُكُمۡ عَلَیۡهِ مِنۡ أَجۡرٍۖ إِنۡ أَجۡرِیَ إِلَّا عَلَىٰ رَبِّ ٱلۡعَٰلَمِینَ ۝١٢٧ وَمَاۤ اَسْـَٔلُكُمْ عَلَیْهِ مِنْ اَجْرٍ ۚ اِنْ اَجْرِیَ اِلَّا عَلٰی رَبِّ الْعٰلَمِیْنَ ۟ؕ أَتَبۡنُونَ بِكُلِّ رِيعٍ ءَايَةٗ تَعۡبَثُونَ ۝١٢٨ أَتَبۡنُونَ بِكُلِّ رِیعٍ ءَایَةࣰ تَعۡبَثُونَ ۝١٢٨ اَتَبْنُوْنَ بِكُلِّ رِیْعٍ مِنَ ٱلۡجِبَالِ بُیُوتࣰا فَٰرِهِینَ ۝١٤٩ وَتَنْحِتُوْنَ مِنَ الْجِبَالِ بُیُوْتًا فٰرِهِیْنَ ۟ۚ فَٱتَّقُواْ ٱللَّهَ وَأَطِيعُونِ ۝١٥٠ فَٱتَّقُوا۟ ٱللَّهَ وَأَطِیعُونِ ۝١٥٠ فَاتَّقُوا اللّٰهَ يُصۡلِحُونَ ۝١٥٢ ٱلَّذِینَ یُفۡسِدُونَ فِی ٱلۡأَرۡضِ وَلَا یُصۡلِحُونَ ۝١٥٢ الَّذِیْنَ یُفْسِدُوْنَ فِی الْاَرْضِ وَلَا یُصْلِحُوْنَ ۟ قَالُوٓاْ إِنَّمَآ أَنتَ مِنَ ٱلۡمُسَحَّرِينَ ۝١٥٣ قَالُوۤا۟ إِنَّمَاۤ أَنتَ مِنَ ٱلۡمُسَحَّرِینَ ۝١٥٣ قَالُوْۤا اِنَّمَاۤ اَنْتَ مِنَ الْمُسَحَّرِیْنَ ۟ۚ مَآ أَنتَ إِلَّا بَشَرٞ مِّثۡلُنَا فَأۡتِ بِـَٔايَةٍ إِن كُنتَ مِنَ ٱلصَّٰدِقِينَ ۝١٥٤ مَاۤ أَنتَ إِلَّا بَشَرࣱ مِّثۡلُنَا فَأۡتِ بِءَایَةٍ إِن كُنتَ مِنَ ٱلصَّٰدِقِینَ ۝١٥٤ مَاۤ اَنْتَ اِلَّا بَشَرٌ مِّثْلُنَا ۖۚ فَاْتِ بِاٰیَةٍ اِنْ كُنْتَ مِنَ الصّٰدِقِیْنَ ۟ قَالَ هَٰذِهِۦ نَاقَةٞ لَّهَا شِرۡبٞ وَلَكُمۡ شِرۡبُ يَوۡمٖ مَّعۡلُومٖ ۝١٥٥ قَالَ هَٰذِهِۦ نَاقَةࣱ لَّهَا شِرۡبࣱ وَلَكُمۡ شِرۡبُ یَوۡمࣲ مَّعۡلُومࣲ ۝١٥٥ قَالَ هٰذِهٖ نَاقَةٌ لَّهَا شِرْبٌ وَّلَكُمْ شِرْبُ یَوْمٍ مَّعْلُوْمٍ ۟ۚ وَلَا عَظِیْمٍ ۟ فَعَقَرُوهَا فَأَصۡبَحُواْ نَٰدِمِينَ ۝١٥٧ فَعَقَرُوهَا فَأَصۡبَحُوا۟ نَٰدِمِینَ ۝١٥٧ فَعَقَرُوْهَا فَاَصْبَحُوْا نٰدِمِیْنَ ۟ۙ فَأَخَذَهُمُ ٱلۡعَذَابُۚ إِنَّ فِي ذَٰلِكَ لَأٓيَةٗۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِينَ ۝١٥٨ فَأَخَذَهُمُ ٱلۡعَذَابُۚ إِنَّ فِی ذَٰلِكَ لَءَایَةࣰۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِینَ ۝١٥٨ فَاَخَذَهُمُ الْعَذَابُ ؕ اِنَّ فِیْ ذٰلِكَ لَاٰیَةً ؕ وَمَا كَانَ لُوطٌ أَلَا تَتَّقُونَ ۝١٦١ اِذْ قَالَ لَهُمْ اَخُوْهُمْ لُوْطٌ اَلَا تَتَّقُوْنَ ۟ۚ إِنِّي لَكُمۡ رَسُولٌ أَمِينٞ ۝١٦٢ إِنِّی لَكُمۡ رَسُولٌ أَمِینࣱ ۝١٦٢ اِنِّیْ لَكُمْ رَسُوْلٌ اَمِیْنٌ ۟ۙ وَٱلۡجِبِلَّةَ ٱلۡأَوَّلِينَ ۝١٨٤ وَٱتَّقُوا۟ ٱلَّذِی خَلَقَكُمۡ وَٱلۡجِبِلَّةَ ٱلۡأَوَّلِینَ ۝١٨٤ وَاتَّقُوا الَّذِیْ خَلَقَكُمْ وَالْجِبِلَّةَ الْاَوَّلِیْنَ ۟ؕ قَالُوٓاْ إِنَّمَآ أَنتَ مِنَ ٱلۡمُسَحَّرِينَ ۝١٨٥ قَالُوۤا۟ إِنَّمَاۤ أَنتَ مِنَ ٱلۡمُسَحَّرِینَ ۝١٨٥ قَالُوْۤا اِنَّمَاۤ اَنْتَ مِنَ الْمُسَحَّرِیْنَ ۟ۙ وَمَآ أَنتَ إِلَّا بَشَرٞ مِّثۡلُنَا وَإِن نَّظُنُّكَ لَمِنَ ٱلۡكَٰذِبِينَ ۝١٨٦ وَمَاۤ أَنتَ إِلَّا بَشَرࣱ مِّثۡلُنَا وَإِن نَّظُنُّكَ لَمِنَ ٱلۡكَٰذِبِینَ ۝١٨٦ وَمَاۤ اَنْتَ اِلَّا بَشَرٌ مِّثْلُنَا وَاِنْ نَّظُنُّكَ لَمِنَ الْكٰذِبِیْنَ ۟ۚ فَأَسۡقِطۡ عَلَيۡنَا كِسَفٗا مِّنَ ٱلسَّمَآءِ إِن كُنتَ مِنَ ٱلصَّٰدِقِينَ ۝١٨٧ فَأَسۡقِطۡ عَلَیۡنَا كِسَفࣰا مِّنَ ٱلسَّمَاۤءِ إِن كُنتَ مِنَ ٱلصَّٰدِقِینَ ۝١٨٧ فَاَسْقِطْ عَلَیْنَا كِسَفًا مِّنَ السَّمَآءِ اِنْ كُنْتَ مِنَ الصّٰدِقِیْنَ ۟ؕ قَالَ رَبِّيٓ أَعۡلَمُ بِمَا تَعۡمَلُونَ ۝١٨٨ قَالَ رَبِّیۤ أَعۡلَمُ بِمَا تَعۡمَلُونَ ۝١٨٨ قَالَ رَبِّیْۤ اَعْلَمُ بِمَا تَعْمَلُوْنَ ۟ فَكَذَّبُوهُ فَأَخَذَهُمۡ عَذَابُ يَوۡمِ ٱلظُّلَّةِۚ إِنَّهُۥ كَانَ عَذَابَ يَوۡمٍ عَظِيمٍ ۝١٨٩ فَكَذَّبُوهُ فَأَخَذَهُمۡ عَذَابُ یَوۡمِ ٱلظُّلَّةِۚ إِنَّهُۥ كَانَ عَذَابَ یَوۡمٍ عَظِیمٍ ۝١٨٩ فَكَذَّبُوْهُ فَاَخَذَهُمْ عَذَابُ یَوْمِ الظُّلَّةِ ؕ اِنَّهٗ كَانَ عَذَابَ یَوْمٍ عَظِیْمٍ ۟ إِنَّ فِي ذَٰلِكَ لَأٓيَةٗۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِينَ ۝١٩٠ إِنَّ فِی ذَٰلِكَ لَءَایَةࣰۖ وَمَا كَانَ أَكۡثَرُهُم مُّؤۡمِنِینَ ۝١٩٠ اِنَّ فِیْ ذٰلِكَ لَاٰیَةً ؕ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِیْنَ ۟ وَإِنَّ رَبَّكَ لَهُوَ ٱلۡعَزِيزُ ٱلرَّحِيمُ ۝١٩١ وَإِنَّ رَبَّكَ لَهُوَ ٱلۡعَزِیزُ ٱلرَّحِیمُ ۝١٩١ وَاِنَّ رَبَّكَ لَهُوَ الْعَزِیْزُ الرَّحِیْمُ ۟۠ وَإِنَّهُۥ لَتَنزِيلُ رَبِّ ٱلۡعَٰلَمِينَ ۝١٩٢ وَإِنَّهُۥ لَتَنزِیلُ رَبِّ ٱلۡعَٰلَمِینَ ۝١٩٢ وَاِنَّهٗ لَتَنْزِیْلُ رَبِّ الْعٰلَمِیْنَ ۟ؕ نَزَلَ بِهِ ٱلرُّوحُ ٱلۡأَمِينُ ۝١٩٣ نَزَلَ بِهِ ٱلرُّوحُ ٱلۡأَمِینُ ۝١٩٣ نَزَلَ بِهِ الرُّوْحُ الْاَمِیْنُ ۟ۙ عَلَىٰ قَلۡبِكَ لِتَكُونَ مِنَ ٱلۡمُنذِرِينَ ۝١٩٤ عَلَىٰ قَلۡبِكَ لِتَكُونَ مِنَ ٱلۡمُنذِرِینَ ۝١٩٤ عَلٰی قَلْبِكَ لِتَكُوْنَ مِنَ الْمُنْذِرِیْنَ ۟ۙ بِلِسَانٍ عَرَبِيّٖ مُّبِينٖ ۝١٩٥ بِلِسَانٍ عَرَبِیࣲّ مُّبِینࣲ ۝١٩٥ بِلِسَانٍ عَرَبِیٍّ مُّبِیْنٍ ۟ؕ وَإِنَّهُۥ لَفِي زُبُرِ ٱلۡأَوَّلِينَ ۝١٩٦ وَإِنَّهُۥ لَفِی زُبُرِ ٱلۡأَوَّلِینَ ۝١٩٦ وَاِنَّهٗ لَفِیْ زُبُرِ الْاَوَّلِیْنَ ۟ أَوَلَمۡ يَكُن لَّهُمۡ ءَايَةً أَن يَعۡلَمَهُۥ عُلَمَٰٓؤُاْ بَنِيٓ إِسۡرَٰٓءِيلَ ۝١٩٧ أَوَلَمۡ یَكُن لَّهُمۡ ءَایَةً أَن یَعۡلَمَهُۥ عُلَمَٰۤؤُا۟ بَنِیۤ إِسۡرَٰۤءِیلَ ۝١٩٧ اَوَلَمْ یَكُنْ لَّهُمْ اٰیَةً اَنْ یَّعْلَمَهٗ عُلَمٰٓؤُا بَنِیْۤ اِسْرَآءِیْلَ ۟ؕ وَلَوۡ نَزَّلۡنَٰهُ عَلَىٰ بَعۡضِ ٱلۡأَعۡجَمِينَ ۝١٩٨ وَلَوۡ نَزَّلۡنَٰهُ عَلَىٰ بَعۡضِ ٱلۡأَعۡجَمِینَ ۝١٩٨ وَلَوْ نَزَّلْنٰهُ عَلٰی بَعْضِ الْاَعْجَمِیْنَ ۟ۙ فَقَرَأَهُۥ عَلَيۡهِم مَّا كَانُواْ بِهِۦ مُؤۡمِنِينَ ۝١٩٩ فَقَرَأَهُۥ عَلَیۡهِم مَّا كَانُوا۟ بِهِۦ مُؤۡمِنِینَ ۝١٩٩ فَقَرَاَهٗ عَلَیْهِمْ مَّا كَانُوْا بِهٖ مُؤْمِنِیْنَ ۟ؕ كَذَٰلِكَ سَلَكۡنَٰهُ فِي قُلُوبِ ٱلۡمُجۡرِمِينَ ۝٢٠٠ كَذَٰلِكَ سَلَكۡنَٰهُ فِی قُلُوبِ ٱلۡمُجۡرِمِینَ ۝٢٠٠ كَذٰلِكَ سَلَكْنٰهُ فِیْ قُلُوْبِ الْمُجْرِمِیْنَ ۟ؕ لَا يُؤۡمِنُونَ بِهِۦ حَتَّىٰ يَرَوُاْ ٱلۡعَذَابَ ٱلۡأَلِيمَ ۝٢٠١ لَا یُؤۡمِنُونَ بِهِۦ حَتَّىٰ یَرَوُا۟ ٱلۡعَذَابَ ٱلۡأَلِیمَ ۝٢٠١ لَا یُؤْمِنُوْنَ بِهٖ حَتّٰی یَرَوُا الْعَذَابَ الْاَلِیْمَ ۟ۙ فَيَأۡتِيَهُم بَغۡتَةٗ وَهُمۡ لَا يَشۡعُرُونَ ۝٢٠٢ فَیَأۡتِیَهُم بَغۡتَةࣰ وَهُمۡ لَا یَشۡعُرُونَ ۝٢٠٢ فَیَاْتِیَهُمْ بَغْتَةً وَّهُمْ لَا یَشْعُرُوْنَ ۟ۙ فَيَقُولُواْ هَلۡ نَحۡنُ مُنظَرُونَ ۝٢٠٣ فَیَقُولُوا۟ هَلۡ نَحۡنُ مُنظَرُونَ ۝٢٠٣ فَیَقُوْلُوْا هَلْ نَحْنُ مُنْظَرُوْنَ ۟ؕ أَفَبِعَذَابِنَا يَسۡتَعۡجِلُونَ ۝٢٠٤ أَفَبِعَذَابِنَا یَسۡتَعۡجِلُونَ ۝٢٠٤ اَفَبِعَذَابِنَا یَسْتَعْجِلُوْنَ ۟ أَفَرَءَيۡتَ إِن مَّتَّعۡنَٰهُمۡ سِنِينَ ۝٢٠٥ أَفَرَءَیۡتَ إِن مَّتَّعۡنَٰهُمۡ سِنِینَ ۝٢٠٥ اَفَرَءَیْتَ اِنْ مَّتَّعْنٰهُمْ سِنِیْنَ ۟ۙ ثُمَّ جَآءَهُم مَّا كَانُواْ يُوعَدُونَ ۝٢٠٦ ثُمَّ جَاۤءَهُم مَّا كَانُوا۟ یُوعَدُونَ ۝٢٠٦ ثُمَّ جَآءَهُمْ مَّا كَانُوْا یُوْعَدُوْنَ ۟ۙ مَآ أَغۡنَىٰ عَنۡهُم مَّا كَانُواْ يُمَتَّعُونَ ۝٢٠٧ مَاۤ أَغۡنَىٰ عَنۡهُم مَّا كَانُوا۟ یُمَتَّعُونَ ۝٢٠٧ مَاۤ اَغْنٰی عَنْهُمْ مَّا كَانُوْا یُمَتَّعُوْنَ ۟ؕ وَمَآ أَهۡلَكۡنَا مِن قَرۡيَةٍ إِلَّا لَهَا مُنذِرُونَ ۝٢٠٨ وَمَاۤ أَهۡلَكۡنَا مِن قَرۡیَةٍ إِلَّا لَهَا مُنذِرُونَ ۝٢٠٨ وَمَاۤ اَهْلَكْنَا مِنْ قَرْیَةٍ اِلَّا لَهَا مُنْذِرُوْنَ ۟ ذِكۡرَىٰ وَمَا كُنَّا ظَٰلِمِينَ ۝٢٠٩ ذِكۡرَىٰ وَمَا كُنَّا ظَٰلِمِینَ ۝٢٠٩ ذِكْرٰی ۛ۫ وَمَا كُنَّا ظٰلِمِیْنَ ۟ وَمَا تَنَزَّلَتۡ بِهِ ٱلشَّيَٰطِينُ ۝٢١٠ وَمَا تَنَزَّلَتۡ بِهِ ٱلشَّیَٰطِینُ ۝٢١٠ وَمَا تَنَزَّلَتْ بِهِ الشَّیٰطِیْنُ ۟ وَمَا يَنۢبَغِي لَهُمۡ وَمَا يَسۡتَطِيعُونَ ۝٢١١ وَمَا یَنۢبَغِی لَهُمۡ وَمَا یَسۡتَطِیعُونَ ۝٢١١ وَمَا یَنْۢبَغِیْ لَهُمْ وَمَا یَسْتَطِیْعُوْنَ ۟ؕ إِنَّهُمۡ عَنِ ٱلسَّمۡعِ لَمَعۡزُولُونَ ۝٢١٢ إِنَّهُمۡ عَنِ ٱلسَّمۡعِ لَمَعۡزُولُونَ ۝٢١٢ اِنَّهُمْ عَنِ السَّمْعِ لَمَعْزُوْلُوْنَ ۟ؕ فَلَا تَدۡعُ مِنَ الْمُعَذَّبِیْنَ ۟ۚ وَأَنذِرۡ عَشِيرَتَكَ ٱلۡأَقۡرَبِينَ ۝٢١٤ وَأَنذِرۡ عَشِیرَتَكَ ٱلۡأَقۡرَبِینَ ۝٢١٤ وَاَنْذِرْ عَشِیْرَتَكَ الْاَقْرَبِیْنَ ۟ۙ وَٱخۡفِضۡ جَنَاحَكَ لِمَنِ ٱتَّبَعَكَ مِنَ ٱلۡمُؤۡمِنِينَ ۝٢١٥ وَٱخۡفِضۡ جَنَاحَكَ لِمَنِ ٱتَّبَعَكَ مِنَ ٱلۡمُؤۡمِنِینَ ۝٢١٥ وَاخْفِضْ جَنَاحَكَ لِمَنِ اتَّبَعَكَ مِنَ الْمُؤْمِنِیْنَ ۟ۚ فَإِنۡ عَصَوۡكَ فَقُلۡ إِنِّي بَرِيٓءٞ مِّمَّا وَتَقَلُّبَكَ فِي ٱلسَّٰجِدِينَ ۝٢١٩ وَتَقَلُّبَكَ فِی ٱلسَّٰجِدِینَ ۝٢١٩ وَتَقَلُّبَكَ فِی السّٰجِدِیْنَ ۟ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلۡعَلِيمُ ۝٢٢٠ إِنَّهُۥ هُوَ ٱلسَّمِیعُ ٱلۡعَلِیمُ ۝٢٢٠ اِنَّهٗ هُوَ السَّمِیْعُ الْعَلِیْمُ ۟ هَلۡ أُنَبِّئُكُمۡ عَلَىٰ مَن تَنَزَّلُ ٱلشَّيَٰطِينُ ۝٢٢١ هَلۡ أُنَبِّئُكُمۡ عَلَىٰ مَن تَنَزَّلُ ٱلشَّیَٰطِینُ ۝٢٢١ هَلْ اُنَبِّئُكُمْ عَلٰی مَنْ تَنَزَّلُ الشَّیٰطِیْنُ ۟ؕ تَنَزَّلُ عَلَىٰ كُلِّ أَفَّاكٍ أَثِيمٖ ۝٢٢٢ تَنَزَّلُ عَلَىٰ كُلِّ أَفَّاكٍ أَثِیمࣲ ۝٢٢٢ تَنَزَّلُ عَلٰی كُلِّ اَفَّاكٍ اَثِیْمٍ ۟ۙ يُلۡقُونَ ٱلسَّمۡعَ وَأَكۡثَرُهُمۡ كَٰذِبُونَ ۝٢٢٣ یُلۡقُونَ ٱلسَّمۡعَ وَأَكۡثَرُهُمۡ كَٰذِبُونَ ۝٢٢٣ یُّلْقُوْنَ السَّمْعَ وَاَكْثَرُهُمْ كٰذِبُوْنَ ۟ؕ وَٱلشُّعَرَآءُ يَتَّبِعُهُمُ ٱلۡغَاوُۥنَ ۝٢٢٤ وَٱلشُّعَرَاۤءُ یَتَّبِعُهُمُ ٱلۡغَاوُۥنَ ۝٢٢٤ وَالشُّعَرَآءُ یَتَّبِعُهُمُ الْغَاوٗنَ ۟ؕ أَلَمۡ تَرَ أَنَّهُمۡ فِي كُلِّ وَادٖ يَهِيمُونَ ۝٢٢٥ أَلَمۡ تَرَ أَنَّهُمۡ فِی كُلِّ وَادࣲ یَهِیمُونَ ۝٢٢٥ اَلَمْ تَرَ اَنَّهُمْ فِیْ كُلِّ وَادٍ یَّهِیْمُوْنَ ۟ۙ وَأَنَّهُمۡ يَقُولُونَ مَا لَا يَفۡعَلُونَ ۝٢٢٦ وَأَنَّهُمۡ یَقُولُونَ مَا لَا یَفۡعَلُونَ ۝٢٢٦ وَاَنَّهُمْ یَقُوْلُوْنَ مَا لَا یَفْعَلُوْنَ ۟ۙ إِلَّا ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ وَذَكَرُواْ ٱللَّهَ كَثِيرٗا وَٱنتَصَرُواْ مِنۢ بَعۡدِ مَا ظُلِمُواْۗ وَسَيَعۡلَمُ ٱلَّذِينَ ظَلَمُوٓاْ أَيَّ مُنقَلَبٖ يَنقَلِبُونَ ۝٢٢٧ إِلَّا ٱلَّذِینَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ وَذَكَرُوا۟ ٱللَّهَ كَثِیرࣰا وَٱنتَصَرُوا۟ مِنۢ بَعۡدِ مَا ظُلِمُوا۟ۗ وَسَیَعۡلَمُ ٱلَّذِینَ ظَلَمُوۤا۟ أَیَّ مُنقَلَبࣲ یَنقَلِبُونَ ۝٢٢٧ اِلَّا الَّذِیْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَذَكَرُوا اللّٰهَ كَثِیْرًا وَّانْتَصَرُوْا مِنْ بَعْدِ مَا ظُلِمُوْا ؕ وَسَیَعْلَمُ الَّذِیْنَ ظَلَمُوْۤا اَیَّ مُنْقَلَبٍ یَّنْقَلِبُوْنَ ۟۠
