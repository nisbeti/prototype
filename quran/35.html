بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

ٱلۡحَمۡدُ لِلَّهِ فَاطِرِ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ جَاعِلِ ٱلۡمَلَٰٓئِكَةِ رُسُلًا أُوْلِيٓ أَجۡنِحَةٖ مَّثۡنَىٰ وَثُلَٰثَ وَرُبَٰعَۚ يَزِيدُ فِي ٱلۡخَلۡقِ مَا يَشَآءُۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرٞ ۝١ ٱلۡحَمۡدُ لِلَّهِ فَاطِرِ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ جَاعِلِ ٱلۡمَلَٰۤئِكَةِ رُسُلًا أُو۟لِیۤ أَجۡنِحَةࣲ مَّثۡنَىٰ وَثُلَٰثَ وَرُبَٰعَۚ یَزِیدُ فِی ٱلۡخَلۡقِ مَا یَشَاۤءُۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَیۡءࣲ قَدِیرࣱ ۝١ اَلْحَمْدُ لِلّٰهِ فَاطِرِ السَّمٰوٰتِ وَالْاَرْضِ جَاعِلِ الْمَلٰٓىِٕكَةِ رُسُلًا اُولِیْۤ اَجْنِحَةٍ مَّثْنٰی وَثُلٰثَ وَرُبٰعَ ؕ یَزِیْدُ فِی الْخَلْقِ مَا یَشَآءُ ؕ اِنَّ اللّٰهَ عَلٰی كُلِّ شَیْءٍ قَدِیْرٌ ۟ مَّا يَفۡتَحِ ٱللَّهُ لِلنَّاسِ مِن رَّحۡمَةٖ فَلَا مُمۡسِكَ لَهَاۖ وَمَا يُمۡسِكۡ فَلَا مُرۡسِلَ لَهُۥ مِنۢ بَعۡدِهِۦۚ وَهُوَ ٱلۡعَزِيزُ ٱلۡحَكِيمُ ۝٢ مَّا یَفۡتَحِ ٱللَّهُ لِلنَّاسِ مِن رَّحۡمَةࣲ فَلَا مُمۡسِكَ لَهَاۖ وَمَا یُمۡسِكۡ فَلَا مُرۡسِلَ لَهُۥ مِنۢ بَعۡدِهِۦۚ وَهُوَ ٱلۡعَزِیزُ ٱلۡحَكِیمُ ۝٢ مَا یَفْتَحِ اللّٰهُ لِلنَّاسِ مِنْ رَّحْمَةٍ فَلَا مُمْسِكَ لَهَا ۚ وَمَا یُمْسِكْ ۙ فَلَا مُرْسِلَ لَهٗ مِنْ بَعْدِهٖ ؕ وَهُوَ الْعَزِیْزُ الْحَكِیْمُ ۟ يَٰٓأَيُّهَا ٱلنَّاسُ ٱذۡكُرُواْ نِعۡمَتَ ٱللَّهِ عَلَيۡكُمۡۚ هَلۡ مِنۡ خَٰلِقٍ غَيۡرُ ٱللَّهِ يَرۡزُقُكُم مِّنَ ٱلسَّمَآءِ وَٱلۡأَرۡضِۚ لَآ إِلَٰهَ إِلَّا هُوَۖ فَأَنَّىٰ تُؤۡفَكُونَ ۝٣ یَٰۤأَیُّهَا ٱلنَّاسُ ٱذۡكُرُوا۟ نِعۡمَتَ ٱللَّهِ عَلَیۡكُمۡۚ هَلۡ مِنۡ خَٰلِقٍ غَیۡرُ ٱللَّهِ یَرۡزُقُكُم مِّنَ ٱلسَّمَاۤءِ وَٱلۡأَرۡضِۚ لَاۤ إِلَٰهَ إِلَّا هُوَۖ فَأَنَّىٰ تُؤۡفَكُونَ ۝٣ یٰۤاَیُّهَا النَّاسُ اذْكُرُوْا نِعْمَتَ اللّٰهِ عَلَیْكُمْ ؕ هَلْ مِنْ خَالِقٍ غَیْرُ اللّٰهِ یَرْزُقُكُمْ مِّنَ السَّمَآءِ وَالْاَرْضِ ؕ لَاۤ اِلٰهَ اِلَّا هُوَ ۖؗ فَاَنّٰی تُؤْفَكُوْنَ ۟ وَإِن يُكَذِّبُوكَ فَقَدۡ كُذِّبَتۡ رُسُلٞ مِّن قَبۡلِكَۚ وَإِلَى ٱللَّهِ تُرۡجَعُ ٱلۡأُمُورُ ۝٤ وَإِن یُكَذِّبُوكَ فَقَدۡ كُذِّبَتۡ رُسُلࣱ مِّن قَبۡلِكَۚ وَإِلَى ٱللَّهِ تُرۡجَعُ ٱلۡأُمُورُ ۝٤ وَاِنْ یُّكَذِّبُوْكَ فَقَدْ كُذِّبَتْ رُسُلٌ مِّنْ قَبْلِكَ ؕ وَاِلَی اللّٰهِ تُرْجَعُ الْاُمُوْرُ ۟ يَٰٓأَيُّهَا ٱلنَّاسُ إِنَّ وَعۡدَ ٱللَّهِ حَقّٞۖ فَلَا تَغُرَّنَّكُمُ ٱلۡحَيَوٰةُ ٱلدُّنۡيَا وَلَا يَغُرَّنَّكُم بِٱللَّهِ ٱلۡغَرُورُ ۝٥ یَٰۤأَیُّهَا ٱلنَّاسُ إِنَّ وَعۡدَ ٱللَّهِ حَقࣱّۖ فَلَا تَغُرَّنَّكُمُ ٱلۡحَیَوٰةُ ٱلدُّنۡیَا وَلَا یَغُرَّنَّكُم بِٱللَّهِ ٱلۡغَرُورُ ۝٥ یٰۤاَیُّهَا النَّاسُ اِنَّ وَعْدَ اللّٰهِ حَقٌّ فَلَا تَغُرَّنَّكُمُ الْحَیٰوةُ الدُّنْیَا ۥ وَلَا یَغُرَّنَّكُمْ بِاللّٰهِ الْغَرُوْرُ ۟ إِنَّ ٱلشَّيۡطَٰنَ لَكُمۡ عَدُوّٞ فَٱتَّخِذُوهُ عَدُوًّاۚ إِنَّمَا يَدۡعُواْ حِزۡبَهُۥ لِيَكُونُواْ مِنۡ أَصۡحَٰبِ ٱلسَّعِيرِ ۝٦ إِنَّ ٱلشَّیۡطَٰنَ لَكُمۡ عَدُوࣱّ فَٱتَّخِذُوهُ عَدُوًّاۚ إِنَّمَا یَدۡعُوا۟ حِزۡبَهُۥ لِیَكُونُوا۟ مِنۡ أَصۡحَٰبِ ٱلسَّعِیرِ ۝٦ اِنَّ الشَّیْطٰنَ لَكُمْ عَدُوٌّ فَاتَّخِذُوْهُ عَدُوًّا ؕ اِنَّمَا یَدْعُوْا حِزْبَهٗ لِیَكُوْنُوْا مِنْ اَصْحٰبِ السَّعِیْرِ ۟ؕ ٱلَّذِينَ كَفَرُواْ لَهُمۡ عَذَابٞ شَدِيدٞۖ وَٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ لَهُم مَّغۡفِرَةٞ وَأَجۡرٞ كَبِيرٌ ۝٧ ٱلَّذِینَ كَفَرُوا۟ لَهُمۡ عَذَابࣱ شَدِیدࣱۖ وَٱلَّذِینَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ لَهُم مَّغۡفِرَةࣱ وَأَجۡرࣱ كَبِیرٌ ۝٧ اَلَّذِیْنَ كَفَرُوْا لَهُمْ عَذَابٌ شَدِیْدٌ ؕ۬ وَالَّذِیْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ لَهُمْ مَّغْفِرَةٌ وَّاَجْرٌ كَبِیْرٌ ۟۠ أَفَمَن زُيِّنَ لَهُۥ سُوٓءُ عَمَلِهِۦ فَرَءَاهُ حَسَنٗاۖ فَإِنَّ ٱللَّهَ يُضِلُّ مَن يَشَآءُ وَيَهۡدِي مَن يَشَآءُۖ فَلَا تَذۡهَبۡ نَفۡسُكَ عَلَيۡهِمۡ حَسَرَٰتٍۚ إِنَّ ٱللَّهَ عَلِيمُۢ بِمَا يَصۡنَعُونَ ۝٨ أَفَمَن زُیِّنَ لَهُۥ سُوۤءُ عَمَلِهِۦ فَرَءَاهُ حَسَنࣰاۖ فَإِنَّ ٱللَّهَ یُضِلُّ مَن یَشَاۤءُ وَیَهۡدِی مَن یَشَاۤءُۖ فَلَا تَذۡهَبۡ نَفۡسُكَ عَلَیۡهِمۡ حَسَرَٰتٍۚ إِنَّ ٱللَّهَ عَلِیمُۢ بِمَا یَصۡنَعُونَ ۝٨ اَفَمَنْ زُیِّنَ لَهٗ سُوْٓءُ عَمَلِهٖ فَرَاٰهُ حَسَنًا ؕ فَاِنَّ اللّٰهَ یُضِلُّ مَنْ یَّشَآءُ وَیَهْدِیْ مَنْ یَّشَآءُ ۖؗ فَلَا تَذْهَبْ نَفْسُكَ عَلَیْهِمْ حَسَرٰتٍ ؕ اِنَّ اللّٰهَ عَلِیْمٌۢ بِمَا یَصْنَعُوْنَ ۟ وَٱللَّهُ ٱلَّذِيٓ أَرۡسَلَ ٱلرِّيَٰحَ فَتُثِيرُ سَحَابٗا فَسُقۡنَٰهُ إِلَىٰ بَلَدٖ مَّيِّتٖ فَأَحۡيَيۡنَا بِهِ ٱلۡأَرۡضَ بَعۡدَ مَوۡتِهَاۚ كَذَٰلِكَ ٱلنُّشُورُ ۝٩ وَٱللَّهُ ٱلَّذِیۤ أَرۡسَلَ ٱلرِّیَٰحَ فَتُثِیرُ سَحَابࣰا فَسُقۡنَٰهُ إِلَىٰ بَلَدࣲ مَّیِّتࣲ فَأَحۡیَیۡنَا بِهِ ٱلۡأَرۡضَ بَعۡدَ مَوۡتِهَاۚ كَذَٰلِكَ ٱلنُّشُورُ ۝٩ وَاللّٰهُ الَّذِیْۤ اَرْسَلَ الرِّیٰحَ فَتُثِیْرُ سَحَابًا فَسُقْنٰهُ اِلٰی بَلَدٍ مَّیِّتٍ فَاَحْیَیْنَا بِهِ الْاَرْضَ بَعْدَ مَوْتِهَا ؕ كَذٰلِكَ النُّشُوْرُ ۟ مَن كَانَ يُرِيدُ ٱلۡعِزَّةَ فَلِلَّهِ ٱلۡعِزَّةُ جَمِيعًاۚ إِلَيۡهِ يَصۡعَدُ ٱلۡكَلِمُ ٱلطَّيِّبُ وَٱلۡعَمَلُ ٱلصَّٰلِحُ يَرۡفَعُهُۥۚ وَٱلَّذِينَ يَمۡكُرُونَ ٱلسَّيِّـَٔاتِ لَهُمۡ عَذَابٞ شَدِيدٞۖ وَمَكۡرُ أُوْلَٰٓئِكَ هُوَ يَبُورُ ۝١٠ مَن كَانَ یُرِیدُ ٱلۡعِزَّةَ فَلِلَّهِ ٱلۡعِزَّةُ جَمِیعًاۚ إِلَیۡهِ یَصۡعَدُ ٱلۡكَلِمُ ٱلطَّیِّبُ وَٱلۡعَمَلُ ٱلصَّٰلِحُ یَرۡفَعُهُۥۚ وَٱلَّذِینَ یَمۡكُرُونَ ٱلسَّیِّءَاتِ لَهُمۡ عَذَابࣱ شَدِیدࣱۖ وَمَكۡرُ أُو۟لَٰۤئِكَ هُوَ یَبُورُ ۝١٠ مَنْ كَانَ یُرِیْدُ الْعِزَّةَ فَلِلّٰهِ الْعِزَّةُ جَمِیْعًا ؕ اِلَیْهِ یَصْعَدُ الْكَلِمُ الطَّیِّبُ وَالْعَمَلُ الصَّالِحُ یَرْفَعُهٗ ؕ وَالَّذِیْنَ یَمْكُرُوْنَ السَّیِّاٰتِ لَهُمْ عَذَابٌ شَدِیْدٌ ؕ وَمَكْرُ اُولٰٓىِٕكَ هُوَ یَبُوْرُ ۟ وَٱللَّهُ خَلَقَكُم مِّن تُرَابٖ ثُمَّ مِن نُّطۡفَةٖ ثُمَّ جَعَلَكُمۡ أَزۡوَٰجٗاۚ وَمَا تَحۡمِلُ مِنۡ أُنثَىٰ وَلَا تَضَعُ إِلَّا بِعِلۡمِهِۦۚ وَمَا يُعَمَّرُ مِن مُّعَمَّرٖ وَلَا يُنقَصُ مِنۡ عُمُرِهِۦٓ إِلَّا فِي كِتَٰبٍۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٞ ۝١١ وَٱللَّهُ خَلَقَكُم مِّن تُرَابࣲ ثُمَّ مِن نُّطۡفَةࣲ ثُمَّ جَعَلَكُمۡ أَزۡوَٰجࣰاۚ وَمَا تَحۡمِلُ مِنۡ أُنثَىٰ وَلَا تَضَعُ إِلَّا بِعِلۡمِهِۦۚ وَمَا یُعَمَّرُ مِن مُّعَمَّرࣲ وَلَا یُنقَصُ مِنۡ عُمُرِهِۦۤ إِلَّا فِی كِتَٰبٍۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ یَسِیرࣱ ۝١١ وَاللّٰهُ خَلَقَكُمْ مِّنْ تُرَابٍ ثُمَّ مِنْ نُّطْفَةٍ ثُمَّ جَعَلَكُمْ اَزْوَاجًا ؕ وَمَا تَحْمِلُ مِنْ اُ وَلَا تَضَعُ اِلَّا بِعِلْمِهٖ ؕ وَمَا یُعَمَّرُ مِنْ مُّعَمَّرٍ وَّلَا یُنْقَصُ مِنْ عُمُرِهٖۤ اِلَّا فِیْ كِتٰبٍ ؕ اِنَّ ذٰلِكَ عَلَی اللّٰهِ یَسِیْرٌ ۟ وَمَا يَسۡتَوِي ٱلۡبَحۡرَانِ هَٰذَا عَذۡبٞ فُرَاتٞ سَآئِغٞ شَرَابُهُۥ وَهَٰذَا مِلۡحٌ أُجَاجٞۖ وَمِن كُلّٖ تَأۡكُلُونَ لَحۡمٗا طَرِيّٗا وَتَسۡتَخۡرِجُونَ حِلۡيَةٗ تَلۡبَسُونَهَاۖ وَتَرَى ٱلۡفُلۡكَ فِيهِ مَوَاخِرَ لِتَبۡتَغُواْ مِن فَضۡلِهِۦ وَلَعَلَّكُمۡ تَشۡكُرُونَ ۝١٢ وَمَا یَسۡتَوِی ٱلۡبَحۡرَانِ هَٰذَا عَذۡبࣱ فُرَاتࣱ سَاۤئِغࣱ شَرَابُهُۥ وَهَٰذَا مِلۡحٌ أُجَاجࣱۖ وَمِن كُلࣲّ تَأۡكُلُونَ لَحۡمࣰا طَرِیࣰّا وَتَسۡتَخۡرِجُونَ حِلۡیَةࣰ تَلۡبَسُونَهَاۖ وَتَرَى ٱلۡفُلۡكَ فِیهِ مَوَاخِرَ لِتَبۡتَغُوا۟ مِن فَضۡلِهِۦ وَلَعَلَّكُمۡ تَشۡكُرُونَ ۝١٢ وَمَا یَسْتَوِی الْبَحْرٰنِ ۖۗ هٰذَا عَذْبٌ فُرَاتٌ سَآىِٕغٌ شَرَابُهٗ وَهٰذَا مِلْحٌ اُجَاجٌ ؕ وَمِنْ كُلٍّ تَاْكُلُوْنَ لَحْمًا طَرِیًّا وَّتَسْتَخْرِجُوْنَ حِلْیَةً تَلْبَسُوْنَهَا ۚ وَتَرَی الْفُلْكَ فِیْهِ مَوَاخِرَ لِتَبْتَغُوْا مِنْ فَضْلِهٖ وَلَعَلَّكُمْ تَشْكُرُوْنَ ۟ يُولِجُ ٱلَّيۡلَ فِي ٱلنَّهَارِ وَيُولِجُ ٱلنَّهَارَ فِي ٱلَّيۡلِ وَسَخَّرَ ٱلشَّمۡسَ وَٱلۡقَمَرَۖ كُلّٞ يَجۡرِي لِأَجَلٖ مُّسَمّٗىۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمۡ لَهُ ٱلۡمُلۡكُۚ وَٱلَّذِينَ تَدۡعُونَ مِن دُونِهِۦ مَا يَمۡلِكُونَ مِن قِطۡمِيرٍ ۝١٣ یُولِجُ ٱلَّیۡلَ فِی ٱلنَّهَارِ وَیُولِجُ ٱلنَّهَارَ فِی ٱلَّیۡلِ وَسَخَّرَ ٱلشَّمۡسَ وَٱلۡقَمَرَۖ كُلࣱّ یَجۡرِی لِأَجَلࣲ مُّسَمࣰّىۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمۡ لَهُ ٱلۡمُلۡكُۚ وَٱلَّذِینَ تَدۡعُونَ مِن دُونِهِۦ مَا یَمۡلِكُونَ مِن قِطۡمِیرٍ ۝١٣ یُوْلِجُ الَّیْلَ فِی النَّهَارِ وَیُوْلِجُ النَّهَارَ فِی الَّیْلِ ۙ وَسَخَّرَ الشَّمْسَ وَالْقَمَرَ ۖؗ كُلٌّ یَّجْرِیْ لِاَجَلٍ مُّسَمًّی ؕ ذٰلِكُمُ اللّٰهُ رَبُّكُمْ لَهُ الْمُلْكُ ؕ وَالَّذِیْنَ تَدْعُوْنَ مِنْ دُوْنِهٖ مَا یَمْلِكُوْنَ مِنْ قِطْمِیْرٍ ۟ؕ إِن تَدۡعُوهُمۡ لَا يَسۡمَعُواْ دُعَآءَكُمۡ وَلَوۡ سَمِعُواْ مَا ٱسۡتَجَابُواْ لَكُمۡۖ وَيَوۡمَ ٱلۡقِيَٰمَةِ يَكۡفُرُونَ بِشِرۡكِكُمۡۚ وَلَا يُنَبِّئُكَ مِثۡلُ خَبِيرٖ ۝١٤ إِن تَدۡعُوهُمۡ لَا یَسۡمَعُوا۟ دُعَاۤءَكُمۡ وَلَوۡ سَمِعُوا۟ مَا ٱسۡتَجَابُوا۟ لَكُمۡۖ وَیَوۡمَ ٱلۡقِیَٰمَةِ یَكۡفُرُونَ بِشِرۡكِكُمۡۚ وَلَا یُنَبِّئُكَ مِثۡلُ خَبِیرࣲ ۝١٤ اِنْ تَدْعُوْهُمْ لَا یَسْمَعُوْا دُعَآءَكُمْ ۚ وَلَوْ سَمِعُوْا مَا اسْتَجَابُوْا لَكُمْ ؕ وَیَوْمَ الْقِیٰمَةِ یَكْفُرُوْنَ بِشِرْكِكُمْ ؕ وَلَا یُنَبِّئُكَ مِثْلُ خَبِیْرٍ ۟۠ يَٰٓأَيُّهَا ٱلنَّاسُ أَنتُمُ ٱلۡفُقَرَآءُ إِلَى ٱللَّهِۖ وَٱللَّهُ هُوَ ٱلۡغَنِيُّ ٱلۡحَمِيدُ ۝١٥ یَٰۤأَیُّهَا ٱلنَّاسُ أَنتُمُ ٱلۡفُقَرَاۤءُ إِلَى ٱللَّهِۖ وَٱللَّهُ هُوَ ٱلۡغَنِیُّ ٱلۡحَمِیدُ ۝١٥ یٰۤاَیُّهَا النَّاسُ اَنْتُمُ الْفُقَرَآءُ اِلَی اللّٰهِ ۚ وَاللّٰهُ هُوَ الْغَنِیُّ الْحَمِیْدُ ۟ إِن يَشَأۡ يُذۡهِبۡكُمۡ وَيَأۡتِ بِخَلۡقٖ جَدِيدٖ ۝١٦ إِن یَشَأۡ یُذۡهِبۡكُمۡ وَیَأۡتِ بِخَلۡقࣲ جَدِیدࣲ ۝١٦ اِنْ یَّشَاْ یُذْهِبْكُمْ وَیَاْتِ بِخَلْقٍ جَدِیْدٍ ۟ۚ وَمَا ذَٰلِكَ عَلَى ٱللَّهِ بِعَزِيزٖ ۝١٧ وَمَا ذَٰلِكَ عَلَى ٱللَّهِ بِعَزِیزࣲ ۝١٧ وَمَا ذٰلِكَ عَلَی اللّٰهِ بِعَزِیْزٍ ۟ وَلَا تَزِرُ وَازِرَةٞ وِزۡرَ أُخۡرَىٰۚ وَإِن تَدۡعُ مُثۡقَلَةٌ إِلَىٰ حِمۡلِهَا لَا يُحۡمَلۡ مِنۡهُ شَيۡءٞ وَلَوۡ كَانَ ذَا قُرۡبَىٰٓۗ إِنَّمَا تُنذِرُ ٱلَّذِينَ يَخۡشَوۡنَ رَبَّهُم بِٱلۡغَيۡبِ وَأَقَامُواْ ٱلصَّلَوٰةَۚ وَمَن تَزَكَّىٰ فَإِنَّمَا يَتَزَكَّىٰ لِنَفۡسِهِۦۚ وَإِلَى ٱللَّهِ ٱلۡمَصِيرُ ۝١٨ وَلَا تَزِرُ وَازِرَةࣱ وِزۡرَ أُخۡرَىٰۚ وَإِن تَدۡعُ مُثۡقَلَةٌ إِلَىٰ حِمۡلِهَا لَا یُحۡمَلۡ مِنۡهُ شَیۡءࣱ وَلَوۡ كَانَ ذَا قُرۡبَىٰۤۗ إِنَّمَا تُنذِرُ ٱلَّذِینَ یَخۡشَوۡنَ رَبَّهُم بِٱلۡغَیۡبِ وَأَقَامُوا۟ ٱلصَّلَوٰةَۚ وَمَن تَزَكَّىٰ فَإِنَّمَا یَتَزَكَّىٰ لِنَفۡسِهِۦۚ وَإِلَى ٱللَّهِ ٱلۡمَصِیرُ ۝١٨ وَلَا تَزِرُ وَازِرَةٌ وِّزْرَ اُخْرٰی ؕ وَاِنْ تَدْعُ مُثْقَلَةٌ اِلٰی حِمْلِهَا لَا یُحْمَلْ مِنْهُ شَیْءٌ وَّلَوْ كَانَ ذَا قُرْبٰی ؕ اِنَّمَا تُنْذِرُ الَّذِیْنَ یَخْشَوْنَ رَبَّهُمْ بِالْغَیْبِ وَاَقَامُوا الصَّلٰوةَ ؕ وَمَنْ تَزَكّٰی فَاِنَّمَا یَتَزَكّٰی لِنَفْسِهٖ ؕ وَاِلَی اللّٰهِ الْمَصِیْرُ ۟ وَمَا يَسۡتَوِي ٱلۡأَعۡمَىٰ وَٱلۡبَصِيرُ ۝١٩ وَمَا یَسۡتَوِی ٱلۡأَعۡمَىٰ وَٱلۡبَصِیرُ ۝١٩ وَمَا یَسْتَوِی الْاَعْمٰی وَالْبَصِیْرُ ۟ۙ وَلَا ٱلظُّلُمَٰتُ وَلَا ٱلنُّورُ ۝٢٠ وَلَا ٱلظُّلُمَٰتُ وَلَا ٱلنُّورُ ۝٢٠ وَلَا الظُّلُمٰتُ وَلَا النُّوْرُ ۟ۙ وَلَا ٱلظِّلُّ وَلَا ٱلۡحَرُورُ ۝٢١ وَلَا ٱلظِّلُّ وَلَا ٱلۡحَرُورُ ۝٢١ وَلَا الظِّلُّ وَلَا الْحَرُوْرُ ۟ۚ وَمَا يَسۡتَوِي ٱلۡأَحۡيَآءُ وَلَا ٱلۡأَمۡوَٰتُۚ إِنَّ ٱللَّهَ يُسۡمِعُ مَن يَشَآءُۖ وَمَآ أَنتَ بِمُسۡمِعٖ مَّن فِي ٱلۡقُبُورِ ۝٢٢ وَمَا یَسۡتَوِی ٱلۡأَحۡیَاۤءُ وَلَا ٱلۡأَمۡوَٰتُۚ إِنَّ ٱللَّهَ یُسۡمِعُ مَن یَشَاۤءُۖ وَمَاۤ أَنتَ بِمُسۡمِعࣲ مَّن فِی ٱلۡقُبُورِ ۝٢٢ وَمَا یَسْتَوِی الْاَحْیَآءُ وَلَا الْاَمْوَاتُ ؕ اِنَّ اللّٰهَ یُسْمِعُ مَنْ یَّشَآءُ ۚ وَمَاۤ اَنْتَ بِمُسْمِعٍ مَّنْ فِی الْقُبُوْرِ ۟ ٱلَّذِینَ مِن قَبۡلِهِمۡ جَاۤءَتۡهُمۡ رُسُلُهُم بِٱلۡبَیِّنَٰتِ وَبِٱلزُّبُرِ وَبِٱلۡكِتَٰبِ ٱلۡمُنِیرِ ۝٢٥ وَاِنْ یُّكَذِّبُوْكَ فَقَدْ كَذَّبَ الَّذِیْنَ مِنْ قَبْلِهِمْ ۚ جَآءَتْهُمْ رُسُلُهُمْ مِمَّا رَزَقْنٰهُمْ سِرًّا وَّعَلَانِیَةً یَّرْجُوْنَ تِجَارَةً لَّنْ تَبُوْرَ ۟ۙ لِيُوَفِّيَهُمۡ أُجُورَهُمۡ وَيَزِيدَهُم مِّن فَضۡلِهِۦٓۚ إِنَّهُۥ غَفُورٞ شَكُورٞ ۝٣٠ لِیُوَفِّیَهُمۡ أُجُورَهُمۡ وَیَزِیدَهُم مِّن فَضۡلِهِۦۤۚ إِنَّهُۥ غَفُورࣱ شَكُورࣱ ۝٣٠ لِیُوَفِّیَهُمْ اُجُوْرَهُمْ وَیَزِیْدَهُمْ مِّنْ فَضْلِهٖ ؕ اِنَّهٗ غَفُوْرٌ شَكُوْرٌ ۟ وَٱلَّذِيٓ أَوۡحَيۡنَآ إِلَيۡكَ مِنَ ٱلۡكِتَٰبِ هُوَ ٱلۡحَقُّ مُصَدِّقٗا لِّمَا بَيۡنَ يَدَيۡهِۗ إِنَّ ٱللَّهَ بِعِبَادِهِۦ لَخَبِيرُۢ بَصِيرٞ ۝٣١ وَٱلَّذِیۤ أَوۡحَیۡنَاۤ إِلَیۡكَ مِنَ ٱلۡكِتَٰبِ هُوَ ٱلۡحَقُّ مُصَدِّقࣰا لِّمَا بَیۡنَ یَدَیۡهِۗ إِنَّ ٱللَّهَ بِعِبَادِهِۦ لَخَبِیرُۢ بَصِیرࣱ ۝٣١ وَالَّذِیْۤ اَوْحَیْنَاۤ اِلَیْكَ مِنَ الْكِتٰبِ هُوَ الْحَقُّ مُصَدِّقًا لِّمَا بَیْنَ یَدَیْهِ ؕ اِنَّ اللّٰهَ بِعِبَادِهٖ لَخَبِیْرٌ بَصِیْرٌ ۟ ثُمَّ أَوۡرَثۡنَا ٱلۡكِتَٰبَ ٱلَّذِينَ ٱصۡطَفَيۡنَا مِنۡ عِبَادِنَاۖ فَمِنۡهُمۡ ظَالِمٞ لِّنَفۡسِهِۦ وَمِنۡهُم مُّقۡتَصِدٞ وَمِنۡهُمۡ سَابِقُۢ بِٱلۡخَيۡرَٰتِ بِإِذۡنِ ٱللَّهِۚ ذَٰلِكَ هُوَ ٱلۡفَضۡلُ ٱلۡكَبِيرُ ۝٣٢ ثُمَّ أَوۡرَثۡنَا ٱلۡكِتَٰبَ ٱلَّذِینَ ٱصۡطَفَیۡنَا مِنۡ عِبَادِنَاۖ فَمِنۡهُمۡ ظَالِمࣱ لِّنَفۡسِهِۦ وَمِنۡهُم مُّقۡتَصِدࣱ وَمِنۡهُمۡ سَابِقُۢ بِٱلۡخَیۡرَٰتِ بِإِذۡنِ ٱللَّهِۚ ذَٰلِكَ هُوَ ٱلۡفَضۡلُ ٱلۡكَبِیرُ ۝٣٢ ثُمَّ اَوْرَثْنَا الْكِتٰبَ الَّذِیْنَ اصْطَفَیْنَا مِنْ عِبَادِنَا ۚ فَمِنْهُمْ ظَالِمٌ لِّنَفْسِهٖ ۚ وَمِنْهُمْ مُّقْتَصِدٌ ۚ وَمِنْهُمْ سَابِقٌ بِالْخَیْرٰتِ بِاِذْنِ اللّٰهِ ؕ ذٰلِكَ هُوَ الْفَضْلُ الْكَبِیْرُ ۟ؕ جَنَّٰتُ عَدۡنٖ يَدۡخُلُونَهَا يُحَلَّوۡنَ فِيهَا مِنۡ أَسَاوِرَ مِن ذَهَبٖ وَلُؤۡلُؤٗاۖ وَلِبَاسُهُمۡ فِيهَا حَرِيرٞ ۝٣٣ جَنَّٰتُ عَدۡنࣲ یَدۡخُلُونَهَا یُحَلَّوۡنَ فِیهَا مِنۡ أَسَاوِرَ مِن ذَهَبࣲ وَلُؤۡلُؤࣰاۖ وَلِبَاسُهُمۡ فِیهَا حَرِیرࣱ ۝٣٣ جَنّٰتُ عَدْنٍ یَّدْخُلُوْنَهَا یُحَلَّوْنَ فِیْهَا مِنْ اَسَاوِرَ مِنْ ذَهَبٍ وَّلُؤْلُؤًا ۚ وَلِبَاسُهُمْ فِیْهَا حَرِیْرٌ ۟ وَقَالُواْ ٱلۡحَمۡدُ لِلَّهِ ٱلَّذِيٓ أَذۡهَبَ عَنَّا ٱلۡحَزَنَۖ إِنَّ رَبَّنَا لَغَفُورٞ شَكُورٌ ۝٣٤ وَقَالُوا۟ ٱلۡحَمۡدُ لِلَّهِ ٱلَّذِیۤ أَذۡهَبَ عَنَّا ٱلۡحَزَنَۖ إِنَّ رَبَّنَا لَغَفُورࣱ شَكُورٌ ۝٣٤ وَقَالُوا الْحَمْدُ لِلّٰهِ الَّذِیْۤ اَذْهَبَ عَنَّا الْحَزَنَ ؕ اِنَّ رَبَّنَا لَغَفُوْرٌ شَكُوْرُ ۟ۙ ٱلَّذِيٓ أَحَلَّنَا دَارَ ٱلۡمُقَامَةِ مِن فَضۡلِهِۦ لَا يَمَسُّنَا فِيهَا نَصَبٞ وَلَا يَمَسُّنَا فِيهَا لُغُوبٞ ۝٣٥ ٱلَّذِیۤ أَحَلَّنَا دَارَ ٱلۡمُقَامَةِ مِن فَضۡلِهِۦ لَا یَمَسُّنَا فِیهَا نَصَبࣱ وَلَا یَمَسُّنَا فِیهَا لُغُوبࣱ ۝٣٥ لَّذِیْۤ اَحَلَّنَا دَارَ الْمُقَامَةِ مِنْ فَضْلِهٖ ۚ لَا یَمَسُّنَا فِیْهَا نَصَبٌ وَّلَا یَمَسُّنَا فِیْهَا لُغُوْبٌ ۟ وَٱلَّذِينَ كَفَرُواْ لَهُمۡ نَارُ جَهَنَّمَ لَا يُقۡضَىٰ عَلَيۡهِمۡ فَيَمُوتُواْ وَلَا يُخَفَّفُ عَنۡهُم مِّنۡ عَذَابِهَاۚ كَذَٰلِكَ نَجۡزِي كُلَّ كَفُورٖ ۝٣٦ وَٱلَّذِینَ كَفَرُوا۟ لَهُمۡ نَارُ جَهَنَّمَ لَا یُقۡضَىٰ عَلَیۡهِمۡ فَیَمُوتُوا۟ وَلَا یُخَفَّفُ عَنۡهُم مِّنۡ عَذَابِهَاۚ كَذَٰلِكَ نَجۡزِی كُلَّ كَفُورࣲ ۝٣٦ وَالَّذِیْنَ كَفَرُوْا لَهُمْ نَارُ جَهَنَّمَ ۚ لَا یُقْضٰی عَلَیْهِمْ فَیَمُوْتُوْا وَلَا یُخَفَّفُ عَنْهُمْ مِّنْ عَذَابِهَا ؕ كَذٰلِكَ نَجْزِیْ كُلَّ كَفُوْرٍ ۟ۚ وَهُمۡ يَصۡطَرِخُونَ فِيهَا رَبَّنَآ أَخۡرِجۡنَا نَعۡمَلۡ صَٰلِحًا غَيۡرَ ٱلَّذِي كُنَّا نَعۡمَلُۚ أَوَلَمۡ نُعَمِّرۡكُم مَّا يَتَذَكَّرُ فِيهِ مَن تَذَكَّرَ وَجَآءَكُمُ ٱلنَّذِيرُۖ فَذُوقُواْ فَمَا لِلظَّٰلِمِينَ مِن نَّصِيرٍ ۝٣٧ وَهُمۡ یَصۡطَرِخُونَ فِیهَا رَبَّنَاۤ أَخۡرِجۡنَا نَعۡمَلۡ صَٰلِحًا غَیۡرَ ٱلَّذِی كُنَّا نَعۡمَلُۚ أَوَلَمۡ نُعَمِّرۡكُم مَّا یَتَذَكَّرُ فِیهِ مَن تَذَكَّرَ وَجَاۤءَكُمُ ٱلنَّذِیرُۖ فَذُوقُوا۟ فَمَا لِلظَّٰلِمِینَ مِن نَّصِیرٍ ۝٣٧ وَهُمْ یَصْطَرِخُوْنَ فِیْهَا ۚ رَبَّنَاۤ اَخْرِجْنَا نَعْمَلْ صَالِحًا غَیْرَ الَّذِیْ كُنَّا نَعْمَلُ ؕ اَوَلَمْ نُعَمِّرْكُمْ مَّا یَتَذَكَّرُ فِیْهِ مَنْ تَذَكَّرَ وَجَآءَكُمُ النَّذِیْرُ ؕ فَذُوْقُوْا فَمَا لِلظّٰلِمِیْنَ مِنْ نَّصِیْرٍ ۟۠ إِنَّ ٱللَّهَ عَٰلِمُ غَيۡبِ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۚ إِنَّهُۥ عَلِيمُۢ بِذَاتِ ٱلصُّدُورِ ۝٣٨ إِنَّ ٱللَّهَ عَٰلِمُ غَیۡبِ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۚ إِنَّهُۥ عَلِیمُۢ بِذَاتِ ٱلصُّدُورِ ۝٣٨ اِنَّ اللّٰهَ عٰلِمُ غَیْبِ السَّمٰوٰتِ وَالْاَرْضِ ؕ اِنَّهٗ عَلِیْمٌۢ بِذَاتِ الصُّدُوْرِ ۟ هُوَ ٱلَّذِي جَعَلَكُمۡ خَلَٰٓئِفَ فِي ٱلۡأَرۡضِۚ فَمَن كَفَرَ فَعَلَيۡهِ كُفۡرُهُۥۖ وَلَا يَزِيدُ ٱلۡكَٰفِرِينَ كُفۡرُهُمۡ عِندَ رَبِّهِمۡ إِلَّا مَقۡتٗاۖ وَلَا يَزِيدُ ٱلۡكَٰفِرِينَ كُفۡرُهُمۡ إِلَّا خَسَارٗا ۝٣٩ هُوَ ٱلَّذِی جَعَلَكُمۡ خَلَٰۤئِفَ فِی ٱلۡأَرۡضِۚ فَمَن كَفَرَ فَعَلَیۡهِ كُفۡرُهُۥۖ وَلَا یَزِیدُ ٱلۡكَٰفِرِینَ كُفۡرُهُمۡ عِندَ رَبِّهِمۡ إِلَّا مَقۡتࣰاۖ وَلَا یَزِیدُ ٱلۡكَٰفِرِینَ كُفۡرُهُمۡ إِلَّا خَسَارࣰا ۝٣٩ هُوَ الَّذِیْ جَعَلَكُمْ خَلٰٓىِٕفَ فِی الْاَرْضِ ؕ فَمَنْ كَفَرَ فَعَلَیْهِ كُفْرُهٗ ؕ وَلَا یَزِیْدُ الْكٰفِرِیْنَ كُفْرُهُمْ عِنْدَ رَبِّهِمْ اِلَّا مَقْتًا ۚ وَلَا یَزِیْدُ الْكٰفِرِیْنَ كُفْرُهُمْ اِلَّا خَسَارًا ۟ قُلۡ أَرَءَيۡتُمۡ شُرَكَآءَكُمُ ٱلَّذِينَ تَدۡعُونَ مِن دُونِ ٱللَّهِ أَرُونِي مَاذَا خَلَقُواْ مِنَ ٱلۡأَرۡضِ أَمۡ لَهُمۡ شِرۡكٞ فِي ٱلسَّمَٰوَٰتِ أَمۡ ءَاتَيۡنَٰهُمۡ كِتَٰبٗا فَهُمۡ عَلَىٰ بَيِّنَتٖ مِّنۡهُۚ بَلۡ إِن يَعِدُ ٱلظَّٰلِمُونَ بَعۡضُهُم بَعۡضًا إِلَّا غُرُورًا ۝٤٠ قُلۡ أَرَءَیۡتُمۡ شُرَكَاۤءَكُمُ ٱلَّذِینَ تَدۡعُونَ مِن دُونِ ٱللَّهِ أَرُونِی مَاذَا خَلَقُوا۟ مِنَ ٱلۡأَرۡضِ أَمۡ لَهُمۡ شِرۡكࣱ فِی ٱلسَّمَٰوَٰتِ أَمۡ ءَاتَیۡنَٰهُمۡ كِتَٰبࣰا فَهُمۡ عَلَىٰ بَیِّنَتࣲ مِّنۡهُۚ بَلۡ إِن یَعِدُ ٱلظَّٰلِمُونَ بَعۡضُهُم بَعۡضًا إِلَّا غُرُورًا ۝٤٠ قُلْ اَرَءَیْتُمْ شُرَكَآءَكُمُ الَّذِیْنَ تَدْعُوْنَ مِنْ دُوْنِ اللّٰهِ ؕ اَرُوْنِیْ مَاذَا خَلَقُوْا مِنَ الْاَرْضِ اَمْ لَهُمْ شِرْكٌ فِی السَّمٰوٰتِ ۚ اَمْ اٰتَیْنٰهُمْ كِتٰبًا فَهُمْ عَلٰی بَیِّنَتٍ مِّنْهُ ۚ بَلْ اِنْ یَّعِدُ الظّٰلِمُوْنَ بَعْضُهُمْ بَعْضًا اِلَّا غُرُوْرًا ۟ إِنَّ ٱللَّهَ يُمۡسِكُ ٱلسَّمَٰوَٰتِ مِنْ اَحَدٍ مِّنْ بَعْدِهٖ ؕ اِنَّهٗ كَانَ حَلِیْمًا غَفُوْرًا ۟ وَأَقۡسَمُواْ بِٱللَّهِ جَهۡدَ أَيۡمَٰنِهِمۡ لَئِن جَآءَهُمۡ نَذِيرٞ لَّيَكُونُنَّ أَهۡدَىٰ مِنۡ إِحۡدَى ٱلۡأُمَمِۖ فَلَمَّا جَآءَهُمۡ نَذِيرٞ مَّا زَادَهُمۡ إِلَّا نُفُورًا ۝٤٢ وَأَقۡسَمُوا۟ بِٱللَّهِ جَهۡدَ أَیۡمَٰنِهِمۡ لَئِن جَاۤءَهُمۡ نَذِیرࣱ لَّیَكُونُنَّ أَهۡدَىٰ مِنۡ إِحۡدَى ٱلۡأُمَمِۖ فَلَمَّا جَاۤءَهُمۡ نَذِیرࣱ مَّا زَادَهُمۡ إِلَّا نُفُورًا ۝٤٢ وَاَقْسَمُوْا بِاللّٰهِ جَهْدَ اَیْمَانِهِمْ لَىِٕنْ جَآءَهُمْ نَذِیْرٌ لَّیَكُوْنُنَّ اَهْدٰی مِنْ اِحْدَی الْاُمَمِ ۚ فَلَمَّا جَآءَهُمْ نَذِیْرٌ مَّا زَادَهُمْ اِلَّا نُفُوْرَا ۟ۙ ٱسۡتِكۡبَارٗا فِي ٱلۡأَرۡضِ وَمَكۡرَ ٱلسَّيِّيِٕۚ وَلَا يَحِيقُ ٱلۡمَكۡرُ ٱلسَّيِّئُ إِلَّا بِأَهۡلِهِۦۚ فَهَلۡ يَنظُرُونَ إِلَّا سُنَّتَ ٱلۡأَوَّلِينَۚ فَلَن تَجِدَ لِسُنَّتِ ٱللَّهِ تَبۡدِيلٗاۖ وَلَن تَجِدَ لِسُنَّتِ ٱللَّهِ تَحۡوِيلًا ۝٤٣ ٱسۡتِكۡبَارࣰا فِی ٱلۡأَرۡضِ وَمَكۡرَ ٱلسَّیِّئِۚ وَلَا یَحِیقُ ٱلۡمَكۡرُ ٱلسَّیِّئُ إِلَّا بِأَهۡلِهِۦۚ فَهَلۡ یَنظُرُونَ إِلَّا سُنَّتَ ٱلۡأَوَّلِینَۚ فَلَن تَجِدَ لِسُنَّتِ ٱللَّهِ تَبۡدِیلࣰاۖ وَلَن تَجِدَ لِسُنَّتِ ٱللَّهِ تَحۡوِیلًا ۝٤٣ سْتِكْبَارًا فِی الْاَرْضِ وَمَكْرَ السَّیِّىءِ ؕ وَلَا یَحِیْقُ الْمَكْرُ السَّیِّئُ اِلَّا بِاَهْلِهٖ ؕ فَهَلْ یَنْظُرُوْنَ اِلَّا سُنَّتَ الْاَوَّلِیْنَ ۚ فَلَنْ تَجِدَ لِسُنَّتِ اللّٰهِ تَبْدِیْلًا ۚ۬ وَلَنْ تَجِدَ لِسُنَّتِ اللّٰهِ تَحْوِیْلًا ۟ أَوَلَمۡ يَسِيرُواْ فِي ٱلۡأَرۡضِ فَيَنظُرُواْ كَيۡفَ كَانَ عَٰقِبَةُ ٱلَّذِينَ مِن قَبۡلِهِمۡ وَكَانُوٓاْ أَشَدَّ مِنۡهُمۡ قُوَّةٗۚ وَمَا كَانَ ٱللَّهُ لِيُعۡجِزَهُۥ مِن شَيۡءٖ فِي ٱلسَّمَٰوَٰتِ وَلَا فِي ٱلۡأَرۡضِۚ إِنَّهُۥ كَانَ عَلِيمٗا قَدِيرٗا ۝٤٤ أَوَلَمۡ یَسِیرُوا۟ فِی ٱلۡأَرۡضِ فَیَنظُرُوا۟ كَیۡفَ كَانَ عَٰقِبَةُ ٱلَّذِینَ مِن قَبۡلِهِمۡ وَكَانُوۤا۟ أَشَدَّ مِنۡهُمۡ قُوَّةࣰۚ وَمَا كَانَ ٱللَّهُ لِیُعۡجِزَهُۥ مِن شَیۡءࣲ فِی ٱلسَّمَٰوَٰتِ وَلَا فِی ٱلۡأَرۡضِۚ إِنَّهُۥ كَانَ عَلِیمࣰا قَدِیرࣰا ۝٤٤ اَوَلَمْ یَسِیْرُوْا فِی الْاَرْضِ فَیَنْظُرُوْا كَیْفَ كَانَ عَاقِبَةُ الَّذِیْنَ مِنْ قَبْلِهِمْ وَكَانُوْۤا اَشَدَّ مِنْهُمْ قُوَّةً ؕ وَمَا كَانَ اللّٰهُ لِیُعْجِزَهٗ مِنْ شَیْءٍ فِی السَّمٰوٰتِ وَلَا فِی الْاَرْضِ ؕ اِنَّهٗ كَانَ عَلِیْمًا قَدِیْرًا ۟ وَلَوۡ يُؤَاخِذُ ٱللَّهُ ٱلنَّاسَ بِمَا كَسَبُواْ مَا تَرَكَ عَلَىٰ ظَهۡرِهَا مِن دَآبَّةٖ وَلَٰكِن يُؤَخِّرُهُمۡ إِلَىٰٓ أَجَلٖ مُّسَمّٗىۖ فَإِذَا جَآءَ أَجَلُهُمۡ فَإِنَّ ٱللَّهَ كَانَ بِعِبَادِهِۦ بَصِيرَۢا ۝٤٥ وَلَوۡ یُؤَاخِذُ ٱللَّهُ ٱلنَّاسَ بِمَا كَسَبُوا۟ مَا تَرَكَ عَلَىٰ ظَهۡرِهَا مِن دَاۤبَّةࣲ وَلَٰكِن یُؤَخِّرُهُمۡ إِلَىٰۤ أَجَلࣲ مُّسَمࣰّىۖ فَإِذَا جَاۤءَ أَجَلُهُمۡ فَإِنَّ ٱللَّهَ كَانَ بِعِبَادِهِۦ بَصِیرَۢا ۝٤٥ وَلَوْ یُؤَاخِذُ اللّٰهُ النَّاسَ بِمَا كَسَبُوْا مَا تَرَكَ عَلٰی ظَهْرِهَا مِنْ دَآبَّةٍ وَّلٰكِنْ یُّؤَخِّرُهُمْ اِلٰۤی اَجَلٍ مُّسَمًّی ۚ فَاِذَا جَآءَ اَجَلُهُمْ فَاِنَّ اللّٰهَ كَانَ بِعِبَادِهٖ بَصِیْرًا ۟۠
