Surah 37 · Meccan · 182 ayat

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

وَٱلصَّٰٓفَّٰتِ صَفّٗا ۝١ وَٱلصَّٰۤفَّٰتِ صَفࣰّا ۝١ وَالصّٰٓفّٰتِ صَفًّا ۟ۙ فَٱلزَّٰجِرَٰتِ زَجۡرٗا ۝٢ فَٱلزَّٰجِرَٰتِ زَجۡرࣰا ۝٢ فَالزّٰجِرٰتِ زَجْرًا ۟ۙ فَٱلتَّٰلِيَٰتِ ذِكۡرًا ۝٣ فَٱلتَّٰلِیَٰتِ ذِكۡرًا ۝٣ فَالتّٰلِیٰتِ ذِكْرًا ۟ۙ إِنَّ إِلَٰهَكُمۡ لَوَٰحِدٞ ۝٤ إِنَّ إِلَٰهَكُمۡ لَوَٰحِدࣱ ۝٤ اِنَّ اِلٰهَكُمْ لَوَاحِدٌ ۟ؕ رَّبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَيۡنَهُمَا وَرَبُّ ٱلۡمَشَٰرِقِ ۝٥ رَّبُّ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا بَیۡنَهُمَا وَرَبُّ ٱلۡمَشَٰرِقِ ۝٥ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ وَمَا بَیْنَهُمَا وَرَبُّ الْمَشَارِقِ ۟ؕ إِنَّا زَيَّنَّا ٱلسَّمَآءَ ٱلدُّنۡيَا بِزِينَةٍ ٱلۡكَوَاكِبِ ۝٦ إِنَّا زَیَّنَّا ٱلسَّمَاۤءَ ٱلدُّنۡیَا بِزِینَةٍ ٱلۡكَوَاكِبِ ۝٦ اِنَّا زَیَّنَّا السَّمَآءَ الدُّنْیَا بِزِیْنَةِ لْكَوَاكِبِ ۟ۙ وَحِفۡظٗا مِّن كُلِّ شَيۡطَٰنٖ مَّارِدٖ ۝٧ وَحِفۡظࣰا مِّن كُلِّ شَیۡطَٰنࣲ مَّارِدࣲ ۝٧ وَحِفْظًا مِّنْ كُلِّ شَیْطٰنٍ مَّارِدٍ ۟ۚ لَّا يَسَّمَّعُونَ إِلَى ٱلۡمَلَإِ ٱلۡأَعۡلَىٰ وَيُقۡذَفُونَ مِن كُلِّ جَانِبٖ ۝٨ لَّا یَسَّمَّعُونَ إِلَى ٱلۡمَلَإِ ٱلۡأَعۡلَىٰ وَیُقۡذَفُونَ مِن كُلِّ جَانِبࣲ ۝٨ لَا یَسَّمَّعُوْنَ اِلَی الْمَلَاِ الْاَعْلٰی وَیُقْذَفُوْنَ مِنْ كُلِّ جَانِبٍ ۟ۗۖ دُحُورٗاۖ وَلَهُمۡ عَذَابٞ وَاصِبٌ ۝٩ دُحُورࣰاۖ وَلَهُمۡ عَذَابࣱ وَاصِبٌ ۝٩ دُحُوْرًا وَّلَهُمْ عَذَابٌ وَّاصِبٌ ۟ۙ إِلَّا مَنۡ خَطِفَ ٱلۡخَطۡفَةَ فَأَتۡبَعَهُۥ شِهَابٞ ثَاقِبٞ ۝١٠ إِلَّا مَنۡ خَطِفَ ٱلۡخَطۡفَةَ فَأَتۡبَعَهُۥ شِهَابࣱ ثَاقِبࣱ ۝١٠ اِلَّا مَنْ خَطِفَ الْخَطْفَةَ فَاَتْبَعَهٗ شِهَابٌ ثَاقِبٌ ۟ فَٱسۡتَفۡتِهِمۡ أَهُمۡ أَشَدُّ خَلۡقًا أَم مَّنۡ خَلَقۡنَآۚ إِنَّا خَلَقۡنَٰهُم مِّن طِينٖ لَّازِبِۭ ۝١١ فَٱسۡتَفۡتِهِمۡ أَهُمۡ أَشَدُّ خَلۡقًا أَم مَّنۡ خَلَقۡنَاۤۚ إِنَّا خَلَقۡنَٰهُم مِّن طِینࣲ لَّازِبِۭ ۝١١ فَاسْتَفْتِهِمْ اَهُمْ اَشَدُّ خَلْقًا اَمْ مَّنْ خَلَقْنَا ؕ اِنَّا خَلَقْنٰهُمْ مِّنْ طِیْنٍ لَّازِبٍ ۟ بَلۡ عَجِبۡتَ وَيَسۡخَرُونَ ۝١٢ بَلۡ عَجِبۡتَ وَیَسۡخَرُونَ ۝١٢ بَلْ عَجِبْتَ وَیَسْخَرُوْنَ ۪۟ وَإِذَا ذُكِّرُواْ لَا يَذۡكُرُونَ ۝١٣ وَإِذَا ذُكِّرُوا۟ لَا یَذۡكُرُونَ ۝١٣ وَاِذَا ذُكِّرُوْا لَا یَذْكُرُوْنَ ۪۟ وَإِذَا رَأَوۡاْ ءَايَةٗ يَسۡتَسۡخِرُونَ ۝١٤ وَإِذَا رَأَوۡا۟ ءَایَةࣰ یَسۡتَسۡخِرُونَ ۝١٤ وَاِذَا رَاَوْا اٰیَةً یَّسْتَسْخِرُوْنَ ۪۟ وَقَالُوٓاْ إِنۡ هَٰذَآ إِلَّا سِحۡرٞ مُّبِينٌ ۝١٥ وَقَالُوۤا۟ إِنۡ هَٰذَاۤ إِلَّا سِحۡرࣱ مُّبِینٌ ۝١٥ وَقَالُوْۤا اِنْ هٰذَاۤ اِلَّا سِحْرٌ مُّبِیْنٌ ۟ۚۖ أَءِذَا مِتۡنَا وَكُنَّا تُرَابٗا وَعِظَٰمًا أَءِنَّا لَمَبۡعُوثُونَ ۝١٦ أَءِذَا مِتۡنَا وَكُنَّا تُرَابࣰا وَعِظَٰمًا أَءِنَّا لَمَبۡعُوثُونَ ۝١٦ ءَاِذَا مِتْنَا وَكُنَّا تُرَابًا وَّعِظَامًا ءَاِنَّا لَمَبْعُوْثُوْنَ ۟ۙ أَوَءَابَآؤُنَا ٱلۡأَوَّلُونَ ۝١٧ أَوَءَابَاۤؤُنَا ٱلۡأَوَّلُونَ ۝١٧ اَوَاٰبَآؤُنَا الْاَوَّلُوْنَ ۟ؕ قُلۡ نَعَمۡ وَأَنتُمۡ دَٰخِرُونَ ۝١٨ قُلۡ نَعَمۡ وَأَنتُمۡ دَٰخِرُونَ ۝١٨ قُلْ نَعَمْ وَاَنْتُمْ دَاخِرُوْنَ ۟ۚ فَإِنَّمَا هِيَ زَجۡرَةٞ وَٰحِدَةٞ فَإِذَا هُمۡ يَنظُرُونَ ۝١٩ فَإِنَّمَا هِیَ زَجۡرَةࣱ وَٰحِدَةࣱ فَإِذَا هُمۡ یَنظُرُونَ ۝١٩ فَاِنَّمَا هِیَ زَجْرَةٌ وَّاحِدَةٌ فَاِذَا هُمْ یَنْظُرُوْنَ ۟ وَقَالُواْ يَٰوَيۡلَنَا هَٰذَا يَوۡمُ ٱلدِّينِ ۝٢٠ وَقَالُوا۟ یَٰوَیۡلَنَا هَٰذَا یَوۡمُ ٱلدِّینِ ۝٢٠ وَقَالُوْا یٰوَیْلَنَا هٰذَا یَوْمُ الدِّیْنِ ۟ هَٰذَا يَوۡمُ ٱلۡفَصۡلِ ٱلَّذِي كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢١ هَٰذَا یَوۡمُ ٱلۡفَصۡلِ ٱلَّذِی كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢١ هٰذَا یَوْمُ الْفَصْلِ الَّذِیْ كُنْتُمْ بِهٖ تُكَذِّبُوْنَ ۟۠ ٱحۡشُرُواْ ٱلَّذِينَ ظَلَمُواْ وَأَزۡوَٰجَهُمۡ وَمَا كَانُواْ يَعۡبُدُونَ ۝٢٢ ٱحۡشُرُوا۟ ٱلَّذِینَ ظَلَمُوا۟ وَأَزۡوَٰجَهُمۡ وَمَا كَانُوا۟ یَعۡبُدُونَ ۝٢٢ اُحْشُرُوا الَّذِیْنَ ظَلَمُوْا وَاَزْوَاجَهُمْ وَمَا كَانُوْا یَعْبُدُوْنَ ۟ۙ مِن دُونِ ٱللَّهِ فَٱهۡدُوهُمۡ إِلَىٰ صِرَٰطِ ٱلۡجَحِيمِ ۝٢٣ مِن دُونِ ٱللَّهِ فَٱهۡدُوهُمۡ إِلَىٰ صِرَٰطِ ٱلۡجَحِیمِ ۝٢٣ مِنْ دُوْنِ اللّٰهِ فَاهْدُوْهُمْ اِلٰی صِرَاطِ الْجَحِیْمِ ۟ وَقِفُوهُمۡۖ إِنَّهُم مَّسۡـُٔولُونَ ۝٢٤ وَقِفُوهُمۡۖ إِنَّهُم مَّسۡءُولُونَ ۝٢٤ وَقِفُوْهُمْ اِنَّهُمْ مَّسْـُٔوْلُوْنَ ۟ۙ مَا لَكُمۡ لَا تَنَاصَرُونَ ۝٢٥ مَا لَكُمۡ لَا تَنَاصَرُونَ ۝٢٥ مَا لَكُمْ لَا تَنَاصَرُوْنَ ۟ بَلۡ هُمُ ٱلۡيَوۡمَ مُسۡتَسۡلِمُونَ ۝٢٦ بَلۡ هُمُ ٱلۡیَوۡمَ مُسۡتَسۡلِمُونَ ۝٢٦ بَلْ هُمُ الْیَوْمَ مُسْتَسْلِمُوْنَ ۟ وَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضٖ يَتَسَآءَلُونَ ۝٢٧ وَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضࣲ یَتَسَاۤءَلُونَ ۝٢٧ وَاَقْبَلَ بَعْضُهُمْ عَلٰی بَعْضٍ یَّتَسَآءَلُوْنَ ۟ قَالُوٓاْ إِنَّكُمۡ كُنتُمۡ تَأۡتُونَنَا عَنِ ٱلۡيَمِينِ ۝٢٨ قَالُوۤا۟ إِنَّكُمۡ كُنتُمۡ تَأۡتُونَنَا عَنِ ٱلۡیَمِینِ ۝٢٨ قَالُوْۤا اِنَّكُمْ كُنْتُمْ تَاْتُوْنَنَا عَنِ الْیَمِیْنِ ۟ قَالُواْ بَل لَّمۡ تَكُونُواْ مُؤۡمِنِينَ ۝٢٩ قَالُوا۟ بَل لَّمۡ تَكُونُوا۟ مُؤۡمِنِینَ ۝٢٩ قَالُوْا بَلْ لَّمْ تَكُوْنُوْا مُؤْمِنِیْنَ ۟ۚ وَمَا كَانَ لَنَا عَلَيۡكُم مِّن سُلۡطَٰنِۭۖ بَلۡ كُنتُمۡ قَوۡمٗا طَٰغِينَ ۝٣٠ وَمَا كَانَ لَنَا عَلَیۡكُم مِّن سُلۡطَٰنِۭۖ بَلۡ كُنتُمۡ قَوۡمࣰا طَٰغِینَ ۝٣٠ وَمَا كَانَ لَنَا عَلَیْكُمْ مِّنْ سُلْطٰنٍ ۚ بَلْ كُنْتُمْ قَوْمًا طٰغِیْنَ ۟ فَحَقَّ عَلَيۡنَا قَوۡلُ رَبِّنَآۖ إِنَّا لَذَآئِقُونَ ۝٣١ فَحَقَّ عَلَیۡنَا قَوۡلُ رَبِّنَاۤۖ إِنَّا لَذَاۤئِقُونَ ۝٣١ فَحَقَّ عَلَیْنَا قَوْلُ رَبِّنَاۤ ۖۗ اِنَّا لَذَآىِٕقُوْنَ ۟ فَأَغۡوَيۡنَٰكُمۡ إِنَّا كُنَّا غَٰوِينَ ۝٣٢ فَأَغۡوَیۡنَٰكُمۡ إِنَّا كُنَّا غَٰوِینَ ۝٣٢ فَاَغْوَیْنٰكُمْ اِنَّا كُنَّا غٰوِیْنَ ۟ فَإِنَّهُمۡ يَوۡمَئِذٖ فِي ٱلۡعَذَابِ مُشۡتَرِكُونَ ۝٣٣ فَإِنَّهُمۡ یَوۡمَئِذࣲ فِی ٱلۡعَذَابِ مُشۡتَرِكُونَ ۝٣٣ فَاِنَّهُمْ یَوْمَىِٕذٍ فِی الْعَذَابِ مُشْتَرِكُوْنَ ۟ إِنَّا كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِينَ ۝٣٤ إِنَّا كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِینَ ۝٣٤ اِنَّا كَذٰلِكَ نَفْعَلُ بِالْمُجْرِمِیْنَ ۟ إِنَّهُمۡ كَانُوٓاْ إِذَا قِيلَ لَهُمۡ لَآ إِلَٰهَ إِلَّا ٱللَّهُ يَسۡتَكۡبِرُونَ ۝٣٥ إِنَّهُمۡ كَانُوۤا۟ إِذَا قِیلَ لَهُمۡ لَاۤ إِلَٰهَ إِلَّا ٱللَّهُ یَسۡتَكۡبِرُونَ ۝٣٥ اِنَّهُمْ كَانُوْۤا اِذَا قِیْلَ لَهُمْ لَاۤ اِلٰهَ اِلَّا اللّٰهُ یَسْتَكْبِرُوْنَ ۟ۙ وَيَقُولُونَ أَئِنَّا لَتَارِكُوٓاْ ءَالِهَتِنَا لِشَاعِرٖ مَّجۡنُونِۭ ۝٣٦ وَیَقُولُونَ أَئِنَّا لَتَارِكُوۤا۟ ءَالِهَتِنَا لِشَاعِرࣲ مَّجۡنُونِۭ ۝٣٦ وَیَقُوْلُوْنَ اَىِٕنَّا لَتَارِكُوْۤا اٰلِهَتِنَا لِشَاعِرٍ مَّجْنُوْنٍ ۟ؕ بَلۡ جَآءَ بِٱلۡحَقِّ وَصَدَّقَ ٱلۡمُرۡسَلِينَ ۝٣٧ بَلۡ جَاۤءَ بِٱلۡحَقِّ وَصَدَّقَ ٱلۡمُرۡسَلِینَ ۝٣٧ بَلْ جَآءَ بِالْحَقِّ وَصَدَّقَ الْمُرْسَلِیْنَ ۟ إِنَّكُمۡ لَذَآئِقُواْ ٱلۡعَذَابِ ٱلۡأَلِيمِ ۝٣٨ إِنَّكُمۡ لَذَاۤئِقُوا۟ ٱلۡعَذَابِ ٱلۡأَلِیمِ ۝٣٨ اِنَّكُمْ لَذَآىِٕقُوا الْعَذَابِ الْاَلِیْمِ ۟ۚ وَمَا تُجۡزَوۡنَ إِلَّا مَا كُنتُمۡ تَعۡمَلُونَ ۝٣٩ وَمَا تُجۡزَوۡنَ إِلَّا مَا كُنتُمۡ تَعۡمَلُونَ ۝٣٩ وَمَا تُجْزَوْنَ اِلَّا مَا كُنْتُمْ تَعْمَلُوْنَ ۟ۙ إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ ۝٤٠ إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِینَ ۝٤٠ اِلَّا عِبَادَ اللّٰهِ الْمُخْلَصِیْنَ ۟ أُوْلَٰٓئِكَ لَهُمۡ رِزۡقٞ مَّعۡلُومٞ ۝٤١ أُو۟لَٰۤئِكَ لَهُمۡ رِزۡقࣱ مَّعۡلُومࣱ ۝٤١ اُولٰٓىِٕكَ لَهُمْ رِزْقٌ مَّعْلُوْمٌ ۟ۙ فَوَٰكِهُ وَهُم مُّكۡرَمُونَ ۝٤٢ فَوَٰكِهُ وَهُم مُّكۡرَمُونَ ۝٤٢ فَوَاكِهُ ۚ وَهُمْ مُّكْرَمُوْنَ ۟ۙ فِي جَنَّٰتِ ٱلنَّعِيمِ ۝٤٣ فِی جَنَّٰتِ ٱلنَّعِیمِ ۝٤٣ فِیْ جَنّٰتِ النَّعِیْمِ ۟ۙ عَلَىٰ سُرُرٖ مُّتَقَٰبِلِينَ ۝٤٤ عَلَىٰ سُرُرࣲ مُّتَقَٰبِلِینَ ۝٤٤ عَلٰی سُرُرٍ مُّتَقٰبِلِیْنَ ۟ يُطَافُ عَلَيۡهِم بِكَأۡسٖ مِّن مَّعِينِۭ ۝٤٥ یُطَافُ عَلَیۡهِم بِكَأۡسࣲ مِّن مَّعِینِۭ ۝٤٥ یُطَافُ عَلَیْهِمْ بِكَاْسٍ مِّنْ مَّعِیْنٍ ۟ۙ بَيۡضَآءَ لَذَّةٖ لِّلشَّٰرِبِينَ ۝٤٦ بَیۡضَاۤءَ لَذَّةࣲ لِّلشَّٰرِبِینَ ۝٤٦ بَیْضَآءَ لَذَّةٍ لِّلشّٰرِبِیْنَ ۟ۚۖ لَا فِيهَا غَوۡلٞ وَلَا هُمۡ عَنۡهَا يُنزَفُونَ ۝٤٧ لَا فِیهَا غَوۡلࣱ وَلَا هُمۡ عَنۡهَا یُنزَفُونَ ۝٤٧ لَا فِیْهَا غَوْلٌ وَّلَا هُمْ عَنْهَا یُنْزَفُوْنَ ۟ وَعِندَهُمۡ قَٰصِرَٰتُ ٱلطَّرۡفِ عِينٞ ۝٤٨ وَعِندَهُمۡ قَٰصِرَٰتُ ٱلطَّرۡفِ عِینࣱ ۝٤٨ وَعِنْدَهُمْ قٰصِرٰتُ الطَّرْفِ عِیْنٌ ۟ۙ كَأَنَّهُنَّ بَيۡضٞ مَّكۡنُونٞ ۝٤٩ كَأَنَّهُنَّ بَیۡضࣱ مَّكۡنُونࣱ ۝٤٩ كَاَنَّهُنَّ بَیْضٌ مَّكْنُوْنٌ ۟ فَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضٖ يَتَسَآءَلُونَ ۝٥٠ فَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضࣲ یَتَسَاۤءَلُونَ ۝٥٠ فَاَقْبَلَ بَعْضُهُمْ عَلٰی بَعْضٍ یَّتَسَآءَلُوْنَ ۟ قَالَ قَآئِلٞ مِّنۡهُمۡ إِنِّي كَانَ لِي قَرِينٞ ۝٥١ قَالَ قَاۤئِلࣱ مِّنۡهُمۡ إِنِّی كَانَ لِی قَرِینࣱ ۝٥١ قَالَ قَآىِٕلٌ مِّنْهُمْ اِنِّیْ كَانَ لِیْ قَرِیْنٌ ۟ۙ يَقُولُ أَءِنَّكَ لَمِنَ ٱلۡمُصَدِّقِينَ ۝٥٢ یَقُولُ أَءِنَّكَ لَمِنَ ٱلۡمُصَدِّقِینَ ۝٥٢ یَّقُوْلُ اَىِٕنَّكَ لَمِنَ الْمُصَدِّقِیْنَ ۟ أَءِذَا مِتۡنَا وَكُنَّا تُرَابٗا وَعِظَٰمًا أَءِنَّا لَمَدِينُونَ ۝٥٣ أَءِذَا مِتۡنَا وَكُنَّا تُرَابࣰا وَعِظَٰمًا أَءِنَّا لَمَدِینُونَ ۝٥٣ ءَاِذَا مِتْنَا وَكُنَّا تُرَابًا وَّعِظَامًا ءَاِنَّا لَمَدِیْنُوْنَ ۟ قَالَ هَلۡ أَنتُم مُّطَّلِعُونَ ۝٥٤ قَالَ هَلۡ أَنتُم مُّطَّلِعُونَ ۝٥٤ قَالَ هَلْ اَنْتُمْ مُّطَّلِعُوْنَ ۟ فَٱطَّلَعَ فَرَءَاهُ فِي سَوَآءِ ٱلۡجَحِيمِ ۝٥٥ فَٱطَّلَعَ فَرَءَاهُ فِی سَوَاۤءِ ٱلۡجَحِیمِ ۝٥٥ فَاطَّلَعَ فَرَاٰهُ فِیْ سَوَآءِ الْجَحِیْمِ ۟ قَالَ تَٱللَّهِ إِن كِدتَّ لَتُرۡدِينِ ۝٥٦ قَالَ تَٱللَّهِ إِن كِدتَّ لَتُرۡدِینِ ۝٥٦ قَالَ تَاللّٰهِ اِنْ كِدْتَّ لَتُرْدِیْنِ ۟ۙ وَلَوۡلَا نِعۡمَةُ رَبِّي لَكُنتُ مِنَ ٱلۡمُحۡضَرِينَ ۝٥٧ وَلَوۡلَا نِعۡمَةُ رَبِّی لَكُنتُ مِنَ ٱلۡمُحۡضَرِینَ ۝٥٧ وَلَوْلَا نِعْمَةُ رَبِّیْ لَكُنْتُ مِنَ الْمُحْضَرِیْنَ ۟ أَفَمَا نَحۡنُ بِمَيِّتِينَ ۝٥٨ أَفَمَا نَحۡنُ بِمَیِّتِینَ ۝٥٨ اَفَمَا نَحْنُ بِمَیِّتِیْنَ ۟ۙ إِلَّا مَوۡتَتَنَا ٱلۡأُولَىٰ وَمَا نَحۡنُ بِمُعَذَّبِينَ ۝٥٩ إِلَّا مَوۡتَتَنَا ٱلۡأُولَىٰ وَمَا نَحۡنُ بِمُعَذَّبِینَ ۝٥٩ اِلَّا مَوْتَتَنَا الْاُوْلٰی وَمَا نَحْنُ بِمُعَذَّبِیْنَ ۟ إِنَّ هَٰذَا لَهُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ ۝٦٠ إِنَّ هَٰذَا لَهُوَ ٱلۡفَوۡزُ ٱلۡعَظِیمُ ۝٦٠ اِنَّ هٰذَا لَهُوَ الْفَوْزُ الْعَظِیْمُ ۟ لِمِثۡلِ هَٰذَا فَلۡيَعۡمَلِ ٱلۡعَٰمِلُونَ ۝٦١ لِمِثۡلِ هَٰذَا فَلۡیَعۡمَلِ ٱلۡعَٰمِلُونَ ۝٦١ لِمِثْلِ هٰذَا فَلْیَعْمَلِ الْعٰمِلُوْنَ ۟ أَذَٰلِكَ خَيۡرٞ نُّزُلًا أَمۡ شَجَرَةُ ٱلزَّقُّومِ ۝٦٢ أَذَٰلِكَ خَیۡرࣱ نُّزُلًا أَمۡ شَجَرَةُ ٱلزَّقُّومِ ۝٦٢ اَذٰلِكَ خَیْرٌ نُّزُلًا اَمْ شَجَرَةُ الزَّقُّوْمِ ۟ إِنَّا جَعَلۡنَٰهَا فِتۡنَةٗ لِّلظَّٰلِمِينَ ۝٦٣ إِنَّا جَعَلۡنَٰهَا فِتۡنَةࣰ لِّلظَّٰلِمِینَ ۝٦٣ اِنَّا جَعَلْنٰهَا فِتْنَةً لِّلظّٰلِمِیْنَ ۟ إِنَّهَا شَجَرَةٞ تَخۡرُجُ فِيٓ أَصۡلِ ٱلۡجَحِيمِ ۝٦٤ إِنَّهَا شَجَرَةࣱ تَخۡرُجُ فِیۤ أَصۡلِ ٱلۡجَحِیمِ ۝٦٤ اِنَّهَا شَجَرَةٌ تَخْرُجُ فِیْۤ اَصْلِ الْجَحِیْمِ ۟ۙ طَلۡعُهَا كَأَنَّهُۥ رُءُوسُ ٱلشَّيَٰطِينِ ۝٦٥ طَلۡعُهَا كَأَنَّهُۥ رُءُوسُ ٱلشَّیَٰطِینِ ۝٦٥ طَلْعُهَا كَاَنَّهٗ رُءُوْسُ الشَّیٰطِیْنِ ۟ فَإِنَّهُمۡ لَأٓكِلُونَ مِنۡهَا فَمَالِـُٔونَ مِنۡهَا ٱلۡبُطُونَ ۝٦٦ فَإِنَّهُمۡ لَءَاكِلُونَ مِنۡهَا فَمَالِءُونَ مِنۡهَا ٱلۡبُطُونَ ۝٦٦ فَاِنَّهُمْ لَاٰكِلُوْنَ مِنْهَا فَمَالِـُٔوْنَ مِنْهَا الْبُطُوْنَ ۟ؕ ثُمَّ إِنَّ لَهُمۡ عَلَيۡهَا لَشَوۡبٗا مِّنۡ حَمِيمٖ ۝٦٧ ثُمَّ إِنَّ لَهُمۡ عَلَیۡهَا لَشَوۡبࣰا مِّنۡ حَمِیمࣲ ۝٦٧ ثُمَّ اِنَّ لَهُمْ عَلَیْهَا لَشَوْبًا مِّنْ حَمِیْمٍ ۟ۚ ثُمَّ إِنَّ مَرۡجِعَهُمۡ لَإِلَى ٱلۡجَحِيمِ ۝٦٨ ثُمَّ إِنَّ مَرۡجِعَهُمۡ لَإِلَى ٱلۡجَحِیمِ ۝٦٨ ثُمَّ اِنَّ مَرْجِعَهُمْ لَاۡاِلَی الْجَحِیْمِ ۟ إِنَّهُمۡ أَلۡفَوۡاْ ءَابَآءَهُمۡ ضَآلِّينَ ۝٦٩ إِنَّهُمۡ أَلۡفَوۡا۟ ءَابَاۤءَهُمۡ ضَاۤلِّینَ ۝٦٩ اِنَّهُمْ اَلْفَوْا اٰبَآءَهُمْ ضَآلِّیْنَ ۟ۙ فَهُمۡ عَلَىٰٓ ءَاثَٰرِهِمۡ يُهۡرَعُونَ ۝٧٠ فَهُمۡ عَلَىٰۤ ءَاثَٰرِهِمۡ یُهۡرَعُونَ ۝٧٠ فَهُمْ عَلٰۤی اٰثٰرِهِمْ یُهْرَعُوْنَ ۟ وَلَقَدۡ ضَلَّ قَبۡلَهُمۡ أَكۡثَرُ ٱلۡأَوَّلِينَ ۝٧١ وَلَقَدۡ ضَلَّ قَبۡلَهُمۡ أَكۡثَرُ ٱلۡأَوَّلِینَ ۝٧١ وَلَقَدْ ضَلَّ قَبْلَهُمْ اَكْثَرُ الْاَوَّلِیْنَ ۟ۙ وَلَقَدۡ أَرۡسَلۡنَا فِيهِم مُّنذِرِينَ ۝٧٢ وَلَقَدۡ أَرۡسَلۡنَا فِیهِم مُّنذِرِینَ ۝٧٢ وَلَقَدْ اَرْسَلْنَا فِیْهِمْ مُّنْذِرِیْنَ ۟ فَٱنظُرۡ كَيۡفَ كَانَ عَٰقِبَةُ ٱلۡمُنذَرِينَ ۝٧٣ فَٱنظُرۡ كَیۡفَ كَانَ عَٰقِبَةُ ٱلۡمُنذَرِینَ ۝٧٣ فَانْظُرْ كَیْفَ كَانَ عَاقِبَةُ الْمُنْذَرِیْنَ ۟ۙ إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ ۝٧٤ إِلَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِینَ ۝٧٤ اِلَّا عِبَادَ اللّٰهِ الْمُخْلَصِیْنَ ۟۠ وَلَقَدۡ نَادَىٰنَا نُوحٞ فَلَنِعۡمَ ٱلۡمُجِيبُونَ ۝٧٥ وَلَقَدۡ نَادَىٰنَا نُوحࣱ فَلَنِعۡمَ ٱلۡمُجِیبُونَ ۝٧٥ وَلَقَدْ نَادٰىنَا نُوْحٌ فَلَنِعْمَ الْمُجِیْبُوْنَ ۟ؗۖ وَنَجَّيۡنَٰهُ وَأَهۡلَهُۥ مِنَ ٱلۡكَرۡبِ ٱلۡعَظِيمِ ۝٧٦ وَنَجَّیۡنَٰهُ وَأَهۡلَهُۥ مِنَ ٱلۡكَرۡبِ ٱلۡعَظِیمِ ۝٧٦ وَنَجَّیْنٰهُ وَاَهْلَهٗ مِنَ الْكَرْبِ الْعَظِیْمِ ۟ؗۖ وَجَعَلۡنَا ذُرِّيَّتَهُۥ هُمُ ٱلۡبَاقِينَ ۝٧٧ وَجَعَلۡنَا ذُرِّیَّتَهُۥ هُمُ ٱلۡبَاقِینَ ۝٧٧ وَجَعَلْنَا ذُرِّیَّتَهٗ هُمُ الْبٰقِیْنَ ۟ؗۖ وَتَرَكۡنَا عَلَيۡهِ فِي ٱلۡأٓخِرِينَ ۝٧٨ وَتَرَكۡنَا عَلَیۡهِ فِی ٱلۡءَاخِرِینَ ۝٧٨ وَتَرَكْنَا عَلَیْهِ فِی الْاٰخِرِیْنَ ۟ؗۖ سَلَٰمٌ عَلَىٰ نُوحٖ فِي ٱلۡعَٰلَمِينَ ۝٧٩ سَلَٰمٌ عَلَىٰ نُوحࣲ فِی ٱلۡعَٰلَمِینَ ۝٧٩ سَلٰمٌ عَلٰی نُوْحٍ فِی الْعٰلَمِیْنَ ۟ إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ ۝٨٠ إِنَّا كَذَٰلِكَ نَجۡزِی ٱلۡمُحۡسِنِینَ ۝٨٠ اِنَّا كَذٰلِكَ نَجْزِی الْمُحْسِنِیْنَ ۟ إِنَّهُۥ مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِينَ ۝٨١ إِنَّهُۥ مِنۡ عِبَادِنَا ٱلۡمُؤۡمِنِینَ ۝٨١ اِنَّهٗ مِنْ عِبَادِنَا الْمُؤْمِنِیْنَ ۟ ثُمَّ أَغۡرَقۡنَا ٱلۡأٓخَرِينَ ۝٨٢ ثُمَّ أَغۡرَقۡنَا ٱلۡءَاخَرِینَ ۝٨٢ ثُمَّ اَغْرَقْنَا الْاٰخَرِیْنَ ۟ وَإِنَّ مِن شِيعَتِهِۦ لَإِبۡرَٰهِيمَ ۝٨٣ وَإِنَّ مِن شِیعَتِهِۦ لَإِبۡرَٰهِیمَ ۝٨٣ وَاِنَّ مِنْ شِیْعَتِهٖ لَاِبْرٰهِیْمَ ۟ۘ إِذۡ جَآءَ رَبَّهُۥ بِقَلۡبٖ سَلِيمٍ ۝٨٤ إِذۡ جَاۤءَ رَبَّهُۥ بِقَلۡبࣲ سَلِیمٍ ۝٨٤ اِذْ جَآءَ رَبَّهٗ بِقَلْبٍ سَلِیْمٍ ۟ إِذۡ قَالَ لِأَبِيهِ وَقَوۡمِهِۦ مَاذَا تَعۡبُدُونَ ۝٨٥ إِذۡ قَالَ لِأَبِیهِ وَقَوۡمِهِۦ مَاذَا تَعۡبُدُونَ ۝٨٥ اِذْ قَالَ لِاَبِیْهِ وَقَوْمِهٖ مَاذَا تَعْبُدُوْنَ ۟ۚ أَئِفۡكًا ءَالِهَةٗ دُونَ ٱللَّهِ تُرِيدُونَ ۝٨٦ أَئِفۡكًا ءَالِهَةࣰ دُونَ ٱللَّهِ تُرِیدُونَ ۝٨٦ اَىِٕفْكًا اٰلِهَةً دُوْنَ اللّٰهِ تُرِیْدُوْنَ ۟ؕ فَمَا ظَنُّكُم بِرَبِّ ٱلۡعَٰلَمِينَ ۝٨٧ فَمَا ظَنُّكُم بِرَبِّ ٱلۡعَٰلَمِینَ ۝٨٧ فَمَا ظَنُّكُمْ بِرَبِّ الْعٰلَمِیْنَ ۟ فَنَظَرَ نَظۡرَةٗ فِي ٱلنُّجُومِ ۝٨٨ فَنَظَرَ نَظۡرَةࣰ فِی ٱلنُّجُومِ ۝٨٨ فَنَظَرَ نَظْرَةً فِی النُّجُوْمِ ۟ۙ فَقَالَ إِنِّي سَقِيمٞ ۝٨٩ فَقَالَ إِنِّی سَقِیمࣱ ۝٨٩ فَقَالَ اِنِّیْ سَقِیْمٌ ۟ فَتَوَلَّوۡاْ عَنۡهُ مُدۡبِرِينَ ۝٩٠ فَتَوَلَّوۡا۟ عَنۡهُ مُدۡبِرِینَ ۝٩٠ فَتَوَلَّوْا عَنْهُ مُدْبِرِیْنَ ۟ فَرَاغَ إِلَىٰٓ ءَالِهَتِهِمۡ فَقَالَ أَلَا تَأۡكُلُونَ ۝٩١ فَرَاغَ إِلَىٰۤ ءَالِهَتِهِمۡ فَقَالَ أَلَا تَأۡكُلُونَ ۝٩١ فَرَاغَ اِلٰۤی اٰلِهَتِهِمْ فَقَالَ اَلَا تَاْكُلُوْنَ ۟ۚ مَا لَكُمۡ لَا تَنطِقُونَ ۝٩٢ مَا لَكُمۡ لَا تَنطِقُونَ ۝٩٢ مَا لَكُمْ لَا تَنْطِقُوْنَ ۟ فَرَاغَ عَلَيۡهِمۡ ضَرۡبَۢا بِٱلۡيَمِينِ ۝٩٣ فَرَاغَ عَلَیۡهِمۡ ضَرۡبَۢا بِٱلۡیَمِینِ ۝٩٣ فَرَاغَ عَلَیْهِمْ ضَرْبًا بِالْیَمِیْنِ ۟ فَأَقۡبَلُوٓاْ إِلَيۡهِ يَزِفُّونَ ۝٩٤ فَأَقۡبَلُوۤا۟ إِلَیۡهِ یَزِفُّونَ ۝٩٤ فَاَقْبَلُوْۤا اِلَیْهِ یَزِفُّوْنَ ۟ قَالَ أَتَعۡبُدُونَ مَا تَنۡحِتُونَ ۝٩٥ قَالَ أَتَعۡبُدُونَ مَا تَنۡحِتُونَ ۝٩٥ قَالَ اَتَعْبُدُوْنَ مَا تَنْحِتُوْنَ ۟ۙ وَٱللَّهُ خَلَقَكُمۡ وَمَا تَعۡمَلُونَ ۝٩٦ وَٱللَّهُ خَلَقَكُمۡ وَمَا تَعۡمَلُونَ ۝٩٦ وَاللّٰهُ خَلَقَكُمْ وَمَا تَعْمَلُوْنَ ۟ قَالُواْ ٱبۡنُواْ لَهُۥ بُنۡيَٰنٗا فَأَلۡقُوهُ فِي ٱلۡجَحِيمِ ۝٩٧ قَالُوا۟ ٱبۡنُوا۟ لَهُۥ بُنۡیَٰنࣰا فَأَلۡقُوهُ فِی ٱلۡجَحِیمِ ۝٩٧ قَالُوا ابْنُوْا لَهٗ بُنْیَانًا فَاَلْقُوْهُ فِی الْجَحِیْمِ ۟ فَأَرَادُواْ بِهِۦ كَيۡدٗا فَجَعَلۡنَٰهُمُ ٱلۡأَسۡفَلِينَ ۝٩٨ فَأَرَادُوا۟ بِهِۦ كَیۡدࣰا فَجَعَلۡنَٰهُمُ ٱلۡأَسۡفَلِینَ ۝٩٨ فَاَرَادُوْا بِهٖ كَیْدًا فَجَعَلْنٰهُمُ الْاَسْفَلِیْنَ ۟ وَقَالَ إِنِّي ذَاهِبٌ إِلَىٰ رَبِّي سَيَهۡدِينِ ۝٩٩ وَقَالَ إِنِّی ذَاهِبٌ إِلَىٰ رَبِّی سَیَهۡدِینِ ۝٩٩ وَقَالَ اِنِّیْ ذَاهِبٌ اِلٰی رَبِّیْ سَیَهْدِیْنِ ۟ رَبِّ هَبۡ لِي مِنَ ٱلصَّٰلِحِينَ ۝١٠٠ رَبِّ هَبۡ لِی مِنَ ٱلصَّٰلِحِینَ ۝١٠٠ رَبِّ هَبْ لِیْ مِنَ الصّٰلِحِیْنَ ۟ فَبَشَّرۡنَٰهُ بِغُلَٰمٍ حَلِيمٖ ۝١٠١ فَبَشَّرۡنَٰهُ بِغُلَٰمٍ حَلِیمࣲ ۝١٠١ فَبَشَّرْنٰهُ بِغُلٰمٍ حَلِیْمٍ ۟ فَلَمَّا بَلَغَ مَعَهُ ٱلسَّعۡيَ قَالَ يَٰبُنَيَّ إِنِّيٓ أَرَىٰ فِي ٱلۡمَنَامِ أَنِّيٓ أَذۡبَحُكَ فَٱنظُرۡ مَاذَا تَرَىٰۚ قَالَ يَٰٓأَبَتِ ٱفۡعَلۡ مَا تُؤۡمَرُۖ سَتَجِدُنِيٓ إِن شَآءَ ٱللَّهُ مِنَ ٱلصَّٰبِرِينَ ۝١٠٢ فَلَمَّا بَلَغَ مَعَهُ ٱلسَّعۡیَ قَالَ یَٰبُنَیَّ إِنِّیۤ أَرَىٰ فِی ٱلۡمَنَامِ أَنِّیۤ أَذۡبَحُكَ فَٱنظُرۡ مَاذَا تَرَىٰۚ قَالَ یَٰۤأَبَتِ ٱفۡعَلۡ مَا تُؤۡمَرُۖ سَتَجِدُنِیۤ إِن شَاۤءَ ٱللَّهُ مِنَ ٱلصَّٰبِرِینَ ۝١٠٢ فَلَمَّا بَلَغَ مَعَهُ السَّعْیَ قَالَ یٰبُنَیَّ اِنِّیْۤ اَرٰی فِی الْمَنَامِ اَنِّیْۤ اَذْبَحُكَ فَانْظُرْ مَاذَا تَرٰی ؕ قَالَ یٰۤاَبَتِ افْعَلْ مَا تُؤْمَرُ ؗ سَتَجِدُنِیْۤ اِنْ شَآءَ اللّٰهُ مِنَ الصّٰبِرِیْنَ ۟ فَلَمَّآ أَسۡلَمَا وَتَلَّهُۥ لِلۡجَبِينِ ۝١٠٣ فَلَمَّاۤ أَسۡلَمَا وَتَلَّهُۥ لِلۡجَبِینِ ۝١٠٣ فَلَمَّاۤ اَسْلَمَا وَتَلَّهٗ لِلْجَبِیْنِ ۟ۚ وَنَٰدَيۡنَٰهُ أَن يَٰٓإِبۡرَٰهِيمُ ۝١٠٤ وَنَٰدَیۡنَٰهُ أَن یَٰۤإِبۡرَٰهِیمُ ۝١٠٤ وَنَادَیْنٰهُ اَنْ یّٰۤاِبْرٰهِیْمُ ۟ۙ قَدۡ صَدَّقۡتَ ٱلرُّءۡيَآۚ إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ ۝١٠٥ قَدۡ الْمُخْلَصِیْنَ ۟ وَتَرَكۡنَا عَلَيۡهِ فِي ٱلۡأٓخِرِينَ ۝١٢٩ وَتَرَكۡنَا عَلَیۡهِ فِی ٱلۡءَاخِرِینَ ۝١٢٩ وَتَرَكْنَا عَلَیْهِ فِی الْاٰخِرِیْنَ ۟ۙ سَلَٰمٌ عَلَىٰٓ إِلۡ يَاسِينَ ۝١٣٠ سَلَٰمٌ عَلَىٰۤ عَجُوزٗا فِي ٱلۡغَٰبِرِينَ ۝١٣٥ إِلَّا عَجُوزࣰا فِی ٱلۡغَٰبِرِینَ ۝١٣٥ اِلَّا عَجُوْزًا فِی الْغٰبِرِیْنَ ۟ ثُمَّ دَمَّرۡنَا ٱلۡأٓخَرِينَ ۝١٣٦ ثُمَّ دَمَّرۡنَا ٱلۡءَاخَرِینَ ۝١٣٦ ثُمَّ دَمَّرْنَا الْاٰخَرِیْنَ ۟ وَإِنَّكُمۡ لَتَمُرُّونَ عَلَيۡهِم مُّصۡبِحِينَ ۝١٣٧ وَإِنَّكُمۡ لَتَمُرُّونَ عَلَیۡهِم مُّصۡبِحِینَ ۝١٣٧ وَاِنَّكُمْ لَتَمُرُّوْنَ عَلَیْهِمْ مُّصْبِحِیْنَ ۟ۙ وَبِٱلَّيۡلِۚ أَفَلَا ٱلۡمُدۡحَضِينَ ۝١٤١ فَسَاهَمَ فَكَانَ مِنَ ٱلۡمُدۡحَضِینَ ۝١٤١ فَسَاهَمَ فَكَانَ مِنَ الْمُدْحَضِیْنَ ۟ۚ فَٱلۡتَقَمَهُ ٱلۡحُوتُ وَهُوَ مُلِيمٞ ۝١٤٢ فَٱلۡتَقَمَهُ ٱلۡحُوتُ وَهُوَ مُلِیمࣱ ۝١٤٢ یَقۡطِینࣲ ۝١٤٦ وَاَنْۢبَتْنَا عَلَیْهِ شَجَرَةً مِّنْ یَّقْطِیْنٍ ۟ۚ وَأَرۡسَلۡنَٰهُ إِلَىٰ مِاْئَةِ أَلۡفٍ أَوۡ يَزِيدُونَ ۝١٤٧ وَأَرۡسَلۡنَٰهُ إِلَىٰ مِا۟ئَةِ أَلۡفٍ أَوۡ یَزِیدُونَ ۝١٤٧ إِلَّا لَهُۥ مَقَامٞ مَّعۡلُومٞ ۝١٦٤ وَمَا مِنَّاۤ إِلَّا لَهُۥ مَقَامࣱ مَّعۡلُومࣱ ۝١٦٤ وَمَا مِنَّاۤ اِلَّا لَهٗ مَقَامٌ مَّعْلُوْمٌ ۟ۙ وَإِنَّا لَنَحۡنُ ٱلصَّآفُّونَ ۝١٦٥ وَإِنَّا لَنَحۡنُ ٱلصَّاۤفُّونَ ۝١٦٥ وَّاِنَّا لَنَحْنُ الصَّآفُّوْنَ ۟ۚ وَإِنَّا لَنَحۡنُ ٱلۡمُسَبِّحُونَ ۝١٦٦ وَإِنَّا لَنَحۡنُ ٱلۡمُسَبِّحُونَ ۝١٦٦ وَاِنَّا لَنَحْنُ الْمُسَبِّحُوْنَ ۟ وَإِن كَانُواْ لَيَقُولُونَ ۝١٦٧ وَإِن كَانُوا۟ لَیَقُولُونَ ۝١٦٧ وَاِنْ كَانُوْا لَیَقُوْلُوْنَ ۟ۙ لَوۡ أَنَّ عِندَنَا ذِكۡرٗا مِّنَ ٱلۡأَوَّلِينَ ۝١٦٨ لَوۡ أَنَّ عِندَنَا ذِكۡرࣰا مِّنَ ٱلۡأَوَّلِینَ ۝١٦٨ لَوْ اَنَّ عِنْدَنَا ذِكْرًا مِّنَ الْاَوَّلِیْنَ ۟ۙ لَكُنَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِينَ ۝١٦٩ لَكُنَّا عِبَادَ ٱللَّهِ ٱلۡمُخۡلَصِینَ ۝١٦٩ لَكُنَّا عِبَادَ اللّٰهِ الْمُخْلَصِیْنَ ۟ فَكَفَرُواْ بِهِۦۖ فَسَوۡفَ يَعۡلَمُونَ ۝١٧٠ فَكَفَرُوا۟ بِهِۦۖ فَسَوۡفَ یَعۡلَمُونَ ۝١٧٠ فَكَفَرُوْا بِهٖ فَسَوْفَ یَعْلَمُوْنَ ۟ وَلَقَدۡ سَبَقَتۡ كَلِمَتُنَا لِعِبَادِنَا ٱلۡمُرۡسَلِينَ ۝١٧١ وَلَقَدۡ سَبَقَتۡ كَلِمَتُنَا لِعِبَادِنَا ٱلۡمُرۡسَلِینَ ۝١٧١ وَلَقَدْ سَبَقَتْ كَلِمَتُنَا لِعِبَادِنَا الْمُرْسَلِیْنَ ۟ۚۖ إِنَّهُمۡ لَهُمُ ٱلۡمَنصُورُونَ ۝١٧٢ إِنَّهُمۡ لَهُمُ ٱلۡمَنصُورُونَ ۝١٧٢ اِنَّهُمْ لَهُمُ الْمَنْصُوْرُوْنَ ۪۟ وَإِنَّ جُندَنَا لَهُمُ ٱلۡغَٰلِبُونَ ۝١٧٣ وَإِنَّ جُندَنَا لَهُمُ ٱلۡغَٰلِبُونَ ۝١٧٣ وَاِنَّ جُنْدَنَا لَهُمُ الْغٰلِبُوْنَ ۟ فَتَوَلَّ عَنۡهُمۡ حَتَّىٰ حِينٖ ۝١٧٤ فَتَوَلَّ عَنۡهُمۡ حَتَّىٰ حِینࣲ ۝١٧٤ فَتَوَلَّ عَنْهُمْ حَتّٰی حِیْنٍ ۟ۙ وَأَبۡصِرۡهُمۡ فَسَوۡفَ يُبۡصِرُونَ ۝١٧٥ وَأَبۡصِرۡهُمۡ فَسَوۡفَ یُبۡصِرُونَ ۝١٧٥ وَّاَبْصِرْهُمْ فَسَوْفَ یُبْصِرُوْنَ ۟ أَفَبِعَذَابِنَا يَسۡتَعۡجِلُونَ ۝١٧٦ أَفَبِعَذَابِنَا یَسۡتَعۡجِلُونَ ۝١٧٦ اَفَبِعَذَابِنَا یَسْتَعْجِلُوْنَ ۟ فَإِذَا نَزَلَ بِسَاحَتِهِمۡ فَسَآءَ صَبَاحُ ٱلۡمُنذَرِينَ ۝١٧٧ فَإِذَا نَزَلَ بِسَاحَتِهِمۡ فَسَاۤءَ صَبَاحُ ٱلۡمُنذَرِینَ ۝١٧٧ فَاِذَا نَزَلَ بِسَاحَتِهِمْ فَسَآءَ صَبَاحُ الْمُنْذَرِیْنَ ۟ وَتَوَلَّ عَنۡهُمۡ حَتَّىٰ حِينٖ ۝١٧٨ وَتَوَلَّ عَنۡهُمۡ حَتَّىٰ حِینࣲ ۝١٧٨ وَتَوَلَّ عَنْهُمْ حَتّٰی حِیْنٍ ۟ۙ وَأَبۡصِرۡ فَسَوۡفَ يُبۡصِرُونَ ۝١٧٩ وَأَبۡصِرۡ فَسَوۡفَ یُبۡصِرُونَ ۝١٧٩ وَّاَبْصِرْ فَسَوْفَ یُبْصِرُوْنَ ۟ سُبۡحَٰنَ رَبِّكَ رَبِّ ٱلۡعِزَّةِ عَمَّا يَصِفُونَ ۝١٨٠ سُبۡحَٰنَ رَبِّكَ رَبِّ ٱلۡعِزَّةِ عَمَّا یَصِفُونَ ۝١٨٠ سُبْحٰنَ رَبِّكَ رَبِّ الْعِزَّةِ عَمَّا یَصِفُوْنَ ۟ۚ وَسَلَٰمٌ عَلَى ٱلۡمُرۡسَلِينَ ۝١٨١ وَسَلَٰمٌ عَلَى ٱلۡمُرۡسَلِینَ ۝١٨١ وَسَلٰمٌ عَلَی الْمُرْسَلِیْنَ ۟ۚ وَٱلۡحَمۡدُ لِلَّهِ رَبِّ ٱلۡعَٰلَمِينَ ۝١٨٢ وَٱلۡحَمۡدُ لِلَّهِ رَبِّ ٱلۡعَٰلَمِینَ ۝١٨٢ وَالْحَمْدُ لِلّٰهِ رَبِّ الْعٰلَمِیْنَ ۟۠
