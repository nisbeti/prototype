Surah 40 · Meccan · 85 ayat

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

حمٓ ۝١ حمۤ ۝١ حٰمٓ ۟ۚ تَنزِيلُ ٱلۡكِتَٰبِ مِنَ ٱللَّهِ ٱلۡعَزِيزِ ٱلۡعَلِيمِ ۝٢ تَنزِیلُ ٱلۡكِتَٰبِ مِنَ ٱللَّهِ ٱلۡعَزِیزِ ٱلۡعَلِیمِ ۝٢ تَنْزِیْلُ الْكِتٰبِ مِنَ اللّٰهِ الْعَزِیْزِ الْعَلِیْمِ ۟ۙ غَافِرِ ٱلذَّنۢبِ وَقَابِلِ ٱلتَّوۡبِ شَدِيدِ ٱلۡعِقَابِ ذِي ٱلطَّوۡلِۖ لَآ إِلَٰهَ إِلَّا هُوَۖ إِلَيۡهِ ٱلۡمَصِيرُ ۝٣ غَافِرِ ٱلذَّنۢبِ وَقَابِلِ ٱلتَّوۡبِ شَدِیدِ ٱلۡعِقَابِ ذِی ٱلطَّوۡلِۖ لَاۤ إِلَٰهَ إِلَّا هُوَۖ إِلَیۡهِ ٱلۡمَصِیرُ ۝٣ غَافِرِ الذَّنْۢبِ وَقَابِلِ التَّوْبِ شَدِیْدِ الْعِقَابِ ۙ ذِی الطَّوْلِ ؕ لَاۤ اِلٰهَ اِلَّا هُوَ ؕ اِلَیْهِ الْمَصِیْرُ ۟ مَا يُجَٰدِلُ فِيٓ ءَايَٰتِ ٱللَّهِ إِلَّا ٱلَّذِينَ كَفَرُواْ فَلَا يَغۡرُرۡكَ تَقَلُّبُهُمۡ فِي ٱلۡبِلَٰدِ ۝٤ مَا یُجَٰدِلُ فِیۤ ءَایَٰتِ ٱللَّهِ إِلَّا ٱلَّذِینَ كَفَرُوا۟ فَلَا یَغۡرُرۡكَ تَقَلُّبُهُمۡ فِی ٱلۡبِلَٰدِ ۝٤ مَا یُجَادِلُ فِیْۤ اٰیٰتِ اللّٰهِ اِلَّا الَّذِیْنَ كَفَرُوْا فَلَا یَغْرُرْكَ تَقَلُّبُهُمْ فِی الْبِلَادِ ۟ كَذَّبَتۡ قَبۡلَهُمۡ قَوۡمُ نُوحٖ وَٱلۡأَحۡزَابُ مِنۢ بَعۡدِهِمۡۖ وَهَمَّتۡ كُلُّ أُمَّةِۭ بِرَسُولِهِمۡ لِيَأۡخُذُوهُۖ وَجَٰدَلُواْ بِٱلۡبَٰطِلِ لِيُدۡحِضُواْ بِهِ ٱلۡحَقَّ فَأَخَذۡتُهُمۡۖ فَكَيۡفَ كَانَ عِقَابِ ۝٥ كَذَّبَتۡ قَبۡلَهُمۡ قَوۡمُ نُوحࣲ وَٱلۡأَحۡزَابُ مِنۢ بَعۡدِهِمۡۖ وَهَمَّتۡ كُلُّ أُمَّةِۭ بِرَسُولِهِمۡ لِیَأۡخُذُوهُۖ وَجَٰدَلُوا۟ بِٱلۡبَٰطِلِ لِیُدۡحِضُوا۟ بِهِ ٱلۡحَقَّ فَأَخَذۡتُهُمۡۖ فَكَیۡفَ كَانَ عِقَابِ ۝٥ كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوْحٍ وَّالْاَحْزَابُ مِنْ بَعْدِهِمْ ۪ وَهَمَّتْ كُلُّ اُمَّةٍ بِرَسُوْلِهِمْ لِیَاْخُذُوْهُ وَجٰدَلُوْا بِالْبَاطِلِ لِیُدْحِضُوْا بِهِ الْحَقَّ فَاَخَذْتُهُمْ ۫ فَكَیْفَ كَانَ عِقَابِ ۟ وَكَذَٰلِكَ حَقَّتۡ كَلِمَتُ رَبِّكَ عَلَى ٱلَّذِينَ كَفَرُوٓاْ أَنَّهُمۡ أَصۡحَٰبُ ٱلنَّارِ ۝٦ وَكَذَٰلِكَ حَقَّتۡ كَلِمَتُ رَبِّكَ عَلَى ٱلَّذِینَ كَفَرُوۤا۟ أَنَّهُمۡ أَصۡحَٰبُ ٱلنَّارِ ۝٦ وَكَذٰلِكَ حَقَّتْ كَلِمَتُ رَبِّكَ عَلَی الَّذِیْنَ كَفَرُوْۤا اَنَّهُمْ اَصْحٰبُ النَّارِ ۟ ٱلَّذِينَ يَحۡمِلُونَ ٱلۡعَرۡشَ وَمَنۡ حَوۡلَهُۥ يُسَبِّحُونَ بِحَمۡدِ رَبِّهِمۡ وَيُؤۡمِنُونَ بِهِۦ وَيَسۡتَغۡفِرُونَ لِلَّذِينَ ءَامَنُواْۖ رَبَّنَا وَسِعۡتَ كُلَّ شَيۡءٖ رَّحۡمَةٗ وَعِلۡمٗا فَٱغۡفِرۡ لِلَّذِينَ تَابُواْ وَٱتَّبَعُواْ سَبِيلَكَ وَقِهِمۡ عَذَابَ ٱلۡجَحِيمِ ۝٧ ٱلَّذِینَ یَحۡمِلُونَ ٱلۡعَرۡشَ وَمَنۡ حَوۡلَهُۥ یُسَبِّحُونَ بِحَمۡدِ رَبِّهِمۡ وَیُؤۡمِنُونَ بِهِۦ وَیَسۡتَغۡفِرُونَ لِلَّذِینَ ءَامَنُوا۟ۖ رَبَّنَا وَسِعۡتَ كُلَّ شَیۡءࣲ رَّحۡمَةࣰ وَعِلۡمࣰا فَٱغۡفِرۡ لِلَّذِینَ تَابُوا۟ وَٱتَّبَعُوا۟ سَبِیلَكَ وَقِهِمۡ عَذَابَ ٱلۡجَحِیمِ ۝٧ اَلَّذِیْنَ یَحْمِلُوْنَ الْعَرْشَ وَمَنْ حَوْلَهٗ یُسَبِّحُوْنَ بِحَمْدِ رَبِّهِمْ وَیُؤْمِنُوْنَ بِهٖ وَیَسْتَغْفِرُوْنَ لِلَّذِیْنَ اٰمَنُوْا ۚ رَبَّنَا وَسِعْتَ كُلَّ شَیْءٍ رَّحْمَةً وَّعِلْمًا فَاغْفِرْ لِلَّذِیْنَ تَابُوْا وَاتَّبَعُوْا سَبِیْلَكَ وَقِهِمْ عَذَابَ الْجَحِیْمِ ۟ رَبَّنَا وَأَدۡخِلۡهُمۡ جَنَّٰتِ عَدۡنٍ ٱلَّتِي وَعَدتَّهُمۡ وَمَن صَلَحَ مِنۡ ءَابَآئِهِمۡ وَأَزۡوَٰجِهِمۡ وَذُرِّيَّٰتِهِمۡۚ إِنَّكَ أَنتَ ٱلۡعَزِيزُ ٱلۡحَكِيمُ ۝٨ رَبَّنَا وَأَدۡخِلۡهُمۡ جَنَّٰتِ عَدۡنٍ ٱلَّتِی وَعَدتَّهُمۡ وَمَن صَلَحَ مِنۡ ءَابَاۤئِهِمۡ وَأَزۡوَٰجِهِمۡ وَذُرِّیَّٰتِهِمۡۚ إِنَّكَ أَنتَ ٱلۡعَزِیزُ ٱلۡحَكِیمُ ۝٨ رَبَّنَا وَاَدْخِلْهُمْ جَنّٰتِ عَدْنِ لَّتِیْ وَعَدْتَّهُمْ وَمَنْ صَلَحَ مِنْ اٰبَآىِٕهِمْ وَاَزْوَاجِهِمْ وَذُرِّیّٰتِهِمْ ؕ اِنَّكَ اَنْتَ الْعَزِیْزُ الْحَكِیْمُ ۟ۙ وَقِهِمُ ٱلسَّيِّـَٔاتِۚ وَمَن تَقِ ٱلسَّيِّـَٔاتِ يَوۡمَئِذٖ فَقَدۡ رَحِمۡتَهُۥۚ وَذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ ۝٩ وَقِهِمُ ٱلسَّیِّءَاتِۚ وَمَن تَقِ ٱلسَّیِّءَاتِ یَوۡمَئِذࣲ فَقَدۡ رَحِمۡتَهُۥۚ وَذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِیمُ ۝٩ وَقِهِمُ السَّیِّاٰتِ ؕ وَمَنْ تَقِ السَّیِّاٰتِ یَوْمَىِٕذٍ فَقَدْ رَحِمْتَهٗ ؕ وَذٰلِكَ هُوَ الْفَوْزُ الْعَظِیْمُ ۟۠ إِنَّ ٱلَّذِينَ كَفَرُواْ يُنَادَوۡنَ لَمَقۡتُ ٱللَّهِ أَكۡبَرُ مِن مَّقۡتِكُمۡ أَنفُسَكُمۡ إِذۡ تُدۡعَوۡنَ إِلَى ٱلۡإِيمَٰنِ فَتَكۡفُرُونَ ۝١٠ إِنَّ ٱلَّذِینَ كَفَرُوا۟ یُنَادَوۡنَ لَمَقۡتُ ٱللَّهِ أَكۡبَرُ مِن مَّقۡتِكُمۡ أَنفُسَكُمۡ إِذۡ تُدۡعَوۡنَ إِلَى ٱلۡإِیمَٰنِ فَتَكۡفُرُونَ ۝١٠ اِنَّ الَّذِیْنَ كَفَرُوْا یُنَادَوْنَ لَمَقْتُ اللّٰهِ اَكْبَرُ مِنْ مَّقْتِكُمْ اَنْفُسَكُمْ اِذْ تُدْعَوْنَ اِلَی الْاِیْمَانِ فَتَكْفُرُوْنَ ۟ قَالُواْ رَبَّنَآ أَمَتَّنَا ٱثۡنَتَيۡنِ وَأَحۡيَيۡتَنَا ٱثۡنَتَيۡنِ فَٱعۡتَرَفۡنَا بِذُنُوبِنَا فَهَلۡ إِلَىٰ خُرُوجٖ مِّن سَبِيلٖ ۝١١ قَالُوا۟ رَبَّنَاۤ أَمَتَّنَا ٱثۡنَتَیۡنِ وَأَحۡیَیۡتَنَا ٱثۡنَتَیۡنِ فَٱعۡتَرَفۡنَا بِذُنُوبِنَا فَهَلۡ إِلَىٰ خُرُوجࣲ مِّن سَبِیلࣲ ۝١١ قَالُوْا رَبَّنَاۤ اَمَتَّنَا اثْنَتَیْنِ وَاَحْیَیْتَنَا اثْنَتَیْنِ فَاعْتَرَفْنَا بِذُنُوْبِنَا فَهَلْ اِلٰی خُرُوْجٍ مِّنْ سَبِیْلٍ ۟ ذَٰلِكُم بِأَنَّهُۥٓ إِذَا دُعِيَ ٱللَّهُ وَحۡدَهُۥ كَفَرۡتُمۡ وَإِن يُشۡرَكۡ بِهِۦ تُؤۡمِنُواْۚ فَٱلۡحُكۡمُ لِلَّهِ ٱلۡعَلِيِّ ٱلۡكَبِيرِ ۝١٢ ذَٰلِكُم بِأَنَّهُۥۤ إِذَا دُعِیَ ٱللَّهُ وَحۡدَهُۥ كَفَرۡتُمۡ وَإِن یُشۡرَكۡ بِهِۦ تُؤۡمِنُوا۟ۚ فَٱلۡحُكۡمُ لِلَّهِ ٱلۡعَلِیِّ ٱلۡكَبِیرِ ۝١٢ ذٰلِكُمْ بِاَنَّهٗۤ اِذَا دُعِیَ اللّٰهُ وَحْدَهٗ كَفَرْتُمْ ۚ وَاِنْ یُّشْرَكْ بِهٖ تُؤْمِنُوْا ؕ فَالْحُكْمُ لِلّٰهِ الْعَلِیِّ الْكَبِیْرِ ۟ هُوَ ٱلَّذِي يُرِيكُمۡ ءَايَٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزۡقٗاۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ۝١٣ هُوَ ٱلَّذِی یُرِیكُمۡ ءَایَٰتِهِۦ وَیُنَزِّلُ لَكُم مِّنَ ٱلسَّمَاۤءِ رِزۡقࣰاۚ وَمَا یَتَذَكَّرُ إِلَّا مَن یُنِیبُ ۝١٣ هُوَ الَّذِیْ یُرِیْكُمْ اٰیٰتِهٖ وَیُنَزِّلُ لَكُمْ مِّنَ السَّمَآءِ رِزْقًا ؕ وَمَا یَتَذَكَّرُ اِلَّا مَنْ یُّنِیْبُ ۟ فَٱدۡعُواْ ٱللَّهَ مُخۡلِصِينَ لَهُ ٱلدِّينَ وَلَوۡ كَرِهَ ٱلۡكَٰفِرُونَ ۝١٤ فَٱدۡعُوا۟ ٱللَّهَ مُخۡلِصِینَ لَهُ ٱلدِّینَ وَلَوۡ كَرِهَ ٱلۡكَٰفِرُونَ ۝١٤ فَادْعُوا اللّٰهَ مُخْلِصِیْنَ لَهُ الدِّیْنَ وَلَوْ كَرِهَ الْكٰفِرُوْنَ ۟ رَفِيعُ ٱلدَّرَجَٰتِ ذُو ٱلۡعَرۡشِ يُلۡقِي ٱلرُّوحَ مِنۡ أَمۡرِهِۦ عَلَىٰ مَن يَشَآءُ مِنۡ عِبَادِهِۦ لِيُنذِرَ يَوۡمَ ٱلتَّلَاقِ ۝١٥ رَفِیعُ ٱلدَّرَجَٰتِ ذُو ٱلۡعَرۡشِ یُلۡقِی ٱلرُّوحَ مِنۡ أَمۡرِهِۦ عَلَىٰ مَن یَشَاۤءُ مِنۡ عِبَادِهِۦ لِیُنذِرَ یَوۡمَ ٱلتَّلَاقِ ۝١٥ رَفِیْعُ الدَّرَجٰتِ ذُو الْعَرْشِ ۚ یُلْقِی الرُّوْحَ مِنْ اَمْرِهٖ عَلٰی مَنْ یَّشَآءُ مِنْ عِبَادِهٖ لِیُنْذِرَ یَوْمَ التَّلَاقِ ۟ۙ يَوۡمَ هُم بَٰرِزُونَۖ لَا يَخۡفَىٰ عَلَى ٱللَّهِ مِنۡهُمۡ شَيۡءٞۚ لِّمَنِ ٱلۡمُلۡكُ ٱلۡيَوۡمَۖ لِلَّهِ ٱلۡوَٰحِدِ ٱلۡقَهَّارِ ۝١٦ یَوۡمَ هُم بَٰرِزُونَۖ لَا یَخۡفَىٰ عَلَى ٱللَّهِ مِنۡهُمۡ شَیۡءࣱۚ لِّمَنِ ٱلۡمُلۡكُ ٱلۡیَوۡمَۖ لِلَّهِ ٱلۡوَٰحِدِ ٱلۡقَهَّارِ ۝١٦ یَوْمَ هُمْ بٰرِزُوْنَ ۚ۬ لَا یَخْفٰی عَلَی اللّٰهِ مِنْهُمْ شَیْءٌ ؕ لِمَنِ الْمُلْكُ الْیَوْمَ ؕ لِلّٰهِ الْوَاحِدِ الْقَهَّارِ ۟ ٱلۡيَوۡمَ تُجۡزَىٰ كُلُّ نَفۡسِۭ بِمَا كَسَبَتۡۚ لَا ظُلۡمَ ٱلۡيَوۡمَۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلۡحِسَابِ ۝١٧ ٱلۡیَوۡمَ تُجۡزَىٰ كُلُّ نَفۡسِۭ بِمَا كَسَبَتۡۚ لَا ظُلۡمَ ٱلۡیَوۡمَۚ إِنَّ ٱللَّهَ سَرِیعُ ٱلۡحِسَابِ ۝١٧ اَلْیَوْمَ تُجْزٰی كُلُّ نَفْسٍ بِمَا ٱللَّهِ مِن وَاقࣲ ۝٢١ اَوَلَمْ یَسِیْرُوْا فِی الْاَرْضِ فَیَنْظُرُوْا كَیْفَ كَانَ عَاقِبَةُ الَّذِیْنَ كَانُوْا مِنْ قَبْلِهِمْ ؕ كَانُوْا هُمْ اَشَدَّ مِنْهُمْ قُوَّةً وَّاٰثَارًا فِی الْاَرْضِ الَّذِیْنَ اٰمَنُوْا مَعَهٗ وَاسْتَحْیُوْا نِسَآءَهُمْ ؕ وَمَا كَیْدُ الْكٰفِرِیْنَ اِلَّا فِیْ ضَلٰلٍ ۟ وَقَالَ فِرۡعَوۡنُ ذَرُونِيٓ أَقۡتُلۡ مُوسَىٰ وَلۡيَدۡعُ رَبَّهُۥٓۖ إِنِّيٓ أَخَافُ أَن ٱلنَّارِ فَيَقُولُ ٱلضُّعَفَٰٓؤُاْ لِلَّذِينَ ٱسۡتَكۡبَرُوٓاْ إِنَّا كُنَّا لَكُمۡ تَبَعٗا فَهَلۡ أَنتُم مُّغۡنُونَ عَنَّا نَصِيبٗا مِّنَ ٱلنَّارِ ۝٤٧ وَإِذۡ یَتَحَاۤجُّونَ فِی ٱلنَّارِ فَیَقُولُ ٱلضُّعَفَٰۤؤُا۟ لِلَّذِینَ ٱسۡتَكۡبَرُوۤا۟ إِنَّا كُنَّا لَكُمۡ تَبَعࣰا فَهَلۡ أَنتُم مُّغۡنُونَ عَنَّا نَصِیبࣰا مِّنَ ٱلنَّارِ ۝٤٧ وَاِذْ یَتَحَآجُّوْنَ فِی النَّارِ فَیَقُوْلُ الضُّعَفٰٓؤُا لِلَّذِیْنَ اسْتَكْبَرُوْۤا اِنَّا كُنَّا لَكُمْ تَبَعًا فَهَلْ اَنْتُمْ مُّغْنُوْنَ عَنَّا نَصِیْبًا مِّنَ النَّارِ ۟ قَالَ ٱلَّذِينَ ٱسۡتَكۡبَرُوٓاْ إِنَّا كُلّٞ فِيهَآ إِنَّ ٱللَّهَ قَدۡ حَكَمَ بَيۡنَ ٱلۡعِبَادِ ۝٤٨ قَالَ ٱلَّذِینَ ٱسۡتَكۡبَرُوۤا۟ إِنَّا كُلࣱّ فِیهَاۤ إِنَّ ٱللَّهَ قَدۡ حَكَمَ بَیۡنَ ٱلۡعِبَادِ ۝٤٨ قَالَ الَّذِیْنَ اسْتَكْبَرُوْۤا اِنَّا كُلٌّ فِیْهَاۤ ۙ اِنَّ اللّٰهَ قَدْ حَكَمَ بَیْنَ الْعِبَادِ ۟ وَقَالَ ٱلَّذِينَ فِي ٱلنَّارِ لِخَزَنَةِ جَهَنَّمَ ٱدۡعُواْ رَبَّكُمۡ يُخَفِّفۡ عَنَّا يَوۡمٗا مِّنَ ٱلۡعَذَابِ ۝٤٩ وَقَالَ ٱلَّذِینَ فِی ٱلنَّارِ لِخَزَنَةِ جَهَنَّمَ ٱدۡعُوا۟ رَبَّكُمۡ یُخَفِّفۡ عَنَّا یَوۡمࣰا مِّنَ ٱلۡعَذَابِ ۝٤٩ وَقَالَ الَّذِیْنَ فِی النَّارِ لِخَزَنَةِ جَهَنَّمَ ادْعُوْا رَبَّكُمْ یُخَفِّفْ عَنَّا یَوْمًا مِّنَ الْعَذَابِ ۟ قَالُوٓاْ أَوَلَمۡ تَكُ تَأۡتِيكُمۡ رُسُلُكُم بِٱلۡبَيِّنَٰتِۖ قَالُواْ بَلَىٰۚ قَالُواْ فَٱدۡعُواْۗ وَمَا دُعَٰٓؤُاْ ٱلۡكَٰفِرِينَ إِلَّا فِي ضَلَٰلٍ ۝٥٠ قَالُوۤا۟ أَوَلَمۡ تَكُ تَأۡتِیكُمۡ رُسُلُكُم بِٱلۡبَیِّنَٰتِۖ قَالُوا۟ بَلَىٰۚ قَالُوا۟ فَٱدۡعُوا۟ۗ وَمَا دُعَٰۤؤُا۟ ٱلۡكَٰفِرِینَ إِلَّا فِی ضَلَٰلٍ ۝٥٠ قَالُوْۤا اَوَلَمْ تَكُ تَاْتِیْكُمْ رُسُلُكُمْ بِالْبَیِّنٰتِ ؕ قَالُوْا بَلٰی ؕ قَالُوْا فَادْعُوْا ۚ وَمَا دُعٰٓؤُا الْكٰفِرِیْنَ اِلَّا فِیْ ضَلٰلٍ ۟۠ إِنَّا لَنَنصُرُ رُسُلَنَا وَٱلَّذِينَ ءَامَنُواْ فِي ٱلۡحَيَوٰةِ ٱلدُّنۡيَا وَيَوۡمَ يَقُومُ ٱلۡأَشۡهَٰدُ ۝٥١ إِنَّا لَنَنصُرُ رُسُلَنَا وَٱلَّذِینَ ءَامَنُوا۟ فِی ٱلۡحَیَوٰةِ ٱلدُّنۡیَا وَیَوۡمَ یَقُومُ ٱلۡأَشۡهَٰدُ ۝٥١ اِنَّا لَنَنْصُرُ رُسُلَنَا وَالَّذِیْنَ اٰمَنُوْا فِی الْحَیٰوةِ الدُّنْیَا وَیَوْمَ یَقُوْمُ الْاَشْهَادُ ۟ۙ يَوۡمَ لَا يَنفَعُ ٱلظَّٰلِمِينَ مَعۡذِرَتُهُمۡۖ وَلَهُمُ ٱللَّعۡنَةُ وَلَهُمۡ سُوٓءُ ٱلدَّارِ ۝٥٢ یَوۡمَ لَا یَنفَعُ ٱلظَّٰلِمِینَ مَعۡذِرَتُهُمۡۖ وَلَهُمُ ٱللَّعۡنَةُ وَلَهُمۡ سُوۤءُ ٱلدَّارِ ۝٥٢ یَوْمَ لَا یَنْفَعُ الظّٰلِمِیْنَ مَعْذِرَتُهُمْ وَلَهُمُ اللَّعْنَةُ وَلَهُمْ سُوْٓءُ الدَّارِ ۟ وَلَقَدۡ ءَاتَيۡنَا مُوسَى ٱلۡهُدَىٰ وَأَوۡرَثۡنَا بَنِيٓ إِسۡرَٰٓءِيلَ ٱلۡكِتَٰبَ ۝٥٣ وَلَقَدۡ ءَاتَیۡنَا مُوسَى ٱلۡهُدَىٰ وَأَوۡرَثۡنَا بَنِیۤ إِسۡرَٰۤءِیلَ ٱلۡكِتَٰبَ ۝٥٣ وَلَقَدْ اٰتَیْنَا مُوْسَی الْهُدٰی وَاَوْرَثْنَا بَنِیْۤ اِسْرَآءِیْلَ الْكِتٰبَ ۟ۙ هُدٗى وَذِكۡرَىٰ لِأُوْلِي ٱلۡأَلۡبَٰبِ ۝٥٤ هُدࣰى وَذِكۡرَىٰ لِأُو۟لِی ٱلۡأَلۡبَٰبِ ۝٥٤ هُدًی وَّذِكْرٰی لِاُولِی الْاَلْبَابِ ۟ فَٱصۡبِرۡ إِنَّ وَعۡدَ ٱللَّهِ حَقّٞ وَٱسۡتَغۡفِرۡ لِذَنۢبِكَ وَسَبِّحۡ بِحَمۡدِ رَبِّكَ بِٱلۡعَشِيِّ وَٱلۡإِبۡكَٰرِ ۝٥٥ فَٱصۡبِرۡ إِنَّ وَعۡدَ ٱللَّهِ حَقࣱّ وَٱسۡتَغۡفِرۡ لِذَنۢبِكَ وَسَبِّحۡ بِحَمۡدِ رَبِّكَ بِٱلۡعَشِیِّ وَٱلۡإِبۡكَٰرِ ۝٥٥ فَاصْبِرْ اِنَّ وَعْدَ اللّٰهِ حَقٌّ وَّاسْتَغْفِرْ لِذَنْۢبِكَ وَسَبِّحْ بِحَمْدِ رَبِّكَ بِالْعَشِیِّ وَالْاِبْكَارِ ۟ إِنَّ ٱلَّذِينَ يُجَٰدِلُونَ فِيٓ ءَايَٰتِ ٱللَّهِ بِغَيۡرِ سُلۡطَٰنٍ أَتَىٰهُمۡ إِن فِي صُدُورِهِمۡ إِلَّا كِبۡرٞ مَّا هُم بِبَٰلِغِيهِۚ فَٱسۡتَعِذۡ بِٱللَّهِۖ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلۡبَصِيرُ ۝٥٦ إِنَّ ٱلَّذِینَ یُجَٰدِلُونَ فِیۤ ءَایَٰتِ ٱللَّهِ بِغَیۡرِ سُلۡطَٰنٍ أَتَىٰهُمۡ إِن فِی صُدُورِهِمۡ إِلَّا كِبۡرࣱ مَّا هُم بِبَٰلِغِیهِۚ فَٱسۡتَعِذۡ بِٱللَّهِۖ إِنَّهُۥ هُوَ ٱلسَّمِیعُ ٱلۡبَصِیرُ ۝٥٦ اِنَّ الَّذِیْنَ یُجَادِلُوْنَ فِیْۤ اٰیٰتِ اللّٰهِ بِغَیْرِ سُلْطٰنٍ اَتٰىهُمْ ۙ اِنْ فِیْ صُدُوْرِهِمْ اِلَّا كِبْرٌ مَّا هُمْ بِبَالِغِیْهِ ۚ فَاسْتَعِذْ بِاللّٰهِ ؕ اِنَّهٗ هُوَ السَّمِیْعُ الْبَصِیْرُ ۟ لَخَلۡقُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ أَكۡبَرُ مِنۡ خَلۡقِ ٱلنَّاسِ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يَعۡلَمُونَ ۝٥٧ لَخَلۡقُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ أَكۡبَرُ مِنۡ خَلۡقِ ٱلنَّاسِ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا یَعۡلَمُونَ ۝٥٧ لَخَلْقُ السَّمٰوٰتِ وَالْاَرْضِ اَكْبَرُ مِنْ خَلْقِ النَّاسِ وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا یَعْلَمُوْنَ ۟ وَمَا يَسۡتَوِي ٱلۡأَعۡمَىٰ وَٱلۡبَصِيرُ وَٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ وَلَا ٱلۡمُسِيٓءُۚ قَلِيلٗا مَّا تَتَذَكَّرُونَ ۝٥٨ وَمَا یَسۡتَوِی ٱلۡأَعۡمَىٰ وَٱلۡبَصِیرُ وَٱلَّذِینَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ وَلَا ٱلۡمُسِیۤءُۚ قَلِیلࣰا مَّا تَتَذَكَّرُونَ ۝٥٨ وَمَا یَسْتَوِی الْاَعْمٰی وَالْبَصِیْرُ ۙ۬ وَالَّذِیْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَلَا الْمُسِیْٓءُ ؕ قَلِیْلًا مَّا تَتَذَكَّرُوْنَ ۟ إِنَّ ٱلسَّاعَةَ لَأٓتِيَةٞ لَّا رَيۡبَ فِيهَا وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يُؤۡمِنُونَ ۝٥٩ إِنَّ ٱلسَّاعَةَ لَءَاتِیَةࣱ لَّا رَیۡبَ فِیهَا وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا یُؤۡمِنُونَ ۝٥٩ اِنَّ السَّاعَةَ لَاٰتِیَةٌ لَّا رَیْبَ فِیْهَا وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا یُؤْمِنُوْنَ ۟ وَقَالَ رَبُّكُمُ ٱدۡعُونِيٓ أَسۡتَجِبۡ لَكُمۡۚ إِنَّ ٱلَّذِينَ يَسۡتَكۡبِرُونَ عَنۡ عِبَادَتِي سَيَدۡخُلُونَ جَهَنَّمَ دَاخِرِينَ ۝٦٠ وَقَالَ رَبُّكُمُ ٱدۡعُونِیۤ أَسۡتَجِبۡ لَكُمۡۚ إِنَّ ٱلَّذِینَ یَسۡتَكۡبِرُونَ عَنۡ عِبَادَتِی سَیَدۡخُلُونَ جَهَنَّمَ دَاخِرِینَ ۝٦٠ وَقَالَ رَبُّكُمُ ادْعُوْنِیْۤ اَسْتَجِبْ لَكُمْ ؕ اِنَّ الَّذِیْنَ یَسْتَكْبِرُوْنَ عَنْ عِبَادَتِیْ سَیَدْخُلُوْنَ جَهَنَّمَ دٰخِرِیْنَ ۟۠ ٱللَّهُ ٱلَّذِي جَعَلَ لَكُمُ ٱلَّيۡلَ لِتَسۡكُنُواْ فِيهِ وَٱلنَّهَارَ مُبۡصِرًاۚ إِنَّ ٱللَّهَ لَذُو فَضۡلٍ عَلَى ٱلنَّاسِ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا يَشۡكُرُونَ ۝٦١ ٱللَّهُ ٱلَّذِی جَعَلَ لَكُمُ ٱلَّیۡلَ لِتَسۡكُنُوا۟ فِیهِ وَٱلنَّهَارَ مُبۡصِرًاۚ إِنَّ ٱللَّهَ لَذُو فَضۡلٍ عَلَى ٱلنَّاسِ وَلَٰكِنَّ أَكۡثَرَ ٱلنَّاسِ لَا یَشۡكُرُونَ ۝٦١ اَللّٰهُ الَّذِیْ جَعَلَ لَكُمُ الَّیْلَ لَكُمُ ٱلۡأَنۡعَٰمَ لِتَرۡكَبُوا۟ مِنۡهَا وَمِنۡهَا تَأۡكُلُونَ ۝٧٩ اَللّٰهُ الَّذِیْ جَعَلَ لَكُمُ الْاَنْعَامَ لِتَرْكَبُوْا مِنْهَا وَمِنْهَا تَاْكُلُوْنَ ۟ؗ وَلَكُمۡ فِيهَا مَنَٰفِعُ وَلِتَبۡلُغُواْ عَلَيۡهَا حَاجَةٗ فِي صُدُورِكُمۡ وَعَلَيۡهَا وَعَلَى ٱلۡفُلۡكِ تُحۡمَلُونَ ۝٨٠ وَلَكُمۡ فِیهَا مَنَٰفِعُ وَلِتَبۡلُغُوا۟ عَلَیۡهَا حَاجَةࣰ فِی صُدُورِكُمۡ وَعَلَیۡهَا وَعَلَى ٱلۡفُلۡكِ تُحۡمَلُونَ ۝٨٠ وَلَكُمْ فِیْهَا مَنَافِعُ وَلِتَبْلُغُوْا عَلَیْهَا حَاجَةً فِیْ صُدُوْرِكُمْ وَعَلَیْهَا وَعَلَی الْفُلْكِ تُحْمَلُوْنَ ۟ؕ وَيُرِيكُمۡ ءَايَٰتِهِۦ فَأَيَّ ءَايَٰتِ ٱللَّهِ تُنكِرُونَ ۝٨١ وَیُرِیكُمۡ ءَایَٰتِهِۦ فَأَیَّ ءَایَٰتِ ٱللَّهِ تُنكِرُونَ ۝٨١ وَیُرِیْكُمْ اٰیٰتِهٖ ۖۗ فَاَیَّ اٰیٰتِ اللّٰهِ تُنْكِرُوْنَ ۟ أَفَلَمۡ يَسِيرُواْ فِي ٱلۡأَرۡضِ فَيَنظُرُواْ كَيۡفَ كَانَ عَٰقِبَةُ ٱلَّذِينَ مِن قَبۡلِهِمۡۚ كَانُوٓاْ أَكۡثَرَ مِنۡهُمۡ وَأَشَدَّ قُوَّةٗ وَءَاثَارٗا فِي ٱلۡأَرۡضِ فَمَآ أَغۡنَىٰ عَنۡهُم مَّا كَانُواْ يَكۡسِبُونَ ۝٨٢ أَفَلَمۡ یَسِیرُوا۟ فِی ٱلۡأَرۡضِ فَیَنظُرُوا۟ كَیۡفَ كَانَ عَٰقِبَةُ ٱلَّذِینَ مِن قَبۡلِهِمۡۚ كَانُوۤا۟ أَكۡثَرَ مِنۡهُمۡ وَأَشَدَّ قُوَّةࣰ وَءَاثَارࣰا فِی ٱلۡأَرۡضِ فَمَاۤ أَغۡنَىٰ عَنۡهُم مَّا كَانُوا۟ یَكۡسِبُونَ ۝٨٢ اَفَلَمْ یَسِیْرُوْا فِی الْاَرْضِ فَیَنْظُرُوْا كَیْفَ كَانَ عَاقِبَةُ الَّذِیْنَ مِنْ قَبْلِهِمْ ؕ كَانُوْۤا اَكْثَرَ مِنْهُمْ وَاَشَدَّ قُوَّةً وَّاٰثَارًا فِی الْاَرْضِ فَمَاۤ اَغْنٰی عَنْهُمْ مَّا كَانُوْا یَكْسِبُوْنَ ۟ فَلَمَّا جَآءَتۡهُمۡ رُسُلُهُم بِٱلۡبَيِّنَٰتِ فَرِحُواْ بِمَا عِندَهُم مِّنَ ٱلۡعِلۡمِ وَحَاقَ بِهِم مَّا كَانُواْ بِهِۦ يَسۡتَهۡزِءُونَ ۝٨٣ فَلَمَّا جَاۤءَتۡهُمۡ رُسُلُهُم بِٱلۡبَیِّنَٰتِ فَرِحُوا۟ بِمَا عِندَهُم مِّنَ ٱلۡعِلۡمِ وَحَاقَ بِهِم مَّا كَانُوا۟ بِهِۦ یَسۡتَهۡزِءُونَ ۝٨٣ فَلَمَّا جَآءَتْهُمْ رُسُلُهُمْ بِالْبَیِّنٰتِ فَرِحُوْا بِمَا عِنْدَهُمْ مِّنَ الْعِلْمِ وَحَاقَ بِهِمْ مَّا كَانُوْا بِهٖ یَسْتَهْزِءُوْنَ ۟ فَلَمَّا رَأَوۡاْ بَأۡسَنَا قَالُوٓاْ ءَامَنَّا بِٱللَّهِ وَحۡدَهُۥ وَكَفَرۡنَا بِمَا كُنَّا بِهِۦ مُشۡرِكِينَ ۝٨٤ فَلَمَّا رَأَوۡا۟ بَأۡسَنَا قَالُوۤا۟ ءَامَنَّا بِٱللَّهِ وَحۡدَهُۥ وَكَفَرۡنَا بِمَا كُنَّا بِهِۦ مُشۡرِكِینَ ۝٨٤ فَلَمَّا رَاَوْا بَاْسَنَا قَالُوْۤا اٰمَنَّا بِاللّٰهِ وَحْدَهٗ وَكَفَرْنَا بِمَا كُنَّا بِهٖ مُشْرِكِیْنَ ۟ فَلَمۡ يَكُ يَنفَعُهُمۡ إِيمَٰنُهُمۡ لَمَّا رَأَوۡاْ بَأۡسَنَاۖ سُنَّتَ ٱللَّهِ ٱلَّتِي قَدۡ خَلَتۡ فِي عِبَادِهِۦۖ وَخَسِرَ هُنَالِكَ ٱلۡكَٰفِرُونَ ۝٨٥ فَلَمۡ یَكُ یَنفَعُهُمۡ إِیمَٰنُهُمۡ لَمَّا رَأَوۡا۟ بَأۡسَنَاۖ سُنَّتَ ٱللَّهِ ٱلَّتِی قَدۡ خَلَتۡ فِی عِبَادِهِۦۖ وَخَسِرَ هُنَالِكَ ٱلۡكَٰفِرُونَ ۝٨٥ فَلَمْ یَكُ یَنْفَعُهُمْ اِیْمَانُهُمْ لَمَّا رَاَوْا بَاْسَنَا ؕ سُنَّتَ اللّٰهِ الَّتِیْ قَدْ خَلَتْ فِیْ عِبَادِهٖ ۚ وَخَسِرَ هُنَالِكَ الْكٰفِرُوْنَ ۟۠
