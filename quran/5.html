بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ أَوۡفُواْ بِٱلۡعُقُودِۚ أُحِلَّتۡ لَكُم بَهِيمَةُ ٱلۡأَنۡعَٰمِ إِلَّا مَا يُتۡلَىٰ عَلَيۡكُمۡ غَيۡرَ مُحِلِّي ٱلصَّيۡدِ وَأَنتُمۡ حُرُمٌۗ إِنَّ ٱللَّهَ يَحۡكُمُ مَا يُرِيدُ ۝١ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوۤا۟ أَوۡفُوا۟ بِٱلۡعُقُودِۚ أُحِلَّتۡ لَكُم بَهِیمَةُ ٱلۡأَنۡعَٰمِ إِلَّا مَا یُتۡلَىٰ عَلَیۡكُمۡ غَیۡرَ مُحِلِّی ٱلصَّیۡدِ وَأَنتُمۡ حُرُمٌۗ إِنَّ ٱللَّهَ یَحۡكُمُ مَا یُرِیدُ ۝١ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْۤا اَوْفُوْا بِالْعُقُوْدِ ؕ۬ اُحِلَّتْ لَكُمْ بَهِیْمَةُ الْاَنْعَامِ اِلَّا مَا یُتْلٰی عَلَیْكُمْ غَیْرَ مُحِلِّی الصَّیْدِ وَاَنْتُمْ حُرُمٌ ؕ اِنَّ اللّٰهَ یَحْكُمُ مَا یُرِیْدُ ۟ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تُحِلُّواْ شَعَٰٓئِرَ ٱللَّهِ وَلَا ٱلشَّهۡرَ ٱلۡحَرَامَ وَلَا ٱلۡهَدۡيَ وَلَا ٱلۡقَلَٰٓئِدَ وَلَآ ءَآمِّينَ ٱلۡبَيۡتَ ٱلۡحَرَامَ يَبۡتَغُونَ فَضۡلٗا مِّن رَّبِّهِمۡ وَرِضۡوَٰنٗاۚ وَإِذَا حَلَلۡتُمۡ فَٱصۡطَادُواْۚ وَلَا يَجۡرِمَنَّكُمۡ شَنَـَٔانُ قَوۡمٍ أَن صَدُّوكُمۡ عَنِ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ أَن تَعۡتَدُواْۘ وَتَعَاوَنُواْ عَلَى ٱلۡبِرِّ وَٱلتَّقۡوَىٰۖ وَلَا تَعَاوَنُواْ عَلَى ٱلۡإِثۡمِ وَٱلۡعُدۡوَٰنِۚ وَٱتَّقُواْ ٱللَّهَۖ إِنَّ ٱللَّهَ شَدِيدُ ٱلۡعِقَابِ ۝٢ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ لَا تُحِلُّوا۟ شَعَٰۤئِرَ ٱللَّهِ وَلَا ٱلشَّهۡرَ ٱلۡحَرَامَ وَلَا ٱلۡهَدۡیَ وَلَا ٱلۡقَلَٰۤئِدَ وَلَاۤ ءَاۤمِّینَ ٱلۡبَیۡتَ ٱلۡحَرَامَ یَبۡتَغُونَ فَضۡلࣰا مِّن رَّبِّهِمۡ وَرِضۡوَٰنࣰاۚ وَإِذَا حَلَلۡتُمۡ فَٱصۡطَادُوا۟ۚ وَلَا یَجۡرِمَنَّكُمۡ شَنَءَانُ قَوۡمٍ أَن صَدُّوكُمۡ عَنِ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ أَن تَعۡتَدُوا۟ۘ وَتَعَاوَنُوا۟ عَلَى ٱلۡبِرِّ وَٱلتَّقۡوَىٰۖ وَلَا تَعَاوَنُوا۟ عَلَى ٱلۡإِثۡمِ وَٱلۡعُدۡوَٰنِۚ وَٱتَّقُوا۟ ٱللَّهَۖ إِنَّ ٱللَّهَ شَدِیدُ ٱلۡعِقَابِ ۝٢ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا لَا تُحِلُّوْا شَعَآىِٕرَ اللّٰهِ وَلَا الشَّهْرَ الْحَرَامَ وَلَا الْهَدْیَ وَلَا الْقَلَآىِٕدَ وَلَاۤ آٰمِّیْنَ الْبَیْتَ الْحَرَامَ یَبْتَغُوْنَ فَضْلًا مِّنْ رَّبِّهِمْ وَرِضْوَانًا ؕ وَاِذَا حَلَلْتُمْ فَاصْطَادُوْا ؕ وَلَا یَجْرِمَنَّكُمْ شَنَاٰنُ قَوْمٍ اَنْ صَدُّوْكُمْ عَنِ الْمَسْجِدِ الْحَرَامِ اَنْ تَعْتَدُوْا ۘ وَتَعَاوَنُوْا عَلَی الْبِرِّ وَالتَّقْوٰی ۪ وَلَا تَعَاوَنُوْا عَلَی الْاِثْمِ وَالْعُدْوَانِ ۪ وَاتَّقُوا اللّٰهَ ؕ اِنَّ اللّٰهَ شَدِیْدُ الْعِقَابِ ۟ حُرِّمَتۡ عَلَيۡكُمُ ٱلۡمَيۡتَةُ وَٱلدَّمُ وَلَحۡمُ ٱلۡخِنزِيرِ وَمَآ أُهِلَّ لِغَيۡرِ ٱللَّهِ بِهِۦ وَٱلۡمُنۡخَنِقَةُ وَٱلۡمَوۡقُوذَةُ وَٱلۡمُتَرَدِّيَةُ وَٱلنَّطِيحَةُ وَمَآ أَكَلَ ٱلسَّبُعُ إِلَّا مَا ذَكَّيۡتُمۡ وَمَا ذُبِحَ عَلَى ٱلنُّصُبِ وَأَن تَسۡتَقۡسِمُواْ بِٱلۡأَزۡلَٰمِۚ ذَٰلِكُمۡ فِسۡقٌۗ ٱلۡيَوۡمَ يَئِسَ ٱلَّذِينَ كَفَرُواْ مِن دِينِكُمۡ فَلَا تَخۡشَوۡهُمۡ وَٱخۡشَوۡنِۚ ٱلۡيَوۡمَ أَكۡمَلۡتُ لَكُمۡ دِينَكُمۡ وَأَتۡمَمۡتُ عَلَيۡكُمۡ نِعۡمَتِي وَرَضِيتُ لَكُمُ ٱلۡإِسۡلَٰمَ دِينٗاۚ فَمَنِ ٱضۡطُرَّ فِي مَخۡمَصَةٍ غَيۡرَ مُتَجَانِفٖ لِّإِثۡمٖ فَإِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٞ ۝٣ حُرِّمَتۡ عَلَیۡكُمُ ٱلۡمَیۡتَةُ وَٱلدَّمُ وَلَحۡمُ ٱلۡخِنزِیرِ وَمَاۤ أُهِلَّ لِغَیۡرِ ٱللَّهِ بِهِۦ وَٱلۡمُنۡخَنِقَةُ وَٱلۡمَوۡقُوذَةُ وَٱلۡمُتَرَدِّیَةُ وَٱلنَّطِیحَةُ وَمَاۤ أَكَلَ ٱلسَّبُعُ إِلَّا مَا ذَكَّیۡتُمۡ وَمَا ذُبِحَ عَلَى ٱلنُّصُبِ وَأَن تَسۡتَقۡسِمُوا۟ بِٱلۡأَزۡلَٰمِۚ ذَٰلِكُمۡ فِسۡقٌۗ ٱلۡیَوۡمَ یَئِسَ ٱلَّذِینَ كَفَرُوا۟ مِن دِینِكُمۡ فَلَا تَخۡشَوۡهُمۡ وَٱخۡشَوۡنِۚ ٱلۡیَوۡمَ أَكۡمَلۡتُ لَكُمۡ دِینَكُمۡ وَأَتۡمَمۡتُ عَلَیۡكُمۡ نِعۡمَتِی وَرَضِیتُ لَكُمُ ٱلۡإِسۡلَٰمَ دِینࣰاۚ فَمَنِ ٱضۡطُرَّ فِی مَخۡمَصَةٍ غَیۡرَ مُتَجَانِفࣲ لِّإِثۡمࣲ فَإِنَّ ٱللَّهَ غَفُورࣱ رَّحِیمࣱ ۝٣ حُرِّمَتْ عَلَیْكُمُ الْمَیْتَةُ وَالدَّمُ وَلَحْمُ الْخِنْزِیْرِ وَمَاۤ اُهِلَّ لِغَیْرِ اللّٰهِ بِهٖ وَالْمُنْخَنِقَةُ وَالْمَوْقُوْذَةُ وَالْمُتَرَدِّیَةُ وَالنَّطِیْحَةُ وَمَاۤ اَكَلَ السَّبُعُ اِلَّا مَا ذَكَّیْتُمْ ۫ وَمَا ذُبِحَ عَلَی النُّصُبِ وَاَنْ تَسْتَقْسِمُوْا بِالْاَزْلَامِ ؕ ذٰلِكُمْ فِسْقٌ ؕ اَلْیَوْمَ یَىِٕسَ الَّذِیْنَ كَفَرُوْا مِنْ دِیْنِكُمْ فَلَا تَخْشَوْهُمْ وَاخْشَوْنِ ؕ اَلْیَوْمَ اَكْمَلْتُ لَكُمْ دِیْنَكُمْ وَاَتْمَمْتُ عَلَیْكُمْ نِعْمَتِیْ وَرَضِیْتُ لَكُمُ الْاِسْلَامَ دِیْنًا ؕ فَمَنِ اضْطُرَّ فِیْ مَخْمَصَةٍ غَیْرَ مُتَجَانِفٍ لِّاِثْمٍ ۙ فَاِنَّ اللّٰهَ غَفُوْرٌ رَّحِیْمٌ ۟ يَسۡـَٔلُونَكَ مَاذَآ أُحِلَّ لَهُمۡۖ قُلۡ أُحِلَّ لَكُمُ ٱلطَّيِّبَٰتُ وَمَا عَلَّمۡتُم مِّنَ ٱلۡجَوَارِحِ مُكَلِّبِينَ تُعَلِّمُونَهُنَّ مِمَّا عَلَّمَكُمُ ٱللَّهُۖ فَكُلُواْ مِمَّآ أَمۡسَكۡنَ عَلَيۡكُمۡ وَٱذۡكُرُواْ ٱسۡمَ ٱللَّهِ عَلَيۡهِۖ وَٱتَّقُواْ ٱللَّهَۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلۡحِسَابِ ۝٤ یَسۡءَلُونَكَ مَاذَاۤ أُحِلَّ لَهُمۡۖ قُلۡ أُحِلَّ لَكُمُ ٱلطَّیِّبَٰتُ وَمَا عَلَّمۡتُم مِّنَ ٱلۡجَوَارِحِ مُكَلِّبِینَ تُعَلِّمُونَهُنَّ مِمَّا عَلَّمَكُمُ ٱللَّهُۖ فَكُلُوا۟ مِمَّاۤ أَمۡسَكۡنَ عَلَیۡكُمۡ وَٱذۡكُرُوا۟ ٱسۡمَ ٱللَّهِ عَلَیۡهِۖ وَٱتَّقُوا۟ ٱللَّهَۚ إِنَّ ٱللَّهَ سَرِیعُ ٱلۡحِسَابِ ۝٤ یَسْـَٔلُوْنَكَ مَاذَاۤ اُحِلَّ لَهُمْ ؕ قُلْ اُحِلَّ لَكُمُ الطَّیِّبٰتُ ۙ وَمَا عَلَّمْتُمْ مِّنَ الْجَوَارِحِ مُكَلِّبِیْنَ تُعَلِّمُوْنَهُنَّ مِمَّا عَلَّمَكُمُ اللّٰهُ ؗ فَكُلُوْا مِمَّاۤ اَمْسَكْنَ عَلَیْكُمْ وَاذْكُرُوا اسْمَ اللّٰهِ عَلَیْهِ ۪ وَاتَّقُوا اللّٰهَ ؕ اِنَّ اللّٰهَ سَرِیْعُ الْحِسَابِ ۟ ٱلۡيَوۡمَ أُحِلَّ لَكُمُ ٱلطَّيِّبَٰتُۖ وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلۡكِتَٰبَ حِلّٞ لَّكُمۡ وَطَعَامُكُمۡ حِلّٞ لَّهُمۡۖ وَٱلۡمُحۡصَنَٰتُ مِنَ ٱلۡمُؤۡمِنَٰتِ وَٱلۡمُحۡصَنَٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلۡكِتَٰبَ مِن قَبۡلِكُمۡ إِذَآ ءَاتَيۡتُمُوهُنَّ أُجُورَهُنَّ مُحۡصِنِينَ غَيۡرَ مُسَٰفِحِينَ وَلَا مُتَّخِذِيٓ أَخۡدَانٖۗ وَمَن يَكۡفُرۡ بِٱلۡإِيمَٰنِ فَقَدۡ حَبِطَ عَمَلُهُۥ وَهُوَ فِي ٱلۡأٓخِرَةِ مِنَ ٱلۡخَٰسِرِينَ ۝٥ ٱلۡیَوۡمَ أُحِلَّ لَكُمُ ٱلطَّیِّبَٰتُۖ وَطَعَامُ ٱلَّذِینَ أُوتُوا۟ ٱلۡكِتَٰبَ حِلࣱّ لَّكُمۡ وَطَعَامُكُمۡ حِلࣱّ لَّهُمۡۖ وَٱلۡمُحۡصَنَٰتُ مِنَ ٱلۡمُؤۡمِنَٰتِ وَٱلۡمُحۡصَنَٰتُ مِنَ ٱلَّذِینَ أُوتُوا۟ ٱلۡكِتَٰبَ مِن قَبۡلِكُمۡ إِذَاۤ ءَاتَیۡتُمُوهُنَّ أُجُورَهُنَّ مُحۡصِنِینَ غَیۡرَ مُسَٰفِحِینَ وَلَا مُتَّخِذِیۤ أَخۡدَانࣲۗ وَمَن یَكۡفُرۡ بِٱلۡإِیمَٰنِ فَقَدۡ حَبِطَ عَمَلُهُۥ وَهُوَ فِی ٱلۡءَاخِرَةِ مِنَ ٱلۡخَٰسِرِینَ ۝٥ اَلْیَوْمَ اُحِلَّ لَكُمُ الطَّیِّبٰتُ ؕ وَطَعَامُ الَّذِیْنَ اُوْتُوا الْكِتٰبَ حِلٌّ لَّكُمْ ۪ وَطَعَامُكُمْ حِلٌّ لَّهُمْ ؗ وَالْمُحْصَنٰتُ مِنَ الْمُؤْمِنٰتِ وَالْمُحْصَنٰتُ مِنَ الَّذِیْنَ اُوْتُوا الْكِتٰبَ مِنْ قَبْلِكُمْ اِذَاۤ اٰتَیْتُمُوْهُنَّ اُجُوْرَهُنَّ مُحْصِنِیْنَ غَیْرَ مُسٰفِحِیْنَ وَلَا مُتَّخِذِیْۤ اَخْدَانٍ ؕ وَمَنْ یَّكْفُرْ بِالْاِیْمَانِ فَقَدْ حَبِطَ عَمَلُهٗ ؗ وَهُوَ فِی الْاٰخِرَةِ مِنَ الْخٰسِرِیْنَ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا قُمۡتُمۡ إِلَى ٱلصَّلَوٰةِ فَٱغۡسِلُواْ وُجُوهَكُمۡ وَأَيۡدِيَكُمۡ إِلَى ٱلۡمَرَافِقِ وَٱمۡسَحُواْ بِرُءُوسِكُمۡ وَأَرۡجُلَكُمۡ إِلَى ٱلۡكَعۡبَيۡنِۚ وَإِن كُنتُمۡ جُنُبٗا فَٱطَّهَّرُواْۚ وَإِن كُنتُم مَّرۡضَىٰٓ أَوۡ عَلَىٰ سَفَرٍ أَوۡ جَآءَ أَحَدٞ مِّنكُم مِّنَ ٱلۡغَآئِطِ أَوۡ لَٰمَسۡتُمُ ٱلنِّسَآءَ فَلَمۡ تَجِدُواْ مَآءٗ فَتَيَمَّمُواْ صَعِيدٗا طَيِّبٗا فَٱمۡسَحُواْ بِوُجُوهِكُمۡ وَأَيۡدِيكُم مِّنۡهُۚ مَا يُرِيدُ ٱللَّهُ لِيَجۡعَلَ عَلَيۡكُم مِّنۡ حَرَجٖ ٱلَّذِینَ ءَامَنُوا۟ كُونُوا۟ قَوَّٰمِینَ لِلَّهِ شُهَدَاۤءَ بِٱلۡقِسۡطِۖ وَلَا یَجۡرِمَنَّكُمۡ شَنَءَانُ قَوۡمٍ عَلَىٰۤ أَلَّا تَعۡدِلُوا۟ۚ ٱعۡدِلُوا۟ هُوَ أَقۡرَبُ لِلتَّقۡوَىٰۖ وَٱتَّقُوا۟ ٱللَّهَۚ دَخَلۡتُمُوهُ فَإِنَّكُمۡ غَٰلِبُونَۚ وَعَلَى ٱللَّهِ فَتَوَكَّلُوٓاْ إِن كُنتُم مُّؤۡمِنِينَ ۝٢٣ قَالَ رَجُلَانِ مِنَ ٱلَّذِینَ یَخَافُونَ أَنۡعَمَ ٱللَّهُ عَلَیۡهِمَا ٱدۡخُلُوا۟ عَلَیۡهِمُ ٱلۡبَابَ الْبَابَ ۚ فَاِذَا دَخَلْتُمُوْهُ فَاِنَّكُمْ غٰلِبُوْنَ ۚ۬ وَعَلَی اللّٰهِ فَتَوَكَّلُوْۤا اِنْ كُنْتُمْ مُّؤْمِنِیْنَ ۟ قَالُواْ يَٰمُوسَىٰٓ إِنَّا لَن نَّدۡخُلَهَآ أَبَدٗا مَّا دَامُواْ فِيهَا فَٱذۡهَبۡ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ ۝٢٤ قَالُوا۟ یَٰمُوسَىٰۤ إِنَّا لَن نَّدۡخُلَهَاۤ أَبَدࣰا مَّا دَامُوا۟ فِیهَا فَٱذۡهَبۡ أَنتَ وَرَبُّكَ فَقَٰتِلَاۤ إِنَّا هَٰهُنَا قَٰعِدُونَ ۝٢٤ قَالُوْا یٰمُوْسٰۤی اِنَّا لَنْ نَّدْخُلَهَاۤ اَبَدًا مَّا دَامُوْا فِیْهَا فَاذْهَبْ اَنْتَ وَرَبُّكَ فَقَاتِلَاۤ اِنَّا هٰهُنَا قٰعِدُوْنَ ۟ قَالَ رَبِّ إِنِّي لَآ أَمۡلِكُ إِلَّا نَفۡسِي وَأَخِيۖ فَٱفۡرُقۡ بَيۡنَنَا وَبَيۡنَ ٱلۡقَوۡمِ ٱلۡفَٰسِقِينَ ۝٢٥ قَالَ رَبِّ إِنِّی لَاۤ أَمۡلِكُ إِلَّا نَفۡسِی وَأَخِیۖ فَٱفۡرُقۡ بَیۡنَنَا وَبَیۡنَ ٱلۡقَوۡمِ ٱلۡفَٰسِقِینَ ۝٢٥ قَالَ رَبِّ اِنِّیْ لَاۤ اَمْلِكُ اِلَّا نَفْسِیْ وَاَخِیْ فَافْرُقْ بَیْنَنَا وَبَیْنَ الْقَوْمِ الْفٰسِقِیْنَ ۟ قَالَ فَإِنَّهَا مُحَرَّمَةٌ عَلَيۡهِمۡۛ أَرۡبَعِينَ سَنَةٗۛ يَتِيهُونَ فِي ٱلۡأَرۡضِۚ فَلَا تَأۡسَ عَلَى ٱلۡقَوۡمِ ٱلۡفَٰسِقِينَ ۝٢٦ قَالَ فَإِنَّهَا مُحَرَّمَةٌ عَلَیۡهِمۡۛ أَرۡبَعِینَ سَنَةࣰۛ یَتِیهُونَ فِی ٱلۡأَرۡضِۚ فَلَا تَأۡسَ عَلَى ٱلۡقَوۡمِ ٱلۡفَٰسِقِینَ ۝٢٦ قَالَ فَاِنَّهَا مُحَرَّمَةٌ عَلَیْهِمْ اَرْبَعِیْنَ سَنَةً ۚ یَتِیْهُوْنَ فِی الْاَرْضِ ؕ فَلَا تَاْسَ عَلَی الْقَوْمِ الْفٰسِقِیْنَ ۟۠ وَٱتۡلُ عَلَيۡهِمۡ نَبَأَ ٱبۡنَيۡ ءَادَمَ بِٱلۡحَقِّ إِذۡ قَرَّبَا قُرۡبَانٗا فَتُقُبِّلَ مِنۡ أَحَدِهِمَا وَلَمۡ يُتَقَبَّلۡ مِنَ ٱلۡأٓخَرِ قَالَ لَأَقۡتُلَنَّكَۖ قَالَ إِنَّمَا يَتَقَبَّلُ ٱللَّهُ مِنَ ٱلۡمُتَّقِينَ ۝٢٧ وَٱتۡلُ عَلَیۡهِمۡ نَبَأَ ٱبۡنَیۡ ءَادَمَ بِٱلۡحَقِّ إِذۡ قَرَّبَا قُرۡبَانࣰا فَتُقُبِّلَ مِنۡ أَحَدِهِمَا وَلَمۡ یُتَقَبَّلۡ مِنَ ٱلۡءَاخَرِ قَالَ لَأَقۡتُلَنَّكَۖ قَالَ إِنَّمَا یَتَقَبَّلُ ٱللَّهُ مِنَ ٱلۡمُتَّقِینَ ۝٢٧ وَاتْلُ عَلَیْهِمْ نَبَاَ ابْنَیْ اٰدَمَ بِالْحَقِّ ۘ اِذْ قَرَّبَا قُرْبَانًا فَتُقُبِّلَ مِنْ اَحَدِهِمَا وَلَمْ یُتَقَبَّلْ مِنَ الْاٰخَرِ ؕ قَالَ لَاَقْتُلَنَّكَ ؕ قَالَ اِنَّمَا یَتَقَبَّلُ اللّٰهُ مِنَ الْمُتَّقِیْنَ ۟ لَئِنۢ بَسَطتَ إِلَيَّ يَدَكَ لِتَقۡتُلَنِي مَآ أَنَا۠ بِبَاسِطٖ يَدِيَ إِلَيۡكَ لِأَقۡتُلَكَۖ إِنِّيٓ أَخَافُ ٱللَّهَ رَبَّ ٱلۡعَٰلَمِينَ ۝٢٨ لَئِنۢ بَسَطتَ إِلَیَّ یَدَكَ لِتَقۡتُلَنِی مَاۤ أَنَا۠ بِبَاسِطࣲ یَدِیَ إِلَیۡكَ لِأَقۡتُلَكَۖ إِنِّیۤ أَخَافُ ٱللَّهَ رَبَّ ٱلۡعَٰلَمِینَ ۝٢٨ لَىِٕنْۢ بَسَطْتَّ اِلَیَّ یَدَكَ لِتَقْتُلَنِیْ مَاۤ اَنَا بِبَاسِطٍ یَّدِیَ اِلَیْكَ لِاَقْتُلَكَ ۚ اِنِّیْۤ اَخَافُ اللّٰهَ رَبَّ الْعٰلَمِیْنَ ۟ إِنِّيٓ أُرِيدُ أَن تَبُوٓأَ بِإِثۡمِي وَإِثۡمِكَ فَتَكُونَ مِنۡ أَصۡحَٰبِ ٱلنَّارِۚ وَذَٰلِكَ جَزَٰٓؤُاْ ٱلظَّٰلِمِينَ ۝٢٩ إِنِّیۤ أُرِیدُ أَن تَبُوۤأَ بِإِثۡمِی وَإِثۡمِكَ فَتَكُونَ مِنۡ أَصۡحَٰبِ ٱلنَّارِۚ وَذَٰلِكَ جَزَٰۤؤُا۟ ٱلظَّٰلِمِینَ ۝٢٩ اِنِّیْۤ اُرِیْدُ اَنْ تَبُوْٓءَاۡ بِاِثْمِیْ وَاِثْمِكَ فَتَكُوْنَ مِنْ اَصْحٰبِ النَّارِ ۚ وَذٰلِكَ جَزٰٓؤُا الظّٰلِمِیْنَ ۟ۚ فَطَوَّعَتۡ لَهُۥ نَفۡسُهُۥ قَتۡلَ أَخِيهِ فَقَتَلَهُۥ فَأَصۡبَحَ مِنَ ٱلۡخَٰسِرِينَ ۝٣٠ فَطَوَّعَتۡ لَهُۥ نَفۡسُهُۥ قَتۡلَ أَخِیهِ فَقَتَلَهُۥ فَأَصۡبَحَ مِنَ ٱلۡخَٰسِرِینَ ۝٣٠ فَطَوَّعَتْ لَهٗ نَفْسُهٗ قَتْلَ اَخِیْهِ فَقَتَلَهٗ فَاَصْبَحَ مِنَ الْخٰسِرِیْنَ ۟ فَبَعَثَ ٱللَّهُ غُرَابٗا يَبۡحَثُ فِي ٱلۡأَرۡضِ لِيُرِيَهُۥ كَيۡفَ يُوَٰرِي سَوۡءَةَ أَخِيهِۚ قَالَ يَٰوَيۡلَتَىٰٓ أَعَجَزۡتُ أَنۡ أَكُونَ مِثۡلَ هَٰذَا ٱلۡغُرَابِ فَأُوَٰرِيَ سَوۡءَةَ أَخِيۖ فَأَصۡبَحَ مِنَ ٱلنَّٰدِمِينَ ۝٣١ فَبَعَثَ ٱللَّهُ غُرَابࣰا یَبۡحَثُ فِی ٱلۡأَرۡضِ لِیُرِیَهُۥ كَیۡفَ یُوَٰرِی سَوۡءَةَ أَخِیهِۚ قَالَ یَٰوَیۡلَتَىٰۤ أَعَجَزۡتُ أَنۡ أَكُونَ مِثۡلَ هَٰذَا ٱلۡغُرَابِ فَأُوَٰرِیَ سَوۡءَةَ أَخِیۖ فَأَصۡبَحَ مِنَ ٱلنَّٰدِمِینَ ۝٣١ فَبَعَثَ اللّٰهُ غُرَابًا یَّبْحَثُ فِی الْاَرْضِ لِیُرِیَهٗ كَیْفَ یُوَارِیْ سَوْءَةَ اَخِیْهِ ؕ قَالَ یٰوَیْلَتٰۤی اَعَجَزْتُ اَنْ اَكُوْنَ مِثْلَ هٰذَا الْغُرَابِ فَاُوَارِیَ سَوْءَةَ اَخِیْ ۚ فَاَصْبَحَ مِنَ النّٰدِمِیْنَ ۟ مِنۡ عَذَابٌ عَظِیْمٌ ۟ۙ إِلَّا ٱلَّذِينَ تَابُواْ مِن قَبۡلِ أَن تَقۡدِرُواْ عَلَيۡهِمۡۖ فَٱعۡلَمُوٓاْ أَنَّ ٱللَّهَ غَفُورٞ رَّحِيمٞ ۝٣٤ إِلَّا ٱلَّذِینَ تَابُوا۟ مِن قَبۡلِ أَن تَقۡدِرُوا۟ عَلَیۡهِمۡۖ ءَامَنُواْ ٱتَّقُواْ ٱللَّهَ وَٱبۡتَغُوٓاْ إِلَيۡهِ ٱلۡوَسِيلَةَ وَجَٰهِدُواْ فِي سَبِيلِهِۦ لَعَلَّكُمۡ تُفۡلِحُونَ ۝٣٥ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَٱبۡتَغُوۤا۟ إِلَیۡهِ ٱلۡوَسِیلَةَ وَجَٰهِدُوا۟ فِی سَبِیلِهِۦ لَعَلَّكُمۡ تُفۡلِحُونَ ۝٣٥ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوا اتَّقُوا اللّٰهَ وَابْتَغُوْۤا اِلَیْهِ الْوَسِیْلَةَ وَجَاهِدُوْا فِیْ سَبِیْلِهٖ لَعَلَّكُمْ تُفْلِحُوْنَ ۟ إِنَّ ٱلَّذِينَ كَفَرُواْ لَوۡ أَنَّ لَهُم مَّا فِي ٱلۡأَرۡضِ جَمِيعٗا وَمِثۡلَهُۥ مَعَهُۥ لِيَفۡتَدُواْ بِهِۦ مِنۡ عَذَابِ يَوۡمِ ٱلۡقِيَٰمَةِ مَا تُقُبِّلَ مِنۡهُمۡۖ وَلَهُمۡ عَذَابٌ أَلِيمٞ ۝٣٦ إِنَّ ٱلَّذِینَ كَفَرُوا۟ لَوۡ أَنَّ لَهُم مَّا فِی ٱلۡأَرۡضِ جَمِیعࣰا وَمِثۡلَهُۥ مَعَهُۥ لِیَفۡتَدُوا۟ بِهِۦ مِنۡ عَذَابِ یَوۡمِ ٱلۡقِیَٰمَةِ مَا تُقُبِّلَ مِنۡهُمۡۖ وَلَهُمۡ عَذَابٌ أَلِیمࣱ ۝٣٦ اِنَّ الَّذِیْنَ كَفَرُوْا لَوْ اَنَّ لَهُمْ مَّا فِی الْاَرْضِ جَمِیْعًا وَّمِثْلَهٗ مَعَهٗ لِیَفْتَدُوْا بِهٖ مِنْ عَذَابِ یَوْمِ الْقِیٰمَةِ مَا تُقُبِّلَ مِنْهُمْ ۚ وَلَهُمْ عَذَابٌ اَلِیْمٌ ۟ يُرِيدُونَ أَن يَخۡرُجُواْ مِنَ ٱلنَّارِ وَمَا هُم بِخَٰرِجِينَ مِنۡهَاۖ وَلَهُمۡ عَذَابٞ مُّقِيمٞ ۝٣٧ یُرِیدُونَ أَن یَخۡرُجُوا۟ مِنَ ٱلنَّارِ وَمَا هُم بِخَٰرِجِینَ مِنۡهَاۖ وَلَهُمۡ عَذَابࣱ مُّقِیمࣱ ۝٣٧ یُرِیْدُوْنَ اَنْ یَّخْرُجُوْا مِنَ النَّارِ وَمَا هُمْ بِخٰرِجِیْنَ مِنْهَا ؗ وَلَهُمْ عَذَابٌ مُّقِیْمٌ ۟ وَٱلسَّارِقُ وَٱلسَّارِقَةُ فَٱقۡطَعُوٓاْ أَيۡدِيَهُمَا جَزَآءَۢ بِمَا كَسَبَا نَكَٰلٗا مِّنَ ٱللَّهِۗ وَٱللَّهُ عَزِيزٌ حَكِيمٞ ۝٣٨ وَٱلسَّارِقُ وَٱلسَّارِقَةُ فَٱقۡطَعُوۤا۟ أَیۡدِیَهُمَا جَزَاۤءَۢ بِمَا كَسَبَا نَكَٰلࣰا مِّنَ ٱللَّهِۗ وَٱللَّهُ عَزِیزٌ حَكِیمࣱ ۝٣٨ وَالسَّارِقُ وَالسَّارِقَةُ فَاقْطَعُوْۤا اَیْدِیَهُمَا جَزَآءً بِمَا كَسَبَا نَكَالًا مِّنَ اللّٰهِ ؕ وَاللّٰهُ عَزِیْزٌ حَكِیْمٌ ۟ فَمَن تَابَ مِنۢ بَعۡدِ ظُلۡمِهِۦ وَأَصۡلَحَ فَإِنَّ ٱللَّهَ يَتُوبُ عَلَيۡهِۚ إِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٌ ۝٣٩ فَمَن تَابَ مِنۢ بَعۡدِ ظُلۡمِهِۦ وَأَصۡلَحَ فَإِنَّ ٱللَّهَ یَتُوبُ عَلَیۡهِۚ إِنَّ ٱللَّهَ غَفُورࣱ رَّحِیمٌ ۝٣٩ فَمَنْ تَابَ مِنْ بَعْدِ ظُلْمِهٖ وَاَصْلَحَ فَاِنَّ اللّٰهَ یَتُوْبُ عَلَیْهِ ؕ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِیْمٌ ۟ أَلَمۡ تَعۡلَمۡ أَنَّ ٱللَّهَ لَهُۥ مُلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ يُعَذِّبُ مَن يَشَآءُ وَيَغۡفِرُ لِمَن يَشَآءُۗ وَٱللَّهُ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرٞ ۝٤٠ أَلَمۡ تَعۡلَمۡ أَنَّ ٱللَّهَ لَهُۥ مُلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ یُعَذِّبُ مَن یَشَاۤءُ وَیَغۡفِرُ لِمَن یَشَاۤءُۗ وَٱللَّهُ عَلَىٰ كُلِّ شَیۡءࣲ قَدِیرࣱ ۝٤٠ اَلَمْ تَعْلَمْ اَنَّ اللّٰهَ لَهٗ مُلْكُ السَّمٰوٰتِ وَالْاَرْضِ ؕ یُعَذِّبُ مَنْ یَّشَآءُ وَیَغْفِرُ لِمَنْ یَّشَآءُ ؕ وَاللّٰهُ عَلٰی كُلِّ شَیْءٍ قَدِیْرٌ ۟ يَٰٓأَيُّهَا ٱلرَّسُولُ لَا يَحۡزُنكَ ٱلَّذِينَ يُسَٰرِعُونَ فِي ٱلۡكُفۡرِ مِنَ ٱلَّذِينَ قَالُوٓاْ ءَامَنَّا بِأَفۡوَٰهِهِمۡ وَلَمۡ تُؤۡمِن قُلُوبُهُمۡۛ وَمِنَ ٱلَّذِينَ هَادُواْۛ سَمَّٰعُونَ لِلۡكَذِبِ سَمَّٰعُونَ لِقَوۡمٍ ءَاخَرِينَ لَمۡ يَأۡتُوكَۖ يُحَرِّفُونَ ٱلۡكَلِمَ مِنۢ بَعۡدِ مَوَاضِعِهِۦۖ يَقُولُونَ إِنۡ أُوتِيتُمۡ هَٰذَا فَخُذُوهُ وَإِن لَّمۡ تُؤۡتَوۡهُ فَٱحۡذَرُواْۚ وَمَن يُرِدِ ٱللَّهُ فِتۡنَتَهُۥ فَلَن تَمۡلِكَ لَهُۥ مِنَ ٱللَّهِ شَيۡـًٔاۚ أُوْلَٰٓئِكَ ٱلَّذِينَ لَمۡ يُرِدِ ٱللَّهُ أَن يُطَهِّرَ قُلُوبَهُمۡۚ لَهُمۡ فِي ٱلدُّنۡيَا خِزۡيٞۖ وَلَهُمۡ فِي ٱلۡأٓخِرَةِ عَذَابٌ عَظِيمٞ ۝٤١ یَٰۤأَیُّهَا ٱلرَّسُولُ لَا یَحۡزُنكَ ٱلَّذِینَ یُسَٰرِعُونَ فِی ٱلۡكُفۡرِ مِنَ ٱلَّذِینَ قَالُوۤا۟ ءَامَنَّا بِأَفۡوَٰهِهِمۡ وَلَمۡ تُؤۡمِن قُلُوبُهُمۡۛ وَمِنَ ٱلَّذِینَ هَادُوا۟ۛ سَمَّٰعُونَ لِلۡكَذِبِ سَمَّٰعُونَ لِقَوۡمٍ ءَاخَرِینَ لَمۡ یَأۡتُوكَۖ یُحَرِّفُونَ ٱلۡكَلِمَ مِنۢ بَعۡدِ مَوَاضِعِهِۦۖ یَقُولُونَ إِنۡ أُوتِیتُمۡ هَٰذَا فَخُذُوهُ وَإِن لَّمۡ تُؤۡتَوۡهُ فَٱحۡذَرُوا۟ۚ وَمَن یُرِدِ ٱللَّهُ فِتۡنَتَهُۥ فَلَن تَمۡلِكَ لَهُۥ مِنَ ٱللَّهِ شَیۡءًاۚ أُو۟لَٰۤئِكَ ٱلَّذِینَ لَمۡ یُرِدِ ٱللَّهُ أَن یُطَهِّرَ قُلُوبَهُمۡۚ لَهُمۡ فِی ٱلدُّنۡیَا خِزۡیࣱۖ وَلَهُمۡ فِی ٱلۡءَاخِرَةِ عَذَابٌ عَظِیمࣱ ۝٤١ یٰۤاَیُّهَا الرَّسُوْلُ لَا یَحْزُنْكَ الَّذِیْنَ یُسَارِعُوْنَ فِی الْكُفْرِ مِنَ الَّذِیْنَ قَالُوْۤا اٰمَنَّا بِاَفْوَاهِهِمْ وَلَمْ تُؤْمِنْ قُلُوْبُهُمْ ۛۚ وَمِنَ الَّذِیْنَ هَادُوْا ۛۚ سَمّٰعُوْنَ لِلْكَذِبِ سَمّٰعُوْنَ لِقَوْمٍ اٰخَرِیْنَ ۙ لَمْ یَاْتُوْكَ ؕ یُحَرِّفُوْنَ الْكَلِمَ مِنْ بَعْدِ مَوَاضِعِهٖ ۚ یَقُوْلُوْنَ اِنْ اُوْتِیْتُمْ هٰذَا فَخُذُوْهُ وَاِنْ لَّمْ تُؤْتَوْهُ فَاحْذَرُوْا ؕ وَمَنْ یُّرِدِ اللّٰهُ فِتْنَتَهٗ فَلَنْ تَمْلِكَ لَهٗ مِنَ اللّٰهِ شَیْـًٔا ؕ اُولٰٓىِٕكَ الَّذِیْنَ لَمْ یُرِدِ اللّٰهُ اَنْ یُّطَهِّرَ قُلُوْبَهُمْ ؕ لَهُمْ فِی الدُّنْیَا خِزْیٌ ۖۚ وَّلَهُمْ فِی الْاٰخِرَةِ عَذَابٌ عَظِیْمٌ ۟ سَمَّٰعُونَ لِلۡكَذِبِ أَكَّٰلُونَ لِلسُّحۡتِۚ فَإِن جَآءُوكَ فَٱحۡكُم بَيۡنَهُمۡ أَوۡ أَعۡرِضۡ عَنۡهُمۡۖ وَإِن تُعۡرِضۡ عَنۡهُمۡ فَلَن يَضُرُّوكَ شَيۡـٔٗاۖ وَإِنۡ حَكَمۡتَ فَٱحۡكُم بَيۡنَهُم بِٱلۡقِسۡطِۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلۡمُقۡسِطِينَ ۝٤٢ سَمَّٰعُونَ لِلۡكَذِبِ أَكَّٰلُونَ لِلسُّحۡتِۚ فَإِن جَاۤءُوكَ فَٱحۡكُم بَیۡنَهُمۡ أَوۡ أَعۡرِضۡ عَنۡهُمۡۖ وَإِن تُعۡرِضۡ عَنۡهُمۡ فَلَن یَضُرُّوكَ شَیۡءࣰاۖ وَإِنۡ حَكَمۡتَ فَٱحۡكُم بَیۡنَهُم بِٱلۡقِسۡطِۚ إِنَّ ٱللَّهَ یُحِبُّ ٱلۡمُقۡسِطِینَ ۝٤٢ سَمّٰعُوْنَ لِلْكَذِبِ اَكّٰلُوْنَ لِلسُّحْتِ ؕ فَاِنْ جَآءُوْكَ فَاحْكُمْ بَیْنَهُمْ اَوْ اَعْرِضْ عَنْهُمْ ۚ وَاِنْ تُعْرِضْ عَنْهُمْ فَلَنْ یَّضُرُّوْكَ شَیْـًٔا ؕ وَاِنْ حَكَمْتَ فَاحْكُمْ بَیْنَهُمْ بِالْقِسْطِ ؕ اِنَّ اللّٰهَ یُحِبُّ الْمُقْسِطِیْنَ ۟ وَكَيۡفَ يُحَكِّمُونَكَ وَعِندَهُمُ ٱلتَّوۡرَىٰةُ فِيهَا حُكۡمُ ٱللَّهِ ثُمَّ يَتَوَلَّوۡنَ مِنۢ بَعۡدِ ذَٰلِكَۚ وَمَآ أُوْلَٰٓئِكَ بِٱلۡمُؤۡمِنِينَ ۝٤٣ وَكَیۡفَ یُحَكِّمُونَكَ وَعِندَهُمُ ٱلتَّوۡرَىٰةُ فِیهَا حُكۡمُ ٱللَّهِ ثُمَّ یَتَوَلَّوۡنَ مِنۢ بَعۡدِ ذَٰلِكَۚ وَمَاۤ أُو۟لَٰۤئِكَ بِٱلۡمُؤۡمِنِینَ ۝٤٣ وَكَیْفَ یُحَكِّمُوْنَكَ وَعِنْدَهُمُ التَّوْرٰىةُ فِیْهَا حُكْمُ اللّٰهِ ثُمَّ یَتَوَلَّوْنَ مِنْ بَعْدِ ذٰلِكَ ؕ وَمَاۤ اُولٰٓىِٕكَ بِالْمُؤْمِنِیْنَ ۟۠ إِنَّآ أَنزَلۡنَا ٱلتَّوۡرَىٰةَ فِيهَا هُدٗى وَنُورٞۚ يَحۡكُمُ بِهَا ٱلنَّبِيُّونَ ٱلَّذِينَ أَسۡلَمُواْ لِلَّذِينَ هَادُواْ وَٱلرَّبَّٰنِيُّونَ وَٱلۡأَحۡبَارُ بِمَا ٱسۡتُحۡفِظُواْ مِن كِتَٰبِ ٱللَّهِ وَكَانُواْ عَلَيۡهِ شُهَدَآءَۚ فَلَا تَخۡشَوُاْ ٱلنَّاسَ وَٱخۡشَوۡنِ وَلَا تَشۡتَرُواْ بِـَٔايَٰتِي ثَمَنٗا قَلِيلٗاۚ وَمَن لَّمۡ يَحۡكُم بِمَآ أَنزَلَ ٱللَّهُ فَأُوْلَٰٓئِكَ هُمُ ٱلۡكَٰفِرُونَ ۝٤٤ إِنَّاۤ أَنزَلۡنَا ٱلتَّوۡرَىٰةَ فِیهَا هُدࣰى وَنُورࣱۚ یَحۡكُمُ بِهَا ٱلنَّبِیُّونَ ٱلَّذِینَ أَسۡلَمُوا۟ لِلَّذِینَ هَادُوا۟ وَٱلرَّبَّٰنِیُّونَ وَٱلۡأَحۡبَارُ بِمَا ٱسۡتُحۡفِظُوا۟ مِن كِتَٰبِ ٱللَّهِ وَكَانُوا۟ عَلَیۡهِ شُهَدَاۤءَۚ فَلَا تَخۡشَوُا۟ ٱلنَّاسَ وَٱخۡشَوۡنِ وَلَا تَشۡتَرُوا۟ بِءَایَٰتِی ثَمَنࣰا قَلِیلࣰاۚ وَمَن لَّمۡ یَحۡكُم بِمَاۤ أَنزَلَ ٱللَّهُ فَأُو۟لَٰۤئِكَ هُمُ ٱلۡكَٰفِرُونَ ۝٤٤ اِنَّاۤ اَنْزَلْنَا التَّوْرٰىةَ فِیْهَا هُدًی وَّنُوْرٌ ۚ یَحْكُمُ بِهَا النَّبِیُّوْنَ الَّذِیْنَ اَسْلَمُوْا لِلَّذِیْنَ هَادُوْا وَالرَّبّٰنِیُّوْنَ وَالْاَحْبَارُ بِمَا اسْتُحْفِظُوْا مِنْ كِتٰبِ اللّٰهِ وَكَانُوْا عَلَیْهِ شُهَدَآءَ ۚ فَلَا تَخْشَوُا النَّاسَ وَاخْشَوْنِ وَلَا تَشْتَرُوْا بِاٰیٰتِیْ ثَمَنًا قَلِیْلًا ؕ وَمَنْ لَّمْ یَحْكُمْ بِمَاۤ اَنْزَلَ اللّٰهُ فَاُولٰٓىِٕكَ هُمُ الْكٰفِرُوْنَ ۟ وَكَتَبۡنَا عَلَيۡهِمۡ فِيهَآ أَنَّ ٱلنَّفۡسَ بِٱلنَّفۡسِ وَٱلۡعَيۡنَ بِٱلۡعَيۡنِ وَٱلۡأَنفَ بِٱلۡأَنفِ وَٱلۡأُذُنَ بِٱلۡأُذُنِ وَٱلسِّنَّ بِٱلسِّنِّ وَٱلۡجُرُوحَ قِصَاصٞۚ فَمَن تَصَدَّقَ بِهِۦ فَهُوَ كَفَّارَةٞ لَّهُۥۚ وَمَن لَّمۡ يَحۡكُم بِمَآ أَنزَلَ ٱللَّهُ فَأُوْلَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ۝٤٥ وَكَتَبۡنَا عَلَیۡهِمۡ فِیهَاۤ أَنَّ ٱلنَّفۡسَ بِٱلنَّفۡسِ وَٱلۡعَیۡنَ بِٱلۡعَیۡنِ وَٱلۡأَنفَ بِٱلۡأَنفِ وَٱلۡأُذُنَ بِٱلۡأُذُنِ وَٱلسِّنَّ بِٱلسِّنِّ وَٱلۡجُرُوحَ قِصَاصࣱۚ فَمَن تَصَدَّقَ بِهِۦ فَهُوَ كَفَّارَةࣱ لَّهُۥۚ وَمَن لَّمۡ یَحۡكُم بِمَاۤ أَنزَلَ ٱللَّهُ فَأُو۟لَٰۤئِكَ هُمُ ٱلظَّٰلِمُونَ ۝٤٥ وَكَتَبْنَا عَلَیْهِمْ فِیْهَاۤ اَنَّ النَّفْسَ بِالنَّفْسِ ۙ وَالْعَیْنَ بِالْعَیْنِ وَالْاَنْفَ بِالْاَنْفِ وَالْاُذُنَ بِالْاُذُنِ وَالسِّنَّ بِالسِّنِّ ۙ وَالْجُرُوْحَ قِصَاصٌ ؕ فَمَنْ تَصَدَّقَ بِهٖ فَهُوَ كَفَّارَةٌ لَّهٗ ؕ وَمَنْ لَّمْ یَحْكُمْ بِمَاۤ اَنْزَلَ اللّٰهُ فَاُولٰٓىِٕكَ هُمُ الظّٰلِمُوْنَ ۟ وَقَفَّيۡنَا عَلَىٰٓ ءَاثَٰرِهِم بِعِيسَى ٱبۡنِ مَرۡيَمَ مُصَدِّقٗا لِّمَا بَيۡنَ يَدَيۡهِ مِنَ ٱلتَّوۡرَىٰةِۖ وَءَاتَيۡنَٰهُ ٱلۡإِنجِيلَ فِيهِ هُدٗى وَنُورٞ وَمُصَدِّقٗا لِّمَا بَيۡنَ يَدَيۡهِ مِنَ ٱلتَّوۡرَىٰةِ وَهُدٗى وَمَوۡعِظَةٗ لِّلۡمُتَّقِينَ ۝٤٦ وَقَفَّیۡنَا عَلَىٰۤ ءَاثَٰرِهِم بِعِیسَى ٱبۡنِ مَرۡیَمَ مُصَدِّقࣰا لِّمَا بَیۡنَ یَدَیۡهِ مِنَ ٱلتَّوۡرَىٰةِۖ وَءَاتَیۡنَٰهُ ٱلۡإِنجِیلَ فِیهِ هُدࣰى وَنُورࣱ وَمُصَدِّقࣰا لِّمَا بَیۡنَ یَدَیۡهِ مِنَ ٱلتَّوۡرَىٰةِ وَهُدࣰى وَمَوۡعِظَةࣰ لِّلۡمُتَّقِینَ ۝٤٦ وَقَفَّیْنَا عَلٰۤی اٰثَارِهِمْ بِعِیْسَی ابْنِ مَرْیَمَ مُصَدِّقًا لِّمَا بَیْنَ یَدَیْهِ مِنَ التَّوْرٰىةِ ۪ وَاٰتَیْنٰهُ الْاِنْجِیْلَ فِیْهِ هُدًی وَّنُوْرٌ ۙ وَّمُصَدِّقًا لِّمَا بَیْنَ یَدَیْهِ مِنَ التَّوْرٰىةِ وَهُدًی وَّمَوْعِظَةً لِّلْمُتَّقِیْنَ ۟ؕ وَلۡيَحۡكُمۡ أَهۡلُ ٱلۡإِنجِيلِ بِمَآ أَنزَلَ ٱللَّهُ فِيهِۚ وَمَن لَّمۡ يَحۡكُم بِمَآ أَنزَلَ ٱللَّهُ فَأُوْلَٰٓئِكَ هُمُ ٱلۡفَٰسِقُونَ ۝٤٧ وَلۡیَحۡكُمۡ أَهۡلُ ٱلۡإِنجِیلِ بِمَاۤ أَنزَلَ ٱللَّهُ فِیهِۚ وَمَن لَّمۡ یَحۡكُم بِمَاۤ أَنزَلَ ٱللَّهُ فَأُو۟لَٰۤئِكَ هُمُ ٱلۡفَٰسِقُونَ ۝٤٧ وَلْیَحْكُمْ اَهْلُ الْاِنْجِیْلِ بِمَاۤ اَنْزَلَ اللّٰهُ فِیْهِ ؕ وَمَنْ لَّمْ یَحْكُمْ بِمَاۤ اَنْزَلَ اللّٰهُ فَاُولٰٓىِٕكَ هُمُ الْفٰسِقُوْنَ ۟ وَأَنزَلۡنَآ إِلَيۡكَ ٱلۡكِتَٰبَ بِٱلۡحَقِّ مُصَدِّقٗا لِّمَا بَيۡنَ يَدَيۡهِ مِنَ ٱلۡكِتَٰبِ وَمُهَيۡمِنًا عَلَيۡهِۖ فَٱحۡكُم بَيۡنَهُم بِمَآ أَنزَلَ ٱللَّهُۖ وَلَا تَتَّبِعۡ أَهۡوَآءَهُمۡ عَمَّا جَآءَكَ مِنَ ٱلۡحَقِّۚ لِكُلّٖ جَعَلۡنَا مِنكُمۡ شِرۡعَةٗ وَمِنۡهَاجٗاۚ وَلَوۡ شَآءَ ٱللَّهُ لَجَعَلَكُمۡ أُمَّةٗ وَٰحِدَةٗ وَلَٰكِن لِّيَبۡلُوَكُمۡ فِي مَآ ءَاتَىٰكُمۡۖ فَٱسۡتَبِقُواْ ٱلۡخَيۡرَٰتِۚ إِلَى ٱللَّهِ مَرۡجِعُكُمۡ جَمِيعٗا فَيُنَبِّئُكُم بِمَا كُنتُمۡ فِيهِ تَخۡتَلِفُونَ ۝٤٨ وَأَنزَلۡنَاۤ إِلَیۡكَ ٱلۡكِتَٰبَ بِٱلۡحَقِّ مُصَدِّقࣰا لِّمَا بَیۡنَ یَدَیۡهِ مِنَ ٱلۡكِتَٰبِ وَمُهَیۡمِنًا عَلَیۡهِۖ فَٱحۡكُم بَیۡنَهُم بِمَاۤ أَنزَلَ ٱللَّهُۖ وَلَا تَتَّبِعۡ أَهۡوَاۤءَهُمۡ عَمَّا جَاۤءَكَ مِنَ ٱلۡحَقِّۚ لِكُلࣲّ جَعَلۡنَا مِنكُمۡ شِرۡعَةࣰ وَمِنۡهَاجࣰاۚ وَلَوۡ شَاۤءَ ٱللَّهُ لَجَعَلَكُمۡ أُمَّةࣰ وَٰحِدَةࣰ وَلَٰكِن لِّیَبۡلُوَكُمۡ فِی مَاۤ ءَاتَىٰكُمۡۖ فَٱسۡتَبِقُوا۟ ٱلۡخَیۡرَٰتِۚ إِلَى ٱللَّهِ مَرۡجِعُكُمۡ جَمِیعࣰا فَیُنَبِّئُكُم بِمَا كُنتُمۡ فِیهِ تَخۡتَلِفُونَ ۝٤٨ وَاَنْزَلْنَاۤ اِلَیْكَ الْكِتٰبَ بِالْحَقِّ مُصَدِّقًا لِّمَا بَیْنَ یَدَیْهِ مِنَ الْكِتٰبِ وَمُهَیْمِنًا عَلَیْهِ فَاحْكُمْ بَیْنَهُمْ بِمَاۤ اَنْزَلَ اللّٰهُ وَلَا تَتَّبِعْ اَهْوَآءَهُمْ عَمَّا جَآءَكَ مِنَ الْحَقِّ ؕ لِكُلٍّ جَعَلْنَا مِنْكُمْ شِرْعَةً وَّمِنْهَاجًا ؕ وَلَوْ شَآءَ اللّٰهُ لَجَعَلَكُمْ اُمَّةً وَّاحِدَةً وَّلٰكِنْ لِّیَبْلُوَكُمْ فِیْ مَاۤ اٰتٰىكُمْ فَاسْتَبِقُوا الْخَیْرٰتِ ؕ اِلَی اللّٰهِ مَرْجِعُكُمْ جَمِیْعًا فَیُنَبِّئُكُمْ بِمَا كُنْتُمْ فِیْهِ تَخْتَلِفُوْنَ ۟ۙ وَأَنِ ٱحۡكُم بَيۡنَهُم بِمَآ أَنزَلَ ٱللَّهُ وَلَا تَتَّبِعۡ أَهۡوَآءَهُمۡ وَٱحۡذَرۡهُمۡ أَن يَفۡتِنُوكَ عَنۢ بَعۡضِ مَآ أَنزَلَ ٱللَّهُ إِلَيۡكَۖ فَإِن تَوَلَّوۡاْ فَٱعۡلَمۡ أَنَّمَا يُرِيدُ ٱللَّهُ أَن يُصِيبَهُم بِبَعۡضِ ذُنُوبِهِمۡۗ وَإِنَّ كَثِيرٗا مِّنَ ٱلنَّاسِ لَفَٰسِقُونَ ۝٤٩ وَأَنِ ٱحۡكُم بَیۡنَهُم بِمَاۤ أَنزَلَ ٱللَّهُ وَلَا تَتَّبِعۡ أَهۡوَاۤءَهُمۡ وَٱحۡذَرۡهُمۡ أَن یَفۡتِنُوكَ عَنۢ بَعۡضِ مَاۤ أَنزَلَ ٱللَّهُ إِلَیۡكَۖ فَإِن تَوَلَّوۡا۟ فَٱعۡلَمۡ أَنَّمَا یُرِیدُ ٱللَّهُ أَن یُصِیبَهُم بِبَعۡضِ ذُنُوبِهِمۡۗ وَإِنَّ كَثِیرࣰا مِّنَ ٱلنَّاسِ لَفَٰسِقُونَ ۝٤٩ وَاَنِ احْكُمْ بَیْنَهُمْ بِمَاۤ اَنْزَلَ اللّٰهُ وَلَا تَتَّبِعْ اَهْوَآءَهُمْ وَاحْذَرْهُمْ اَنْ یَّفْتِنُوْكَ عَنْ بَعْضِ مَاۤ اَنْزَلَ اللّٰهُ اِلَیْكَ ؕ فَاِنْ تَوَلَّوْا فَاعْلَمْ اَنَّمَا یُرِیْدُ اللّٰهُ اَنْ یُّصِیْبَهُمْ بِبَعْضِ ذُنُوْبِهِمْ ؕ وَاِنَّ كَثِیْرًا مِّنَ النَّاسِ لَفٰسِقُوْنَ ۟ أَفَحُكۡمَ ٱلۡجَٰهِلِيَّةِ يَبۡغُونَۚ وَمَنۡ أَحۡسَنُ مِنَ ٱللَّهِ حُكۡمٗا لِّقَوۡمٖ يُوقِنُونَ ۝٥٠ أَفَحُكۡمَ ٱلۡجَٰهِلِیَّةِ یَبۡغُونَۚ وَمَنۡ أَحۡسَنُ مِنَ ٱللَّهِ حُكۡمࣰا لِّقَوۡمࣲ یُوقِنُونَ ۝٥٠ اَفَحُكْمَ الْجَاهِلِیَّةِ یَبْغُوْنَ ؕ وَمَنْ اَحْسَنُ مِنَ اللّٰهِ حُكْمًا لِّقَوْمٍ یُّوْقِنُوْنَ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تَتَّخِذُواْ ٱلۡيَهُودَ وَٱلنَّصَٰرَىٰٓ أَوۡلِيَآءَۘ بَعۡضُهُمۡ أَوۡلِيَآءُ بَعۡضٖۚ وَمَن يَتَوَلَّهُم مِّنكُمۡ فَإِنَّهُۥ مِنۡهُمۡۗ إِنَّ ٱللَّهَ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلظَّٰلِمِينَ ۝٥١ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلۡیَهُودَ وَٱلنَّصَٰرَىٰۤ أَوۡلِیَاۤءَۘ بَعۡضُهُمۡ أَوۡلِیَاۤءُ بَعۡضࣲۚ وَمَن یَتَوَلَّهُم مِّنكُمۡ فَإِنَّهُۥ یُسَٰرِعُونَ فِیهِمۡ یَقُولُونَ نَخۡشَىٰۤ أَن تُصِیبَنَا دَاۤئِرَةࣱۚ فَعَسَى ٱللَّهُ أَن یَأۡتِیَ بِٱلۡفَتۡحِ أَوۡ أَمۡرࣲ مِّنۡ عِندِهِۦ فَیُصۡبِحُوا۟ عَلَىٰ مَاۤ أَسَرُّوا۟ فِیۤ أَنفُسِهِمۡ نَٰدِمِینَ ۝٥٢ فَتَرَی الَّذِیْنَ فِیْ قُلُوْبِهِمْ مَّرَضٌ یُّسَارِعُوْنَ فِیْهِمْ یَقُوْلُوْنَ نَخْشٰۤی اَنْ تُصِیْبَنَا دَآىِٕرَةٌ ؕ فَعَسَی اللّٰهُ اَنْ یَّاْتِیَ بِالْفَتْحِ اَوْ اَمْرٍ مِّنْ عِنْدِهٖ فَیُصْبِحُوْا عَلٰی مَاۤ اَسَرُّوْا فِیْۤ اَنْفُسِهِمْ نٰدِمِیْنَ ۟ؕ وَيَقُولُ ٱلَّذِينَ ءَامَنُوٓاْ أَهَٰٓؤُلَآءِ ٱلَّذِينَ أَقۡسَمُواْ بِٱللَّهِ جَهۡدَ أَيۡمَٰنِهِمۡ إِنَّهُمۡ لَمَعَكُمۡۚ حَبِطَتۡ أَعۡمَٰلُهُمۡ فَأَصۡبَحُواْ خَٰسِرِينَ ۝٥٣ وَیَقُولُ ٱلَّذِینَ ءَامَنُوۤا۟ أَهَٰۤؤُلَاۤءِ ٱلَّذِینَ أَقۡسَمُوا۟ بِٱللَّهِ جَهۡدَ أَیۡمَٰنِهِمۡ إِنَّهُمۡ لَمَعَكُمۡۚ حَبِطَتۡ أَعۡمَٰلُهُمۡ فَأَصۡبَحُوا۟ خَٰسِرِینَ ۝٥٣ وَیَقُوْلُ الَّذِیْنَ اٰمَنُوْۤا اَهٰۤؤُلَآءِ الَّذِیْنَ اَقْسَمُوْا بِاللّٰهِ جَهْدَ اَیْمَانِهِمْ ۙ اِنَّهُمْ لَمَعَكُمْ ؕ حَبِطَتْ اَعْمَالُهُمْ فَاَصْبَحُوْا خٰسِرِیْنَ ۟ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ مَن يَرۡتَدَّ مِنكُمۡ عَن دِينِهِۦ فَسَوۡفَ يَأۡتِي ٱللَّهُ بِقَوۡمٖ يُحِبُّهُمۡ وَيُحِبُّونَهُۥٓ أَذِلَّةٍ عَلَى ٱلۡمُؤۡمِنِينَ أَعِزَّةٍ عَلَى ٱلۡكَٰفِرِينَ يُجَٰهِدُونَ فِي سَبِيلِ ٱللَّهِ وَلَا يَخَافُونَ لَوۡمَةَ لَآئِمٖۚ ذَٰلِكَ فَضۡلُ ٱللَّهِ يُؤۡتِيهِ مَن يَشَآءُۚ وَٱللَّهُ وَٰسِعٌ عَلِيمٌ ۝٥٤ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ مَن یَرۡتَدَّ مِنكُمۡ عَن دِینِهِۦ فَسَوۡفَ یَأۡتِی ٱللَّهُ بِقَوۡمࣲ یُحِبُّهُمۡ وَیُحِبُّونَهُۥۤ أَذِلَّةٍ عَلَى ٱلۡمُؤۡمِنِینَ أَعِزَّةٍ عَلَى ٱلۡكَٰفِرِینَ یُجَٰهِدُونَ فِی سَبِیلِ ٱللَّهِ وَلَا یَخَافُونَ لَوۡمَةَ لَاۤئِمࣲۚ ذَٰلِكَ فَضۡلُ ٱللَّهِ یُؤۡتِیهِ مَن یَشَاۤءُۚ وَٱللَّهُ وَٰسِعٌ عَلِیمٌ ۝٥٤ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا مَنْ یَّرْتَدَّ مِنْكُمْ عَنْ دِیْنِهٖ فَسَوْفَ یَاْتِی اللّٰهُ بِقَوْمٍ یُّحِبُّهُمْ وَیُحِبُّوْنَهٗۤ ۙ اَذِلَّةٍ عَلَی الْمُؤْمِنِیْنَ اَعِزَّةٍ عَلَی الْكٰفِرِیْنَ ؗ یُجَاهِدُوْنَ فِیْ سَبِیْلِ اللّٰهِ وَلَا یَخَافُوْنَ لَوْمَةَ لَآىِٕمٍ ؕ ذٰلِكَ فَضْلُ اللّٰهِ یُؤْتِیْهِ مَنْ یَّشَآءُ ؕ وَاللّٰهُ وَاسِعٌ عَلِیْمٌ ۟ إِنَّمَا وَلِيُّكُمُ ٱللَّهُ وَرَسُولُهُۥ وَٱلَّذِينَ ءَامَنُواْ ٱلَّذِينَ يُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤۡتُونَ ٱلزَّكَوٰةَ وَهُمۡ رَٰكِعُونَ ۝٥٥ إِنَّمَا وَلِیُّكُمُ ٱللَّهُ وَرَسُولُهُۥ وَٱلَّذِینَ ءَامَنُوا۟ ٱلَّذِینَ یُقِیمُونَ ٱلصَّلَوٰةَ وَیُؤۡتُونَ ٱلزَّكَوٰةَ وَهُمۡ رَٰكِعُونَ ۝٥٥ اِنَّمَا وَلِیُّكُمُ اللّٰهُ وَرَسُوْلُهٗ وَالَّذِیْنَ اٰمَنُوا الَّذِیْنَ یُقِیْمُوْنَ الصَّلٰوةَ وَیُؤْتُوْنَ الزَّكٰوةَ وَهُمْ رٰكِعُوْنَ ۟ وَمَن يَتَوَلَّ ٱللَّهَ وَرَسُولَهُۥ وَٱلَّذِينَ ءَامَنُواْ فَإِنَّ حِزۡبَ ٱللَّهِ هُمُ ٱلۡغَٰلِبُونَ ۝٥٦ وَمَن یَتَوَلَّ ٱللَّهَ وَرَسُولَهُۥ وَٱلَّذِینَ ءَامَنُوا۟ فَإِنَّ حِزۡبَ ٱللَّهِ هُمُ ٱلۡغَٰلِبُونَ ۝٥٦ وَمَنْ یَّتَوَلَّ اللّٰهَ وَرَسُوْلَهٗ وَالَّذِیْنَ اٰمَنُوْا فَاِنَّ حِزْبَ اللّٰهِ هُمُ الْغٰلِبُوْنَ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تَتَّخِذُواْ ٱلَّذِينَ ٱتَّخَذُواْ دِينَكُمۡ هُزُوٗا وَلَعِبٗا مِّنَ ٱلَّذِينَ أُوتُواْ ٱلۡكِتَٰبَ مِن قَبۡلِكُمۡ أُوتُوا۟ ٱلۡكِتَٰبَ مِن قَبۡلِكُمۡ وَٱلۡكُفَّارَ أَوۡلِیَاۤءَۚ وَٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤۡمِنِینَ ۝٥٧ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا لَا تَتَّخِذُوا الَّذِیْنَ اتَّخَذُوْا دِیْنَكُمْ هُزُوًا وَّلَعِبًا مِّنَ الَّذِیْنَ اُوْتُوا الْكِتٰبَ مِنْ قَبْلِكُمْ وَالْكُفَّارَ اَوْلِیَآءَ ۚ وَاتَّقُوا اللّٰهَ اِنْ كُنْتُمْ مُّؤْمِنِیْنَ ۟ وَإِذَا نَادَيۡتُمۡ إِلَى ٱلصَّلَوٰةِ ٱتَّخَذُوهَا هُزُوٗا وَلَعِبٗاۚ ذَٰلِكَ بِأَنَّهُمۡ قَوۡمٞ لَّا يَعۡقِلُونَ ۝٥٨ وَإِذَا نَادَیۡتُمۡ إِلَى ٱلصَّلَوٰةِ ٱتَّخَذُوهَا هُزُوࣰا وَلَعِبࣰاۚ ذَٰلِكَ بِأَنَّهُمۡ قَوۡمࣱ لَّا یَعۡقِلُونَ ۝٥٨ وَاِذَا وَاَكْلِهِمُ السُّحْتَ ؕ لَبِئْسَ مَا كَانُوْا یَصْنَعُوْنَ ۟ وَقَالَتِ ٱلۡيَهُودُ يَدُ ٱللَّهِ مَغۡلُولَةٌۚ غُلَّتۡ أَيۡدِيهِمۡ وَلُعِنُواْ بِمَا قَالُواْۘ بَلۡ يَدَاهُ مَبۡسُوطَتَانِ يُنفِقُ كَيۡفَ يَشَآءُۚ وَلَيَزِيدَنَّ كَثِيرٗا مِّنۡهُم مَّآ أُنزِلَ إِلَيۡكَ مِن رَّبِّكَ طُغۡيَٰنٗا وَكُفۡرٗاۚ وَأَلۡقَيۡنَا بَيۡنَهُمُ ٱلۡعَدَٰوَةَ وَٱلۡبَغۡضَآءَ إِلَىٰ يَوۡمِ ٱلۡقِيَٰمَةِۚ كُلَّمَآ أَوۡقَدُواْ نَارٗا لِّلۡحَرۡبِ أَطۡفَأَهَا ٱللَّهُۚ وَيَسۡعَوۡنَ فِي ٱلۡأَرۡضِ فَسَادٗاۚ وَٱللَّهُ لَا يُحِبُّ ٱلۡمُفۡسِدِينَ ۝٦٤ وَقَالَتِ ٱلۡیَهُودُ یَدُ ٱللَّهِ مَغۡلُولَةٌۚ غُلَّتۡ أَیۡدِیهِمۡ وَلُعِنُوا۟ بِمَا قَالُوا۟ۘ بَلۡ یَدَاهُ مَبۡسُوطَتَانِ یُنفِقُ كَیۡفَ یَشَاۤءُۚ وَلَیَزِیدَنَّ كَثِیرࣰا مِّنۡهُم مَّاۤ أُنزِلَ إِلَیۡكَ مِن رَّبِّكَ طُغۡیَٰنࣰا وَكُفۡرࣰاۚ وَأَلۡقَیۡنَا بَیۡنَهُمُ ٱلۡعَدَٰوَةَ وَٱلۡبَغۡضَاۤءَ إِلَىٰ یَوۡمِ ٱلۡقِیَٰمَةِۚ كُلَّمَاۤ أَوۡقَدُوا۟ نَارࣰا لِّلۡحَرۡبِ أَطۡفَأَهَا ٱللَّهُۚ وَیَسۡعَوۡنَ فِی ٱلۡأَرۡضِ فَسَادࣰاۚ وَٱللَّهُ لَا یُحِبُّ ٱلۡمُفۡسِدِینَ ۝٦٤ وَقَالَتِ الْیَهُوْدُ یَدُ اللّٰهِ مَغْلُوْلَةٌ ؕ غُلَّتْ اَیْدِیْهِمْ وَلُعِنُوْا بِمَا قَالُوْا ۘ بَلْ یَدٰهُ مَبْسُوْطَتٰنِ ۙ یُنْفِقُ كَیْفَ یَشَآءُ ؕ وَلَیَزِیْدَنَّ كَثِیْرًا مِّنْهُمْ مَّاۤ اُنْزِلَ اِلَیْكَ مِنْ رَّبِّكَ طُغْیَانًا وَّكُفْرًا ؕ وَاَلْقَیْنَا بَیْنَهُمُ الْعَدَاوَةَ وَالْبَغْضَآءَ اِلٰی یَوْمِ الْقِیٰمَةِ ؕ كُلَّمَاۤ اَوْقَدُوْا نَارًا لِّلْحَرْبِ اَطْفَاَهَا اللّٰهُ ۙ وَیَسْعَوْنَ فِی الْاَرْضِ فَسَادًا ؕ وَاللّٰهُ لَا یُحِبُّ الْمُفْسِدِیْنَ ۟ وَلَوۡ أَنَّ أَهۡلَ ٱلۡكِتَٰبِ ءَامَنُواْ وَٱتَّقَوۡاْ لَكَفَّرۡنَا عَنۡهُمۡ سَيِّـَٔاتِهِمۡ وَلَأَدۡخَلۡنَٰهُمۡ جَنَّٰتِ ٱلنَّعِيمِ ۝٦٥ وَلَوۡ أَنَّ أَهۡلَ ٱلۡكِتَٰبِ ءَامَنُوا۟ وَٱتَّقَوۡا۟ لَكَفَّرۡنَا عَنۡهُمۡ سَیِّءَاتِهِمۡ وَلَأَدۡخَلۡنَٰهُمۡ جَنَّٰتِ ٱلنَّعِیمِ ۝٦٥ وَلَوْ اَنَّ اَهْلَ الْكِتٰبِ اٰمَنُوْا وَاتَّقَوْا لَكَفَّرْنَا عَنْهُمْ سَیِّاٰتِهِمْ وَلَاَدْخَلْنٰهُمْ جَنّٰتِ النَّعِیْمِ ۟ وَلَوۡ أَنَّهُمۡ أَقَامُواْ ٱلتَّوۡرَىٰةَ وَٱلۡإِنجِيلَ وَمَآ أُنزِلَ إِلَيۡهِم مِّن رَّبِّهِمۡ لَأَكَلُواْ مِن فَوۡقِهِمۡ وَمِن تَحۡتِ أَرۡجُلِهِمۚ مِّنۡهُمۡ أُمَّةٞ مُّقۡتَصِدَةٞۖ وَكَثِيرٞ مِّنۡهُمۡ سَآءَ مَا يَعۡمَلُونَ ۝٦٦ وَلَوۡ أَنَّهُمۡ أَقَامُوا۟ ٱلتَّوۡرَىٰةَ وَٱلۡإِنجِیلَ وَمَاۤ أُنزِلَ إِلَیۡهِم مِّن رَّبِّهِمۡ لَأَكَلُوا۟ مِن فَوۡقِهِمۡ وَمِن تَحۡتِ أَرۡجُلِهِمۚ مِّنۡهُمۡ أُمَّةࣱ مُّقۡتَصِدَةࣱۖ وَكَثِیرࣱ مِّنۡهُمۡ سَاۤءَ مَا یَعۡمَلُونَ ۝٦٦ وَلَوْ اَنَّهُمْ اَقَامُوا التَّوْرٰىةَ وَالْاِنْجِیْلَ وَمَاۤ اُنْزِلَ اِلَیْهِمْ مِّنْ رَّبِّهِمْ لَاَكَلُوْا مِنْ فَوْقِهِمْ وَمِنْ تَحْتِ اَرْجُلِهِمْ ؕ مِنْهُمْ اُمَّةٌ مُّقْتَصِدَةٌ ؕ وَكَثِیْرٌ مِّنْهُمْ سَآءَ مَا یَعْمَلُوْنَ ۟۠ يَٰٓأَيُّهَا ٱلرَّسُولُ بَلِّغۡ مَآ أُنزِلَ إِلَيۡكَ مِن رَّبِّكَۖ وَإِن لَّمۡ تَفۡعَلۡ فَمَا بَلَّغۡتَ رِسَالَتَهُۥۚ وَٱللَّهُ يَعۡصِمُكَ مِنَ ٱلنَّاسِۗ إِنَّ ٱللَّهَ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلۡكَٰفِرِينَ ۝٦٧ یَٰۤأَیُّهَا ٱلرَّسُولُ بَلِّغۡ مَاۤ أُنزِلَ إِلَیۡكَ مِن رَّبِّكَۖ وَإِن لَّمۡ تَفۡعَلۡ فَمَا بَلَّغۡتَ رِسَالَتَهُۥۚ وَٱللَّهُ یَعۡصِمُكَ مِنَ ٱلنَّاسِۗ إِنَّ ٱللَّهَ لَا یَهۡدِی ٱلۡقَوۡمَ ٱلۡكَٰفِرِینَ ۝٦٧ یٰۤاَیُّهَا الرَّسُوْلُ بَلِّغْ مَاۤ اُنْزِلَ اِلَیْكَ مِنْ رَّبِّكَ ؕ وَاِنْ لَّمْ تَفْعَلْ فَمَا بَلَّغْتَ رِسَالَتَهٗ ؕ وَاللّٰهُ یَعْصِمُكَ مِنَ النَّاسِ ؕ اِنَّ اللّٰهَ لَا یَهْدِی الْقَوْمَ الْكٰفِرِیْنَ ۟ قُلۡ يَٰٓأَهۡلَ ٱلۡكِتَٰبِ لَسۡتُمۡ عَلَىٰ شَيۡءٍ حَتَّىٰ تُقِيمُواْ ٱلتَّوۡرَىٰةَ وَٱلۡإِنجِيلَ وَمَآ أُنزِلَ إِلَيۡكُم مِّن رَّبِّكُمۡۗ وَلَيَزِيدَنَّ كَثِيرٗا مِّنۡهُم مَّآ أُنزِلَ إِلَيۡكَ مِن رَّبِّكَ طُغۡيَٰنٗا وَكُفۡرٗاۖ فَلَا تَأۡسَ عَلَى ٱلۡقَوۡمِ ٱلۡكَٰفِرِينَ ۝٦٨ قُلۡ یَٰۤأَهۡلَ ٱلۡكِتَٰبِ لَسۡتُمۡ عَلَىٰ شَیۡءٍ حَتَّىٰ تُقِیمُوا۟ ٱلتَّوۡرَىٰةَ وَٱلۡإِنجِیلَ وَمَاۤ أُنزِلَ إِلَیۡكُم مِّن رَّبِّكُمۡۗ وَلَیَزِیدَنَّ كَثِیرࣰا مِّنۡهُم مَّاۤ أُنزِلَ إِلَیۡكَ مِن رَّبِّكَ طُغۡیَٰنࣰا وَكُفۡرࣰاۖ فَلَا تَأۡسَ عَلَى ٱلۡقَوۡمِ ٱلۡكَٰفِرِینَ ۝٦٨ قُلْ یٰۤاَهْلَ الْكِتٰبِ لَسْتُمْ عَلٰی شَیْءٍ حَتّٰی تُقِیْمُوا التَّوْرٰىةَ وَالْاِنْجِیْلَ وَمَاۤ اُنْزِلَ اِلَیْكُمْ مِّنْ رَّبِّكُمْ ؕ وَلَیَزِیْدَنَّ كَثِیْرًا مِّنْهُمْ مَّاۤ اُنْزِلَ اِلَیْكَ مِنْ رَّبِّكَ طُغْیَانًا وَّكُفْرًا ۚ فَلَا تَاْسَ عَلَی الْقَوْمِ الْكٰفِرِیْنَ ۟ إِنَّ ٱلَّذِينَ ءَامَنُواْ وَٱلَّذِينَ هَادُواْ وَٱلصَّٰبِـُٔونَ وَٱلنَّصَٰرَىٰ مَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡيَوۡمِ ٱلۡأٓخِرِ وَعَمِلَ صَٰلِحٗا فَلَا خَوۡفٌ عَلَيۡهِمۡ وَلَا هُمۡ يَحۡزَنُونَ ۝٦٩ إِنَّ ٱلَّذِینَ ءَامَنُوا۟ وَٱلَّذِینَ هَادُوا۟ وَٱلصَّٰبِءُونَ وَٱلنَّصَٰرَىٰ مَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡیَوۡمِ ٱلۡءَاخِرِ وَعَمِلَ صَٰلِحࣰا فَلَا خَوۡفٌ عَلَیۡهِمۡ وَلَا هُمۡ یَحۡزَنُونَ ۝٦٩ اِنَّ الَّذِیْنَ اٰمَنُوْا وَالَّذِیْنَ هَادُوْا وَالصّٰبِـُٔوْنَ وَالنَّصٰرٰی مَنْ اٰمَنَ بِاللّٰهِ وَالْیَوْمِ الْاٰخِرِ وَعَمِلَ صَالِحًا فَلَا خَوْفٌ عَلَیْهِمْ وَلَا هُمْ یَحْزَنُوْنَ ۟ لَقَدۡ أَخَذۡنَا مِيثَٰقَ بَنِيٓ إِسۡرَٰٓءِيلَ وَأَرۡسَلۡنَآ إِلَيۡهِمۡ رُسُلٗاۖ كُلَّمَا جَآءَهُمۡ رَسُولُۢ بِمَا لَا تَهۡوَىٰٓ أَنفُسُهُمۡ فَرِيقٗا كَذَّبُواْ وَفَرِيقٗا يَقۡتُلُونَ ۝٧٠ لَقَدۡ أَخَذۡنَا مِیثَٰقَ بَنِیۤ إِسۡرَٰۤءِیلَ وَأَرۡسَلۡنَاۤ إِلَیۡهِمۡ رُسُلࣰاۖ كُلَّمَا جَاۤءَهُمۡ رَسُولُۢ بِمَا لَا تَهۡوَىٰۤ أَنفُسُهُمۡ فَرِیقࣰا كَذَّبُوا۟ وَفَرِیقࣰا یَقۡتُلُونَ ۝٧٠ لَقَدْ اَخَذْنَا مِیْثَاقَ بَنِیْۤ اِسْرَآءِیْلَ وَاَرْسَلْنَاۤ اِلَیْهِمْ رُسُلًا ؕ كُلَّمَا جَآءَهُمْ رَسُوْلٌۢ بِمَا لَا تَهْوٰۤی اَنْفُسُهُمْ ۙ فَرِیْقًا كَذَّبُوْا وَفَرِیْقًا یَّقْتُلُوْنَ ۟ۗ وَحَسِبُوٓاْ أَلَّا تَكُونَ فِتۡنَةٞ فَعَمُواْ وَصَمُّواْ ثُمَّ تَابَ ٱللَّهُ عَلَيۡهِمۡ ثُمَّ عَمُواْ وَصَمُّواْ كَثِيرٞ مِّنۡهُمۡۚ وَٱللَّهُ بَصِيرُۢ بِمَا يَعۡمَلُونَ ۝٧١ وَحَسِبُوۤا۟ أَلَّا تَكُونَ فِتۡنَةࣱ فَعَمُوا۟ وَصَمُّوا۟ ثُمَّ تَابَ ٱللَّهُ عَلَیۡهِمۡ ثُمَّ عَمُوا۟ وَصَمُّوا۟ كَثِیرࣱ مِّنۡهُمۡۚ وَٱللَّهُ بَصِیرُۢ بِمَا یَعۡمَلُونَ ۝٧١ وَحَسِبُوْۤا اَلَّا تَكُوْنَ فِتْنَةٌ فَعَمُوْا وَصَمُّوْا ثُمَّ تَابَ اللّٰهُ عَلَیْهِمْ ثُمَّ عَمُوْا وَصَمُّوْا كَثِیْرٌ مِّنْهُمْ ؕ وَاللّٰهُ بَصِیْرٌ بِمَا یَعْمَلُوْنَ ۟ لَقَدۡ كَفَرَ ٱلَّذِينَ قَالُوٓاْ إِنَّ ٱللَّهَ هُوَ ٱلۡمَسِيحُ ٱبۡنُ مَرۡيَمَۖ وَقَالَ ٱلۡمَسِيحُ يَٰبَنِيٓ إِسۡرَٰٓءِيلَ ٱعۡبُدُواْ ٱللَّهَ رَبِّي وَرَبَّكُمۡۖ إِنَّهُۥ مَن يُشۡرِكۡ بِٱللَّهِ فَقَدۡ حَرَّمَ ٱللَّهُ عَلَيۡهِ ٱلۡجَنَّةَ وَمَأۡوَىٰهُ ٱلنَّارُۖ وَمَا لِلظَّٰلِمِينَ مِنۡ أَنصَارٖ ۝٧٢ لَقَدۡ كَفَرَ ٱلَّذِینَ قَالُوۤا۟ إِنَّ ٱللَّهَ هُوَ ٱلۡمَسِیحُ ٱبۡنُ مَرۡیَمَۖ وَقَالَ ٱلۡمَسِیحُ یَٰبَنِیۤ إِسۡرَٰۤءِیلَ ٱعۡبُدُوا۟ ٱللَّهَ رَبِّی وَرَبَّكُمۡۖ إِنَّهُۥ مَن یُشۡرِكۡ بِٱللَّهِ فَقَدۡ حَرَّمَ ٱللَّهُ عَلَیۡهِ ٱلۡجَنَّةَ وَمَأۡوَىٰهُ ٱلنَّارُۖ وَمَا لِلظَّٰلِمِینَ مِنۡ أَنصَارࣲ ۝٧٢ لَقَدْ كَفَرَ الَّذِیْنَ قَالُوْۤا اِنَّ اللّٰهَ هُوَ الْمَسِیْحُ ابْنُ مَرْیَمَ ؕ وَقَالَ الْمَسِیْحُ یٰبَنِیْۤ اِسْرَآءِیْلَ اعْبُدُوا اللّٰهَ رَبِّیْ وَرَبَّكُمْ ؕ اِنَّهٗ مَنْ یُّشْرِكْ بِاللّٰهِ فَقَدْ حَرَّمَ اللّٰهُ عَلَیْهِ الْجَنَّةَ وَمَاْوٰىهُ النَّارُ ؕ وَمَا لِلظّٰلِمِیْنَ مِنْ اَنْصَارٍ ۟ لَّقَدۡ كَفَرَ ٱلَّذِينَ قَالُوٓاْ إِنَّ ٱللَّهَ ثَالِثُ ثَلَٰثَةٖۘ وَمَا مِنۡ إِلَٰهٍ إِلَّآ إِلَٰهٞ وَٰحِدٞۚ وَإِن لَّمۡ يَنتَهُواْ عَمَّا يَقُولُونَ لَيَمَسَّنَّ ٱلَّذِينَ كَفَرُواْ مِنۡهُمۡ عَذَابٌ أَلِيمٌ ۝٧٣ لَّقَدۡ كَفَرَ ٱلَّذِینَ قَالُوۤا۟ إِنَّ ٱللَّهَ ثَالِثُ ثَلَٰثَةࣲۘ وَمَا مِنۡ إِلَٰهٍ إِلَّاۤ إِلَٰهࣱ وَٰحِدࣱۚ وَإِن لَّمۡ یَنتَهُوا۟ عَمَّا یَقُولُونَ لَیَمَسَّنَّ ٱلَّذِینَ كَفَرُوا۟ مِنۡهُمۡ عَذَابٌ أَلِیمٌ ۝٧٣ لَقَدْ كَفَرَ الَّذِیْنَ قَالُوْۤا اِنَّ اللّٰهَ ثَالِثُ ثَلٰثَةٍ ۘ وَمَا مِنْ اِلٰهٍ اِلَّاۤ اِلٰهٌ وَّاحِدٌ ؕ وَاِنْ لَّمْ یَنْتَهُوْا عَمَّا یَقُوْلُوْنَ لَیَمَسَّنَّ الَّذِیْنَ كَفَرُوْا مِنْهُمْ عَذَابٌ اَلِیْمٌ ۟ أَفَلَا يَتُوبُونَ إِلَى ٱللَّهِ وَيَسۡتَغۡفِرُونَهُۥۚ وَٱللَّهُ غَفُورٞ رَّحِيمٞ ۝٧٤ أَفَلَا یَتُوبُونَ إِلَى ٱللَّهِ وَیَسۡتَغۡفِرُونَهُۥۚ وَٱللَّهُ غَفُورࣱ رَّحِیمࣱ ۝٧٤ اَفَلَا یَتُوْبُوْنَ اِلَی اللّٰهِ وَیَسْتَغْفِرُوْنَهٗ ؕ وَاللّٰهُ غَفُوْرٌ رَّحِیْمٌ ۟ مَّا ٱلۡمَسِيحُ ٱبۡنُ مَرۡيَمَ إِلَّا رَسُولٞ قَدۡ خَلَتۡ مِن قَبۡلِهِ ٱلرُّسُلُ وَأُمُّهُۥ صِدِّيقَةٞۖ كَانَا يَأۡكُلَانِ ٱلطَّعَامَۗ ٱنظُرۡ كَيۡفَ نُبَيِّنُ لَهُمُ ٱلۡأٓيَٰتِ ثُمَّ ٱنظُرۡ أَنَّىٰ يُؤۡفَكُونَ ۝٧٥ مَّا ٱلۡمَسِیحُ ٱبۡنُ مَرۡیَمَ إِلَّا رَسُولࣱ قَدۡ خَلَتۡ مِن قَبۡلِهِ ٱلرُّسُلُ وَأُمُّهُۥ صِدِّیقَةࣱۖ كَانَا یَأۡكُلَانِ ٱلطَّعَامَۗ ٱنظُرۡ كَیۡفَ نُبَیِّنُ لَهُمُ ٱلۡءَایَٰتِ ثُمَّ ٱنظُرۡ أَنَّىٰ یُؤۡفَكُونَ ۝٧٥ مَا الْمَسِیْحُ ابْنُ مَرْیَمَ اِلَّا رَسُوْلٌ ۚ قَدْ خَلَتْ مِنْ قَبْلِهِ الرُّسُلُ ؕ وَاُمُّهٗ صِدِّیْقَةٌ ؕ كَانَا یَاْكُلٰنِ الطَّعَامَ ؕ اُنْظُرْ كَیْفَ نُبَیِّنُ لَهُمُ الْاٰیٰتِ ثُمَّ انْظُرْ اَنّٰی یُؤْفَكُوْنَ ۟ قُلۡ أَتَعۡبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَمۡلِكُ لَكُمۡ ضَرّٗا وَلَا نَفۡعٗاۚ وَٱللَّهُ هُوَ ٱلسَّمِيعُ ٱلۡعَلِيمُ ۝٧٦ قُلۡ أَتَعۡبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا یَمۡلِكُ لَكُمۡ ضَرࣰّا وَلَا نَفۡعࣰاۚ وَٱللَّهُ هُوَ ٱلسَّمِیعُ ٱلۡعَلِیمُ ۝٧٦ قُلْ اَتَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ مَا لَا یَمْلِكُ لَكُمْ ضَرًّا وَّلَا نَفْعًا ؕ وَاللّٰهُ هُوَ السَّمِیْعُ الْعَلِیْمُ ۟ قُلۡ يَٰٓأَهۡلَ ٱلۡكِتَٰبِ لَا تَغۡلُواْ فِي دِينِكُمۡ غَيۡرَ ٱلۡحَقِّ وَلَا تَتَّبِعُوٓاْ أَهۡوَآءَ قَوۡمٖ قَدۡ ضَلُّواْ مِن قَبۡلُ وَأَضَلُّواْ كَثِيرٗا وَضَلُّواْ عَن سَوَآءِ ٱلسَّبِيلِ ۝٧٧ قُلۡ یَٰۤأَهۡلَ ٱلۡكِتَٰبِ لَا تَغۡلُوا۟ فِی دِینِكُمۡ غَیۡرَ ٱلۡحَقِّ وَلَا تَتَّبِعُوۤا۟ أَهۡوَاۤءَ قَوۡمࣲ قَدۡ ضَلُّوا۟ مِن قَبۡلُ وَأَضَلُّوا۟ كَثِیرࣰا وَضَلُّوا۟ عَن سَوَاۤءِ ٱلسَّبِیلِ ۝٧٧ قُلْ یٰۤاَهْلَ الْكِتٰبِ لَا تَغْلُوْا فِیْ دِیْنِكُمْ غَیْرَ الْحَقِّ وَلَا تَتَّبِعُوْۤا اَهْوَآءَ قَوْمٍ قَدْ ضَلُّوْا مِنْ قَبْلُ وَاَضَلُّوْا كَثِیْرًا وَّضَلُّوْا عَنْ سَوَآءِ السَّبِیْلِ ۟۠ لُعِنَ ٱلَّذِينَ كَفَرُواْ مِنۢ بَنِيٓ إِسۡرَٰٓءِيلَ عَلَىٰ لِسَانِ دَاوُۥدَ وَعِيسَى ٱبۡنِ مَرۡيَمَۚ ذَٰلِكَ بِمَا عَصَواْ وَّكَانُواْ يَعۡتَدُونَ ۝٧٨ لُعِنَ ٱلَّذِینَ كَفَرُوا۟ مِنۢ بَنِیۤ إِسۡرَٰۤءِیلَ عَلَىٰ لِسَانِ دَاوُۥدَ وَعِیسَى ٱبۡنِ مَرۡیَمَۚ ذَٰلِكَ بِمَا عَصَوا۟ وَّكَانُوا۟ یَعۡتَدُونَ ۝٧٨ لُعِنَ الَّذِیْنَ كَفَرُوْا مِنْ بَنِیْۤ اِسْرَآءِیْلَ عَلٰی لِسَانِ دَاوٗدَ وَعِیْسَی ابْنِ مَرْیَمَ ؕ ذٰلِكَ بِمَا عَصَوْا وَّكَانُوْا یَعْتَدُوْنَ ۟ كَانُواْ لَا يَتَنَاهَوۡنَ عَن مُّنكَرٖ فَعَلُوهُۚ لَبِئۡسَ مَا كَانُواْ يَفۡعَلُونَ ۝٧٩ كَانُوا۟ لَا یَتَنَاهَوۡنَ عَن مُّنكَرࣲ فَعَلُوهُۚ لَبِئۡسَ مَا كَانُوا۟ یَفۡعَلُونَ ۝٧٩ كَانُوْا لَا یَتَنَاهَوْنَ عَنْ مُّنْكَرٍ فَعَلُوْهُ ؕ لَبِئْسَ مَا كَانُوْا یَفْعَلُوْنَ ۟ تَرَىٰ كَثِيرٗا مِّنۡهُمۡ يَتَوَلَّوۡنَ ٱلَّذِينَ كَفَرُواْۚ لَبِئۡسَ مَا قَدَّمَتۡ لَهُمۡ أَنفُسُهُمۡ أَن سَخِطَ ٱللَّهُ عَلَيۡهِمۡ وَفِي ٱلۡعَذَابِ هُمۡ خَٰلِدُونَ ۝٨٠ تَرَىٰ كَثِیرࣰا مِّنۡهُمۡ یَتَوَلَّوۡنَ ٱلَّذِینَ كَفَرُوا۟ۚ لَبِئۡسَ مَا قَدَّمَتۡ لَهُمۡ أَنفُسُهُمۡ أَن سَخِطَ ٱللَّهُ عَلَیۡهِمۡ وَفِی ٱلۡعَذَابِ هُمۡ خَٰلِدُونَ ۝٨٠ تَرٰی كَثِیْرًا مِّنْهُمْ یَتَوَلَّوْنَ الَّذِیْنَ كَفَرُوْا ؕ لَبِئْسَ مَا قَدَّمَتْ لَهُمْ اَنْفُسُهُمْ اَنْ سَخِطَ اللّٰهُ عَلَیْهِمْ وَفِی الْعَذَابِ هُمْ خٰلِدُوْنَ ۟ وَلَوۡ كَانُواْ يُؤۡمِنُونَ بِٱللَّهِ وَٱلنَّبِيِّ وَمَآ أُنزِلَ إِلَيۡهِ مَا ٱتَّخَذُوهُمۡ أَوۡلِيَآءَ وَلَٰكِنَّ كَثِيرٗا مِّنۡهُمۡ فَٰسِقُونَ ۝٨١ وَلَوۡ كَانُوا۟ یُؤۡمِنُونَ بِٱللَّهِ وَٱلنَّبِیِّ وَمَاۤ أُنزِلَ إِلَیۡهِ كَثِیْرًا مِّنْهُمْ فٰسِقُوْنَ ۟ لَتَجِدَنَّ أَشَدَّ ٱلنَّاسِ عَدَٰوَةٗ لِّلَّذِينَ ءَامَنُواْ ٱلۡيَهُودَ وَٱلَّذِينَ أَشۡرَكُواْۖ وَلَتَجِدَنَّ أَقۡرَبَهُم مَّوَدَّةٗ لِّلَّذِينَ ءَامَنُواْ ٱلۡقَوۡمِ ٱلصَّٰلِحِينَ ۝٨٤ وَمَا لَنَا لَا نُؤۡمِنُ بِٱللَّهِ وَمَا جَاۤءَنَا مِنَ ٱلۡحَقِّ وَنَطۡمَعُ أَن یُدۡخِلَنَا رَبُّنَا مَعَ ٱلۡقَوۡمِ ٱلصَّٰلِحِینَ ۝٨٤ وَمَا لَنَا لَا نُؤْمِنُ بِاللّٰهِ وَمَا جَآءَنَا مِنَ الْحَقِّ ۙ وَنَطْمَعُ اَنْ یُّدْخِلَنَا رَبُّنَا مَعَ الْقَوْمِ الصّٰلِحِیْنَ ۟ فَأَثَٰبَهُمُ ٱللَّهُ بِمَا قَالُواْ جَنَّٰتٖ تَجۡرِي مِن تَحۡتِهَا ٱلۡأَنۡهَٰرُ خَٰلِدِينَ فِيهَاۚ وَذَٰلِكَ جَزَآءُ ٱلۡمُحۡسِنِينَ ۝٨٥ فَأَثَٰبَهُمُ ٱللَّهُ بِمَا قَالُوا۟ جَنَّٰتࣲ تَجۡرِی مِن تَحۡتِهَا ٱلۡأَنۡهَٰرُ خَٰلِدِینَ فِیهَاۚ وَذَٰلِكَ جَزَاۤءُ ٱلۡمُحۡسِنِینَ ۝٨٥ فَاَثَابَهُمُ اللّٰهُ بِمَا قَالُوْا جَنّٰتٍ تَجْرِیْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِیْنَ فِیْهَا ؕ وَذٰلِكَ جَزَآءُ الْمُحْسِنِیْنَ ۟ وَٱلَّذِينَ كَفَرُواْ وَكَذَّبُواْ بِـَٔايَٰتِنَآ أُوْلَٰٓئِكَ أَصۡحَٰبُ ٱلۡجَحِيمِ ۝٨٦ وَٱلَّذِینَ كَفَرُوا۟ وَكَذَّبُوا۟ بِءَایَٰتِنَاۤ أُو۟لَٰۤئِكَ أَصۡحَٰبُ ٱلۡجَحِیمِ ۝٨٦ وَالَّذِیْنَ كَفَرُوْا وَكَذَّبُوْا بِاٰیٰتِنَاۤ اُولٰٓىِٕكَ اَصْحٰبُ الْجَحِیْمِ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تُحَرِّمُواْ طَيِّبَٰتِ مَآ أَحَلَّ ٱللَّهُ لَكُمۡ وَلَا تَعۡتَدُوٓاْۚ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلۡمُعۡتَدِينَ ۝٨٧ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ لَا تُحَرِّمُوا۟ طَیِّبَٰتِ مَاۤ أَحَلَّ ٱللَّهُ لَكُمۡ وَلَا تَعۡتَدُوۤا۟ۚ إِنَّ ٱللَّهَ لَا یُحِبُّ ٱلۡمُعۡتَدِینَ ۝٨٧ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا لَا تُحَرِّمُوْا طَیِّبٰتِ مَاۤ اَحَلَّ اللّٰهُ لَكُمْ وَلَا تَعْتَدُوْا ؕ اِنَّ اللّٰهَ لَا یُحِبُّ الْمُعْتَدِیْنَ ۟ وَكُلُواْ مِمَّا رَزَقَكُمُ ٱللَّهُ حَلَٰلٗا طَيِّبٗاۚ وَٱتَّقُواْ ٱللَّهَ ٱلَّذِيٓ أَنتُم بِهِۦ مُؤۡمِنُونَ ۝٨٨ وَكُلُوا۟ مِمَّا رَزَقَكُمُ ٱللَّهُ حَلَٰلࣰا طَیِّبࣰاۚ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِیۤ أَنتُم بِهِۦ مُؤۡمِنُونَ ۝٨٨ وَكُلُوْا مِمَّا رَزَقَكُمُ اللّٰهُ حَلٰلًا طَیِّبًا ۪ وَّاتَّقُوا اللّٰهَ الَّذِیْۤ اَنْتُمْ بِهٖ مُؤْمِنُوْنَ ۟ لَا يُؤَاخِذُكُمُ ٱللَّهُ بِٱللَّغۡوِ فِيٓ أَيۡمَٰنِكُمۡ وَلَٰكِن يُؤَاخِذُكُم بِمَا عَقَّدتُّمُ ٱلۡأَيۡمَٰنَۖ فَكَفَّٰرَتُهُۥٓ إِطۡعَامُ عَشَرَةِ مَسَٰكِينَ مِنۡ أَوۡسَطِ مَا تُطۡعِمُونَ أَهۡلِيكُمۡ أَوۡ كِسۡوَتُهُمۡ أَوۡ تَحۡرِيرُ رَقَبَةٖۖ فَمَن لَّمۡ يَجِدۡ فَصِيَامُ ثَلَٰثَةِ أَيَّامٖۚ ذَٰلِكَ كَفَّٰرَةُ أَيۡمَٰنِكُمۡ إِذَا حَلَفۡتُمۡۚ وَٱحۡفَظُوٓاْ أَيۡمَٰنَكُمۡۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمۡ ءَايَٰتِهِۦ لَعَلَّكُمۡ تَشۡكُرُونَ ۝٨٩ لَا یُؤَاخِذُكُمُ ٱللَّهُ بِٱللَّغۡوِ فِیۤ أَیۡمَٰنِكُمۡ وَلَٰكِن یُؤَاخِذُكُم بِمَا عَقَّدتُّمُ ٱلۡأَیۡمَٰنَۖ فَكَفَّٰرَتُهُۥۤ إِطۡعَامُ عَشَرَةِ مَسَٰكِینَ مِنۡ أَوۡسَطِ مَا تُطۡعِمُونَ أَهۡلِیكُمۡ أَوۡ كِسۡوَتُهُمۡ أَوۡ تَحۡرِیرُ رَقَبَةࣲۖ فَمَن لَّمۡ یَجِدۡ فَصِیَامُ ثَلَٰثَةِ أَیَّامࣲۚ ذَٰلِكَ كَفَّٰرَةُ أَیۡمَٰنِكُمۡ إِذَا حَلَفۡتُمۡۚ وَٱحۡفَظُوۤا۟ أَیۡمَٰنَكُمۡۚ كَذَٰلِكَ یُبَیِّنُ ٱللَّهُ لَكُمۡ ءَایَٰتِهِۦ لَعَلَّكُمۡ تَشۡكُرُونَ ۝٨٩ لَا یُؤَاخِذُكُمُ اللّٰهُ بِاللَّغْوِ فِیْۤ اَیْمَانِكُمْ وَلٰكِنْ یُّؤَاخِذُكُمْ بِمَا عَقَّدْتُّمُ الْاَیْمَانَ ۚ فَكَفَّارَتُهٗۤ اِطْعَامُ عَشَرَةِ مَسٰكِیْنَ مِنْ اَوْسَطِ مَا تُطْعِمُوْنَ اَهْلِیْكُمْ اَوْ كِسْوَتُهُمْ اَوْ تَحْرِیْرُ رَقَبَةٍ ؕ فَمَنْ لَّمْ یَجِدْ فَصِیَامُ ثَلٰثَةِ اَیَّامٍ ؕ ذٰلِكَ كَفَّارَةُ اَیْمَانِكُمْ اِذَا حَلَفْتُمْ ؕ وَاحْفَظُوْۤا اَیْمَانَكُمْ ؕ كَذٰلِكَ یُبَیِّنُ اللّٰهُ لَكُمْ اٰیٰتِهٖ لَعَلَّكُمْ تَشْكُرُوْنَ ۟ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِنَّمَا ٱلۡخَمۡرُ وَٱلۡمَيۡسِرُ وَٱلۡأَنصَابُ وَٱلۡأَزۡلَٰمُ رِجۡسٞ مِّنۡ عَمَلِ ٱلشَّيۡطَٰنِ فَٱجۡتَنِبُوهُ لَعَلَّكُمۡ تُفۡلِحُونَ ۝٩٠ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوۤا۟ إِنَّمَا ٱلۡخَمۡرُ وَٱلۡمَیۡسِرُ وَٱلۡأَنصَابُ وَٱلۡأَزۡلَٰمُ رِجۡسࣱ مِّنۡ عَمَلِ ٱلشَّیۡطَٰنِ فَٱجۡتَنِبُوهُ لَعَلَّكُمۡ تُفۡلِحُونَ ۝٩٠ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْۤا اِنَّمَا الْخَمْرُ وَالْمَیْسِرُ وَالْاَنْصَابُ وَالْاَزْلَامُ رِجْسٌ مِّنْ عَمَلِ الشَّیْطٰنِ فَاجْتَنِبُوْهُ لَعَلَّكُمْ تُفْلِحُوْنَ ۟ إِنَّمَا يُرِيدُ ٱلشَّيۡطَٰنُ أَن يُوقِعَ بَيۡنَكُمُ ٱلۡعَدَٰوَةَ وَٱلۡبَغۡضَآءَ فِي ٱلۡخَمۡرِ وَٱلۡمَيۡسِرِ وَيَصُدَّكُمۡ عَن ذِكۡرِ ٱللَّهِ وَعَنِ ٱلصَّلَوٰةِۖ فَهَلۡ أَنتُم مُّنتَهُونَ ۝٩١ إِنَّمَا یُرِیدُ ٱلشَّیۡطَٰنُ أَن یُوقِعَ بَیۡنَكُمُ ٱلۡعَدَٰوَةَ وَٱلۡبَغۡضَاۤءَ فِی ٱلۡخَمۡرِ وَٱلۡمَیۡسِرِ وَیَصُدَّكُمۡ عَن ذِكۡرِ ٱللَّهِ وَعَنِ ٱلصَّلَوٰةِۖ فَهَلۡ أَنتُم مُّنتَهُونَ ۝٩١ اِنَّمَا یُرِیْدُ الشَّیْطٰنُ اَنْ یُّوْقِعَ بَیْنَكُمُ الْعَدَاوَةَ وَالْبَغْضَآءَ فِی الْخَمْرِ وَالْمَیْسِرِ وَیَصُدَّكُمْ عَنْ ذِكْرِ اللّٰهِ وَعَنِ الصَّلٰوةِ ۚ فَهَلْ اَنْتُمْ مُّنْتَهُوْنَ ۟ وَأَطِيعُواْ ٱللَّهَ وَأَطِيعُواْ ٱلرَّسُولَ وَٱحۡذَرُواْۚ فَإِن تَوَلَّيۡتُمۡ فَٱعۡلَمُوٓاْ أَنَّمَا عَلَىٰ رَسُولِنَا ٱلۡبَلَٰغُ ٱلۡمُبِينُ ۝٩٢ وَأَطِیعُوا۟ ٱللَّهَ وَأَطِیعُوا۟ ٱلرَّسُولَ وَٱحۡذَرُوا۟ۚ فَإِن تَوَلَّیۡتُمۡ فَٱعۡلَمُوۤا۟ أَنَّمَا عَلَىٰ رَسُولِنَا ٱلۡبَلَٰغُ ٱلۡمُبِینُ ۝٩٢ وَاَطِیْعُوا اللّٰهَ وَاَطِیْعُوا الرَّسُوْلَ وَاحْذَرُوْا ۚ فَاِنْ تَوَلَّیْتُمْ فَاعْلَمُوْۤا اَنَّمَا عَلٰی رَسُوْلِنَا الْبَلٰغُ الْمُبِیْنُ ۟ لَيۡسَ عَلَى ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ جُنَاحٞ فِيمَا طَعِمُوٓاْ إِذَا مَا ٱتَّقَواْ وَّءَامَنُواْ وَعَمِلُواْ ٱلصَّٰلِحَٰتِ ثُمَّ ٱتَّقَواْ وَّءَامَنُواْ ثُمَّ ٱتَّقَواْ وَّأَحۡسَنُواْۚ وَٱللَّهُ يُحِبُّ ٱلۡمُحۡسِنِينَ ۝٩٣ لَیۡسَ عَلَى ٱلَّذِینَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ جُنَاحࣱ فِیمَا طَعِمُوۤا۟ إِذَا مَا ٱتَّقَوا۟ وَّءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ ثُمَّ ٱتَّقَوا۟ وَّءَامَنُوا۟ ثُمَّ ٱتَّقَوا۟ وَّأَحۡسَنُوا۟ۚ وَٱللَّهُ یُحِبُّ ٱلۡمُحۡسِنِینَ ۝٩٣ لَیْسَ عَلَی الَّذِیْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ جُنَاحٌ فِیْمَا طَعِمُوْۤا اِذَا مَا اتَّقَوْا وَّاٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ ثُمَّ اتَّقَوْا وَّاٰمَنُوْا ثُمَّ اتَّقَوْا وَّاَحْسَنُوْا ؕ وَاللّٰهُ یُحِبُّ الْمُحْسِنِیْنَ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَيَبۡلُوَنَّكُمُ ٱللَّهُ بِشَيۡءٖ مِّنَ ٱلصَّيۡدِ تَنَالُهُۥٓ أَيۡدِيكُمۡ وَرِمَاحُكُمۡ لِيَعۡلَمَ ٱللَّهُ مَن يَخَافُهُۥ بِٱلۡغَيۡبِۚ فَمَنِ ٱعۡتَدَىٰ بَعۡدَ ذَٰلِكَ فَلَهُۥ عَذَابٌ أَلِيمٞ ۝٩٤ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ لَیَبۡلُوَنَّكُمُ ٱللَّهُ بِشَیۡءࣲ مِّنَ ٱلصَّیۡدِ تَنَالُهُۥۤ أَیۡدِیكُمۡ وَرِمَاحُكُمۡ لِیَعۡلَمَ ٱللَّهُ مَن یَخَافُهُۥ بِٱلۡغَیۡبِۚ فَمَنِ ٱعۡتَدَىٰ بَعۡدَ ذَٰلِكَ فَلَهُۥ عَذَابٌ أَلِیمࣱ ۝٩٤ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا لَیَبْلُوَنَّكُمُ اللّٰهُ بِشَیْءٍ مِّنَ الصَّیْدِ تَنَالُهٗۤ اَیْدِیْكُمْ وَرِمَاحُكُمْ لِیَعْلَمَ اللّٰهُ مَنْ یَّخَافُهٗ بِالْغَیْبِ ۚ فَمَنِ اعْتَدٰی بَعْدَ ذٰلِكَ فَلَهٗ عَذَابٌ اَلِیْمٌ ۟ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تَقۡتُلُواْ ٱلصَّيۡدَ وَأَنتُمۡ حُرُمٞۚ وَمَن قَتَلَهُۥ مِنكُم مُّتَعَمِّدٗا فَجَزَآءٞ مِّثۡلُ مَا قَتَلَ مِنَ ٱلنَّعَمِ يَحۡكُمُ نَشۡتَرِی بِهِۦ ثَمَنࣰا وَلَوۡ كَانَ ذَا قُرۡبَىٰ وَلَا نَكۡتُمُ شَهَٰدَةَ ٱللَّهِ إِنَّاۤ إِذࣰا لَّمِنَ ٱلۡءَاثِمِینَ ۝١٠٦ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا شَهَادَةُ بَیْنِكُمْ اِذَا حَضَرَ اَحَدَكُمُ الْمَوْتُ حِیْنَ الْوَصِیَّةِ اثْنٰنِ ذَوَا عَدْلٍ مِّنْكُمْ اَوْ اٰخَرٰنِ مِنْ غَیْرِكُمْ اِنْ اَنْتُمْ ضَرَبْتُمْ فِی الْاَرْضِ فَاَصَابَتْكُمْ مُّصِیْبَةُ الْمَوْتِ ؕ تَحْبِسُوْنَهُمَا مِنْ بَعْدِ الصَّلٰوةِ فَیُقْسِمٰنِ بِاللّٰهِ اِنِ ارْتَبْتُمْ لَا نَشْتَرِیْ بِهٖ ثَمَنًا وَّلَوْ كَانَ ذَا قُرْبٰی ۙ وَلَا نَكْتُمُ شَهَادَةَ ۙ اللّٰهِ اِنَّاۤ اِذًا لَّمِنَ الْاٰثِمِیْنَ ۟ فَإِنۡ عُثِرَ عَلَىٰٓ أَنَّهُمَا ٱسۡتَحَقَّآ إِثۡمٗا فَـَٔاخَرَانِ يَقُومَانِ مَقَامَهُمَا مِنَ ٱلَّذِينَ ٱسۡتَحَقَّ عَلَيۡهِمُ ٱلۡأَوۡلَيَٰنِ فَيُقۡسِمَانِ بِٱللَّهِ لَشَهَٰدَتُنَآ أَحَقُّ مِن شَهَٰدَتِهِمَا وَمَا ٱعۡتَدَيۡنَآ إِنَّآ إِذٗا لَّمِنَ ٱلظَّٰلِمِينَ ۝١٠٧ فَإِنۡ عُثِرَ عَلَىٰۤ أَنَّهُمَا ٱسۡتَحَقَّاۤ إِثۡمࣰا فَءَاخَرَانِ یَقُومَانِ مَقَامَهُمَا مِنَ ٱلَّذِینَ ٱسۡتَحَقَّ عَلَیۡهِمُ ٱلۡأَوۡلَیَٰنِ فَیُقۡسِمَانِ بِٱللَّهِ لَشَهَٰدَتُنَاۤ أَحَقُّ مِن شَهَٰدَتِهِمَا وَمَا ٱعۡتَدَیۡنَاۤ إِنَّاۤ إِذࣰا لَّمِنَ ٱلظَّٰلِمِینَ ۝١٠٧ فَاِنْ عُثِرَ عَلٰۤی اَنَّهُمَا اسْتَحَقَّاۤ اِثْمًا فَاٰخَرٰنِ یَقُوْمٰنِ مَقَامَهُمَا مِنَ الَّذِیْنَ اسْتَحَقَّ عَلَیْهِمُ الْاَوْلَیٰنِ فَیُقْسِمٰنِ بِاللّٰهِ لَشَهَادَتُنَاۤ اَحَقُّ مِنْ شَهَادَتِهِمَا وَمَا اعْتَدَیْنَاۤ ۖؗ اِنَّاۤ اِذًا لَّمِنَ الظّٰلِمِیْنَ ۟ ذَٰلِكَ أَدۡنَىٰٓ أَن يَأۡتُواْ بِٱلشَّهَٰدَةِ عَلَىٰ وَجۡهِهَآ أَوۡ يَخَافُوٓاْ أَن تُرَدَّ أَيۡمَٰنُۢ بَعۡدَ أَيۡمَٰنِهِمۡۗ وَٱتَّقُواْ ٱللَّهَ وَٱسۡمَعُواْۗ وَٱللَّهُ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلۡفَٰسِقِينَ ۝١٠٨ ذَٰلِكَ أَدۡنَىٰۤ أَن یَأۡتُوا۟ بِٱلشَّهَٰدَةِ عَلَىٰ وَجۡهِهَاۤ أَوۡ یَخَافُوۤا۟ أَن تُرَدَّ أَیۡمَٰنُۢ بَعۡدَ أَیۡمَٰنِهِمۡۗ وَٱتَّقُوا۟ ٱللَّهَ وَٱسۡمَعُوا۟ۗ وَٱللَّهُ لَا یَهۡدِی ٱلۡقَوۡمَ ٱلۡفَٰسِقِینَ ۝١٠٨ ذٰلِكَ اَدْنٰۤی اَنْ یَّاْتُوْا بِالشَّهَادَةِ عَلٰی وَجْهِهَاۤ اَوْ یَخَافُوْۤا اَنْ تُرَدَّ اَیْمَانٌ بَعْدَ اَیْمَانِهِمْ ؕ وَاتَّقُوا اللّٰهَ وَاسْمَعُوْا ؕ وَاللّٰهُ لَا یَهْدِی الْقَوْمَ الْفٰسِقِیْنَ ۟۠ يَوۡمَ يَجۡمَعُ ٱللَّهُ ٱلرُّسُلَ فَيَقُولُ مَاذَآ أُجِبۡتُمۡۖ قَالُواْ لَا عِلۡمَ لَنَآۖ إِنَّكَ أَنتَ عَلَّٰمُ ٱلۡغُيُوبِ ۝١٠٩ یَوۡمَ یَجۡمَعُ ٱللَّهُ ٱلرُّسُلَ فَیَقُولُ مَاذَاۤ أُجِبۡتُمۡۖ قَالُوا۟ لَا عِلۡمَ لَنَاۤۖ إِنَّكَ أَنتَ عَلَّٰمُ ٱلۡغُیُوبِ ۝١٠٩ یَوْمَ یَجْمَعُ اللّٰهُ الرُّسُلَ فَیَقُوْلُ مَاذَاۤ اُجِبْتُمْ ؕ قَالُوْا لَا عِلْمَ لَنَا ؕ اِنَّكَ اَنْتَ عَلَّامُ الْغُیُوْبِ ۟ إِذۡ قَالَ ٱللَّهُ يَٰعِيسَى ٱبۡنَ مَرۡيَمَ ٱذۡكُرۡ نِعۡمَتِي عَلَيۡكَ وَعَلَىٰ وَٰلِدَتِكَ إِذۡ أَيَّدتُّكَ بِرُوحِ ٱلۡقُدُسِ فَتَكُونُ طَيۡرَۢا بِإِذۡنِيۖ وَتُبۡرِئُ ٱلۡأَكۡمَهَ وَٱلۡأَبۡرَصَ بِإِذۡنِيۖ وَإِذۡ تُخۡرِجُ ٱلۡمَوۡتَىٰ بِإِذۡنِيۖ وَإِذۡ كَفَفۡتُ بَنِيٓ إِسۡرَٰٓءِيلَ عَنكَ إِذۡ جِئۡتَهُم بِٱلۡبَيِّنَٰتِ فَقَالَ فَتَكُونُ طَیۡرَۢا بِإِذۡنِیۖ وَتُبۡرِئُ ٱلۡأَكۡمَهَ وَٱلۡأَبۡرَصَ بِإِذۡنِیۖ وَإِذۡ تُخۡرِجُ ٱلۡمَوۡتَىٰ بِإِذۡنِیۖ وَإِذۡ كَفَفۡتُ بَنِیۤ إِسۡرَٰۤءِیلَ عَنكَ إِذۡ جِئۡتَهُم بِٱلۡبَیِّنَٰتِ فَقَالَ ٱلسَّمَاۤءِ تَكُونُ لَنَا عِیدࣰا لِّأَوَّلِنَا وَءَاخِرِنَا وَءَایَةࣰ مِّنكَۖ وَٱرۡزُقۡنَا وَأَنتَ خَیۡرُ ٱلرَّٰزِقِینَ ۝١١٤ قَالَ عِیْسَی ابْنُ مَرْیَمَ اللّٰهُمَّ رَبَّنَاۤ اَنْزِلْ عَلَیْنَا مَآىِٕدَةً مِّنَ السَّمَآءِ تَكُوْنُ لَنَا عِیْدًا لِّاَوَّلِنَا وَاٰخِرِنَا وَاٰیَةً مِّنْكَ ۚ وَارْزُقْنَا وَاَنْتَ خَیْرُ الرّٰزِقِیْنَ ۟ قَالَ ٱللَّهُ إِنِّي مُنَزِّلُهَا عَلَيۡكُمۡۖ فَمَن يَكۡفُرۡ بَعۡدُ مِنكُمۡ فَإِنِّيٓ أُعَذِّبُهُۥ عَذَابٗا لَّآ أُعَذِّبُهُۥٓ أَحَدٗا مِّنَ ٱلۡعَٰلَمِينَ ۝١١٥ قَالَ ٱللَّهُ إِنِّی مُنَزِّلُهَا عَلَیۡكُمۡۖ فَمَن یَكۡفُرۡ بَعۡدُ مِنكُمۡ فَإِنِّیۤ أُعَذِّبُهُۥ عَذَابࣰا لَّاۤ أُعَذِّبُهُۥۤ أَحَدࣰا مِّنَ ٱلۡعَٰلَمِینَ ۝١١٥ قَالَ اللّٰهُ اِنِّیْ مُنَزِّلُهَا عَلَیْكُمْ ۚ فَمَنْ یَّكْفُرْ بَعْدُ مِنْكُمْ فَاِنِّیْۤ اُعَذِّبُهٗ عَذَابًا لَّاۤ اُعَذِّبُهٗۤ اَحَدًا مِّنَ الْعٰلَمِیْنَ ۟۠ وَإِذۡ قَالَ ٱللَّهُ يَٰعِيسَى ٱبۡنَ مَرۡيَمَ ءَأَنتَ قُلۡتَ لِلنَّاسِ ٱتَّخِذُونِي وَأُمِّيَ إِلَٰهَيۡنِ مِن دُونِ ٱللَّهِۖ قَالَ سُبۡحَٰنَكَ مَا يَكُونُ لِيٓ أَنۡ أَقُولَ مَا لَيۡسَ لِي بِحَقٍّۚ إِن كُنتُ قُلۡتُهُۥ فَقَدۡ عَلِمۡتَهُۥۚ تَعۡلَمُ مَا فِي نَفۡسِي وَلَآ أَعۡلَمُ مَا فِي نَفۡسِكَۚ إِنَّكَ أَنتَ عَلَّٰمُ ٱلۡغُيُوبِ ۝١١٦ وَإِذۡ قَالَ ٱللَّهُ یَٰعِیسَى ٱبۡنَ مَرۡیَمَ ءَأَنتَ قُلۡتَ لِلنَّاسِ ٱتَّخِذُونِی وَأُمِّیَ إِلَٰهَیۡنِ مِن دُونِ ٱللَّهِۖ قَالَ سُبۡحَٰنَكَ مَا یَكُونُ لِیۤ أَنۡ أَقُولَ مَا لَیۡسَ لِی بِحَقٍّۚ إِن كُنتُ قُلۡتُهُۥ فَقَدۡ عَلِمۡتَهُۥۚ تَعۡلَمُ مَا فِی نَفۡسِی وَلَاۤ أَعۡلَمُ مَا فِی نَفۡسِكَۚ إِنَّكَ أَنتَ عَلَّٰمُ ٱلۡغُیُوبِ ۝١١٦ وَاِذْ قَالَ اللّٰهُ یٰعِیْسَی ابْنَ مَرْیَمَ ءَاَنْتَ قُلْتَ لِلنَّاسِ اتَّخِذُوْنِیْ وَاُمِّیَ اِلٰهَیْنِ مِنْ دُوْنِ اللّٰهِ ؕ قَالَ سُبْحٰنَكَ مَا یَكُوْنُ لِیْۤ اَنْ اَقُوْلَ مَا لَیْسَ لِیْ ۗ بِحَقٍّ ؔؕ اِنْ كُنْتُ قُلْتُهٗ فَقَدْ عَلِمْتَهٗ ؕ تَعْلَمُ مَا فِیْ نَفْسِیْ وَلَاۤ اَعْلَمُ مَا فِیْ نَفْسِكَ ؕ اِنَّكَ اَنْتَ عَلَّامُ الْغُیُوْبِ ۟ مَا قُلۡتُ لَهُمۡ إِلَّا مَآ أَمَرۡتَنِي بِهِۦٓ أَنِ ٱعۡبُدُواْ ٱللَّهَ رَبِّي وَرَبَّكُمۡۚ وَكُنتُ عَلَيۡهِمۡ شَهِيدٗا مَّا دُمۡتُ فِيهِمۡۖ فَلَمَّا تَوَفَّيۡتَنِي كُنتَ أَنتَ ٱلرَّقِيبَ عَلَيۡهِمۡۚ وَأَنتَ عَلَىٰ كُلِّ شَيۡءٖ شَهِيدٌ ۝١١٧ مَا قُلۡتُ لَهُمۡ إِلَّا مَاۤ أَمَرۡتَنِی بِهِۦۤ أَنِ ٱعۡبُدُوا۟ ٱللَّهَ رَبِّی وَرَبَّكُمۡۚ وَكُنتُ عَلَیۡهِمۡ شَهِیدࣰا مَّا دُمۡتُ فِیهِمۡۖ فَلَمَّا تَوَفَّیۡتَنِی كُنتَ أَنتَ ٱلرَّقِیبَ عَلَیۡهِمۡۚ وَأَنتَ عَلَىٰ كُلِّ شَیۡءࣲ شَهِیدٌ ۝١١٧ مَا قُلْتُ لَهُمْ اِلَّا مَاۤ اَمَرْتَنِیْ بِهٖۤ اَنِ اعْبُدُوا اللّٰهَ رَبِّیْ وَرَبَّكُمْ ۚ وَكُنْتُ عَلَیْهِمْ شَهِیْدًا مَّا دُمْتُ فِیْهِمْ ۚ فَلَمَّا تَوَفَّیْتَنِیْ كُنْتَ اَنْتَ الرَّقِیْبَ عَلَیْهِمْ ؕ وَاَنْتَ عَلٰی كُلِّ شَیْءٍ شَهِیْدٌ ۟ إِن تُعَذِّبۡهُمۡ فَإِنَّهُمۡ عِبَادُكَۖ وَإِن تَغۡفِرۡ لَهُمۡ فَإِنَّكَ أَنتَ ٱلۡعَزِيزُ ٱلۡحَكِيمُ ۝١١٨ إِن تُعَذِّبۡهُمۡ فَإِنَّهُمۡ عِبَادُكَۖ وَإِن تَغۡفِرۡ لَهُمۡ فَإِنَّكَ أَنتَ ٱلۡعَزِیزُ ٱلۡحَكِیمُ ۝١١٨ اِنْ تُعَذِّبْهُمْ فَاِنَّهُمْ عِبَادُكَ ۚ وَاِنْ تَغْفِرْ لَهُمْ فَاِنَّكَ اَنْتَ الْعَزِیْزُ الْحَكِیْمُ ۟ قَالَ ٱللَّهُ هَٰذَا يَوۡمُ يَنفَعُ ٱلصَّٰدِقِينَ صِدۡقُهُمۡۚ لَهُمۡ جَنَّٰتٞ تَجۡرِي مِن تَحۡتِهَا ٱلۡأَنۡهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدٗاۖ رَّضِيَ ٱللَّهُ عَنۡهُمۡ وَرَضُواْ عَنۡهُۚ ذَٰلِكَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ ۝١١٩ قَالَ ٱللَّهُ هَٰذَا یَوۡمُ یَنفَعُ ٱلصَّٰدِقِینَ صِدۡقُهُمۡۚ لَهُمۡ جَنَّٰتࣱ تَجۡرِی مِن تَحۡتِهَا ٱلۡأَنۡهَٰرُ خَٰلِدِینَ فِیهَاۤ أَبَدࣰاۖ رَّضِیَ ٱللَّهُ عَنۡهُمۡ وَرَضُوا۟ عَنۡهُۚ ذَٰلِكَ ٱلۡفَوۡزُ ٱلۡعَظِیمُ ۝١١٩ قَالَ اللّٰهُ هٰذَا یَوْمُ یَنْفَعُ الصّٰدِقِیْنَ صِدْقُهُمْ ؕ لَهُمْ جَنّٰتٌ تَجْرِیْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِیْنَ فِیْهَاۤ اَبَدًا ؕ رَضِیَ اللّٰهُ عَنْهُمْ وَرَضُوْا عَنْهُ ؕ ذٰلِكَ الْفَوْزُ الْعَظِیْمُ ۟ لِلَّهِ مُلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا فِيهِنَّۚ وَهُوَ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرُۢ ۝١٢٠ لِلَّهِ مُلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَمَا فِیهِنَّۚ وَهُوَ عَلَىٰ كُلِّ شَیۡءࣲ قَدِیرُۢ ۝١٢٠ لِلّٰهِ مُلْكُ السَّمٰوٰتِ وَالْاَرْضِ وَمَا فِیْهِنَّ ؕ وَهُوَ عَلٰی كُلِّ شَیْءٍ قَدِیْرٌ ۟۠
