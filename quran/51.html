Surah 51 · Meccan · 60 ayat

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

وَٱلذَّٰرِيَٰتِ ذَرۡوٗا ۝١ وَٱلذَّٰرِیَٰتِ ذَرۡوࣰا ۝١ وَالذّٰرِیٰتِ ذَرْوًا ۟ۙ فَٱلۡحَٰمِلَٰتِ وِقۡرٗا ۝٢ فَٱلۡحَٰمِلَٰتِ وِقۡرࣰا ۝٢ فَالْحٰمِلٰتِ وِقْرًا ۟ۙ فَٱلۡجَٰرِيَٰتِ يُسۡرٗا ۝٣ فَٱلۡجَٰرِیَٰتِ یُسۡرࣰا ۝٣ فَالْجٰرِیٰتِ یُسْرًا ۟ۙ فَٱلۡمُقَسِّمَٰتِ أَمۡرًا ۝٤ فَٱلۡمُقَسِّمَٰتِ أَمۡرًا ۝٤ فَالْمُقَسِّمٰتِ اَمْرًا ۟ۙ إِنَّمَا تُوعَدُونَ لَصَادِقٞ ۝٥ إِنَّمَا تُوعَدُونَ لَصَادِقࣱ ۝٥ اِنَّمَا تُوْعَدُوْنَ لَصَادِقٌ ۟ۙ وَإِنَّ ٱلدِّينَ لَوَٰقِعٞ ۝٦ وَإِنَّ ٱلدِّینَ لَوَٰقِعࣱ ۝٦ وَّاِنَّ الدِّیْنَ لَوَاقِعٌ ۟ؕ وَٱلسَّمَآءِ ذَاتِ ٱلۡحُبُكِ ۝٧ وَٱلسَّمَاۤءِ ذَاتِ ٱلۡحُبُكِ ۝٧ وَالسَّمَآءِ ذَاتِ الْحُبُكِ ۟ۙ إِنَّكُمۡ لَفِي قَوۡلٖ مُّخۡتَلِفٖ ۝٨ إِنَّكُمۡ لَفِی قَوۡلࣲ مُّخۡتَلِفࣲ ۝٨ اِنَّكُمْ لَفِیْ قَوْلٍ مُّخْتَلِفٍ ۟ۙ يُؤۡفَكُ عَنۡهُ مَنۡ أُفِكَ ۝٩ یُؤۡفَكُ عَنۡهُ مَنۡ أُفِكَ ۝٩ یُّؤْفَكُ عَنْهُ مَنْ اُفِكَ ۟ؕ قُتِلَ ٱلۡخَرَّٰصُونَ ۝١٠ قُتِلَ ٱلۡخَرَّٰصُونَ ۝١٠ قُتِلَ الْخَرّٰصُوْنَ ۟ۙ ٱلَّذِينَ هُمۡ فِي غَمۡرَةٖ سَاهُونَ ۝١١ ٱلَّذِینَ هُمۡ فِی غَمۡرَةࣲ سَاهُونَ ۝١١ الَّذِیْنَ هُمْ فِیْ غَمْرَةٍ سَاهُوْنَ ۟ۙ يَسۡـَٔلُونَ أَيَّانَ يَوۡمُ ٱلدِّينِ ۝١٢ یَسۡءَلُونَ أَیَّانَ یَوۡمُ ٱلدِّینِ ۝١٢ یَسْـَٔلُوْنَ اَیَّانَ یَوْمُ الدِّیْنِ ۟ؕ يَوۡمَ هُمۡ عَلَى ٱلنَّارِ يُفۡتَنُونَ ۝١٣ یَوۡمَ هُمۡ عَلَى ٱلنَّارِ یُفۡتَنُونَ ۝١٣ یَوْمَ هُمْ عَلَی النَّارِ یُفْتَنُوْنَ ۟ ذُوقُواْ فِتۡنَتَكُمۡ هَٰذَا ٱلَّذِي كُنتُم بِهِۦ تَسۡتَعۡجِلُونَ ۝١٤ ذُوقُوا۟ فِتۡنَتَكُمۡ هَٰذَا ٱلَّذِی كُنتُم بِهِۦ تَسۡتَعۡجِلُونَ ۝١٤ ذُوْقُوْا فِتْنَتَكُمْ ؕ هٰذَا الَّذِیْ كُنْتُمْ بِهٖ تَسْتَعْجِلُوْنَ ۟ إِنَّ ٱلۡمُتَّقِينَ فِي جَنَّٰتٖ وَعُيُونٍ ۝١٥ إِنَّ ٱلۡمُتَّقِینَ فِی جَنَّٰتࣲ وَعُیُونٍ ۝١٥ اِنَّ الْمُتَّقِیْنَ فِیْ جَنّٰتٍ وَّعُیُوْنٍ ۟ۙ ءَاخِذِينَ مَآ ءَاتَىٰهُمۡ رَبُّهُمۡۚ إِنَّهُمۡ كَانُواْ قَبۡلَ ذَٰلِكَ مُحۡسِنِينَ ۝١٦ ءَاخِذِینَ مَاۤ ءَاتَىٰهُمۡ رَبُّهُمۡۚ إِنَّهُمۡ كَانُوا۟ قَبۡلَ ذَٰلِكَ مُحۡسِنِینَ ۝١٦ اٰخِذِیْنَ مَاۤ اٰتٰىهُمْ رَبُّهُمْ ؕ اِنَّهُمْ كَانُوْا قَبْلَ ذٰلِكَ مُحْسِنِیْنَ ۟ؕ كَانُواْ قَلِيلٗا مِّنَ ٱلَّيۡلِ مَا يَهۡجَعُونَ ۝١٧ كَانُوا۟ قَلِیلࣰا مِّنَ ٱلَّیۡلِ مَا یَهۡجَعُونَ ۝١٧ كَانُوْا قَلِیْلًا مِّنَ الَّیْلِ مَا یَهْجَعُوْنَ ۟ وَبِٱلۡأَسۡحَارِ هُمۡ يَسۡتَغۡفِرُونَ ۝١٨ وَبِٱلۡأَسۡحَارِ هُمۡ یَسۡتَغۡفِرُونَ ۝١٨ وَبِالْاَسْحَارِ هُمْ یَسْتَغْفِرُوْنَ ۟ وَفِيٓ أَمۡوَٰلِهِمۡ حَقّٞ لِّلسَّآئِلِ وَٱلۡمَحۡرُومِ ۝١٩ وَفِیۤ أَمۡوَٰلِهِمۡ حَقࣱّ لِّلسَّاۤئِلِ وَٱلۡمَحۡرُومِ ۝١٩ وَفِیْۤ اَمْوَالِهِمْ حَقٌّ لِّلسَّآىِٕلِ وَالْمَحْرُوْمِ ۟ وَفِي ٱلۡأَرۡضِ ءَايَٰتٞ لِّلۡمُوقِنِينَ ۝٢٠ وَفِی ٱلۡأَرۡضِ ءَایَٰتࣱ لِّلۡمُوقِنِینَ ۝٢٠ وَفِی الْاَرْضِ اٰیٰتٌ لِّلْمُوْقِنِیْنَ ۟ۙ وَفِيٓ أَنفُسِكُمۡۚ أَفَلَا تُبۡصِرُونَ ۝٢١ وَفِیۤ أَنفُسِكُمۡۚ أَفَلَا تُبۡصِرُونَ ۝٢١ وَفِیْۤ اَنْفُسِكُمْ ؕ اَفَلَا تُبْصِرُوْنَ ۟ وَفِي ٱلسَّمَآءِ رِزۡقُكُمۡ وَمَا تُوعَدُونَ ۝٢٢ وَفِی ٱلسَّمَاۤءِ رِزۡقُكُمۡ وَمَا تُوعَدُونَ ۝٢٢ وَفِی السَّمَآءِ رِزْقُكُمْ وَمَا تُوْعَدُوْنَ ۟ فَوَرَبِّ ٱلسَّمَآءِ وَٱلۡأَرۡضِ إِنَّهُۥ لَحَقّٞ مِّثۡلَ مَآ أَنَّكُمۡ تَنطِقُونَ ۝٢٣ فَوَرَبِّ ٱلسَّمَاۤءِ وَٱلۡأَرۡضِ إِنَّهُۥ لَحَقࣱّ مِّثۡلَ مَاۤ أَنَّكُمۡ تَنطِقُونَ ۝٢٣ فَوَرَبِّ السَّمَآءِ وَالْاَرْضِ اِنَّهٗ لَحَقٌّ مِّثْلَ مَاۤ اَنَّكُمْ تَنْطِقُوْنَ ۟۠ هَلۡ أَتَىٰكَ حَدِيثُ ضَيۡفِ إِبۡرَٰهِيمَ ٱلۡمُكۡرَمِينَ ۝٢٤ هَلۡ أَتَىٰكَ حَدِیثُ ضَیۡفِ إِبۡرَٰهِیمَ ٱلۡمُكۡرَمِینَ ۝٢٤ هَلْ اَتٰىكَ حَدِیْثُ ضَیْفِ اِبْرٰهِیْمَ الْمُكْرَمِیْنَ ۟ۘ إِذۡ دَخَلُواْ عَلَيۡهِ فَقَالُواْ سَلَٰمٗاۖ قَالَ سَلَٰمٞ قَوۡمٞ مُّنكَرُونَ ۝٢٥ إِذۡ دَخَلُوا۟ عَلَیۡهِ فَقَالُوا۟ سَلَٰمࣰاۖ قَالَ سَلَٰمࣱ قَوۡمࣱ مُّنكَرُونَ ۝٢٥ اِذْ دَخَلُوْا عَلَیْهِ فَقَالُوْا سَلٰمًا ؕ قَالَ سَلٰمٌ ۚ قَوْمٌ مُّنْكَرُوْنَ ۟ فَرَاغَ إِلَىٰٓ أَهۡلِهِۦ فَجَآءَ بِعِجۡلٖ سَمِينٖ ۝٢٦ فَرَاغَ إِلَىٰۤ أَهۡلِهِۦ فَجَاۤءَ بِعِجۡلࣲ سَمِینࣲ ۝٢٦ فَرَاغَ اِلٰۤی اَهْلِهٖ فَجَآءَ بِعِجْلٍ سَمِیْنٍ ۟ۙ فَقَرَّبَهُۥٓ إِلَيۡهِمۡ قَالَ أَلَا تَأۡكُلُونَ ۝٢٧ فَقَرَّبَهُۥۤ إِلَیۡهِمۡ قَالَ أَلَا تَأۡكُلُونَ ۝٢٧ فَقَرَّبَهٗۤ اِلَیْهِمْ قَالَ اَلَا تَاْكُلُوْنَ ۟ؗ فَأَوۡجَسَ مِنۡهُمۡ خِيفَةٗۖ قَالُواْ لَا تَخَفۡۖ وَبَشَّرُوهُ بِغُلَٰمٍ عَلِيمٖ ۝٢٨ فَأَوۡجَسَ مِنۡهُمۡ خِیفَةࣰۖ قَالُوا۟ لَا تَخَفۡۖ وَبَشَّرُوهُ بِغُلَٰمٍ عَلِیمࣲ ۝٢٨ فَاَوْجَسَ مِنْهُمْ خِیْفَةً ؕ قَالُوْا لَا تَخَفْ ؕ وَبَشَّرُوْهُ بِغُلٰمٍ عَلِیْمٍ ۟ فَأَقۡبَلَتِ ٱمۡرَأَتُهُۥ فِي صَرَّةٖ فَصَكَّتۡ وَجۡهَهَا وَقَالَتۡ عَجُوزٌ عَقِيمٞ ۝٢٩ فَأَقۡبَلَتِ ٱمۡرَأَتُهُۥ فِی صَرَّةࣲ فَصَكَّتۡ وَجۡهَهَا وَقَالَتۡ عَجُوزٌ عَقِیمࣱ ۝٢٩ فَاَقْبَلَتِ امْرَاَتُهٗ فِیْ صَرَّةٍ فَصَكَّتْ وَجْهَهَا وَقَالَتْ عَجُوْزٌ عَقِیْمٌ ۟ قَالُواْ كَذَٰلِكِ قَالَ رَبُّكِۖ إِنَّهُۥ هُوَ ٱلۡحَكِيمُ ٱلۡعَلِيمُ ۝٣٠ قَالُوا۟ كَذَٰلِكِ قَالَ رَبُّكِۖ إِنَّهُۥ هُوَ ٱلۡحَكِیمُ ٱلۡعَلِیمُ ۝٣٠ قَالُوْا كَذٰلِكِ ۙ قَالَ رَبُّكِ ؕ اِنَّهٗ هُوَ الْحَكِیْمُ الْعَلِیْمُ ۟ قَالَ فَمَا خَطۡبُكُمۡ أَيُّهَا ٱلۡمُرۡسَلُونَ ۝٣١ قَالَ فَمَا خَطۡبُكُمۡ أَیُّهَا ٱلۡمُرۡسَلُونَ ۝٣١ قَالَ فَمَا خَطْبُكُمْ اَیُّهَا الْمُرْسَلُوْنَ ۟ قَالُوٓاْ إِنَّآ أُرۡسِلۡنَآ إِلَىٰ قَوۡمٖ مُّجۡرِمِينَ ۝٣٢ قَالُوۤا۟ إِنَّاۤ أُرۡسِلۡنَاۤ إِلَىٰ قَوۡمࣲ مُّجۡرِمِینَ ۝٣٢ قَالُوْۤا اِنَّاۤ اُرْسِلْنَاۤ اِلٰی قَوْمٍ مُّجْرِمِیْنَ ۟ۙ لِنُرۡسِلَ عَلَيۡهِمۡ حِجَارَةٗ مِّن طِينٖ ۝٣٣ لِنُرۡسِلَ عَلَیۡهِمۡ حِجَارَةࣰ مِّن طِینࣲ ۝٣٣ لِنُرْسِلَ عَلَیْهِمْ حِجَارَةً مِّنْ طِیْنٍ ۟ۙ مُّسَوَّمَةً عِندَ رَبِّكَ لِلۡمُسۡرِفِينَ ۝٣٤ مُّسَوَّمَةً عِندَ رَبِّكَ لِلۡمُسۡرِفِینَ ۝٣٤ مُّسَوَّمَةً عِنْدَ رَبِّكَ لِلْمُسْرِفِیْنَ ۟ فَأَخۡرَجۡنَا مَن كَانَ فِيهَا مِنَ ٱلۡمُؤۡمِنِينَ ۝٣٥ فَأَخۡرَجۡنَا مَن كَانَ فِیهَا مِنَ ٱلۡمُؤۡمِنِینَ ۝٣٥ فَاَخْرَجْنَا مَنْ كَانَ فِیْهَا مِنَ الْمُؤْمِنِیْنَ ۟ۚ فَمَا وَجَدۡنَا فِيهَا غَيۡرَ بَيۡتٖ مِّنَ ٱلۡمُسۡلِمِينَ ۝٣٦ فَمَا وَجَدۡنَا فِیهَا غَیۡرَ بَیۡتࣲ مِّنَ ٱلۡمُسۡلِمِینَ ۝٣٦ فَمَا وَجَدْنَا فِیْهَا غَیْرَ بَیْتٍ مِّنَ الْمُسْلِمِیْنَ ۟ۚ وَتَرَكۡنَا فِيهَآ ءَايَةٗ لِّلَّذِينَ يَخَافُونَ ٱلۡعَذَابَ ٱلۡأَلِيمَ ۝٣٧ وَتَرَكۡنَا فِیهَاۤ ءَایَةࣰ لِّلَّذِینَ یَخَافُونَ ٱلۡعَذَابَ ٱلۡأَلِیمَ ۝٣٧ وَتَرَكْنَا فِیْهَاۤ اٰیَةً لِّلَّذِیْنَ یَخَافُوْنَ الْعَذَابَ الْاَلِیْمَ ۟ؕ وَفِي مُوسَىٰٓ إِذۡ أَرۡسَلۡنَٰهُ إِلَىٰ فِرۡعَوۡنَ بِسُلۡطَٰنٖ مُّبِينٖ ۝٣٨ وَفِی مُوسَىٰۤ إِذۡ أَرۡسَلۡنَٰهُ إِلَىٰ فِرۡعَوۡنَ بِسُلۡطَٰنࣲ مُّبِینࣲ ۝٣٨ وَفِیْ مُوْسٰۤی اِذْ اَرْسَلْنٰهُ اِلٰی فِرْعَوْنَ بِسُلْطٰنٍ مُّبِیْنٍ ۟ فَتَوَلَّىٰ بِرُكۡنِهِۦ وَقَالَ سَٰحِرٌ أَوۡ مَجۡنُونٞ ۝٣٩ فَتَوَلَّىٰ بِرُكۡنِهِۦ وَقَالَ سَٰحِرٌ أَوۡ مَجۡنُونࣱ ۝٣٩ فَتَوَلّٰی بِرُكْنِهٖ وَقَالَ سٰحِرٌ اَوْ مَجْنُوْنٌ ۟ فَأَخَذۡنَٰهُ وَجُنُودَهُۥ فَنَبَذۡنَٰهُمۡ فِي ٱلۡيَمِّ وَهُوَ مُلِيمٞ ۝٤٠ فَأَخَذۡنَٰهُ وَجُنُودَهُۥ فَنَبَذۡنَٰهُمۡ فِی ٱلۡیَمِّ وَهُوَ مُلِیمࣱ ۝٤٠ فَاَخَذْنٰهُ وَجُنُوْدَهٗ فَنَبَذْنٰهُمْ فِی الْیَمِّ وَهُوَ مُلِیْمٌ ۟ؕ وَفِي عَادٍ إِذۡ أَرۡسَلۡنَا عَلَيۡهِمُ ٱلرِّيحَ ٱلۡعَقِيمَ ۝٤١ وَفِی عَادٍ إِذۡ أَرۡسَلۡنَا عَلَیۡهِمُ ٱلرِّیحَ ٱلۡعَقِیمَ ۝٤١ وَفِیْ عَادٍ اِذْ اَرْسَلْنَا عَلَیْهِمُ الرِّیْحَ الْعَقِیْمَ ۟ۚ مَا تَذَرُ مِن شَيۡءٍ أَتَتۡ عَلَيۡهِ إِلَّا جَعَلَتۡهُ كَٱلرَّمِيمِ ۝٤٢ مَا تَذَرُ مِن شَیۡءٍ أَتَتۡ عَلَیۡهِ إِلَّا جَعَلَتۡهُ كَٱلرَّمِیمِ ۝٤٢ مَا تَذَرُ مِنْ شَیْءٍ اَتَتْ عَلَیْهِ اِلَّا جَعَلَتْهُ كَالرَّمِیْمِ ۟ؕ وَفِي ثَمُودَ إِذۡ قِيلَ لَهُمۡ تَمَتَّعُواْ حَتَّىٰ حِينٖ ۝٤٣ وَفِی ثَمُودَ إِذۡ قِیلَ لَهُمۡ تَمَتَّعُوا۟ حَتَّىٰ حِینࣲ ۝٤٣ وَفِیْ ثَمُوْدَ اِذْ قِیْلَ لَهُمْ تَمَتَّعُوْا حَتّٰی حِیْنٍ ۟ فَعَتَوۡاْ عَنۡ أَمۡرِ رَبِّهِمۡ فَأَخَذَتۡهُمُ ٱلصَّٰعِقَةُ وَهُمۡ يَنظُرُونَ ۝٤٤ فَعَتَوۡا۟ عَنۡ أَمۡرِ رَبِّهِمۡ فَأَخَذَتۡهُمُ ٱلصَّٰعِقَةُ وَهُمۡ یَنظُرُونَ ۝٤٤ فَعَتَوْا عَنْ اَمْرِ رَبِّهِمْ فَاَخَذَتْهُمُ الصّٰعِقَةُ وَهُمْ یَنْظُرُوْنَ ۟ فَمَا ٱسۡتَطَٰعُواْ مِن قِيَامٖ وَمَا كَانُواْ مُنتَصِرِينَ ۝٤٥ فَمَا ٱسۡتَطَٰعُوا۟ مِن قِیَامࣲ وَمَا كَانُوا۟ مُنتَصِرِینَ ۝٤٥ فَمَا اسْتَطَاعُوْا مِنْ قِیَامٍ وَّمَا كَانُوْا مُنْتَصِرِیْنَ ۟ۙ وَقَوۡمَ نُوحٖ مِّن قَبۡلُۖ إِنَّهُمۡ كَانُواْ قَوۡمٗا فَٰسِقِينَ ۝٤٦ وَقَوۡمَ نُوحࣲ مِّن قَبۡلُۖ إِنَّهُمۡ كَانُوا۟ قَوۡمࣰا فَٰسِقِینَ ۝٤٦ وَقَوْمَ نُوْحٍ مِّنْ قَبْلُ ؕ اِنَّهُمْ كَانُوْا قَوْمًا فٰسِقِیْنَ ۟۠ وَٱلسَّمَآءَ بَنَيۡنَٰهَا بِأَيۡيْدٖ وَإِنَّا لَمُوسِعُونَ ۝٤٧ وَٱلسَّمَاۤءَ بَنَیۡنَٰهَا بِأَیۡی۟دࣲ وَإِنَّا لَمُوسِعُونَ ۝٤٧ وَالسَّمَآءَ بَنَیْنٰهَا بِاَیْىدٍ وَّاِنَّا لَمُوْسِعُوْنَ ۟ وَٱلۡأَرۡضَ فَرَشۡنَٰهَا فَنِعۡمَ ٱلۡمَٰهِدُونَ ۝٤٨ وَٱلۡأَرۡضَ فَرَشۡنَٰهَا فَنِعۡمَ ٱلۡمَٰهِدُونَ ۝٤٨ وَالْاَرْضَ فَرَشْنٰهَا فَنِعْمَ الْمٰهِدُوْنَ ۟ وَمِن كُلِّ شَيۡءٍ خَلَقۡنَا زَوۡجَيۡنِ لَعَلَّكُمۡ تَذَكَّرُونَ ۝٤٩ وَمِن كُلِّ شَیۡءٍ خَلَقۡنَا زَوۡجَیۡنِ لَعَلَّكُمۡ تَذَكَّرُونَ ۝٤٩ وَمِنْ كُلِّ شَیْءٍ خَلَقْنَا زَوْجَیْنِ لَعَلَّكُمْ تَذَكَّرُوْنَ ۟ فَفِرُّوٓاْ إِلَى ٱللَّهِۖ إِنِّي لَكُم مِّنۡهُ نَذِيرٞ مُّبِينٞ ۝٥٠ فَفِرُّوۤا۟ إِلَى ٱللَّهِۖ إِنِّی لَكُم مِّنۡهُ نَذِیرࣱ مُّبِینࣱ ۝٥٠ فَفِرُّوْۤا اِلَی اللّٰهِ ؕ اِنِّیْ لَكُمْ مِّنْهُ نَذِیْرٌ مُّبِیْنٌ ۟ۚ وَلَا تَجۡعَلُواْ مَعَ ٱللَّهِ إِلَٰهًا ءَاخَرَۖ إِنِّي لَكُم مِّنۡهُ نَذِيرٞ مُّبِينٞ ۝٥١ وَلَا تَجۡعَلُوا۟ مَعَ ٱللَّهِ إِلَٰهًا ءَاخَرَۖ إِنِّی لَكُم مِّنۡهُ نَذِیرࣱ مُّبِینࣱ ۝٥١ وَلَا تَجْعَلُوْا مَعَ اللّٰهِ اِلٰهًا اٰخَرَ ؕ اِنِّیْ لَكُمْ مِّنْهُ نَذِیْرٌ مُّبِیْنٌ ۟ كَذَٰلِكَ مَآ أَتَى ٱلَّذِينَ مِن قَبۡلِهِم مِّن رَّسُولٍ إِلَّا قَالُواْ سَاحِرٌ أَوۡ مَجۡنُونٌ ۝٥٢ كَذَٰلِكَ مَاۤ أَتَى ٱلَّذِینَ مِن قَبۡلِهِم مِّن رَّسُولٍ إِلَّا قَالُوا۟ سَاحِرٌ أَوۡ مَجۡنُونٌ ۝٥٢ كَذٰلِكَ مَاۤ اَتَی الَّذِیْنَ مِنْ قَبْلِهِمْ مِّنْ رَّسُوْلٍ اِلَّا قَالُوْا سَاحِرٌ اَوْ مَجْنُوْنٌ ۟ۚ أَتَوَاصَوۡاْ بِهِۦۚ بَلۡ هُمۡ قَوۡمٞ طَاغُونَ ۝٥٣ أَتَوَاصَوۡا۟ بِهِۦۚ بَلۡ هُمۡ قَوۡمࣱ طَاغُونَ ۝٥٣ اَتَوَاصَوْا بِهٖ ۚ بَلْ هُمْ قَوْمٌ طَاغُوْنَ ۟ۚ فَتَوَلَّ عَنۡهُمۡ فَمَآ أَنتَ بِمَلُومٖ ۝٥٤ فَتَوَلَّ عَنۡهُمۡ فَمَاۤ أَنتَ بِمَلُومࣲ ۝٥٤ فَتَوَلَّ عَنْهُمْ فَمَاۤ اَنْتَ بِمَلُوْمٍ ۟ وَذَكِّرۡ فَإِنَّ ٱلذِّكۡرَىٰ تَنفَعُ ٱلۡمُؤۡمِنِينَ ۝٥٥ وَذَكِّرۡ فَإِنَّ ٱلذِّكۡرَىٰ تَنفَعُ ٱلۡمُؤۡمِنِینَ ۝٥٥ وَّذَكِّرْ فَاِنَّ الذِّكْرٰی تَنْفَعُ الْمُؤْمِنِیْنَ ۟ وَمَا خَلَقۡتُ ٱلۡجِنَّ وَٱلۡإِنسَ إِلَّا لِيَعۡبُدُونِ ۝٥٦ وَمَا خَلَقۡتُ ٱلۡجِنَّ وَٱلۡإِنسَ إِلَّا لِیَعۡبُدُونِ ۝٥٦ وَمَا خَلَقْتُ الْجِنَّ وَالْاِنْسَ اِلَّا لِیَعْبُدُوْنِ ۟ مَآ أُرِيدُ مِنۡهُم مِّن رِّزۡقٖ وَمَآ أُرِيدُ أَن يُطۡعِمُونِ ۝٥٧ مَاۤ أُرِیدُ مِنۡهُم مِّن رِّزۡقࣲ وَمَاۤ أُرِیدُ أَن یُطۡعِمُونِ ۝٥٧ مَاۤ اُرِیْدُ مِنْهُمْ مِّنْ رِّزْقٍ وَّمَاۤ اُرِیْدُ اَنْ یُّطْعِمُوْنِ ۟ إِنَّ ٱللَّهَ هُوَ ٱلرَّزَّاقُ ذُو ٱلۡقُوَّةِ ٱلۡمَتِينُ ۝٥٨ إِنَّ ٱللَّهَ هُوَ ٱلرَّزَّاقُ ذُو ٱلۡقُوَّةِ ٱلۡمَتِینُ ۝٥٨ اِنَّ اللّٰهَ هُوَ الرَّزَّاقُ ذُو الْقُوَّةِ الْمَتِیْنُ ۟ فَإِنَّ لِلَّذِينَ ظَلَمُواْ ذَنُوبٗا مِّثۡلَ ذَنُوبِ أَصۡحَٰبِهِمۡ فَلَا يَسۡتَعۡجِلُونِ ۝٥٩ فَإِنَّ لِلَّذِینَ ظَلَمُوا۟ ذَنُوبࣰا مِّثۡلَ ذَنُوبِ أَصۡحَٰبِهِمۡ فَلَا یَسۡتَعۡجِلُونِ ۝٥٩ فَاِنَّ لِلَّذِیْنَ ظَلَمُوْا ذَنُوْبًا مِّثْلَ ذَنُوْبِ اَصْحٰبِهِمْ فَلَا یَسْتَعْجِلُوْنِ ۟ فَوَيۡلٞ لِّلَّذِينَ كَفَرُواْ مِن يَوۡمِهِمُ ٱلَّذِي يُوعَدُونَ ۝٦٠ فَوَیۡلࣱ لِّلَّذِینَ كَفَرُوا۟ مِن یَوۡمِهِمُ ٱلَّذِی یُوعَدُونَ ۝٦٠ فَوَیْلٌ لِّلَّذِیْنَ كَفَرُوْا مِنْ یَّوْمِهِمُ الَّذِیْ یُوْعَدُوْنَ ۟۠
