Surah 52 · Meccan · 49 ayat

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

وَٱلطُّورِ ۝١ وَٱلطُّورِ ۝١ وَالطُّوْرِ ۟ۙ وَكِتَٰبٖ مَّسۡطُورٖ ۝٢ وَكِتَٰبࣲ مَّسۡطُورࣲ ۝٢ وَكِتٰبٍ مَّسْطُوْرٍ ۟ۙ فِي رَقّٖ مَّنشُورٖ ۝٣ فِی رَقࣲّ مَّنشُورࣲ ۝٣ فِیْ رَقٍّ مَّنْشُوْرٍ ۟ۙ وَٱلۡبَيۡتِ ٱلۡمَعۡمُورِ ۝٤ وَٱلۡبَیۡتِ ٱلۡمَعۡمُورِ ۝٤ وَّالْبَیْتِ الْمَعْمُوْرِ ۟ۙ وَٱلسَّقۡفِ ٱلۡمَرۡفُوعِ ۝٥ وَٱلسَّقۡفِ ٱلۡمَرۡفُوعِ ۝٥ وَالسَّقْفِ الْمَرْفُوْعِ ۟ۙ وَٱلۡبَحۡرِ ٱلۡمَسۡجُورِ ۝٦ وَٱلۡبَحۡرِ ٱلۡمَسۡجُورِ ۝٦ وَالْبَحْرِ الْمَسْجُوْرِ ۟ۙ إِنَّ عَذَابَ رَبِّكَ لَوَٰقِعٞ ۝٧ إِنَّ عَذَابَ رَبِّكَ لَوَٰقِعࣱ ۝٧ اِنَّ عَذَابَ رَبِّكَ لَوَاقِعٌ ۟ۙ مَّا لَهُۥ مِن دَافِعٖ ۝٨ مَّا لَهُۥ مِن دَافِعࣲ ۝٨ مَّا لَهٗ مِنْ دَافِعٍ ۟ۙ يَوۡمَ تَمُورُ ٱلسَّمَآءُ مَوۡرٗا ۝٩ یَوۡمَ تَمُورُ ٱلسَّمَاۤءُ مَوۡرࣰا ۝٩ یَّوْمَ تَمُوْرُ السَّمَآءُ مَوْرًا ۟ۙ وَتَسِيرُ ٱلۡجِبَالُ سَيۡرٗا ۝١٠ وَتَسِیرُ ٱلۡجِبَالُ سَیۡرࣰا ۝١٠ وَّتَسِیْرُ الْجِبَالُ سَیْرًا ۟ؕ فَوَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝١١ فَوَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝١١ فَوَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ۙ ٱلَّذِينَ هُمۡ فِي خَوۡضٖ يَلۡعَبُونَ ۝١٢ ٱلَّذِینَ هُمۡ فِی خَوۡضࣲ یَلۡعَبُونَ ۝١٢ الَّذِیْنَ هُمْ فِیْ خَوْضٍ یَّلْعَبُوْنَ ۟ۘ يَوۡمَ يُدَعُّونَ إِلَىٰ نَارِ جَهَنَّمَ دَعًّا ۝١٣ یَوۡمَ یُدَعُّونَ إِلَىٰ نَارِ جَهَنَّمَ دَعًّا ۝١٣ یَوْمَ یُدَعُّوْنَ اِلٰی نَارِ جَهَنَّمَ دَعًّا ۟ؕ هَٰذِهِ ٱلنَّارُ ٱلَّتِي كُنتُم بِهَا تُكَذِّبُونَ ۝١٤ هَٰذِهِ ٱلنَّارُ ٱلَّتِی كُنتُم بِهَا تُكَذِّبُونَ ۝١٤ هٰذِهِ النَّارُ الَّتِیْ كُنْتُمْ بِهَا تُكَذِّبُوْنَ ۟ أَفَسِحۡرٌ هَٰذَآ أَمۡ أَنتُمۡ لَا تُبۡصِرُونَ ۝١٥ أَفَسِحۡرٌ هَٰذَاۤ أَمۡ أَنتُمۡ لَا تُبۡصِرُونَ ۝١٥ اَفَسِحْرٌ هٰذَاۤ اَمْ اَنْتُمْ لَا تُبْصِرُوْنَ ۟ۚ ٱصۡلَوۡهَا فَٱصۡبِرُوٓاْ أَوۡ لَا تَصۡبِرُواْ سَوَآءٌ عَلَيۡكُمۡۖ إِنَّمَا تُجۡزَوۡنَ مَا كُنتُمۡ تَعۡمَلُونَ ۝١٦ ٱصۡلَوۡهَا فَٱصۡبِرُوۤا۟ أَوۡ لَا تَصۡبِرُوا۟ سَوَاۤءٌ عَلَیۡكُمۡۖ إِنَّمَا تُجۡزَوۡنَ مَا كُنتُمۡ تَعۡمَلُونَ ۝١٦ اِصْلَوْهَا فَاصْبِرُوْۤا اَوْ لَا تَصْبِرُوْا ۚ سَوَآءٌ عَلَیْكُمْ ؕ اِنَّمَا تُجْزَوْنَ مَا كُنْتُمْ تَعْمَلُوْنَ ۟ إِنَّ ٱلۡمُتَّقِينَ فِي جَنَّٰتٖ وَنَعِيمٖ ۝١٧ إِنَّ ٱلۡمُتَّقِینَ فِی جَنَّٰتࣲ وَنَعِیمࣲ ۝١٧ اِنَّ الْمُتَّقِیْنَ فِیْ جَنّٰتٍ وَّنَعِیْمٍ ۟ۙ فَٰكِهِينَ بِمَآ ءَاتَىٰهُمۡ رَبُّهُمۡ وَوَقَىٰهُمۡ رَبُّهُمۡ عَذَابَ ٱلۡجَحِيمِ ۝١٨ فَٰكِهِینَ بِمَاۤ ءَاتَىٰهُمۡ رَبُّهُمۡ وَوَقَىٰهُمۡ رَبُّهُمۡ عَذَابَ ٱلۡجَحِیمِ ۝١٨ فٰكِهِیْنَ بِمَاۤ اٰتٰىهُمْ رَبُّهُمْ ۚ وَوَقٰىهُمْ رَبُّهُمْ عَذَابَ الْجَحِیْمِ ۟ كُلُواْ وَٱشۡرَبُواْ هَنِيٓـَٔۢا بِمَا كُنتُمۡ تَعۡمَلُونَ ۝١٩ كُلُوا۟ وَٱشۡرَبُوا۟ هَنِیۤءَۢا بِمَا كُنتُمۡ تَعۡمَلُونَ ۝١٩ كُلُوْا وَاشْرَبُوْا هَنِیْٓـًٔا بِمَا كُنْتُمْ تَعْمَلُوْنَ ۟ۙ مُتَّكِـِٔينَ عَلَىٰ سُرُرٖ مَّصۡفُوفَةٖۖ وَزَوَّجۡنَٰهُم بِحُورٍ عِينٖ ۝٢٠ مُتَّكِءِینَ عَلَىٰ سُرُرࣲ مَّصۡفُوفَةࣲۖ وَزَوَّجۡنَٰهُم بِحُورٍ عِینࣲ ۝٢٠ مُتَّكِـِٕیْنَ عَلٰی سُرُرٍ مَّصْفُوْفَةٍ ۚ وَزَوَّجْنٰهُمْ بِحُوْرٍ عِیْنٍ ۟ وَٱلَّذِينَ ءَامَنُواْ وَٱتَّبَعَتۡهُمۡ ذُرِّيَّتُهُم بِإِيمَٰنٍ أَلۡحَقۡنَا بِهِمۡ ذُرِّيَّتَهُمۡ وَمَآ أَلَتۡنَٰهُم مِّنۡ عَمَلِهِم مِّن شَيۡءٖۚ كُلُّ ٱمۡرِيِٕۭ بِمَا كَسَبَ رَهِينٞ ۝٢١ وَٱلَّذِینَ ءَامَنُوا۟ وَٱتَّبَعَتۡهُمۡ ذُرِّیَّتُهُم بِإِیمَٰنٍ أَلۡحَقۡنَا بِهِمۡ ذُرِّیَّتَهُمۡ وَمَاۤ أَلَتۡنَٰهُم مِّنۡ عَمَلِهِم مِّن شَیۡءࣲۚ كُلُّ ٱمۡرِئِۭ بِمَا كَسَبَ رَهِینࣱ ۝٢١ وَالَّذِیْنَ اٰمَنُوْا وَاتَّبَعَتْهُمْ ذُرِّیَّتُهُمْ بِاِیْمَانٍ اَلْحَقْنَا بِهِمْ ذُرِّیَّتَهُمْ وَمَاۤ اَلَتْنٰهُمْ مِّنْ عَمَلِهِمْ مِّنْ شَیْءٍ ؕ كُلُّ امْرِىۢ بِمَا كَسَبَ رَهِیْنٌ ۟ وَأَمۡدَدۡنَٰهُم بِفَٰكِهَةٖ وَلَحۡمٖ مِّمَّا يَشۡتَهُونَ ۝٢٢ وَأَمۡدَدۡنَٰهُم بِفَٰكِهَةࣲ وَلَحۡمࣲ مِّمَّا یَشۡتَهُونَ ۝٢٢ وَاَمْدَدْنٰهُمْ بِفَاكِهَةٍ وَّلَحْمٍ مِّمَّا یَشْتَهُوْنَ ۟ يَتَنَٰزَعُونَ فِيهَا كَأۡسٗا لَّا لَغۡوٞ فِيهَا وَلَا تَأۡثِيمٞ ۝٢٣ یَتَنَٰزَعُونَ فِیهَا كَأۡسࣰا لَّا لَغۡوࣱ فِیهَا وَلَا تَأۡثِیمࣱ ۝٢٣ یَتَنَازَعُوْنَ فِیْهَا كَاْسًا لَّا لَغْوٌ فِیْهَا وَلَا تَاْثِیْمٌ ۟ وَيَطُوفُ عَلَيۡهِمۡ غِلۡمَانٞ لَّهُمۡ كَأَنَّهُمۡ لُؤۡلُؤٞ مَّكۡنُونٞ ۝٢٤ وَیَطُوفُ عَلَیۡهِمۡ غِلۡمَانࣱ لَّهُمۡ كَأَنَّهُمۡ لُؤۡلُؤࣱ مَّكۡنُونࣱ ۝٢٤ وَیَطُوْفُ عَلَیْهِمْ غِلْمَانٌ لَّهُمْ كَاَنَّهُمْ لُؤْلُؤٌ مَّكْنُوْنٌ ۟ وَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضٖ يَتَسَآءَلُونَ ۝٢٥ وَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضࣲ یَتَسَاۤءَلُونَ ۝٢٥ وَاَقْبَلَ بَعْضُهُمْ عَلٰی بَعْضٍ یَّتَسَآءَلُوْنَ ۟ قَالُوٓاْ إِنَّا كُنَّا قَبۡلُ فِيٓ أَهۡلِنَا مُشۡفِقِينَ ۝٢٦ قَالُوۤا۟ إِنَّا كُنَّا قَبۡلُ فِیۤ أَهۡلِنَا مُشۡفِقِینَ ۝٢٦ قَالُوْۤا اِنَّا كُنَّا قَبْلُ فِیْۤ اَهْلِنَا مُشْفِقِیْنَ ۟ فَمَنَّ ٱللَّهُ عَلَيۡنَا وَوَقَىٰنَا عَذَابَ ٱلسَّمُومِ ۝٢٧ فَمَنَّ ٱللَّهُ عَلَیۡنَا وَوَقَىٰنَا عَذَابَ ٱلسَّمُومِ ۝٢٧ فَمَنَّ اللّٰهُ عَلَیْنَا وَوَقٰىنَا عَذَابَ السَّمُوْمِ ۟ إِنَّا كُنَّا مِن قَبۡلُ نَدۡعُوهُۖ إِنَّهُۥ هُوَ ٱلۡبَرُّ ٱلرَّحِيمُ ۝٢٨ إِنَّا كُنَّا مِن قَبۡلُ نَدۡعُوهُۖ إِنَّهُۥ هُوَ ٱلۡبَرُّ ٱلرَّحِیمُ ۝٢٨ اِنَّا كُنَّا مِنْ قَبْلُ نَدْعُوْهُ ؕ اِنَّهٗ هُوَ الْبَرُّ الرَّحِیْمُ ۟۠ فَذَكِّرۡ فَمَآ أَنتَ بِنِعۡمَتِ رَبِّكَ بِكَاهِنٖ وَلَا مَجۡنُونٍ ۝٢٩ فَذَكِّرۡ فَمَاۤ أَنتَ بِنِعۡمَتِ رَبِّكَ بِكَاهِنࣲ وَلَا مَجۡنُونٍ ۝٢٩ فَذَكِّرْ فَمَاۤ اَنْتَ بِنِعْمَتِ رَبِّكَ بِكَاهِنٍ وَّلَا مَجْنُوْنٍ ۟ؕ أَمۡ يَقُولُونَ شَاعِرٞ نَّتَرَبَّصُ بِهِۦ رَيۡبَ ٱلۡمَنُونِ ۝٣٠ أَمۡ یَقُولُونَ شَاعِرࣱ نَّتَرَبَّصُ بِهِۦ رَیۡبَ ٱلۡمَنُونِ ۝٣٠ اَمْ یَقُوْلُوْنَ شَاعِرٌ نَّتَرَبَّصُ بِهٖ رَیْبَ الْمَنُوْنِ ۟ قُلۡ تَرَبَّصُواْ فَإِنِّي مَعَكُم مِّنَ ٱلۡمُتَرَبِّصِينَ ۝٣١ قُلۡ تَرَبَّصُوا۟ فَإِنِّی مَعَكُم مِّنَ ٱلۡمُتَرَبِّصِینَ ۝٣١ قُلْ تَرَبَّصُوْا فَاِنِّیْ مَعَكُمْ مِّنَ الْمُتَرَبِّصِیْنَ ۟ؕ أَمۡ تَأۡمُرُهُمۡ أَحۡلَٰمُهُم بِهَٰذَآۚ أَمۡ هُمۡ قَوۡمٞ طَاغُونَ ۝٣٢ أَمۡ تَأۡمُرُهُمۡ أَحۡلَٰمُهُم بِهَٰذَاۤۚ أَمۡ هُمۡ قَوۡمࣱ طَاغُونَ ۝٣٢ اَمْ تَاْمُرُهُمْ اَحْلَامُهُمْ بِهٰذَاۤ اَمْ هُمْ قَوْمٌ طَاغُوْنَ ۟ۚ أَمۡ يَقُولُونَ تَقَوَّلَهُۥۚ بَل لَّا يُؤۡمِنُونَ ۝٣٣ أَمۡ یَقُولُونَ تَقَوَّلَهُۥۚ بَل لَّا یُؤۡمِنُونَ ۝٣٣ اَمْ یَقُوْلُوْنَ تَقَوَّلَهٗ ۚ بَلْ لَّا یُؤْمِنُوْنَ ۟ۚ فَلۡيَأۡتُواْ بِحَدِيثٖ مِّثۡلِهِۦٓ إِن كَانُواْ صَٰدِقِينَ ۝٣٤ فَلۡیَأۡتُوا۟ بِحَدِیثࣲ مِّثۡلِهِۦۤ إِن كَانُوا۟ صَٰدِقِینَ ۝٣٤ فَلْیَاْتُوْا بِحَدِیْثٍ مِّثْلِهٖۤ اِنْ كَانُوْا صٰدِقِیْنَ ۟ؕ أَمۡ خُلِقُواْ مِنۡ غَيۡرِ شَيۡءٍ أَمۡ هُمُ ٱلۡخَٰلِقُونَ ۝٣٥ أَمۡ خُلِقُوا۟ مِنۡ غَیۡرِ شَیۡءٍ أَمۡ هُمُ ٱلۡخَٰلِقُونَ ۝٣٥ اَمْ خُلِقُوْا مِنْ غَیْرِ شَیْءٍ اَمْ هُمُ الْخٰلِقُوْنَ ۟ؕ أَمۡ خَلَقُواْ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَۚ بَل لَّا يُوقِنُونَ ۝٣٦ أَمۡ خَلَقُوا۟ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَۚ بَل لَّا یُوقِنُونَ ۝٣٦ اَمْ خَلَقُوا السَّمٰوٰتِ وَالْاَرْضَ ۚ بَلْ لَّا یُوْقِنُوْنَ ۟ؕ أَمۡ عِندَهُمۡ خَزَآئِنُ رَبِّكَ أَمۡ هُمُ ٱلۡمُصَۜيۡطِرُونَ ۝٣٧ أَمۡ عِندَهُمۡ خَزَاۤئِنُ رَبِّكَ أَمۡ هُمُ ٱلۡمُصَۣیۡطِرُونَ ۝٣٧ اَمْ عِنْدَهُمْ خَزَآىِٕنُ رَبِّكَ اَمْ هُمُ الْمُصَۜیْطِرُوْنَ ۟ؕ أَمۡ لَهُمۡ سُلَّمٞ يَسۡتَمِعُونَ فِيهِۖ فَلۡيَأۡتِ مُسۡتَمِعُهُم بِسُلۡطَٰنٖ مُّبِينٍ ۝٣٨ أَمۡ لَهُمۡ سُلَّمࣱ یَسۡتَمِعُونَ فِیهِۖ فَلۡیَأۡتِ مُسۡتَمِعُهُم بِسُلۡطَٰنࣲ مُّبِینٍ ۝٣٨ اَمْ لَهُمْ سُلَّمٌ یَّسْتَمِعُوْنَ فِیْهِ ۚ فَلْیَاْتِ مُسْتَمِعُهُمْ بِسُلْطٰنٍ مُّبِیْنٍ ۟ؕ أَمۡ لَهُ ٱلۡبَنَٰتُ وَلَكُمُ ٱلۡبَنُونَ ۝٣٩ أَمۡ لَهُ ٱلۡبَنَٰتُ وَلَكُمُ ٱلۡبَنُونَ ۝٣٩ اَمْ لَهُ الْبَنٰتُ وَلَكُمُ الْبَنُوْنَ ۟ؕ أَمۡ تَسۡـَٔلُهُمۡ أَجۡرٗا فَهُم مِّن مَّغۡرَمٖ مُّثۡقَلُونَ ۝٤٠ أَمۡ تَسۡءَلُهُمۡ أَجۡرࣰا فَهُم مِّن مَّغۡرَمࣲ مُّثۡقَلُونَ ۝٤٠ اَمْ تَسْـَٔلُهُمْ اَجْرًا فَهُمْ مِّنْ مَّغْرَمٍ مُّثْقَلُوْنَ ۟ؕ أَمۡ عِندَهُمُ ٱلۡغَيۡبُ فَهُمۡ يَكۡتُبُونَ ۝٤١ أَمۡ عِندَهُمُ ٱلۡغَیۡبُ فَهُمۡ یَكۡتُبُونَ ۝٤١ اَمْ عِنْدَهُمُ الْغَیْبُ فَهُمْ یَكْتُبُوْنَ ۟ؕ أَمۡ يُرِيدُونَ كَيۡدٗاۖ فَٱلَّذِينَ كَفَرُواْ هُمُ ٱلۡمَكِيدُونَ ۝٤٢ أَمۡ یُرِیدُونَ كَیۡدࣰاۖ فَٱلَّذِینَ كَفَرُوا۟ هُمُ ٱلۡمَكِیدُونَ ۝٤٢ اَمْ یُرِیْدُوْنَ كَیْدًا ؕ فَالَّذِیْنَ كَفَرُوْا هُمُ الْمَكِیْدُوْنَ ۟ؕ أَمۡ لَهُمۡ إِلَٰهٌ غَيۡرُ ٱللَّهِۚ سُبۡحَٰنَ ٱللَّهِ عَمَّا يُشۡرِكُونَ ۝٤٣ أَمۡ لَهُمۡ إِلَٰهٌ غَیۡرُ ٱللَّهِۚ سُبۡحَٰنَ ٱللَّهِ عَمَّا یُشۡرِكُونَ ۝٤٣ اَمْ لَهُمْ اِلٰهٌ غَیْرُ اللّٰهِ ؕ سُبْحٰنَ اللّٰهِ عَمَّا یُشْرِكُوْنَ ۟ وَإِن يَرَوۡاْ كِسۡفٗا مِّنَ ٱلسَّمَآءِ سَاقِطٗا يَقُولُواْ سَحَابٞ مَّرۡكُومٞ ۝٤٤ وَإِن یَرَوۡا۟ كِسۡفࣰا مِّنَ ٱلسَّمَاۤءِ سَاقِطࣰا یَقُولُوا۟ سَحَابࣱ مَّرۡكُومࣱ ۝٤٤ وَاِنْ یَّرَوْا كِسْفًا مِّنَ السَّمَآءِ سَاقِطًا یَّقُوْلُوْا سَحَابٌ مَّرْكُوْمٌ ۟ فَذَرۡهُمۡ حَتَّىٰ يُلَٰقُواْ يَوۡمَهُمُ ٱلَّذِي فِيهِ يُصۡعَقُونَ ۝٤٥ فَذَرۡهُمۡ حَتَّىٰ یُلَٰقُوا۟ یَوۡمَهُمُ ٱلَّذِی فِیهِ یُصۡعَقُونَ ۝٤٥ فَذَرْهُمْ حَتّٰی یُلٰقُوْا یَوْمَهُمُ الَّذِیْ فِیْهِ یُصْعَقُوْنَ ۟ۙ يَوۡمَ لَا يُغۡنِي عَنۡهُمۡ كَيۡدُهُمۡ شَيۡـٔٗا وَلَا هُمۡ يُنصَرُونَ ۝٤٦ یَوۡمَ لَا یُغۡنِی عَنۡهُمۡ كَیۡدُهُمۡ شَیۡءࣰا وَلَا هُمۡ یُنصَرُونَ ۝٤٦ یَوْمَ لَا یُغْنِیْ عَنْهُمْ كَیْدُهُمْ شَیْـًٔا وَّلَا هُمْ یُنْصَرُوْنَ ۟ؕ وَإِنَّ لِلَّذِينَ ظَلَمُواْ عَذَابٗا دُونَ ذَٰلِكَ وَلَٰكِنَّ أَكۡثَرَهُمۡ لَا يَعۡلَمُونَ ۝٤٧ وَإِنَّ لِلَّذِینَ ظَلَمُوا۟ عَذَابࣰا دُونَ ذَٰلِكَ وَلَٰكِنَّ أَكۡثَرَهُمۡ لَا یَعۡلَمُونَ ۝٤٧ وَاِنَّ لِلَّذِیْنَ ظَلَمُوْا عَذَابًا دُوْنَ ذٰلِكَ وَلٰكِنَّ اَكْثَرَهُمْ لَا یَعْلَمُوْنَ ۟ وَٱصۡبِرۡ لِحُكۡمِ رَبِّكَ فَإِنَّكَ بِأَعۡيُنِنَاۖ وَسَبِّحۡ بِحَمۡدِ رَبِّكَ حِينَ تَقُومُ ۝٤٨ وَٱصۡبِرۡ لِحُكۡمِ رَبِّكَ فَإِنَّكَ بِأَعۡیُنِنَاۖ وَسَبِّحۡ بِحَمۡدِ رَبِّكَ حِینَ تَقُومُ ۝٤٨ وَاصْبِرْ لِحُكْمِ رَبِّكَ فَاِنَّكَ بِاَعْیُنِنَا وَسَبِّحْ بِحَمْدِ رَبِّكَ حِیْنَ تَقُوْمُ ۟ۙ وَمِنَ ٱلَّيۡلِ فَسَبِّحۡهُ وَإِدۡبَٰرَ ٱلنُّجُومِ ۝٤٩ وَمِنَ ٱلَّیۡلِ فَسَبِّحۡهُ وَإِدۡبَٰرَ ٱلنُّجُومِ ۝٤٩ وَمِنَ الَّیْلِ فَسَبِّحْهُ وَاِدْبَارَ النُّجُوْمِ ۟۠
