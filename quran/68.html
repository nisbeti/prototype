بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

نٓۚ وَٱلۡقَلَمِ وَمَا يَسۡطُرُونَ ۝١ نۤۚ وَٱلۡقَلَمِ وَمَا یَسۡطُرُونَ ۝١ نٓ وَالْقَلَمِ وَمَا یَسْطُرُوْنَ ۟ۙ مَآ أَنتَ بِنِعۡمَةِ رَبِّكَ بِمَجۡنُونٖ ۝٢ مَاۤ أَنتَ بِنِعۡمَةِ رَبِّكَ بِمَجۡنُونࣲ ۝٢ مَاۤ اَنْتَ بِنِعْمَةِ رَبِّكَ بِمَجْنُوْنٍ ۟ۚ وَإِنَّ لَكَ لَأَجۡرًا غَيۡرَ مَمۡنُونٖ ۝٣ وَإِنَّ لَكَ لَأَجۡرًا غَیۡرَ مَمۡنُونࣲ ۝٣ وَاِنَّ لَكَ لَاَجْرًا غَیْرَ مَمْنُوْنٍ ۟ۚ وَإِنَّكَ لَعَلَىٰ خُلُقٍ عَظِيمٖ ۝٤ وَإِنَّكَ لَعَلَىٰ خُلُقٍ عَظِیمࣲ ۝٤ وَاِنَّكَ لَعَلٰی خُلُقٍ عَظِیْمٍ ۟ فَسَتُبۡصِرُ وَيُبۡصِرُونَ ۝٥ فَسَتُبۡصِرُ وَیُبۡصِرُونَ ۝٥ فَسَتُبْصِرُ وَیُبْصِرُوْنَ ۟ۙ بِأَييِّكُمُ ٱلۡمَفۡتُونُ ۝٦ بِأَییِّكُمُ ٱلۡمَفۡتُونُ ۝٦ بِاَیِّىكُمُ الْمَفْتُوْنُ ۟ إِنَّ رَبَّكَ هُوَ أَعۡلَمُ بِمَن ضَلَّ عَن سَبِيلِهِۦ وَهُوَ أَعۡلَمُ بِٱلۡمُهۡتَدِينَ ۝٧ إِنَّ رَبَّكَ هُوَ أَعۡلَمُ بِمَن ضَلَّ عَن سَبِیلِهِۦ وَهُوَ أَعۡلَمُ بِٱلۡمُهۡتَدِینَ ۝٧ اِنَّ رَبَّكَ هُوَ اَعْلَمُ بِمَنْ ضَلَّ عَنْ سَبِیْلِهٖ ۪ وَهُوَ اَعْلَمُ بِالْمُهْتَدِیْنَ ۟ فَلَا تُطِعِ ٱلۡمُكَذِّبِينَ ۝٨ فَلَا تُطِعِ ٱلۡمُكَذِّبِینَ ۝٨ فَلَا تُطِعِ الْمُكَذِّبِیْنَ ۟ وَدُّواْ لَوۡ تُدۡهِنُ فَيُدۡهِنُونَ ۝٩ وَدُّوا۟ لَوۡ تُدۡهِنُ فَیُدۡهِنُونَ ۝٩ وَدُّوْا لَوْ تُدْهِنُ فَیُدْهِنُوْنَ ۟ وَلَا تُطِعۡ كُلَّ حَلَّافٖ مَّهِينٍ ۝١٠ وَلَا تُطِعۡ كُلَّ حَلَّافࣲ مَّهِینٍ ۝١٠ وَلَا تُطِعْ كُلَّ حَلَّافٍ مَّهِیْنٍ ۟ۙ هَمَّازٖ مَّشَّآءِۭ بِنَمِيمٖ ۝١١ هَمَّازࣲ مَّشَّاۤءِۭ بِنَمِیمࣲ ۝١١ هَمَّازٍ مَّشَّآءٍ بِنَمِیْمٍ ۟ۙ مَّنَّاعٖ لِّلۡخَيۡرِ مُعۡتَدٍ أَثِيمٍ ۝١٢ مَّنَّاعࣲ لِّلۡخَیۡرِ مُعۡتَدٍ أَثِیمٍ ۝١٢ مَّنَّاعٍ لِّلْخَیْرِ مُعْتَدٍ اَثِیْمٍ ۟ۙ عُتُلِّۭ بَعۡدَ ذَٰلِكَ زَنِيمٍ ۝١٣ عُتُلِّۭ بَعۡدَ ذَٰلِكَ زَنِیمٍ ۝١٣ عُتُلٍّ بَعْدَ ذٰلِكَ زَنِیْمٍ ۟ۙ أَن كَانَ ذَا مَالٖ وَبَنِينَ ۝١٤ أَن كَانَ ذَا مَالࣲ وَبَنِینَ ۝١٤ اَنْ كَانَ ذَا مَالٍ وَّبَنِیْنَ ۟ؕ إِذَا تُتۡلَىٰ عَلَيۡهِ ءَايَٰتُنَا قَالَ أَسَٰطِيرُ ٱلۡأَوَّلِينَ ۝١٥ إِذَا تُتۡلَىٰ عَلَیۡهِ ءَایَٰتُنَا قَالَ أَسَٰطِیرُ ٱلۡأَوَّلِینَ ۝١٥ اِذَا تُتْلٰی عَلَیْهِ اٰیٰتُنَا قَالَ اَسَاطِیْرُ الْاَوَّلِیْنَ ۟ سَنَسِمُهُۥ عَلَى ٱلۡخُرۡطُومِ ۝١٦ سَنَسِمُهُۥ عَلَى ٱلۡخُرۡطُومِ ۝١٦ سَنَسِمُهٗ عَلَی الْخُرْطُوْمِ ۟ إِنَّا بَلَوۡنَٰهُمۡ كَمَا بَلَوۡنَآ أَصۡحَٰبَ ٱلۡجَنَّةِ إِذۡ أَقۡسَمُواْ لَيَصۡرِمُنَّهَا مُصۡبِحِينَ ۝١٧ إِنَّا بَلَوۡنَٰهُمۡ كَمَا بَلَوۡنَاۤ أَصۡحَٰبَ ٱلۡجَنَّةِ إِذۡ أَقۡسَمُوا۟ لَیَصۡرِمُنَّهَا مُصۡبِحِینَ ۝١٧ اِنَّا بَلَوْنٰهُمْ كَمَا بَلَوْنَاۤ اَصْحٰبَ الْجَنَّةِ ۚ اِذْ اَقْسَمُوْا لَیَصْرِمُنَّهَا مُصْبِحِیْنَ ۟ۙ وَلَا يَسۡتَثۡنُونَ ۝١٨ وَلَا یَسۡتَثۡنُونَ ۝١٨ وَلَا یَسْتَثْنُوْنَ ۟ فَطَافَ عَلَيۡهَا طَآئِفٞ مِّن رَّبِّكَ وَهُمۡ نَآئِمُونَ ۝١٩ فَطَافَ عَلَیۡهَا طَاۤئِفࣱ مِّن رَّبِّكَ وَهُمۡ نَاۤئِمُونَ ۝١٩ فَطَافَ عَلَیْهَا طَآىِٕفٌ مِّنْ رَّبِّكَ وَهُمْ نَآىِٕمُوْنَ ۟ فَأَصۡبَحَتۡ كَٱلصَّرِيمِ ۝٢٠ فَأَصۡبَحَتۡ كَٱلصَّرِیمِ ۝٢٠ فَاَصْبَحَتْ كَالصَّرِیْمِ ۟ۙ فَتَنَادَوۡاْ مُصۡبِحِينَ ۝٢١ فَتَنَادَوۡا۟ مُصۡبِحِینَ ۝٢١ فَتَنَادَوْا مُصْبِحِیْنَ ۟ۙ أَنِ ٱغۡدُواْ عَلَىٰ حَرۡثِكُمۡ إِن كُنتُمۡ صَٰرِمِينَ ۝٢٢ أَنِ ٱغۡدُوا۟ عَلَىٰ حَرۡثِكُمۡ إِن كُنتُمۡ صَٰرِمِینَ ۝٢٢ اَنِ اغْدُوْا عَلٰی حَرْثِكُمْ اِنْ كُنْتُمْ صٰرِمِیْنَ ۟ فَٱنطَلَقُواْ وَهُمۡ يَتَخَٰفَتُونَ ۝٢٣ فَٱنطَلَقُوا۟ وَهُمۡ یَتَخَٰفَتُونَ ۝٢٣ فَانْطَلَقُوْا وَهُمْ یَتَخَافَتُوْنَ ۟ۙ أَن لَّا يَدۡخُلَنَّهَا ٱلۡيَوۡمَ عَلَيۡكُم مِّسۡكِينٞ ۝٢٤ أَن لَّا یَدۡخُلَنَّهَا ٱلۡیَوۡمَ عَلَیۡكُم مِّسۡكِینࣱ ۝٢٤ اَنْ لَّا یَدْخُلَنَّهَا الْیَوْمَ عَلَیْكُمْ مِّسْكِیْنٌ ۟ۙ وَغَدَوۡاْ عَلَىٰ حَرۡدٖ قَٰدِرِينَ ۝٢٥ وَغَدَوۡا۟ عَلَىٰ حَرۡدࣲ قَٰدِرِینَ ۝٢٥ وَّغَدَوْا عَلٰی حَرْدٍ قٰدِرِیْنَ ۟ فَلَمَّا رَأَوۡهَا قَالُوٓاْ إِنَّا لَضَآلُّونَ ۝٢٦ فَلَمَّا رَأَوۡهَا قَالُوۤا۟ إِنَّا لَضَاۤلُّونَ ۝٢٦ فَلَمَّا رَاَوْهَا قَالُوْۤا اِنَّا لَضَآلُّوْنَ ۟ۙ بَلۡ نَحۡنُ مَحۡرُومُونَ ۝٢٧ بَلۡ نَحۡنُ مَحۡرُومُونَ ۝٢٧ بَلْ نَحْنُ مَحْرُوْمُوْنَ ۟ قَالَ أَوۡسَطُهُمۡ أَلَمۡ أَقُل لَّكُمۡ لَوۡلَا تُسَبِّحُونَ ۝٢٨ قَالَ أَوۡسَطُهُمۡ أَلَمۡ أَقُل لَّكُمۡ لَوۡلَا تُسَبِّحُونَ ۝٢٨ قَالَ اَوْسَطُهُمْ اَلَمْ اَقُلْ لَّكُمْ لَوْلَا تُسَبِّحُوْنَ ۟ قَالُواْ سُبۡحَٰنَ رَبِّنَآ إِنَّا كُنَّا ظَٰلِمِينَ ۝٢٩ قَالُوا۟ سُبۡحَٰنَ رَبِّنَاۤ إِنَّا كُنَّا ظَٰلِمِینَ ۝٢٩ قَالُوْا سُبْحٰنَ رَبِّنَاۤ اِنَّا كُنَّا ظٰلِمِیْنَ ۟ فَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضٖ يَتَلَٰوَمُونَ ۝٣٠ فَأَقۡبَلَ بَعۡضُهُمۡ عَلَىٰ بَعۡضࣲ یَتَلَٰوَمُونَ ۝٣٠ فَاَقْبَلَ بَعْضُهُمْ عَلٰی بَعْضٍ یَّتَلَاوَمُوْنَ ۟ قَالُواْ يَٰوَيۡلَنَآ إِنَّا كُنَّا طَٰغِينَ ۝٣١ قَالُوا۟ یَٰوَیۡلَنَاۤ إِنَّا كُنَّا طَٰغِینَ ۝٣١ قَالُوْا یٰوَیْلَنَاۤ اِنَّا كُنَّا طٰغِیْنَ ۟ عَسَىٰ رَبُّنَآ أَن يُبۡدِلَنَا خَيۡرٗا مِّنۡهَآ إِنَّآ إِلَىٰ رَبِّنَا رَٰغِبُونَ ۝٣٢ عَسَىٰ رَبُّنَاۤ أَن یُبۡدِلَنَا خَیۡرࣰا مِّنۡهَاۤ إِنَّاۤ إِلَىٰ رَبِّنَا رَٰغِبُونَ ۝٣٢ عَسٰی رَبُّنَاۤ اَنْ یُّبْدِلَنَا خَیْرًا مِّنْهَاۤ اِنَّاۤ اِلٰی رَبِّنَا رٰغِبُوْنَ ۟ كَذَٰلِكَ ٱلۡعَذَابُۖ وَلَعَذَابُ ٱلۡأٓخِرَةِ أَكۡبَرُۚ لَوۡ كَانُواْ يَعۡلَمُونَ ۝٣٣ كَذَٰلِكَ ٱلۡعَذَابُۖ وَلَعَذَابُ ٱلۡءَاخِرَةِ أَكۡبَرُۚ لَوۡ كَانُوا۟ یَعۡلَمُونَ ۝٣٣ كَذٰلِكَ الْعَذَابُ ؕ وَلَعَذَابُ الْاٰخِرَةِ اَكْبَرُ ۘ لَوْ كَانُوْا یَعْلَمُوْنَ ۟۠ إِنَّ لِلۡمُتَّقِينَ عِندَ رَبِّهِمۡ جَنَّٰتِ ٱلنَّعِيمِ ۝٣٤ إِنَّ لِلۡمُتَّقِینَ عِندَ رَبِّهِمۡ جَنَّٰتِ ٱلنَّعِیمِ ۝٣٤ اِنَّ لِلْمُتَّقِیْنَ عِنْدَ رَبِّهِمْ جَنّٰتِ النَّعِیْمِ ۟ أَفَنَجۡعَلُ ٱلۡمُسۡلِمِينَ كَٱلۡمُجۡرِمِينَ ۝٣٥ أَفَنَجۡعَلُ ٱلۡمُسۡلِمِینَ كَٱلۡمُجۡرِمِینَ ۝٣٥ اَفَنَجْعَلُ الْمُسْلِمِیْنَ كَالْمُجْرِمِیْنَ ۟ؕ مَا لَكُمۡ كَيۡفَ تَحۡكُمُونَ ۝٣٦ مَا لَكُمۡ كَیۡفَ تَحۡكُمُونَ ۝٣٦ مَا لَكُمْ ۥ كَیْفَ تَحْكُمُوْنَ ۟ۚ أَمۡ لَكُمۡ كِتَٰبٞ فِيهِ تَدۡرُسُونَ ۝٣٧ أَمۡ لَكُمۡ كِتَٰبࣱ فِیهِ تَدۡرُسُونَ ۝٣٧ اَمْ لَكُمْ كِتٰبٌ فِیْهِ تَدْرُسُوْنَ ۟ۙ إِنَّ لَكُمۡ فِيهِ لَمَا تَخَيَّرُونَ ۝٣٨ إِنَّ لَكُمۡ فِیهِ لَمَا تَخَیَّرُونَ ۝٣٨ اِنَّ لَكُمْ فِیْهِ لَمَا تَخَیَّرُوْنَ ۟ۚ أَمۡ لَكُمۡ أَيۡمَٰنٌ عَلَيۡنَا بَٰلِغَةٌ إِلَىٰ يَوۡمِ ٱلۡقِيَٰمَةِ إِنَّ لَكُمۡ لَمَا تَحۡكُمُونَ ۝٣٩ أَمۡ لَكُمۡ أَیۡمَٰنٌ عَلَیۡنَا بَٰلِغَةٌ إِلَىٰ یَوۡمِ ٱلۡقِیَٰمَةِ إِنَّ لَكُمۡ لَمَا تَحۡكُمُونَ ۝٣٩ اَمْ لَكُمْ اَیْمَانٌ عَلَیْنَا بَالِغَةٌ اِلٰی یَوْمِ الْقِیٰمَةِ ۙ اِنَّ لَكُمْ لَمَا تَحْكُمُوْنَ ۟ۚ سَلۡهُمۡ أَيُّهُم بِذَٰلِكَ زَعِيمٌ ۝٤٠ سَلۡهُمۡ أَیُّهُم بِذَٰلِكَ زَعِیمٌ ۝٤٠ سَلْهُمْ اَیُّهُمْ بِذٰلِكَ زَعِیْمٌ ۟ۚۛ أَمۡ لَهُمۡ شُرَكَآءُ فَلۡيَأۡتُواْ بِشُرَكَآئِهِمۡ إِن كَانُواْ صَٰدِقِينَ ۝٤١ أَمۡ لَهُمۡ شُرَكَاۤءُ فَلۡیَأۡتُوا۟ بِشُرَكَاۤئِهِمۡ إِن كَانُوا۟ صَٰدِقِینَ ۝٤١ اَمْ لَهُمْ شُرَكَآءُ ۛۚ فَلْیَاْتُوْا بِشُرَكَآىِٕهِمْ اِنْ كَانُوْا صٰدِقِیْنَ ۟ يَوۡمَ يُكۡشَفُ عَن سَاقٖ وَيُدۡعَوۡنَ إِلَى ٱلسُّجُودِ فَلَا يَسۡتَطِيعُونَ ۝٤٢ یَوۡمَ یُكۡشَفُ عَن سَاقࣲ وَیُدۡعَوۡنَ إِلَى ٱلسُّجُودِ فَلَا یَسۡتَطِیعُونَ ۝٤٢ یَوْمَ یُكْشَفُ عَنْ سَاقٍ وَّیُدْعَوْنَ اِلَی السُّجُوْدِ فَلَا یَسْتَطِیْعُوْنَ ۟ۙ خَٰشِعَةً أَبۡصَٰرُهُمۡ تَرۡهَقُهُمۡ ذِلَّةٞۖ وَقَدۡ كَانُواْ يُدۡعَوۡنَ إِلَى ٱلسُّجُودِ وَهُمۡ سَٰلِمُونَ ۝٤٣ خَٰشِعَةً أَبۡصَٰرُهُمۡ تَرۡهَقُهُمۡ ذِلَّةࣱۖ وَقَدۡ كَانُوا۟ یُدۡعَوۡنَ إِلَى ٱلسُّجُودِ وَهُمۡ سَٰلِمُونَ ۝٤٣ خَاشِعَةً اَبْصَارُهُمْ تَرْهَقُهُمْ ذِلَّةٌ ؕ وَقَدْ كَانُوْا یُدْعَوْنَ اِلَی السُّجُوْدِ وَهُمْ سٰلِمُوْنَ ۟ فَذَرۡنِي وَمَن يُكَذِّبُ بِهَٰذَا ٱلۡحَدِيثِۖ سَنَسۡتَدۡرِجُهُم مِّنۡ حَيۡثُ لَا يَعۡلَمُونَ ۝٤٤ فَذَرۡنِی وَمَن یُكَذِّبُ بِهَٰذَا ٱلۡحَدِیثِۖ سَنَسۡتَدۡرِجُهُم مِّنۡ حَیۡثُ لَا یَعۡلَمُونَ ۝٤٤ فَذَرْنِیْ وَمَنْ یُّكَذِّبُ بِهٰذَا الْحَدِیْثِ ؕ سَنَسْتَدْرِجُهُمْ مِّنْ حَیْثُ لَا یَعْلَمُوْنَ ۟ۙ وَأُمۡلِي لَهُمۡۚ إِنَّ كَيۡدِي مَتِينٌ ۝٤٥ وَأُمۡلِی لَهُمۡۚ إِنَّ كَیۡدِی مَتِینٌ ۝٤٥ وَاُمْلِیْ لَهُمْ ؕ اِنَّ كَیْدِیْ مَتِیْنٌ ۟ أَمۡ تَسۡـَٔلُهُمۡ أَجۡرٗا فَهُم مِّن مَّغۡرَمٖ مُّثۡقَلُونَ ۝٤٦ أَمۡ تَسۡءَلُهُمۡ أَجۡرࣰا فَهُم مِّن مَّغۡرَمࣲ مُّثۡقَلُونَ ۝٤٦ اَمْ تَسْـَٔلُهُمْ اَجْرًا فَهُمْ مِّنْ مَّغْرَمٍ مُّثْقَلُوْنَ ۟ۚ أَمۡ عِندَهُمُ ٱلۡغَيۡبُ فَهُمۡ يَكۡتُبُونَ ۝٤٧ أَمۡ عِندَهُمُ ٱلۡغَیۡبُ فَهُمۡ یَكۡتُبُونَ ۝٤٧ اَمْ عِنْدَهُمُ الْغَیْبُ فَهُمْ یَكْتُبُوْنَ ۟ فَٱصۡبِرۡ لِحُكۡمِ رَبِّكَ وَلَا تَكُن كَصَاحِبِ ٱلۡحُوتِ إِذۡ نَادَىٰ وَهُوَ مَكۡظُومٞ ۝٤٨ فَٱصۡبِرۡ لِحُكۡمِ رَبِّكَ وَلَا تَكُن كَصَاحِبِ ٱلۡحُوتِ إِذۡ نَادَىٰ وَهُوَ مَكۡظُومࣱ ۝٤٨ فَاصْبِرْ لِحُكْمِ رَبِّكَ وَلَا تَكُنْ كَصَاحِبِ الْحُوْتِ ۘ اِذْ نَادٰی وَهُوَ مَكْظُوْمٌ ۟ؕ لَّوۡلَآ أَن تَدَٰرَكَهُۥ نِعۡمَةٞ مِّن رَّبِّهِۦ لَنُبِذَ بِٱلۡعَرَآءِ وَهُوَ مَذۡمُومٞ ۝٤٩ لَّوۡلَاۤ أَن تَدَٰرَكَهُۥ نِعۡمَةࣱ مِّن رَّبِّهِۦ لَنُبِذَ بِٱلۡعَرَاۤءِ وَهُوَ مَذۡمُومࣱ ۝٤٩ لَوْلَاۤ اَنْ تَدٰرَكَهٗ نِعْمَةٌ مِّنْ رَّبِّهٖ لَنُبِذَ بِالْعَرَآءِ وَهُوَ مَذْمُوْمٌ ۟ فَٱجۡتَبَٰهُ رَبُّهُۥ فَجَعَلَهُۥ مِنَ ٱلصَّٰلِحِينَ ۝٥٠ فَٱجۡتَبَٰهُ رَبُّهُۥ فَجَعَلَهُۥ مِنَ ٱلصَّٰلِحِینَ ۝٥٠ فَاجْتَبٰىهُ رَبُّهٗ فَجَعَلَهٗ مِنَ الصّٰلِحِیْنَ ۟ وَإِن يَكَادُ ٱلَّذِينَ كَفَرُواْ لَيُزۡلِقُونَكَ بِأَبۡصَٰرِهِمۡ لَمَّا سَمِعُواْ ٱلذِّكۡرَ وَيَقُولُونَ إِنَّهُۥ لَمَجۡنُونٞ ۝٥١ وَإِن یَكَادُ ٱلَّذِینَ كَفَرُوا۟ لَیُزۡلِقُونَكَ بِأَبۡصَٰرِهِمۡ لَمَّا سَمِعُوا۟ ٱلذِّكۡرَ وَیَقُولُونَ إِنَّهُۥ لَمَجۡنُونࣱ ۝٥١ وَاِنْ یَّكَادُ الَّذِیْنَ كَفَرُوْا لَیُزْلِقُوْنَكَ بِاَبْصَارِهِمْ لَمَّا سَمِعُوا الذِّكْرَ وَیَقُوْلُوْنَ اِنَّهٗ لَمَجْنُوْنٌ ۟ۘ وَمَا هُوَ إِلَّا ذِكۡرٞ لِّلۡعَٰلَمِينَ ۝٥٢ وَمَا هُوَ إِلَّا ذِكۡرࣱ لِّلۡعَٰلَمِینَ ۝٥٢ وَمَا هُوَ اِلَّا ذِكْرٌ لِّلْعٰلَمِیْنَ ۟۠
