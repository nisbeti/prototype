Surah 70 · Meccan · 44 ayat

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

سَأَلَ سَآئِلُۢ بِعَذَابٖ وَاقِعٖ ۝١ سَأَلَ سَاۤئِلُۢ بِعَذَابࣲ وَاقِعࣲ ۝١ سَاَلَ سَآىِٕلٌۢ بِعَذَابٍ وَّاقِعٍ ۟ۙ لِّلۡكَٰفِرِينَ لَيۡسَ لَهُۥ دَافِعٞ ۝٢ لِّلۡكَٰفِرِینَ لَیۡسَ لَهُۥ دَافِعࣱ ۝٢ لِّلْكٰفِرِیْنَ لَیْسَ لَهٗ دَافِعٌ ۟ۙ مِّنَ ٱللَّهِ ذِي ٱلۡمَعَارِجِ ۝٣ مِّنَ ٱللَّهِ ذِی ٱلۡمَعَارِجِ ۝٣ مِّنَ اللّٰهِ ذِی الْمَعَارِجِ ۟ؕ تَعۡرُجُ ٱلۡمَلَٰٓئِكَةُ وَٱلرُّوحُ إِلَيۡهِ فِي يَوۡمٖ كَانَ مِقۡدَارُهُۥ خَمۡسِينَ أَلۡفَ سَنَةٖ ۝٤ تَعۡرُجُ ٱلۡمَلَٰۤئِكَةُ وَٱلرُّوحُ إِلَیۡهِ فِی یَوۡمࣲ كَانَ مِقۡدَارُهُۥ خَمۡسِینَ أَلۡفَ سَنَةࣲ ۝٤ تَعْرُجُ الْمَلٰٓىِٕكَةُ وَالرُّوْحُ اِلَیْهِ فِیْ یَوْمٍ كَانَ مِقْدَارُهٗ خَمْسِیْنَ اَلْفَ سَنَةٍ ۟ۚ فَٱصۡبِرۡ صَبۡرٗا جَمِيلًا ۝٥ فَٱصۡبِرۡ صَبۡرࣰا جَمِیلًا ۝٥ فَاصْبِرْ صَبْرًا جَمِیْلًا ۟ إِنَّهُمۡ يَرَوۡنَهُۥ بَعِيدٗا ۝٦ إِنَّهُمۡ یَرَوۡنَهُۥ بَعِیدࣰا ۝٦ اِنَّهُمْ یَرَوْنَهٗ بَعِیْدًا ۟ۙ وَنَرَىٰهُ قَرِيبٗا ۝٧ وَنَرَىٰهُ قَرِیبࣰا ۝٧ وَّنَرٰىهُ قَرِیْبًا ۟ؕ يَوۡمَ تَكُونُ ٱلسَّمَآءُ كَٱلۡمُهۡلِ ۝٨ یَوۡمَ تَكُونُ ٱلسَّمَاۤءُ كَٱلۡمُهۡلِ ۝٨ یَوْمَ تَكُوْنُ السَّمَآءُ كَالْمُهْلِ ۟ۙ وَتَكُونُ ٱلۡجِبَالُ كَٱلۡعِهۡنِ ۝٩ وَتَكُونُ ٱلۡجِبَالُ كَٱلۡعِهۡنِ ۝٩ وَتَكُوْنُ الْجِبَالُ كَالْعِهْنِ ۟ۙ وَلَا يَسۡـَٔلُ حَمِيمٌ حَمِيمٗا ۝١٠ وَلَا یَسۡءَلُ حَمِیمٌ حَمِیمࣰا ۝١٠ وَلَا یَسْـَٔلُ حَمِیْمٌ حَمِیْمًا ۟ۚۖ يُبَصَّرُونَهُمۡۚ يَوَدُّ ٱلۡمُجۡرِمُ لَوۡ يَفۡتَدِي مِنۡ عَذَابِ يَوۡمِئِذِۭ بِبَنِيهِ ۝١١ یُبَصَّرُونَهُمۡۚ یَوَدُّ ٱلۡمُجۡرِمُ لَوۡ یَفۡتَدِی مِنۡ عَذَابِ یَوۡمِئِذِۭ بِبَنِیهِ ۝١١ یُّبَصَّرُوْنَهُمْ ؕ یَوَدُّ الْمُجْرِمُ لَوْ یَفْتَدِیْ مِنْ عَذَابِ یَوْمِىِٕذٍ بِبَنِیْهِ ۟ۙ وَصَٰحِبَتِهِۦ وَأَخِيهِ ۝١٢ وَصَٰحِبَتِهِۦ وَأَخِیهِ ۝١٢ وَصَاحِبَتِهٖ وَاَخِیْهِ ۟ۙ وَفَصِيلَتِهِ ٱلَّتِي تُـٔۡوِيهِ ۝١٣ وَفَصِیلَتِهِ ٱلَّتِی تُءۡوِیهِ ۝١٣ وَفَصِیْلَتِهِ الَّتِیْ تُـْٔوِیْهِ ۟ۙ وَمَن فِي ٱلۡأَرۡضِ جَمِيعٗا ثُمَّ يُنجِيهِ ۝١٤ وَمَن فِی ٱلۡأَرۡضِ جَمِیعࣰا ثُمَّ یُنجِیهِ ۝١٤ وَمَنْ فِی الْاَرْضِ جَمِیْعًا ۙ ثُمَّ یُنْجِیْهِ ۟ۙ كَلَّآۖ إِنَّهَا لَظَىٰ ۝١٥ كَلَّاۤۖ إِنَّهَا لَظَىٰ ۝١٥ كَلَّا ؕ اِنَّهَا لَظٰی ۟ۙ نَزَّاعَةٗ لِّلشَّوَىٰ ۝١٦ نَزَّاعَةࣰ لِّلشَّوَىٰ ۝١٦ نَزَّاعَةً لِّلشَّوٰی ۟ۚۖ تَدۡعُواْ مَنۡ أَدۡبَرَ وَتَوَلَّىٰ ۝١٧ تَدۡعُوا۟ مَنۡ أَدۡبَرَ وَتَوَلَّىٰ ۝١٧ تَدْعُوْا مَنْ اَدْبَرَ وَتَوَلّٰی ۟ۙ وَجَمَعَ فَأَوۡعَىٰٓ ۝١٨ وَجَمَعَ فَأَوۡعَىٰۤ ۝١٨ وَجَمَعَ فَاَوْعٰی ۟ إِنَّ ٱلۡإِنسَٰنَ خُلِقَ هَلُوعًا ۝١٩ إِنَّ ٱلۡإِنسَٰنَ خُلِقَ هَلُوعًا ۝١٩ اِنَّ الْاِنْسَانَ خُلِقَ هَلُوْعًا ۟ۙ إِذَا مَسَّهُ ٱلشَّرُّ جَزُوعٗا ۝٢٠ إِذَا مَسَّهُ ٱلشَّرُّ جَزُوعࣰا ۝٢٠ اِذَا مَسَّهُ الشَّرُّ جَزُوْعًا ۟ۙ وَإِذَا مَسَّهُ ٱلۡخَيۡرُ مَنُوعًا ۝٢١ وَإِذَا مَسَّهُ ٱلۡخَیۡرُ مَنُوعًا ۝٢١ وَّاِذَا مَسَّهُ الْخَیْرُ مَنُوْعًا ۟ۙ إِلَّا ٱلۡمُصَلِّينَ ۝٢٢ إِلَّا ٱلۡمُصَلِّینَ ۝٢٢ اِلَّا الْمُصَلِّیْنَ ۟ۙ ٱلَّذِينَ هُمۡ عَلَىٰ صَلَاتِهِمۡ دَآئِمُونَ ۝٢٣ ٱلَّذِینَ هُمۡ عَلَىٰ صَلَاتِهِمۡ دَاۤئِمُونَ ۝٢٣ الَّذِیْنَ هُمْ عَلٰی صَلَاتِهِمْ دَآىِٕمُوْنَ ۟ وَٱلَّذِينَ فِيٓ أَمۡوَٰلِهِمۡ حَقّٞ مَّعۡلُومٞ ۝٢٤ وَٱلَّذِینَ فِیۤ أَمۡوَٰلِهِمۡ حَقࣱّ مَّعۡلُومࣱ ۝٢٤ وَالَّذِیْنَ فِیْۤ اَمْوَالِهِمْ حَقٌّ مَّعْلُوْمٌ ۟ لِّلسَّآئِلِ وَٱلۡمَحۡرُومِ ۝٢٥ لِّلسَّاۤئِلِ وَٱلۡمَحۡرُومِ ۝٢٥ لِّلسَّآىِٕلِ وَالْمَحْرُوْمِ ۟ وَٱلَّذِينَ يُصَدِّقُونَ بِيَوۡمِ ٱلدِّينِ ۝٢٦ وَٱلَّذِینَ یُصَدِّقُونَ بِیَوۡمِ ٱلدِّینِ ۝٢٦ وَالَّذِیْنَ یُصَدِّقُوْنَ بِیَوْمِ الدِّیْنِ ۟ وَٱلَّذِينَ هُم مِّنۡ عَذَابِ رَبِّهِم مُّشۡفِقُونَ ۝٢٧ وَٱلَّذِینَ هُم مِّنۡ عَذَابِ رَبِّهِم مُّشۡفِقُونَ ۝٢٧ وَالَّذِیْنَ هُمْ مِّنْ عَذَابِ رَبِّهِمْ مُّشْفِقُوْنَ ۟ۚ إِنَّ عَذَابَ رَبِّهِمۡ غَيۡرُ مَأۡمُونٖ ۝٢٨ إِنَّ عَذَابَ رَبِّهِمۡ غَیۡرُ مَأۡمُونࣲ ۝٢٨ اِنَّ عَذَابَ رَبِّهِمْ غَیْرُ مَاْمُوْنٍ ۟ وَٱلَّذِينَ هُمۡ لِفُرُوجِهِمۡ حَٰفِظُونَ ۝٢٩ وَٱلَّذِینَ هُمۡ لِفُرُوجِهِمۡ حَٰفِظُونَ ۝٢٩ وَالَّذِیْنَ هُمْ لِفُرُوْجِهِمْ حٰفِظُوْنَ ۟ۙ إِلَّا عَلَىٰٓ أَزۡوَٰجِهِمۡ أَوۡ مَا مَلَكَتۡ أَيۡمَٰنُهُمۡ فَإِنَّهُمۡ غَيۡرُ مَلُومِينَ ۝٣٠ إِلَّا عَلَىٰۤ أَزۡوَٰجِهِمۡ أَوۡ مَا مَلَكَتۡ أَیۡمَٰنُهُمۡ فَإِنَّهُمۡ غَیۡرُ مَلُومِینَ ۝٣٠ اِلَّا عَلٰۤی اَزْوَاجِهِمْ اَوْ مَا مَلَكَتْ اَیْمَانُهُمْ فَاِنَّهُمْ غَیْرُ مَلُوْمِیْنَ ۟ۚ فَمَنِ ٱبۡتَغَىٰ وَرَآءَ ذَٰلِكَ فَأُوْلَٰٓئِكَ هُمُ ٱلۡعَادُونَ ۝٣١ فَمَنِ ٱبۡتَغَىٰ وَرَاۤءَ ذَٰلِكَ فَأُو۟لَٰۤئِكَ هُمُ ٱلۡعَادُونَ ۝٣١ فَمَنِ ابْتَغٰی وَرَآءَ ذٰلِكَ فَاُولٰٓىِٕكَ هُمُ الْعٰدُوْنَ ۟ۚ وَٱلَّذِينَ هُمۡ لِأَمَٰنَٰتِهِمۡ وَعَهۡدِهِمۡ رَٰعُونَ ۝٣٢ وَٱلَّذِینَ هُمۡ لِأَمَٰنَٰتِهِمۡ وَعَهۡدِهِمۡ رَٰعُونَ ۝٣٢ وَالَّذِیْنَ هُمْ لِاَمٰنٰتِهِمْ وَعَهْدِهِمْ رٰعُوْنَ ۟ وَٱلَّذِينَ هُم بِشَهَٰدَٰتِهِمۡ قَآئِمُونَ ۝٣٣ وَٱلَّذِینَ هُم بِشَهَٰدَٰتِهِمۡ قَاۤئِمُونَ ۝٣٣ وَالَّذِیْنَ هُمْ بِشَهٰدٰتِهِمْ قَآىِٕمُوْنَ ۟ وَٱلَّذِينَ هُمۡ عَلَىٰ صَلَاتِهِمۡ يُحَافِظُونَ ۝٣٤ وَٱلَّذِینَ هُمۡ عَلَىٰ صَلَاتِهِمۡ یُحَافِظُونَ ۝٣٤ وَالَّذِیْنَ هُمْ عَلٰی صَلَاتِهِمْ یُحَافِظُوْنَ ۟ؕ أُوْلَٰٓئِكَ فِي جَنَّٰتٖ مُّكۡرَمُونَ ۝٣٥ أُو۟لَٰۤئِكَ فِی جَنَّٰتࣲ مُّكۡرَمُونَ ۝٣٥ اُولٰٓىِٕكَ فِیْ جَنّٰتٍ مُّكْرَمُوْنَ ۟ؕ۠ فَمَالِ ٱلَّذِينَ كَفَرُواْ قِبَلَكَ مُهۡطِعِينَ ۝٣٦ فَمَالِ ٱلَّذِینَ كَفَرُوا۟ قِبَلَكَ مُهۡطِعِینَ ۝٣٦ فَمَالِ الَّذِیْنَ كَفَرُوْا قِبَلَكَ مُهْطِعِیْنَ ۟ۙ عَنِ ٱلۡيَمِينِ وَعَنِ ٱلشِّمَالِ عِزِينَ ۝٣٧ عَنِ ٱلۡیَمِینِ وَعَنِ ٱلشِّمَالِ عِزِینَ ۝٣٧ عَنِ الْیَمِیْنِ وَعَنِ الشِّمَالِ عِزِیْنَ ۟ أَيَطۡمَعُ كُلُّ ٱمۡرِيٕٖ مِّنۡهُمۡ أَن يُدۡخَلَ جَنَّةَ نَعِيمٖ ۝٣٨ أَیَطۡمَعُ كُلُّ ٱمۡرِئࣲ مِّنۡهُمۡ أَن یُدۡخَلَ جَنَّةَ نَعِیمࣲ ۝٣٨ اَیَطْمَعُ كُلُّ امْرِئٍ مِّنْهُمْ اَنْ یُّدْخَلَ جَنَّةَ نَعِیْمٍ ۟ۙ كَلَّآۖ إِنَّا خَلَقۡنَٰهُم مِّمَّا يَعۡلَمُونَ ۝٣٩ كَلَّاۤۖ إِنَّا خَلَقۡنَٰهُم مِّمَّا یَعۡلَمُونَ ۝٣٩ كَلَّا ؕ اِنَّا خَلَقْنٰهُمْ مِّمَّا یَعْلَمُوْنَ ۟ فَلَآ أُقۡسِمُ بِرَبِّ ٱلۡمَشَٰرِقِ وَٱلۡمَغَٰرِبِ إِنَّا لَقَٰدِرُونَ ۝٤٠ فَلَاۤ أُقۡسِمُ بِرَبِّ ٱلۡمَشَٰرِقِ وَٱلۡمَغَٰرِبِ إِنَّا لَقَٰدِرُونَ ۝٤٠ فَلَاۤ اُقْسِمُ بِرَبِّ الْمَشٰرِقِ وَالْمَغٰرِبِ اِنَّا لَقٰدِرُوْنَ ۟ۙ عَلَىٰٓ أَن نُّبَدِّلَ خَيۡرٗا مِّنۡهُمۡ وَمَا نَحۡنُ بِمَسۡبُوقِينَ ۝٤١ عَلَىٰۤ أَن نُّبَدِّلَ خَیۡرࣰا مِّنۡهُمۡ وَمَا نَحۡنُ بِمَسۡبُوقِینَ ۝٤١ عَلٰۤی اَنْ نُّبَدِّلَ خَیْرًا مِّنْهُمْ ۙ وَمَا نَحْنُ بِمَسْبُوْقِیْنَ ۟ فَذَرۡهُمۡ يَخُوضُواْ وَيَلۡعَبُواْ حَتَّىٰ يُلَٰقُواْ يَوۡمَهُمُ ٱلَّذِي يُوعَدُونَ ۝٤٢ فَذَرۡهُمۡ یَخُوضُوا۟ وَیَلۡعَبُوا۟ حَتَّىٰ یُلَٰقُوا۟ یَوۡمَهُمُ ٱلَّذِی یُوعَدُونَ ۝٤٢ فَذَرْهُمْ یَخُوْضُوْا وَیَلْعَبُوْا حَتّٰی یُلٰقُوْا یَوْمَهُمُ الَّذِیْ یُوْعَدُوْنَ ۟ۙ يَوۡمَ يَخۡرُجُونَ مِنَ ٱلۡأَجۡدَاثِ سِرَاعٗا كَأَنَّهُمۡ إِلَىٰ نُصُبٖ يُوفِضُونَ ۝٤٣ یَوۡمَ یَخۡرُجُونَ مِنَ ٱلۡأَجۡدَاثِ سِرَاعࣰا كَأَنَّهُمۡ إِلَىٰ نُصُبࣲ یُوفِضُونَ ۝٤٣ یَوْمَ یَخْرُجُوْنَ مِنَ الْاَجْدَاثِ سِرَاعًا كَاَنَّهُمْ اِلٰی نُصُبٍ یُّوْفِضُوْنَ ۟ۙ خَٰشِعَةً أَبۡصَٰرُهُمۡ تَرۡهَقُهُمۡ ذِلَّةٞۚ ذَٰلِكَ ٱلۡيَوۡمُ ٱلَّذِي كَانُواْ يُوعَدُونَ ۝٤٤ خَٰشِعَةً أَبۡصَٰرُهُمۡ تَرۡهَقُهُمۡ ذِلَّةࣱۚ ذَٰلِكَ ٱلۡیَوۡمُ ٱلَّذِی كَانُوا۟ یُوعَدُونَ ۝٤٤ خَاشِعَةً اَبْصَارُهُمْ تَرْهَقُهُمْ ذِلَّةٌ ؕ ذٰلِكَ الْیَوْمُ الَّذِیْ كَانُوْا یُوْعَدُوْنَ ۟۠
