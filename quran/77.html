بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

وَٱلۡمُرۡسَلَٰتِ عُرۡفٗا ۝١وَٱلۡمُرۡسَلَٰتِ عُرۡفࣰا ۝١وَالْمُرْسَلٰتِ عُرْفًا ۟ۙ

فَٱلۡعَٰصِفَٰتِ عَصۡفٗا ۝٢فَٱلۡعَٰصِفَٰتِ عَصۡفࣰا ۝٢فَالْعٰصِفٰتِ عَصْفًا ۟ۙ

وَٱلنَّٰشِرَٰتِ نَشۡرٗا ۝٣وَٱلنَّٰشِرَٰتِ نَشۡرࣰا ۝٣وَّالنّٰشِرٰتِ نَشْرًا ۟ۙ

فَٱلۡفَٰرِقَٰتِ فَرۡقٗا ۝٤فَٱلۡفَٰرِقَٰتِ فَرۡقࣰا ۝٤فَالْفٰرِقٰتِ فَرْقًا ۟ۙ

فَٱلۡمُلۡقِيَٰتِ ذِكۡرًا ۝٥فَٱلۡمُلۡقِیَٰتِ ذِكۡرًا ۝٥فَالْمُلْقِیٰتِ ذِكْرًا ۟ۙ

عُذۡرًا أَوۡ نُذۡرًا ۝٦عُذۡرًا أَوۡ نُذۡرًا ۝٦عُذْرًا اَوْ نُذْرًا ۟ۙ

إِنَّمَا تُوعَدُونَ لَوَٰقِعٞ ۝٧إِنَّمَا تُوعَدُونَ لَوَٰقِعࣱ ۝٧اِنَّمَا تُوْعَدُوْنَ لَوَاقِعٌ ۟ؕ

فَإِذَا ٱلنُّجُومُ طُمِسَتۡ ۝٨فَإِذَا ٱلنُّجُومُ طُمِسَتۡ ۝٨فَاِذَا النُّجُوْمُ طُمِسَتْ ۟ۙ

وَإِذَا ٱلسَّمَآءُ فُرِجَتۡ ۝٩وَإِذَا ٱلسَّمَاۤءُ فُرِجَتۡ ۝٩وَاِذَا السَّمَآءُ فُرِجَتْ ۟ۙ

وَإِذَا ٱلۡجِبَالُ نُسِفَتۡ ۝١٠وَإِذَا ٱلۡجِبَالُ نُسِفَتۡ ۝١٠وَاِذَا الْجِبَالُ نُسِفَتْ ۟ۙ

وَإِذَا ٱلرُّسُلُ أُقِّتَتۡ ۝١١وَإِذَا ٱلرُّسُلُ أُقِّتَتۡ ۝١١وَاِذَا الرُّسُلُ اُقِّتَتْ ۟ؕ

لِأَيِّ يَوۡمٍ أُجِّلَتۡ ۝١٢لِأَیِّ یَوۡمٍ أُجِّلَتۡ ۝١٢لِاَیِّ یَوْمٍ اُجِّلَتْ ۟ؕ

لِيَوۡمِ ٱلۡفَصۡلِ ۝١٣لِیَوۡمِ ٱلۡفَصۡلِ ۝١٣لِیَوْمِ الْفَصْلِ ۟ۚ

وَمَآ أَدۡرَىٰكَ مَا يَوۡمُ ٱلۡفَصۡلِ ۝١٤وَمَاۤ أَدۡرَىٰكَ مَا یَوۡمُ ٱلۡفَصۡلِ ۝١٤وَمَاۤ اَدْرٰىكَ مَا یَوْمُ الْفَصْلِ ۟ؕ

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝١٥وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝١٥وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

أَلَمۡ نُهۡلِكِ ٱلۡأَوَّلِينَ ۝١٦أَلَمۡ نُهۡلِكِ ٱلۡأَوَّلِینَ ۝١٦اَلَمْ نُهْلِكِ الْاَوَّلِیْنَ ۟ؕ

ثُمَّ نُتۡبِعُهُمُ ٱلۡأٓخِرِينَ ۝١٧ثُمَّ نُتۡبِعُهُمُ ٱلۡءَاخِرِینَ ۝١٧ثُمَّ نُتْبِعُهُمُ الْاٰخِرِیْنَ ۟

كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِينَ ۝١٨كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِینَ ۝١٨كَذٰلِكَ نَفْعَلُ بِالْمُجْرِمِیْنَ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝١٩وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝١٩وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

أَلَمۡ نَخۡلُقكُّم مِّن مَّآءٖ مَّهِينٖ ۝٢٠أَلَمۡ نَخۡلُقكُّم مِّن مَّاۤءࣲ مَّهِینࣲ ۝٢٠اَلَمْ نَخْلُقْكُّمْ مِّنْ مَّآءٍ مَّهِیْنٍ ۟ۙ

فَجَعَلۡنَٰهُ فِي قَرَارٖ مَّكِينٍ ۝٢١فَجَعَلۡنَٰهُ فِی قَرَارࣲ مَّكِینٍ ۝٢١فَجَعَلْنٰهُ فِیْ قَرَارٍ مَّكِیْنٍ ۟ۙ

إِلَىٰ قَدَرٖ مَّعۡلُومٖ ۝٢٢إِلَىٰ قَدَرࣲ مَّعۡلُومࣲ ۝٢٢اِلٰی قَدَرٍ مَّعْلُوْمٍ ۟ۙ

فَقَدَرۡنَا فَنِعۡمَ ٱلۡقَٰدِرُونَ ۝٢٣فَقَدَرۡنَا فَنِعۡمَ ٱلۡقَٰدِرُونَ ۝٢٣فَقَدَرْنَا ۖۗ فَنِعْمَ الْقٰدِرُوْنَ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٢٤وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٢٤وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

أَلَمۡ نَجۡعَلِ ٱلۡأَرۡضَ كِفَاتًا ۝٢٥أَلَمۡ نَجۡعَلِ ٱلۡأَرۡضَ كِفَاتًا ۝٢٥اَلَمْ نَجْعَلِ الْاَرْضَ كِفَاتًا ۟ۙ

أَحۡيَآءٗ وَأَمۡوَٰتٗا ۝٢٦أَحۡیَاۤءࣰ وَأَمۡوَٰتࣰا ۝٢٦اَحْیَآءً وَّاَمْوَاتًا ۟ۙ

وَجَعَلۡنَا فِيهَا رَوَٰسِيَ شَٰمِخَٰتٖ وَأَسۡقَيۡنَٰكُم مَّآءٗ فُرَاتٗا ۝٢٧وَجَعَلۡنَا فِیهَا رَوَٰسِیَ شَٰمِخَٰتࣲ وَأَسۡقَیۡنَٰكُم مَّاۤءࣰ فُرَاتࣰا ۝٢٧وَّجَعَلْنَا فِیْهَا رَوَاسِیَ شٰمِخٰتٍ وَّاَسْقَیْنٰكُمْ مَّآءً فُرَاتًا ۟ؕ

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٢٨وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٢٨وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

ٱنطَلِقُوٓاْ إِلَىٰ مَا كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢٩ٱنطَلِقُوۤا۟ إِلَىٰ مَا كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢٩اِنْطَلِقُوْۤا اِلٰی مَا كُنْتُمْ بِهٖ تُكَذِّبُوْنَ ۟ۚ

ٱنطَلِقُوٓاْ إِلَىٰ ظِلّٖ ذِي ثَلَٰثِ شُعَبٖ ۝٣٠ٱنطَلِقُوۤا۟ إِلَىٰ ظِلࣲّ ذِی ثَلَٰثِ شُعَبࣲ ۝٣٠اِنْطَلِقُوْۤا اِلٰی ظِلٍّ ذِیْ ثَلٰثِ شُعَبٍ ۟ۙ

لَّا ظَلِيلٖ وَلَا يُغۡنِي مِنَ ٱللَّهَبِ ۝٣١لَّا ظَلِیلࣲ وَلَا یُغۡنِی مِنَ ٱللَّهَبِ ۝٣١لَّا ظَلِیْلٍ وَّلَا یُغْنِیْ مِنَ اللَّهَبِ ۟ؕ

إِنَّهَا تَرۡمِي بِشَرَرٖ كَٱلۡقَصۡرِ ۝٣٢إِنَّهَا تَرۡمِی بِشَرَرࣲ كَٱلۡقَصۡرِ ۝٣٢اِنَّهَا تَرْمِیْ بِشَرَرٍ كَالْقَصْرِ ۟ۚ

كَأَنَّهُۥ جِمَٰلَتٞ صُفۡرٞ ۝٣٣كَأَنَّهُۥ جِمَٰلَتࣱ صُفۡرࣱ ۝٣٣كَاَنَّهٗ جِمٰلَتٌ صُفْرٌ ۟ؕ

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٣٤وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٣٤وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

هَٰذَا يَوۡمُ لَا يَنطِقُونَ ۝٣٥هَٰذَا یَوۡمُ لَا یَنطِقُونَ ۝٣٥هٰذَا یَوْمُ لَا یَنْطِقُوْنَ ۟ۙ

وَلَا يُؤۡذَنُ لَهُمۡ فَيَعۡتَذِرُونَ ۝٣٦وَلَا یُؤۡذَنُ لَهُمۡ فَیَعۡتَذِرُونَ ۝٣٦وَلَا یُؤْذَنُ لَهُمْ فَیَعْتَذِرُوْنَ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٣٧وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٣٧وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

هَٰذَا يَوۡمُ ٱلۡفَصۡلِۖ جَمَعۡنَٰكُمۡ وَٱلۡأَوَّلِينَ ۝٣٨هَٰذَا یَوۡمُ ٱلۡفَصۡلِۖ جَمَعۡنَٰكُمۡ وَٱلۡأَوَّلِینَ ۝٣٨هٰذَا یَوْمُ الْفَصْلِ ۚ جَمَعْنٰكُمْ وَالْاَوَّلِیْنَ ۟

فَإِن كَانَ لَكُمۡ كَيۡدٞ فَكِيدُونِ ۝٣٩فَإِن كَانَ لَكُمۡ كَیۡدࣱ فَكِیدُونِ ۝٣٩فَاِنْ كَانَ لَكُمْ كَیْدٌ فَكِیْدُوْنِ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٠وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٠وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟۠

إِنَّ ٱلۡمُتَّقِينَ فِي ظِلَٰلٖ وَعُيُونٖ ۝٤١إِنَّ ٱلۡمُتَّقِینَ فِی ظِلَٰلࣲ وَعُیُونࣲ ۝٤١اِنَّ الْمُتَّقِیْنَ فِیْ ظِلٰلٍ وَّعُیُوْنٍ ۟ۙ

وَفَوَٰكِهَ مِمَّا يَشۡتَهُونَ ۝٤٢وَفَوَٰكِهَ مِمَّا یَشۡتَهُونَ ۝٤٢وَّفَوَاكِهَ مِمَّا یَشْتَهُوْنَ ۟ؕ

كُلُواْ وَٱشۡرَبُواْ هَنِيٓـَٔۢا بِمَا كُنتُمۡ تَعۡمَلُونَ ۝٤٣كُلُوا۟ وَٱشۡرَبُوا۟ هَنِیۤءَۢا بِمَا كُنتُمۡ تَعۡمَلُونَ ۝٤٣كُلُوْا وَاشْرَبُوْا هَنِیْٓـًٔا بِمَا كُنْتُمْ تَعْمَلُوْنَ ۟

إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ ۝٤٤إِنَّا كَذَٰلِكَ نَجۡزِی ٱلۡمُحۡسِنِینَ ۝٤٤اِنَّا كَذٰلِكَ نَجْزِی الْمُحْسِنِیْنَ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٥وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٥وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

كُلُواْ وَتَمَتَّعُواْ قَلِيلًا إِنَّكُم مُّجۡرِمُونَ ۝٤٦كُلُوا۟ وَتَمَتَّعُوا۟ قَلِیلًا إِنَّكُم مُّجۡرِمُونَ ۝٤٦كُلُوْا وَتَمَتَّعُوْا قَلِیْلًا اِنَّكُمْ مُّجْرِمُوْنَ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٧وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٧وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

وَإِذَا قِيلَ لَهُمُ ٱرۡكَعُواْ لَا يَرۡكَعُونَ ۝٤٨وَإِذَا قِیلَ لَهُمُ ٱرۡكَعُوا۟ لَا یَرۡكَعُونَ ۝٤٨وَاِذَا قِیْلَ لَهُمُ ارْكَعُوْا لَا یَرْكَعُوْنَ ۟

وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٩وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٩وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟

فَبِأَيِّ حَدِيثِۭ بَعۡدَهُۥ يُؤۡمِنُونَ ۝٥٠فَبِأَیِّ حَدِیثِۭ بَعۡدَهُۥ یُؤۡمِنُونَ ۝٥٠فَبِاَیِّ حَدِیْثٍ بَعْدَهٗ یُؤْمِنُوْنَ ۟۠
