بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

وَٱلۡمُرۡسَلَٰتِ عُرۡفٗا ۝١ وَٱلۡمُرۡسَلَٰتِ عُرۡفࣰا ۝١ وَالْمُرْسَلٰتِ عُرْفًا ۟ۙ فَٱلۡعَٰصِفَٰتِ عَصۡفٗا ۝٢ فَٱلۡعَٰصِفَٰتِ عَصۡفࣰا ۝٢ فَالْعٰصِفٰتِ عَصْفًا ۟ۙ وَٱلنَّٰشِرَٰتِ نَشۡرٗا ۝٣ وَٱلنَّٰشِرَٰتِ نَشۡرࣰا ۝٣ وَّالنّٰشِرٰتِ نَشْرًا ۟ۙ فَٱلۡفَٰرِقَٰتِ فَرۡقٗا ۝٤ فَٱلۡفَٰرِقَٰتِ فَرۡقࣰا ۝٤ فَالْفٰرِقٰتِ فَرْقًا ۟ۙ فَٱلۡمُلۡقِيَٰتِ ذِكۡرًا ۝٥ فَٱلۡمُلۡقِیَٰتِ ذِكۡرًا ۝٥ فَالْمُلْقِیٰتِ ذِكْرًا ۟ۙ عُذۡرًا أَوۡ نُذۡرًا ۝٦ عُذۡرًا أَوۡ نُذۡرًا ۝٦ عُذْرًا اَوْ نُذْرًا ۟ۙ إِنَّمَا تُوعَدُونَ لَوَٰقِعٞ ۝٧ إِنَّمَا تُوعَدُونَ لَوَٰقِعࣱ ۝٧ اِنَّمَا تُوْعَدُوْنَ لَوَاقِعٌ ۟ؕ فَإِذَا ٱلنُّجُومُ طُمِسَتۡ ۝٨ فَإِذَا ٱلنُّجُومُ طُمِسَتۡ ۝٨ فَاِذَا النُّجُوْمُ طُمِسَتْ ۟ۙ وَإِذَا ٱلسَّمَآءُ فُرِجَتۡ ۝٩ وَإِذَا ٱلسَّمَاۤءُ فُرِجَتۡ ۝٩ وَاِذَا السَّمَآءُ فُرِجَتْ ۟ۙ وَإِذَا ٱلۡجِبَالُ نُسِفَتۡ ۝١٠ وَإِذَا ٱلۡجِبَالُ نُسِفَتۡ ۝١٠ وَاِذَا الْجِبَالُ نُسِفَتْ ۟ۙ وَإِذَا ٱلرُّسُلُ أُقِّتَتۡ ۝١١ وَإِذَا ٱلرُّسُلُ أُقِّتَتۡ ۝١١ وَاِذَا الرُّسُلُ اُقِّتَتْ ۟ؕ لِأَيِّ يَوۡمٍ أُجِّلَتۡ ۝١٢ لِأَیِّ یَوۡمٍ أُجِّلَتۡ ۝١٢ لِاَیِّ یَوْمٍ اُجِّلَتْ ۟ؕ لِيَوۡمِ ٱلۡفَصۡلِ ۝١٣ لِیَوۡمِ ٱلۡفَصۡلِ ۝١٣ لِیَوْمِ الْفَصْلِ ۟ۚ وَمَآ أَدۡرَىٰكَ مَا يَوۡمُ ٱلۡفَصۡلِ ۝١٤ وَمَاۤ أَدۡرَىٰكَ مَا یَوۡمُ ٱلۡفَصۡلِ ۝١٤ وَمَاۤ اَدْرٰىكَ مَا یَوْمُ الْفَصْلِ ۟ؕ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝١٥ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝١٥ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ أَلَمۡ نُهۡلِكِ ٱلۡأَوَّلِينَ ۝١٦ أَلَمۡ نُهۡلِكِ ٱلۡأَوَّلِینَ ۝١٦ اَلَمْ نُهْلِكِ الْاَوَّلِیْنَ ۟ؕ ثُمَّ نُتۡبِعُهُمُ ٱلۡأٓخِرِينَ ۝١٧ ثُمَّ نُتۡبِعُهُمُ ٱلۡءَاخِرِینَ ۝١٧ ثُمَّ نُتْبِعُهُمُ الْاٰخِرِیْنَ ۟ كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِينَ ۝١٨ كَذَٰلِكَ نَفۡعَلُ بِٱلۡمُجۡرِمِینَ ۝١٨ كَذٰلِكَ نَفْعَلُ بِالْمُجْرِمِیْنَ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝١٩ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝١٩ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ أَلَمۡ نَخۡلُقكُّم مِّن مَّآءٖ مَّهِينٖ ۝٢٠ أَلَمۡ نَخۡلُقكُّم مِّن مَّاۤءࣲ مَّهِینࣲ ۝٢٠ اَلَمْ نَخْلُقْكُّمْ مِّنْ مَّآءٍ مَّهِیْنٍ ۟ۙ فَجَعَلۡنَٰهُ فِي قَرَارٖ مَّكِينٍ ۝٢١ فَجَعَلۡنَٰهُ فِی قَرَارࣲ مَّكِینٍ ۝٢١ فَجَعَلْنٰهُ فِیْ قَرَارٍ مَّكِیْنٍ ۟ۙ إِلَىٰ قَدَرٖ مَّعۡلُومٖ ۝٢٢ إِلَىٰ قَدَرࣲ مَّعۡلُومࣲ ۝٢٢ اِلٰی قَدَرٍ مَّعْلُوْمٍ ۟ۙ فَقَدَرۡنَا فَنِعۡمَ ٱلۡقَٰدِرُونَ ۝٢٣ فَقَدَرۡنَا فَنِعۡمَ ٱلۡقَٰدِرُونَ ۝٢٣ فَقَدَرْنَا ۖۗ فَنِعْمَ الْقٰدِرُوْنَ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٢٤ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٢٤ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ أَلَمۡ نَجۡعَلِ ٱلۡأَرۡضَ كِفَاتًا ۝٢٥ أَلَمۡ نَجۡعَلِ ٱلۡأَرۡضَ كِفَاتًا ۝٢٥ اَلَمْ نَجْعَلِ الْاَرْضَ كِفَاتًا ۟ۙ أَحۡيَآءٗ وَأَمۡوَٰتٗا ۝٢٦ أَحۡیَاۤءࣰ وَأَمۡوَٰتࣰا ۝٢٦ اَحْیَآءً وَّاَمْوَاتًا ۟ۙ وَجَعَلۡنَا فِيهَا رَوَٰسِيَ شَٰمِخَٰتٖ وَأَسۡقَيۡنَٰكُم مَّآءٗ فُرَاتٗا ۝٢٧ وَجَعَلۡنَا فِیهَا رَوَٰسِیَ شَٰمِخَٰتࣲ وَأَسۡقَیۡنَٰكُم مَّاۤءࣰ فُرَاتࣰا ۝٢٧ وَّجَعَلْنَا فِیْهَا رَوَاسِیَ شٰمِخٰتٍ وَّاَسْقَیْنٰكُمْ مَّآءً فُرَاتًا ۟ؕ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٢٨ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٢٨ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ ٱنطَلِقُوٓاْ إِلَىٰ مَا كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢٩ ٱنطَلِقُوۤا۟ إِلَىٰ مَا كُنتُم بِهِۦ تُكَذِّبُونَ ۝٢٩ اِنْطَلِقُوْۤا اِلٰی مَا كُنْتُمْ بِهٖ تُكَذِّبُوْنَ ۟ۚ ٱنطَلِقُوٓاْ إِلَىٰ ظِلّٖ ذِي ثَلَٰثِ شُعَبٖ ۝٣٠ ٱنطَلِقُوۤا۟ إِلَىٰ ظِلࣲّ ذِی ثَلَٰثِ شُعَبࣲ ۝٣٠ اِنْطَلِقُوْۤا اِلٰی ظِلٍّ ذِیْ ثَلٰثِ شُعَبٍ ۟ۙ لَّا ظَلِيلٖ وَلَا يُغۡنِي مِنَ ٱللَّهَبِ ۝٣١ لَّا ظَلِیلࣲ وَلَا یُغۡنِی مِنَ ٱللَّهَبِ ۝٣١ لَّا ظَلِیْلٍ وَّلَا یُغْنِیْ مِنَ اللَّهَبِ ۟ؕ إِنَّهَا تَرۡمِي بِشَرَرٖ كَٱلۡقَصۡرِ ۝٣٢ إِنَّهَا تَرۡمِی بِشَرَرࣲ كَٱلۡقَصۡرِ ۝٣٢ اِنَّهَا تَرْمِیْ بِشَرَرٍ كَالْقَصْرِ ۟ۚ كَأَنَّهُۥ جِمَٰلَتٞ صُفۡرٞ ۝٣٣ كَأَنَّهُۥ جِمَٰلَتࣱ صُفۡرࣱ ۝٣٣ كَاَنَّهٗ جِمٰلَتٌ صُفْرٌ ۟ؕ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٣٤ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٣٤ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ هَٰذَا يَوۡمُ لَا يَنطِقُونَ ۝٣٥ هَٰذَا یَوۡمُ لَا یَنطِقُونَ ۝٣٥ هٰذَا یَوْمُ لَا یَنْطِقُوْنَ ۟ۙ وَلَا يُؤۡذَنُ لَهُمۡ فَيَعۡتَذِرُونَ ۝٣٦ وَلَا یُؤۡذَنُ لَهُمۡ فَیَعۡتَذِرُونَ ۝٣٦ وَلَا یُؤْذَنُ لَهُمْ فَیَعْتَذِرُوْنَ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٣٧ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٣٧ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ هَٰذَا يَوۡمُ ٱلۡفَصۡلِۖ جَمَعۡنَٰكُمۡ وَٱلۡأَوَّلِينَ ۝٣٨ هَٰذَا یَوۡمُ ٱلۡفَصۡلِۖ جَمَعۡنَٰكُمۡ وَٱلۡأَوَّلِینَ ۝٣٨ هٰذَا یَوْمُ الْفَصْلِ ۚ جَمَعْنٰكُمْ وَالْاَوَّلِیْنَ ۟ فَإِن كَانَ لَكُمۡ كَيۡدٞ فَكِيدُونِ ۝٣٩ فَإِن كَانَ لَكُمۡ كَیۡدࣱ فَكِیدُونِ ۝٣٩ فَاِنْ كَانَ لَكُمْ كَیْدٌ فَكِیْدُوْنِ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٠ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٠ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟۠ إِنَّ ٱلۡمُتَّقِينَ فِي ظِلَٰلٖ وَعُيُونٖ ۝٤١ إِنَّ ٱلۡمُتَّقِینَ فِی ظِلَٰلࣲ وَعُیُونࣲ ۝٤١ اِنَّ الْمُتَّقِیْنَ فِیْ ظِلٰلٍ وَّعُیُوْنٍ ۟ۙ وَفَوَٰكِهَ مِمَّا يَشۡتَهُونَ ۝٤٢ وَفَوَٰكِهَ مِمَّا یَشۡتَهُونَ ۝٤٢ وَّفَوَاكِهَ مِمَّا یَشْتَهُوْنَ ۟ؕ كُلُواْ وَٱشۡرَبُواْ هَنِيٓـَٔۢا بِمَا كُنتُمۡ تَعۡمَلُونَ ۝٤٣ كُلُوا۟ وَٱشۡرَبُوا۟ هَنِیۤءَۢا بِمَا كُنتُمۡ تَعۡمَلُونَ ۝٤٣ كُلُوْا وَاشْرَبُوْا هَنِیْٓـًٔا بِمَا كُنْتُمْ تَعْمَلُوْنَ ۟ إِنَّا كَذَٰلِكَ نَجۡزِي ٱلۡمُحۡسِنِينَ ۝٤٤ إِنَّا كَذَٰلِكَ نَجۡزِی ٱلۡمُحۡسِنِینَ ۝٤٤ اِنَّا كَذٰلِكَ نَجْزِی الْمُحْسِنِیْنَ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٥ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٥ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ كُلُواْ وَتَمَتَّعُواْ قَلِيلًا إِنَّكُم مُّجۡرِمُونَ ۝٤٦ كُلُوا۟ وَتَمَتَّعُوا۟ قَلِیلًا إِنَّكُم مُّجۡرِمُونَ ۝٤٦ كُلُوْا وَتَمَتَّعُوْا قَلِیْلًا اِنَّكُمْ مُّجْرِمُوْنَ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٧ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٧ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ وَإِذَا قِيلَ لَهُمُ ٱرۡكَعُواْ لَا يَرۡكَعُونَ ۝٤٨ وَإِذَا قِیلَ لَهُمُ ٱرۡكَعُوا۟ لَا یَرۡكَعُونَ ۝٤٨ وَاِذَا قِیْلَ لَهُمُ ارْكَعُوْا لَا یَرْكَعُوْنَ ۟ وَيۡلٞ يَوۡمَئِذٖ لِّلۡمُكَذِّبِينَ ۝٤٩ وَیۡلࣱ یَوۡمَئِذࣲ لِّلۡمُكَذِّبِینَ ۝٤٩ وَیْلٌ یَّوْمَىِٕذٍ لِّلْمُكَذِّبِیْنَ ۟ فَبِأَيِّ حَدِيثِۭ بَعۡدَهُۥ يُؤۡمِنُونَ ۝٥٠ فَبِأَیِّ حَدِیثِۭ بَعۡدَهُۥ یُؤۡمِنُونَ ۝٥٠ فَبِاَیِّ حَدِیْثٍ بَعْدَهٗ یُؤْمِنُوْنَ ۟۠
