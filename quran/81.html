التكوير At-Takwir · Surah 81 · Meccan

بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِیمِ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِیْمِ ۟

إِذَا ٱلشَّمۡسُ كُوِّرَتۡ ۝١ إِذَا ٱلشَّمۡسُ كُوِّرَتۡ ۝١ اِذَا الشَّمْسُ كُوِّرَتْ ۟ وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتۡ ۝٢ وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتۡ ۝٢ وَاِذَا النُّجُوْمُ انْكَدَرَتْ ۟ وَإِذَا ٱلۡجِبَالُ سُيِّرَتۡ ۝٣ وَإِذَا ٱلۡجِبَالُ سُیِّرَتۡ ۝٣ وَاِذَا الْجِبَالُ سُیِّرَتْ ۟ وَإِذَا ٱلۡعِشَارُ عُطِّلَتۡ ۝٤ وَإِذَا ٱلۡعِشَارُ عُطِّلَتۡ ۝٤ وَاِذَا الْعِشَارُ عُطِّلَتْ ۟ وَإِذَا ٱلۡوُحُوشُ حُشِرَتۡ ۝٥ وَإِذَا ٱلۡوُحُوشُ حُشِرَتۡ ۝٥ وَاِذَا الْوُحُوْشُ حُشِرَتْ ۟ وَإِذَا ٱلۡبِحَارُ سُجِّرَتۡ ۝٦ وَإِذَا ٱلۡبِحَارُ سُجِّرَتۡ ۝٦ وَاِذَا الْبِحَارُ سُجِّرَتْ ۟ وَإِذَا ٱلنُّفُوسُ زُوِّجَتۡ ۝٧ وَإِذَا ٱلنُّفُوسُ زُوِّجَتۡ ۝٧ وَاِذَا النُّفُوْسُ زُوِّجَتْ ۟ وَإِذَا ٱلۡمَوۡءُۥدَةُ سُئِلَتۡ ۝٨ وَإِذَا ٱلۡمَوۡءُۥدَةُ سُئِلَتۡ ۝٨ وَاِذَا الْمَوْءٗدَةُ سُىِٕلَتْ ۟ بِأَيِّ ذَنۢبٖ قُتِلَتۡ ۝٩ بِأَیِّ ذَنۢبࣲ قُتِلَتۡ ۝٩ بِاَیِّ ذَنْۢبٍ قُتِلَتْ ۟ۚ وَإِذَا ٱلصُّحُفُ نُشِرَتۡ ۝١٠ وَإِذَا ٱلصُّحُفُ نُشِرَتۡ ۝١٠ وَاِذَا الصُّحُفُ نُشِرَتْ ۟ وَإِذَا ٱلسَّمَآءُ كُشِطَتۡ ۝١١ وَإِذَا ٱلسَّمَاۤءُ كُشِطَتۡ ۝١١ وَاِذَا السَّمَآءُ كُشِطَتْ ۟ وَإِذَا ٱلۡجَحِيمُ سُعِّرَتۡ ۝١٢ وَإِذَا ٱلۡجَحِیمُ سُعِّرَتۡ ۝١٢ وَاِذَا الْجَحِیْمُ سُعِّرَتْ ۟ وَإِذَا ٱلۡجَنَّةُ أُزۡلِفَتۡ ۝١٣ وَإِذَا ٱلۡجَنَّةُ أُزۡلِفَتۡ ۝١٣ وَاِذَا الْجَنَّةُ اُزْلِفَتْ ۟ عَلِمَتۡ نَفۡسٞ مَّآ أَحۡضَرَتۡ ۝١٤ عَلِمَتۡ نَفۡسࣱ مَّاۤ أَحۡضَرَتۡ ۝١٤ عَلِمَتْ نَفْسٌ مَّاۤ اَحْضَرَتْ ۟ؕ فَلَآ أُقۡسِمُ بِٱلۡخُنَّسِ ۝١٥ فَلَاۤ أُقۡسِمُ بِٱلۡخُنَّسِ ۝١٥ فَلَاۤ اُقْسِمُ بِالْخُنَّسِ ۟ۙ ٱلۡجَوَارِ ٱلۡكُنَّسِ ۝١٦ ٱلۡجَوَارِ ٱلۡكُنَّسِ ۝١٦ الْجَوَارِ الْكُنَّسِ ۟ۙ وَٱلَّيۡلِ إِذَا عَسۡعَسَ ۝١٧ وَٱلَّیۡلِ إِذَا عَسۡعَسَ ۝١٧ وَالَّیْلِ اِذَا عَسْعَسَ ۟ۙ وَٱلصُّبۡحِ إِذَا تَنَفَّسَ ۝١٨ وَٱلصُّبۡحِ إِذَا تَنَفَّسَ ۝١٨ وَالصُّبْحِ اِذَا تَنَفَّسَ ۟ۙ إِنَّهُۥ لَقَوۡلُ رَسُولٖ كَرِيمٖ ۝١٩ إِنَّهُۥ لَقَوۡلُ رَسُولࣲ كَرِیمࣲ ۝١٩ اِنَّهٗ لَقَوْلُ رَسُوْلٍ كَرِیْمٍ ۟ۙ ذِي قُوَّةٍ عِندَ ذِي ٱلۡعَرۡشِ مَكِينٖ ۝٢٠ ذِی قُوَّةٍ عِندَ ذِی ٱلۡعَرۡشِ مَكِینࣲ ۝٢٠ ذِیْ قُوَّةٍ عِنْدَ ذِی الْعَرْشِ مَكِیْنٍ ۟ۙ مُّطَاعٖ ثَمَّ أَمِينٖ ۝٢١ مُّطَاعࣲ ثَمَّ أَمِینࣲ ۝٢١ مُّطَاعٍ ثَمَّ اَمِیْنٍ ۟ؕ وَمَا صَاحِبُكُم بِمَجۡنُونٖ ۝٢٢ وَمَا صَاحِبُكُم بِمَجۡنُونࣲ ۝٢٢ وَمَا صَاحِبُكُمْ بِمَجْنُوْنٍ ۟ۚ وَلَقَدۡ رَءَاهُ بِٱلۡأُفُقِ ٱلۡمُبِينِ ۝٢٣ وَلَقَدۡ رَءَاهُ بِٱلۡأُفُقِ ٱلۡمُبِینِ ۝٢٣ وَلَقَدْ رَاٰهُ بِالْاُفُقِ الْمُبِیْنِ ۟ۚ وَمَا هُوَ عَلَى ٱلۡغَيۡبِ بِضَنِينٖ ۝٢٤ وَمَا هُوَ عَلَى ٱلۡغَیۡبِ بِضَنِینࣲ ۝٢٤ وَمَا هُوَ عَلَی الْغَیْبِ بِضَنِیْنٍ ۟ۚ وَمَا هُوَ بِقَوۡلِ شَيۡطَٰنٖ رَّجِيمٖ ۝٢٥ وَمَا هُوَ بِقَوۡلِ شَیۡطَٰنࣲ رَّجِیمࣲ ۝٢٥ وَمَا هُوَ بِقَوْلِ شَیْطٰنٍ رَّجِیْمٍ ۟ۙ فَأَيۡنَ تَذۡهَبُونَ ۝٢٦ فَأَیۡنَ تَذۡهَبُونَ ۝٢٦ فَاَیْنَ تَذْهَبُوْنَ ۟ؕ إِنۡ هُوَ إِلَّا ذِكۡرٞ لِّلۡعَٰلَمِينَ ۝٢٧ إِنۡ هُوَ إِلَّا ذِكۡرࣱ لِّلۡعَٰلَمِینَ ۝٢٧ اِنْ هُوَ اِلَّا ذِكْرٌ لِّلْعٰلَمِیْنَ ۟ۙ لِمَن شَآءَ مِنكُمۡ أَن يَسۡتَقِيمَ ۝٢٨ لِمَن شَاۤءَ مِنكُمۡ أَن یَسۡتَقِیمَ ۝٢٨ لِمَنْ شَآءَ مِنْكُمْ اَنْ یَّسْتَقِیْمَ ۟ؕ وَمَا تَشَآءُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ رَبُّ ٱلۡعَٰلَمِينَ ۝٢٩ وَمَا تَشَاۤءُونَ إِلَّاۤ أَن یَشَاۤءَ ٱللَّهُ رَبُّ ٱلۡعَٰلَمِینَ ۝٢٩ وَمَا تَشَآءُوْنَ اِلَّاۤ اَنْ یَّشَآءَ اللّٰهُ رَبُّ الْعٰلَمِیْنَ ۟۠
