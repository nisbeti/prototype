بَرَآءَةٞ مِّنَ ٱللَّهِ وَرَسُولِهِۦٓ إِلَى ٱلَّذِينَ عَٰهَدتُّم مِّنَ ٱلۡمُشۡرِكِينَ ۝١ بَرَاۤءَةࣱ مِّنَ ٱللَّهِ وَرَسُولِهِۦۤ إِلَى ٱلَّذِینَ عَٰهَدتُّم مِّنَ ٱلۡمُشۡرِكِینَ ۝١ بَرَآءَةٌ مِّنَ اللّٰهِ وَرَسُوْلِهٖۤ اِلَی الَّذِیْنَ عٰهَدْتُّمْ مِّنَ الْمُشْرِكِیْنَ ۟ؕ فَسِيحُواْ فِي ٱلۡأَرۡضِ أَرۡبَعَةَ أَشۡهُرٖ وَٱعۡلَمُوٓاْ أَنَّكُمۡ غَيۡرُ مُعۡجِزِي ٱللَّهِ وَأَنَّ ٱللَّهَ مُخۡزِي ٱلۡكَٰفِرِينَ ۝٢ فَسِیحُوا۟ فِی ٱلۡأَرۡضِ أَرۡبَعَةَ أَشۡهُرࣲ وَٱعۡلَمُوۤا۟ أَنَّكُمۡ غَیۡرُ مُعۡجِزِی ٱللَّهِ وَأَنَّ ٱللَّهَ مُخۡزِی ٱلۡكَٰفِرِینَ ۝٢ فَسِیْحُوْا فِی الْاَرْضِ اَرْبَعَةَ اَشْهُرٍ وَّاعْلَمُوْۤا اَنَّكُمْ غَیْرُ مُعْجِزِی اللّٰهِ ۙ وَاَنَّ اللّٰهَ مُخْزِی الْكٰفِرِیْنَ ۟ وَأَذَٰنٞ مِّنَ ٱللَّهِ وَرَسُولِهِۦٓ إِلَى ٱلنَّاسِ يَوۡمَ ٱلۡحَجِّ ٱلۡأَكۡبَرِ أَنَّ ٱللَّهَ بَرِيٓءٞ مِّنَ ٱلۡمُشۡرِكِينَ وَرَسُولُهُۥۚ فَإِن تُبۡتُمۡ فَهُوَ خَيۡرٞ لَّكُمۡۖ وَإِن تَوَلَّيۡتُمۡ فَٱعۡلَمُوٓاْ أَنَّكُمۡ غَيۡرُ مُعۡجِزِي ٱللَّهِۗ وَبَشِّرِ ٱلَّذِينَ كَفَرُواْ بِعَذَابٍ أَلِيمٍ ۝٣ وَأَذَٰنࣱ مِّنَ ٱللَّهِ وَرَسُولِهِۦۤ إِلَى ٱلنَّاسِ یَوۡمَ ٱلۡحَجِّ ٱلۡأَكۡبَرِ أَنَّ ٱللَّهَ بَرِیۤءࣱ مِّنَ ٱلۡمُشۡرِكِینَ وَرَسُولُهُۥۚ فَإِن تُبۡتُمۡ فَهُوَ خَیۡرࣱ لَّكُمۡۖ وَإِن تَوَلَّیۡتُمۡ فَٱعۡلَمُوۤا۟ أَنَّكُمۡ غَیۡرُ مُعۡجِزِی ٱللَّهِۗ وَبَشِّرِ ٱلَّذِینَ كَفَرُوا۟ بِعَذَابٍ أَلِیمٍ ۝٣ وَاَذَانٌ مِّنَ اللّٰهِ وَرَسُوْلِهٖۤ اِلَی النَّاسِ یَوْمَ الْحَجِّ الْاَكْبَرِ اَنَّ اللّٰهَ بَرِیْٓءٌ مِّنَ الْمُشْرِكِیْنَ ۙ۬ وَرَسُوْلُهٗ ؕ فَاِنْ تُبْتُمْ فَهُوَ خَیْرٌ لَّكُمْ ۚ وَاِنْ تَوَلَّیْتُمْ فَاعْلَمُوْۤا اَنَّكُمْ غَیْرُ مُعْجِزِی اللّٰهِ ؕ وَبَشِّرِ الَّذِیْنَ كَفَرُوْا بِعَذَابٍ اَلِیْمٍ ۟ۙ إِلَّا ٱلَّذِينَ عَٰهَدتُّم مِّنَ ٱلۡمُشۡرِكِينَ ثُمَّ لَمۡ يَنقُصُوكُمۡ شَيۡـٔٗا وَلَمۡ يُظَٰهِرُواْ عَلَيۡكُمۡ أَحَدٗا فَأَتِمُّوٓاْ إِلَيۡهِمۡ عَهۡدَهُمۡ إِلَىٰ مُدَّتِهِمۡۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلۡمُتَّقِينَ ۝٤ إِلَّا ٱلَّذِینَ عَٰهَدتُّم مِّنَ ٱلۡمُشۡرِكِینَ ثُمَّ لَمۡ یَنقُصُوكُمۡ شَیۡءࣰا وَلَمۡ یُظَٰهِرُوا۟ عَلَیۡكُمۡ أَحَدࣰا فَأَتِمُّوۤا۟ إِلَیۡهِمۡ عَهۡدَهُمۡ إِلَىٰ مُدَّتِهِمۡۚ إِنَّ ٱللَّهَ یُحِبُّ ٱلۡمُتَّقِینَ ۝٤ اِلَّا الَّذِیْنَ عٰهَدْتُّمْ مِّنَ الْمُشْرِكِیْنَ ثُمَّ لَمْ یَنْقُصُوْكُمْ شَیْـًٔا وَّلَمْ یُظَاهِرُوْا عَلَیْكُمْ اَحَدًا فَاَتِمُّوْۤا اِلَیْهِمْ عَهْدَهُمْ اِلٰی مُدَّتِهِمْ ؕ اِنَّ اللّٰهَ یُحِبُّ الْمُتَّقِیْنَ ۟ فَإِذَا ٱنسَلَخَ ٱلۡأَشۡهُرُ ٱلۡحُرُمُ فَٱقۡتُلُواْ ٱلۡمُشۡرِكِينَ حَيۡثُ وَجَدتُّمُوهُمۡ وَخُذُوهُمۡ وَٱحۡصُرُوهُمۡ وَٱقۡعُدُواْ لَهُمۡ كُلَّ مَرۡصَدٖۚ فَإِن تَابُواْ وَأَقَامُواْ ٱلصَّلَوٰةَ وَءَاتَوُاْ ٱلزَّكَوٰةَ فَخَلُّواْ سَبِيلَهُمۡۚ إِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٞ ۝٥ فَإِذَا ٱنسَلَخَ ٱلۡأَشۡهُرُ ٱلۡحُرُمُ فَٱقۡتُلُوا۟ ٱلۡمُشۡرِكِینَ حَیۡثُ وَجَدتُّمُوهُمۡ وَخُذُوهُمۡ وَٱحۡصُرُوهُمۡ وَٱقۡعُدُوا۟ لَهُمۡ كُلَّ مَرۡصَدࣲۚ فَإِن تَابُوا۟ وَأَقَامُوا۟ ٱلصَّلَوٰةَ وَءَاتَوُا۟ ٱلزَّكَوٰةَ فَخَلُّوا۟ سَبِیلَهُمۡۚ إِنَّ ٱللَّهَ غَفُورࣱ رَّحِیمࣱ ۝٥ فَاِذَا انْسَلَخَ الْاَشْهُرُ الْحُرُمُ فَاقْتُلُوا الْمُشْرِكِیْنَ حَیْثُ وَجَدْتُّمُوْهُمْ وَخُذُوْهُمْ وَاحْصُرُوْهُمْ وَاقْعُدُوْا لَهُمْ كُلَّ مَرْصَدٍ ۚ فَاِنْ تَابُوْا وَاَقَامُوا الصَّلٰوةَ وَاٰتَوُا الزَّكٰوةَ فَخَلُّوْا سَبِیْلَهُمْ ؕ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِیْمٌ ۟ وَإِنۡ أَحَدٞ مِّنَ ٱلۡمُشۡرِكِينَ ٱسۡتَجَارَكَ فَأَجِرۡهُ حَتَّىٰ يَسۡمَعَ كَلَٰمَ ٱللَّهِ ثُمَّ أَبۡلِغۡهُ مَأۡمَنَهُۥۚ ذَٰلِكَ بِأَنَّهُمۡ قَوۡمٞ لَّا يَعۡلَمُونَ ۝٦ وَإِنۡ أَحَدࣱ مِّنَ ٱلۡمُشۡرِكِینَ ٱسۡتَجَارَكَ فَأَجِرۡهُ حَتَّىٰ یَسۡمَعَ كَلَٰمَ ٱللَّهِ ثُمَّ أَبۡلِغۡهُ مَأۡمَنَهُۥۚ ذَٰلِكَ بِأَنَّهُمۡ قَوۡمࣱ لَّا یَعۡلَمُونَ ۝٦ وَاِنْ اَحَدٌ مِّنَ الْمُشْرِكِیْنَ اسْتَجَارَكَ فَاَجِرْهُ حَتّٰی یَسْمَعَ كَلٰمَ اللّٰهِ ثُمَّ اَبْلِغْهُ مَاْمَنَهٗ ؕ ذٰلِكَ بِاَنَّهُمْ قَوْمٌ لَّا یَعْلَمُوْنَ ۟۠ كَيۡفَ يَكُونُ لِلۡمُشۡرِكِينَ عَهۡدٌ عِندَ ٱللَّهِ وَعِندَ رَسُولِهِۦٓ إِلَّا ٱلَّذِينَ عَٰهَدتُّمۡ عِندَ ٱلۡمَسۡجِدِ ٱلۡحَرَامِۖ فَمَا ٱسۡتَقَٰمُواْ لَكُمۡ فَٱسۡتَقِيمُواْ لَهُمۡۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلۡمُتَّقِينَ ۝٧ كَیۡفَ یَكُونُ لِلۡمُشۡرِكِینَ عَهۡدٌ عِندَ ٱللَّهِ وَعِندَ رَسُولِهِۦۤ إِلَّا ٱلَّذِینَ عَٰهَدتُّمۡ عِندَ ٱلۡمَسۡجِدِ ٱلۡحَرَامِۖ فَمَا ٱسۡتَقَٰمُوا۟ لَكُمۡ فَٱسۡتَقِیمُوا۟ لَهُمۡۚ إِنَّ ٱللَّهَ یُحِبُّ ٱلۡمُتَّقِینَ ۝٧ كَیْفَ یَكُوْنُ لِلْمُشْرِكِیْنَ عَهْدٌ عِنْدَ اللّٰهِ وَعِنْدَ رَسُوْلِهٖۤ اِلَّا الَّذِیْنَ عٰهَدْتُّمْ عِنْدَ الْمَسْجِدِ الْحَرَامِ ۚ فَمَا اسْتَقَامُوْا لَكُمْ فَاسْتَقِیْمُوْا لَهُمْ ؕ اِنَّ اللّٰهَ یُحِبُّ الْمُتَّقِیْنَ ۟ كَيۡفَ وَإِن يَظۡهَرُواْ عَلَيۡكُمۡ لَا يَرۡقُبُواْ فِيكُمۡ إِلّٗا وَلَا ذِمَّةٗۚ يُرۡضُونَكُم بِأَفۡوَٰهِهِمۡ وَتَأۡبَىٰ قُلُوبُهُمۡ وَأَكۡثَرُهُمۡ فَٰسِقُونَ ۝٨ كَیۡفَ وَإِن یَظۡهَرُوا۟ عَلَیۡكُمۡ لَا یَرۡقُبُوا۟ فِیكُمۡ إِلࣰّا وَلَا ذِمَّةࣰۚ یُرۡضُونَكُم بِأَفۡوَٰهِهِمۡ وَتَأۡبَىٰ قُلُوبُهُمۡ وَأَكۡثَرُهُمۡ فَٰسِقُونَ ۝٨ كَیْفَ وَاِنْ یَّظْهَرُوْا عَلَیْكُمْ لَا یَرْقُبُوْا فِیْكُمْ اِلًّا وَّلَا ذِمَّةً ؕ یُرْضُوْنَكُمْ بِاَفْوَاهِهِمْ وَتَاْبٰی قُلُوْبُهُمْ ۚ وَاَكْثَرُهُمْ فٰسِقُوْنَ ۟ۚ ٱشۡتَرَوۡاْ بِـَٔايَٰتِ ٱللَّهِ ثَمَنٗا قَلِيلٗا فَصَدُّواْ عَن سَبِيلِهِۦٓۚ إِنَّهُمۡ سَآءَ مَا كَانُواْ يَعۡمَلُونَ ۝٩ ٱشۡتَرَوۡا۟ بِءَایَٰتِ ٱللَّهِ ثَمَنࣰا قَلِیلࣰا فَصَدُّوا۟ عَن سَبِیلِهِۦۤۚ إِنَّهُمۡ سَاۤءَ مَا كَانُوا۟ یَعۡمَلُونَ ۝٩ اِشْتَرَوْا بِاٰیٰتِ اللّٰهِ ثَمَنًا قَلِیْلًا فَصَدُّوْا عَنْ سَبِیْلِهٖ ؕ اِنَّهُمْ سَآءَ مَا كَانُوْا یَعْمَلُوْنَ ۟ لَا يَرۡقُبُونَ فِي مُؤۡمِنٍ إِلّٗا وَلَا ذِمَّةٗۚ وَأُوْلَٰٓئِكَ هُمُ ٱلۡمُعۡتَدُونَ ۝١٠ لَا یَرۡقُبُونَ فِی مُؤۡمِنٍ إِلࣰّا وَلَا ذِمَّةࣰۚ وَأُو۟لَٰۤئِكَ هُمُ ٱلۡمُعۡتَدُونَ ۝١٠ لَا یَرْقُبُوْنَ فِیْ مُؤْمِنٍ اِلًّا وَّلَا ذِمَّةً ؕ وَاُولٰٓىِٕكَ هُمُ الْمُعْتَدُوْنَ ۟ فَإِن تَابُواْ وَأَقَامُواْ ٱلصَّلَوٰةَ وَءَاتَوُاْ ٱلزَّكَوٰةَ فَإِخۡوَٰنُكُمۡ فِي ٱلدِّينِۗ وَنُفَصِّلُ ٱلۡأٓيَٰتِ لِقَوۡمٖ يَعۡلَمُونَ ۝١١ فَإِن تَابُوا۟ وَأَقَامُوا۟ ٱلصَّلَوٰةَ وَءَاتَوُا۟ ٱلزَّكَوٰةَ فَإِخۡوَٰنُكُمۡ فِی ٱلدِّینِۗ وَنُفَصِّلُ ٱلۡءَایَٰتِ لِقَوۡمࣲ یَعۡلَمُونَ ۝١١ فَاِنْ تَابُوْا وَاَقَامُوا الصَّلٰوةَ وَاٰتَوُا الزَّكٰوةَ فَاِخْوَانُكُمْ فِی الدِّیْنِ ؕ وَنُفَصِّلُ الْاٰیٰتِ لِقَوْمٍ یَّعْلَمُوْنَ ۟ وَإِن نَّكَثُوٓاْ أَيۡمَٰنَهُم مِّنۢ بَعۡدِ عَهۡدِهِمۡ وَطَعَنُواْ فِي دِينِكُمۡ فَقَٰتِلُوٓاْ أَئِمَّةَ ٱلۡكُفۡرِ إِنَّهُمۡ لَآ أَيۡمَٰنَ لَهُمۡ لَعَلَّهُمۡ يَنتَهُونَ ۝١٢ وَإِن نَّكَثُوۤا۟ أَیۡمَٰنَهُم مِّنۢ بَعۡدِ عَهۡدِهِمۡ وَطَعَنُوا۟ فِی دِینِكُمۡ فَقَٰتِلُوۤا۟ أَئِمَّةَ ٱلۡكُفۡرِ إِنَّهُمۡ لَاۤ أَیۡمَٰنَ لَهُمۡ لَعَلَّهُمۡ یَنتَهُونَ ۝١٢ وَاِنْ نَّكَثُوْۤا اَیْمَانَهُمْ مِّنْ بَعْدِ عَهْدِهِمْ وَطَعَنُوْا فِیْ دِیْنِكُمْ فَقَاتِلُوْۤا اَىِٕمَّةَ الْكُفْرِ ۙ اِنَّهُمْ لَاۤ اَیْمَانَ لَهُمْ لَعَلَّهُمْ یَنْتَهُوْنَ ۟ أَلَا تُقَٰتِلُونَ قَوۡمٗا نَّكَثُوٓاْ أَيۡمَٰنَهُمۡ وَهَمُّواْ بِإِخۡرَاجِ ٱلرَّسُولِ وَهُم بَدَءُوكُمۡ أَوَّلَ مَرَّةٍۚ أَتَخۡشَوۡنَهُمۡۚ فَٱللَّهُ أَحَقُّ أَن تَخۡشَوۡهُ إِن كُنتُم مُّؤۡمِنِينَ ۝١٣ أَلَا تُقَٰتِلُونَ قَوۡمࣰا نَّكَثُوۤا۟ أَیۡمَٰنَهُمۡ وَهَمُّوا۟ بِإِخۡرَاجِ ٱلرَّسُولِ وَهُم بَدَءُوكُمۡ أَوَّلَ مَرَّةٍۚ أَتَخۡشَوۡنَهُمۡۚ فَٱللَّهُ أَحَقُّ أَن تَخۡشَوۡهُ إِن كُنتُم مُّؤۡمِنِینَ ۝١٣ اَلَا تُقَاتِلُوْنَ قَوْمًا نَّكَثُوْۤا اَیْمَانَهُمْ وَهَمُّوْا بِاِخْرَاجِ الرَّسُوْلِ وَهُمْ بَدَءُوْكُمْ اَوَّلَ مَرَّةٍ ؕ اَتَخْشَوْنَهُمْ ۚ فَاللّٰهُ اَحَقُّ اَنْ تَخْشَوْهُ اِنْ كُنْتُمْ مُّؤْمِنِیْنَ ۟ قَٰتِلُوهُمۡ يُعَذِّبۡهُمُ ٱللَّهُ بِأَيۡدِيكُمۡ وَيُخۡزِهِمۡ وَيَنصُرۡكُمۡ عَلَيۡهِمۡ وَيَشۡفِ صُدُورَ قَوۡمٖ مُّؤۡمِنِينَ ۝١٤ قَٰتِلُوهُمۡ یُعَذِّبۡهُمُ ٱللَّهُ بِأَیۡدِیكُمۡ وَیُخۡزِهِمۡ وَیَنصُرۡكُمۡ عَلَیۡهِمۡ وَیَشۡفِ صُدُورَ قَوۡمࣲ مُّؤۡمِنِینَ ۝١٤ قَاتِلُوْهُمْ یُعَذِّبْهُمُ اللّٰهُ بِاَیْدِیْكُمْ وَیُخْزِهِمْ وَیَنْصُرْكُمْ عَلَیْهِمْ وَیَشْفِ صُدُوْرَ قَوْمٍ مُّؤْمِنِیْنَ ۟ۙ وَيُذۡهِبۡ غَيۡظَ قُلُوبِهِمۡۗ وَيَتُوبُ ٱللَّهُ عَلَىٰ مَن يَشَآءُۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌ ۝١٥ وَیُذۡهِبۡ غَیۡظَ قُلُوبِهِمۡۗ وَیَتُوبُ ٱللَّهُ عَلَىٰ مَن یَشَاۤءُۗ وَٱللَّهُ عَلِیمٌ حَكِیمٌ ۝١٥ وَیُذْهِبْ غَیْظَ قُلُوْبِهِمْ ؕ وَیَتُوْبُ اللّٰهُ عَلٰی مَنْ یَّشَآءُ ؕ وَاللّٰهُ عَلِیْمٌ حَكِیْمٌ ۟ أَمۡ حَسِبۡتُمۡ أَن تُتۡرَكُواْ وَلَمَّا يَعۡلَمِ ٱللَّهُ ٱلَّذِينَ جَٰهَدُواْ مِنكُمۡ وَلَمۡ يَتَّخِذُواْ مِن دُونِ ٱللَّهِ وَلَا رَسُولِهِۦ وَلَا ٱلۡمُؤۡمِنِينَ وَلِيجَةٗۚ وَٱللَّهُ خَبِيرُۢ بِمَا تَعۡمَلُونَ ۝١٦ أَمۡ حَسِبۡتُمۡ أَن تُتۡرَكُوا۟ وَلَمَّا یَعۡلَمِ ٱللَّهُ ٱلَّذِینَ جَٰهَدُوا۟ مِنكُمۡ وَلَمۡ یَتَّخِذُوا۟ مِن دُونِ ٱللَّهِ وَلَا رَسُولِهِۦ وَلَا ٱلۡمُؤۡمِنِینَ وَلِیجَةࣰۚ وَٱللَّهُ خَبِیرُۢ بِمَا تَعۡمَلُونَ ۝١٦ اَمْ حَسِبْتُمْ اَنْ تُتْرَكُوْا وَلَمَّا یَعْلَمِ اللّٰهُ الَّذِیْنَ جٰهَدُوْا مِنْكُمْ وَلَمْ یَتَّخِذُوْا مِنْ دُوْنِ اللّٰهِ وَلَا رَسُوْلِهٖ وَلَا الْمُؤْمِنِیْنَ وَلِیْجَةً ؕ وَاللّٰهُ خَبِیْرٌ بِمَا تَعْمَلُوْنَ ۟۠ مَا كَانَ لِلۡمُشۡرِكِينَ أَن يَعۡمُرُواْ مَسَٰجِدَ ٱللَّهِ شَٰهِدِينَ عَلَىٰٓ أَنفُسِهِم بِٱلۡكُفۡرِۚ أُوْلَٰٓئِكَ حَبِطَتۡ أَعۡمَٰلُهُمۡ وَفِي ٱلنَّارِ هُمۡ خَٰلِدُونَ ۝١٧ مَا كَانَ لِلۡمُشۡرِكِینَ أَن یَعۡمُرُوا۟ مَسَٰجِدَ ٱللَّهِ شَٰهِدِینَ عَلَىٰۤ أَنفُسِهِم بِٱلۡكُفۡرِۚ أُو۟لَٰۤئِكَ حَبِطَتۡ أَعۡمَٰلُهُمۡ وَفِی ٱلنَّارِ هُمۡ خَٰلِدُونَ ۝١٧ مَا كَانَ لِلْمُشْرِكِیْنَ اَنْ یَّعْمُرُوْا مَسٰجِدَ اللّٰهِ شٰهِدِیْنَ عَلٰۤی اَنْفُسِهِمْ بِالْكُفْرِ ؕ اُولٰٓىِٕكَ حَبِطَتْ اَعْمَالُهُمْ ۖۚ وَفِی النَّارِ هُمْ خٰلِدُوْنَ ۟ إِنَّمَا يَعۡمُرُ مَسَٰجِدَ ٱللَّهِ مَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡيَوۡمِ ٱلۡأٓخِرِ وَأَقَامَ ٱلصَّلَوٰةَ وَءَاتَى ٱلزَّكَوٰةَ وَلَمۡ يَخۡشَ إِلَّا ٱللَّهَۖ فَعَسَىٰٓ أُوْلَٰٓئِكَ أَن يَكُونُواْ مِنَ ٱلۡمُهۡتَدِينَ ۝١٨ إِنَّمَا یَعۡمُرُ مَسَٰجِدَ ٱللَّهِ مَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡیَوۡمِ ٱلۡءَاخِرِ وَأَقَامَ ٱلصَّلَوٰةَ وَءَاتَى ٱلزَّكَوٰةَ وَلَمۡ یَخۡشَ إِلَّا ٱللَّهَۖ فَعَسَىٰۤ أُو۟لَٰۤئِكَ أَن یَكُونُوا۟ مِنَ ٱلۡمُهۡتَدِینَ ۝١٨ اِنَّمَا یَعْمُرُ مَسٰجِدَ اللّٰهِ مَنْ اٰمَنَ بِاللّٰهِ وَالْیَوْمِ الْاٰخِرِ وَاَقَامَ الصَّلٰوةَ وَاٰتَی الزَّكٰوةَ وَلَمْ یَخْشَ اِلَّا اللّٰهَ فَعَسٰۤی اُولٰٓىِٕكَ اَنْ یَّكُوْنُوْا مِنَ الْمُهْتَدِیْنَ ۟ أَجَعَلۡتُمۡ سِقَايَةَ ٱلۡحَآجِّ وَعِمَارَةَ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ كَمَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡيَوۡمِ ٱلۡأٓخِرِ وَجَٰهَدَ فِي سَبِيلِ ٱللَّهِۚ لَا يَسۡتَوُۥنَ عِندَ ٱللَّهِۗ وَٱللَّهُ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلظَّٰلِمِينَ ۝١٩ أَجَعَلۡتُمۡ سِقَایَةَ ٱلۡحَاۤجِّ وَعِمَارَةَ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ كَمَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡیَوۡمِ ٱلۡءَاخِرِ وَجَٰهَدَ فِی سَبِیلِ ٱللَّهِۚ لَا یَسۡتَوُۥنَ عِندَ ٱللَّهِۗ وَٱللَّهُ لَا یَهۡدِی ٱلۡقَوۡمَ ٱلظَّٰلِمِینَ ۝١٩ اَجَعَلْتُمْ سِقَایَةَ الْحَآجِّ وَعِمَارَةَ الْمَسْجِدِ الْحَرَامِ كَمَنْ اٰمَنَ بِاللّٰهِ وَالْیَوْمِ الْاٰخِرِ وَجٰهَدَ فِیْ سَبِیْلِ اللّٰهِ ؕ لَا یَسْتَوٗنَ عِنْدَ اللّٰهِ ؕ وَاللّٰهُ لَا یَهْدِی الْقَوْمَ الظّٰلِمِیْنَ ۟ۘ ٱلَّذِينَ ءَامَنُواْ وَهَاجَرُواْ وَجَٰهَدُواْ فِي سَبِيلِ وَرَسُوْلِهٖ وَجِهَادٍ فِیْ سَبِیْلِهٖ فَتَرَبَّصُوْا حَتّٰی یَاْتِیَ اللّٰهُ بِاَمْرِهٖ ؕ وَاللّٰهُ لَا یَهْدِی الْقَوْمَ الْفٰسِقِیْنَ ۟۠ لَقَدۡ نَصَرَكُمُ ٱللَّهُ فِي مَوَاطِنَ كَثِيرَةٖ وَيَوۡمَ حُنَيۡنٍ إِذۡ أَعۡجَبَتۡكُمۡ كَثۡرَتُكُمۡ فَلَمۡ تُغۡنِ عَنكُمۡ شَيۡـٔٗا وَضَاقَتۡ عَلَيۡكُمُ ٱلۡأَرۡضُ بِمَا رَحُبَتۡ ثُمَّ وَلَّيۡتُم مُّدۡبِرِينَ ۝٢٥ لَقَدۡ نَصَرَكُمُ ٱللَّهُ فِی مَوَاطِنَ كَثِیرَةࣲ وَیَوۡمَ حُنَیۡنٍ إِذۡ أَعۡجَبَتۡكُمۡ كَثۡرَتُكُمۡ فَلَمۡ تُغۡنِ عَنكُمۡ شَیۡءࣰا وَضَاقَتۡ عَلَیۡكُمُ ٱلۡأَرۡضُ بِمَا رَحُبَتۡ ثُمَّ وَلَّیۡتُم مُّدۡبِرِینَ ۝٢٥ لَقَدْ نَصَرَكُمُ اللّٰهُ فِیْ مَوَاطِنَ كَثِیْرَةٍ ۙ وَّیَوْمَ حُنَیْنٍ ۙ اِذْ اَعْجَبَتْكُمْ كَثْرَتُكُمْ فَلَمْ تُغْنِ عَنْكُمْ شَیْـًٔا وَّضَاقَتْ عَلَیْكُمُ الْاَرْضُ بِمَا رَحُبَتْ ثُمَّ وَلَّیْتُمْ مُّدْبِرِیْنَ ۟ۚ ثُمَّ أَنزَلَ ٱللَّهُ سَكِينَتَهُۥ عَلَىٰ رَسُولِهِۦ وَعَلَى ٱلۡمُؤۡمِنِينَ وَأَنزَلَ جُنُودٗا لَّمۡ تَرَوۡهَا وَعَذَّبَ ٱلَّذِينَ كَفَرُواْۚ وَذَٰلِكَ جَزَآءُ ٱلۡكَٰفِرِينَ ۝٢٦ ثُمَّ أَنزَلَ ٱللَّهُ سَكِینَتَهُۥ عَلَىٰ رَسُولِهِۦ وَعَلَى ٱلۡمُؤۡمِنِینَ وَأَنزَلَ جُنُودࣰا لَّمۡ تَرَوۡهَا وَعَذَّبَ ٱلَّذِینَ كَفَرُوا۟ۚ وَذَٰلِكَ جَزَاۤءُ ٱلۡكَٰفِرِینَ ۝٢٦ ثُمَّ اَنْزَلَ اللّٰهُ سَكِیْنَتَهٗ عَلٰی رَسُوْلِهٖ وَعَلَی الْمُؤْمِنِیْنَ وَاَنْزَلَ جُنُوْدًا لَّمْ تَرَوْهَا وَعَذَّبَ الَّذِیْنَ كَفَرُوْا ؕ وَذٰلِكَ جَزَآءُ الْكٰفِرِیْنَ ۟ ثُمَّ يَتُوبُ ٱللَّهُ مِنۢ بَعۡدِ ذَٰلِكَ عَلَىٰ مَن يَشَآءُۗ وَٱللَّهُ غَفُورٞ رَّحِيمٞ ۝٢٧ ثُمَّ یَتُوبُ ٱللَّهُ مِنۢ بَعۡدِ ذَٰلِكَ عَلَىٰ مَن یَشَاۤءُۗ وَٱللَّهُ غَفُورࣱ رَّحِیمࣱ ۝٢٧ ثُمَّ یَتُوْبُ اللّٰهُ مِنْ بَعْدِ ذٰلِكَ عَلٰی مَنْ یَّشَآءُ ؕ وَاللّٰهُ غَفُوْرٌ رَّحِیْمٌ ۟ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِنَّمَا ٱلۡمُشۡرِكُونَ نَجَسٞ فَلَا يَقۡرَبُواْ ٱلۡمَسۡجِدَ ٱلۡحَرَامَ بَعۡدَ عَامِهِمۡ هَٰذَاۚ وَإِنۡ خِفۡتُمۡ عَيۡلَةٗ فَسَوۡفَ يُغۡنِيكُمُ ٱللَّهُ مِن فَضۡلِهِۦٓ إِن شَآءَۚ إِنَّ ٱللَّهَ عَلِيمٌ حَكِيمٞ ۝٢٨ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوۤا۟ إِنَّمَا ٱلۡمُشۡرِكُونَ نَجَسࣱ فَلَا یَقۡرَبُوا۟ ٱلۡمَسۡجِدَ ٱلۡحَرَامَ بَعۡدَ عَامِهِمۡ هَٰذَاۚ وَإِنۡ خِفۡتُمۡ عَیۡلَةࣰ فَسَوۡفَ یُغۡنِیكُمُ ٱللَّهُ مِن فَضۡلِهِۦۤ إِن شَاۤءَۚ إِنَّ ٱللَّهَ عَلِیمٌ حَكِیمࣱ ۝٢٨ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْۤا اِنَّمَا الْمُشْرِكُوْنَ نَجَسٌ فَلَا یَقْرَبُوا الْمَسْجِدَ الْحَرَامَ بَعْدَ عَامِهِمْ هٰذَا ۚ وَاِنْ خِفْتُمْ عَیْلَةً فَسَوْفَ یُغْنِیْكُمُ اللّٰهُ مِنْ فَضْلِهٖۤ اِنْ شَآءَ ؕ اِنَّ اللّٰهَ عَلِیْمٌ حَكِیْمٌ ۟ قَٰتِلُواْ ٱلَّذِينَ لَا يُؤۡمِنُونَ بِٱللَّهِ وَلَا بِٱلۡيَوۡمِ ٱلۡأٓخِرِ وَلَا يُحَرِّمُونَ مَا حَرَّمَ ٱللَّهُ وَرَسُولُهُۥ وَلَا يَدِينُونَ دِينَ ٱلۡحَقِّ مِنَ ٱلَّذِينَ أُوتُواْ ٱلۡكِتَٰبَ حَتَّىٰ يُعۡطُواْ ٱلۡجِزۡيَةَ عَن يَدٖ وَهُمۡ صَٰغِرُونَ ۝٢٩ قَٰتِلُوا۟ ٱلَّذِینَ لَا یُؤۡمِنُونَ بِٱللَّهِ وَلَا بِٱلۡیَوۡمِ ٱلۡءَاخِرِ وَلَا یُحَرِّمُونَ مَا حَرَّمَ ٱللَّهُ وَرَسُولُهُۥ وَلَا یَدِینُونَ دِینَ ٱلۡحَقِّ مِنَ ٱلَّذِینَ أُوتُوا۟ ٱلۡكِتَٰبَ حَتَّىٰ یُعۡطُوا۟ ٱلۡجِزۡیَةَ عَن یَدࣲ وَهُمۡ صَٰغِرُونَ ۝٢٩ قَاتِلُوا الَّذِیْنَ لَا یُؤْمِنُوْنَ بِاللّٰهِ وَلَا بِالْیَوْمِ الْاٰخِرِ وَلَا یُحَرِّمُوْنَ مَا حَرَّمَ اللّٰهُ وَرَسُوْلُهٗ وَلَا یَدِیْنُوْنَ دِیْنَ الْحَقِّ مِنَ الَّذِیْنَ اُوْتُوا الْكِتٰبَ حَتّٰی یُعْطُوا الْجِزْیَةَ عَنْ یَّدٍ وَّهُمْ صٰغِرُوْنَ ۟۠ وَقَالَتِ ٱلۡيَهُودُ عُزَيۡرٌ ٱبۡنُ ٱللَّهِ وَقَالَتِ ٱلنَّصَٰرَى ٱلۡمَسِيحُ ٱبۡنُ ٱللَّهِۖ ذَٰلِكَ قَوۡلُهُم بِأَفۡوَٰهِهِمۡۖ يُضَٰهِـُٔونَ قَوۡلَ ٱلَّذِينَ كَفَرُواْ مِن قَبۡلُۚ قَٰتَلَهُمُ ٱللَّهُۖ أَنَّىٰ يُؤۡفَكُونَ ۝٣٠ وَقَالَتِ ٱلۡیَهُودُ عُزَیۡرٌ ٱبۡنُ ٱللَّهِ وَقَالَتِ ٱلنَّصَٰرَى ٱلۡمَسِیحُ ٱبۡنُ ٱللَّهِۖ ذَٰلِكَ قَوۡلُهُم بِأَفۡوَٰهِهِمۡۖ یُضَٰهِءُونَ قَوۡلَ ٱلَّذِینَ كَفَرُوا۟ مِن قَبۡلُۚ قَٰتَلَهُمُ ٱللَّهُۖ أَنَّىٰ یُؤۡفَكُونَ ۝٣٠ وَقَالَتِ الْیَهُوْدُ عُزَیْرُ بْنُ اللّٰهِ وَقَالَتِ النَّصٰرَی الْمَسِیْحُ ابْنُ اللّٰهِ ؕ ذٰلِكَ قَوْلُهُمْ بِاَفْوَاهِهِمْ ۚ یُضَاهِـُٔوْنَ قَوْلَ الَّذِیْنَ كَفَرُوْا مِنْ قَبْلُ مَا كُنتُمۡ تَكۡنِزُونَ ۝٣٥ یَوۡمَ یُحۡمَىٰ عَلَیۡهَا فِی نَارِ جَهَنَّمَ فَتُكۡوَىٰ بِهَا جِبَاهُهُمۡ وَجُنُوبُهُمۡ وَظُهُورُهُمۡۖ هَٰذَا مَا كَنَزۡتُمۡ لِأَنفُسِكُمۡ فَذُوقُوا۟ مَا كُنتُمۡ تَكۡنِزُونَ ۝٣٥ یَّوْمَ یُحْمٰی عَلَیْهَا فِیْ نَارِ جَهَنَّمَ فَتُكْوٰی بِهَا جِبَاهُهُمْ وَجُنُوْبُهُمْ وَظُهُوْرُهُمْ ؕ هٰذَا مَا كَنَزْتُمْ لِاَنْفُسِكُمْ فَذُوْقُوْا مَا كُنْتُمْ تَكْنِزُوْنَ ۟ إِنَّ عِدَّةَ ٱلشُّهُورِ عِندَ ٱللَّهِ ٱثۡنَا عَشَرَ شَهۡرٗا فِي كِتَٰبِ ٱللَّهِ يَوۡمَ خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ مِنۡهَآ أَرۡبَعَةٌ حُرُمٞۚ ذَٰلِكَ ٱلدِّينُ ٱلۡقَيِّمُۚ فَلَا تَظۡلِمُواْ فِيهِنَّ أَنفُسَكُمۡۚ وَقَٰتِلُواْ ٱلۡمُشۡرِكِينَ كَآفَّةٗ كَمَا يُقَٰتِلُونَكُمۡ كَآفَّةٗۚ وَٱعۡلَمُوٓاْ أَنَّ ٱللَّهَ مَعَ ٱلۡمُتَّقِينَ ۝٣٦ إِنَّ عِدَّةَ ٱلشُّهُورِ عِندَ ٱللَّهِ ٱثۡنَا عَشَرَ شَهۡرࣰا فِی كِتَٰبِ ٱللَّهِ یَوۡمَ خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضَ مِنۡهَاۤ أَرۡبَعَةٌ حُرُمࣱۚ ذَٰلِكَ ٱلدِّینُ ٱلۡقَیِّمُۚ فَلَا تَظۡلِمُوا۟ فِیهِنَّ أَنفُسَكُمۡۚ وَقَٰتِلُوا۟ ٱلۡمُشۡرِكِینَ كَاۤفَّةࣰ كَمَا یُقَٰتِلُونَكُمۡ كَاۤفَّةࣰۚ وَٱعۡلَمُوۤا۟ أَنَّ ٱللَّهَ مَعَ ٱلۡمُتَّقِینَ ۝٣٦ اِنَّ عِدَّةَ الشُّهُوْرِ عِنْدَ اللّٰهِ اثْنَا عَشَرَ شَهْرًا فِیْ كِتٰبِ اللّٰهِ یَوْمَ خَلَقَ السَّمٰوٰتِ وَالْاَرْضَ مِنْهَاۤ اَرْبَعَةٌ حُرُمٌ ؕ ذٰلِكَ الدِّیْنُ الْقَیِّمُ ۙ۬ فَلَا تَظْلِمُوْا فِیْهِنَّ اَنْفُسَكُمْ وَقَاتِلُوا الْمُشْرِكِیْنَ كَآفَّةً كَمَا یُقَاتِلُوْنَكُمْ كَآفَّةً ؕ وَاعْلَمُوْۤا اَنَّ اللّٰهَ مَعَ الْمُتَّقِیْنَ ۟ إِنَّمَا ٱلنَّسِيٓءُ زِيَادَةٞ فِي ٱلۡكُفۡرِۖ يُضَلُّ بِهِ ٱلَّذِينَ كَفَرُواْ يُحِلُّونَهُۥ عَامٗا وَيُحَرِّمُونَهُۥ عَامٗا لِّيُوَاطِـُٔواْ عِدَّةَ مَا حَرَّمَ ٱللَّهُ فَيُحِلُّواْ مَا حَرَّمَ ٱللَّهُۚ زُيِّنَ لَهُمۡ سُوٓءُ أَعۡمَٰلِهِمۡۗ وَٱللَّهُ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلۡكَٰفِرِينَ ۝٣٧ إِنَّمَا ٱلنَّسِیۤءُ زِیَادَةࣱ فِی ٱلۡكُفۡرِۖ یُضَلُّ بِهِ ٱلَّذِینَ كَفَرُوا۟ یُحِلُّونَهُۥ عَامࣰا وَیُحَرِّمُونَهُۥ عَامࣰا لِّیُوَاطِءُوا۟ عِدَّةَ مَا حَرَّمَ ٱللَّهُ فَیُحِلُّوا۟ مَا حَرَّمَ ٱللَّهُۚ زُیِّنَ لَهُمۡ سُوۤءُ أَعۡمَٰلِهِمۡۗ وَٱللَّهُ لَا یَهۡدِی ٱلۡقَوۡمَ ٱلۡكَٰفِرِینَ ۝٣٧ اِنَّمَا النَّسِیْٓءُ زِیَادَةٌ فِی الْكُفْرِ یُضَلُّ بِهِ الَّذِیْنَ كَفَرُوْا یُحِلُّوْنَهٗ عَامًا وَّیُحَرِّمُوْنَهٗ عَامًا لِّیُوَاطِـُٔوْا عِدَّةَ مَا حَرَّمَ اللّٰهُ فَیُحِلُّوْا مَا حَرَّمَ اللّٰهُ ؕ زُیِّنَ لَهُمْ سُوْٓءُ اَعْمَالِهِمْ ؕ وَاللّٰهُ لَا یَهْدِی الْقَوْمَ الْكٰفِرِیْنَ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ مَا لَكُمۡ إِذَا قِيلَ لَكُمُ ٱنفِرُواْ فِي سَبِيلِ ٱللَّهِ ٱثَّاقَلۡتُمۡ إِلَى ٱلۡأَرۡضِۚ أَرَضِيتُم بِٱلۡحَيَوٰةِ ٱلدُّنۡيَا مِنَ ٱلۡأٓخِرَةِۚ فَمَا مَتَٰعُ ٱلۡحَيَوٰةِ ٱلدُّنۡيَا فِي ٱلۡأٓخِرَةِ إِلَّا قَلِيلٌ ۝٣٨ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ مَا لَكُمۡ إِذَا قِیلَ لَكُمُ ٱنفِرُوا۟ فِی سَبِیلِ ٱللَّهِ ٱثَّاقَلۡتُمۡ إِلَى ٱلۡأَرۡضِۚ أَرَضِیتُم بِٱلۡحَیَوٰةِ ٱلدُّنۡیَا مِنَ ٱلۡءَاخِرَةِۚ فَمَا مَتَٰعُ ٱلۡحَیَوٰةِ ٱلدُّنۡیَا فِی ٱلۡءَاخِرَةِ إِلَّا قَلِیلٌ ۝٣٨ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا مَا لَكُمْ اِذَا قِیْلَ لَكُمُ انْفِرُوْا فِیْ سَبِیْلِ اللّٰهِ اثَّاقَلْتُمْ اِلَی الْاَرْضِ ؕ اَرَضِیْتُمْ بِالْحَیٰوةِ الدُّنْیَا مِنَ الْاٰخِرَةِ ۚ فَمَا مَتَاعُ الْحَیٰوةِ الدُّنْیَا فِی الْاٰخِرَةِ اِلَّا قَلِیْلٌ ۟ إِلَّا تَنفِرُواْ يُعَذِّبۡكُمۡ عَذَابًا أَلِيمٗا وَيَسۡتَبۡدِلۡ قَوۡمًا غَيۡرَكُمۡ وَلَا تَضُرُّوهُ شَيۡـٔٗاۗ وَٱللَّهُ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرٌ ۝٣٩ إِلَّا تَنفِرُوا۟ یُعَذِّبۡكُمۡ عَذَابًا أَلِیمࣰا وَیَسۡتَبۡدِلۡ قَوۡمًا غَیۡرَكُمۡ وَلَا تَضُرُّوهُ شَیۡءࣰاۗ وَٱللَّهُ عَلَىٰ كُلِّ شَیۡءࣲ قَدِیرٌ ۝٣٩ اِلَّا تَنْفِرُوْا یُعَذِّبْكُمْ عَذَابًا اَلِیْمًا ۙ۬ وَّیَسْتَبْدِلْ قَوْمًا غَیْرَكُمْ وَلَا تَضُرُّوْهُ شَیْـًٔا ؕ وَاللّٰهُ عَلٰی كُلِّ شَیْءٍ قَدِیْرٌ ۟ إِلَّا تَنصُرُوهُ فَقَدۡ نَصَرَهُ ٱللَّهُ إِذۡ أَخۡرَجَهُ ٱلَّذِينَ كَفَرُواْ ثَانِيَ ٱثۡنَيۡنِ إِذۡ هُمَا فِي ٱلۡغَارِ إِذۡ يَقُولُ لِصَٰحِبِهِۦ لَا تَحۡزَنۡ إِنَّ ٱللَّهَ مَعَنَاۖ فَأَنزَلَ ٱللَّهُ عَلَیۡهِ وَأَیَّدَهُۥ بِجُنُودࣲ لَّمۡ تَرَوۡهَا وَجَعَلَ كَلِمَةَ ٱلَّذِینَ كَفَرُوا۟ ٱلسُّفۡلَىٰۗ وَكَلِمَةُ ٱللَّهِ هِیَ ٱلۡعُلۡیَاۗ وَٱللَّهُ عَزِیزٌ حَكِیمٌ ۝٤٠ اِلَّا تَنْصُرُوْهُ فَقَدْ نَصَرَهُ اللّٰهُ اِذْ اَخْرَجَهُ الَّذِیْنَ كَفَرُوْا ثَانِیَ اثْنَیْنِ اِذْ هُمَا فِی الْغَارِ اِذْ یَقُوْلُ لِصَاحِبِهٖ لَا تَحْزَنْ اِنَّ اللّٰهَ مَعَنَا ۚ فَاَنْزَلَ اللّٰهُ سَكِیْنَتَهٗ عَلَیْهِ زَادُوكُمۡ إِلَّا خَبَالٗا وَلَأَوۡضَعُواْ خِلَٰلَكُمۡ يَبۡغُونَكُمُ ٱلۡفِتۡنَةَ وَفِيكُمۡ سَمَّٰعُونَ لَهُمۡۗ وَٱللَّهُ عَلِيمُۢ بِٱلظَّٰلِمِينَ ۝٤٧ لَوۡ خَرَجُوا۟ فِیكُم مَّا زَادُوكُمۡ إِلَّا خَبَالࣰا وَلَأَوۡضَعُوا۟ خِلَٰلَكُمۡ یَبۡغُونَكُمُ ٱلۡفِتۡنَةَ وَفِیكُمۡ سَمَّٰعُونَ لَهُمۡۗ وَٱللَّهُ عَلِیمُۢ بِٱلظَّٰلِمِینَ ۝٤٧ لَوْ خَرَجُوْا فِیْكُمْ مَّا زَادُوْكُمْ اِلَّا خَبَالًا ٱلَّذِینَ مِن قَبۡلِكُم بِخَلَٰقِهِمۡ وَخُضۡتُمۡ كَٱلَّذِی خَاضُوۤا۟ۚ أُو۟لَٰۤئِكَ حَبِطَتۡ أَعۡمَٰلُهُمۡ فِی ٱلدُّنۡیَا وَٱلۡءَاخِرَةِۖ وَأُو۟لَٰۤئِكَ هُمُ ٱلۡخَٰسِرُونَ ۝٦٩ كَالَّذِیْنَ مِنْ قَبْلِكُمْ كَانُوْۤا اَشَدَّ مِنْكُمْ قُوَّةً وَّاَكْثَرَ اَمْوَالًا وَّاَوْلَادًا ؕ فَاسْتَمْتَعُوْا بِخَلَاقِهِمْ فَاسْتَمْتَعْتُمْ بِخَلَاقِكُمْ كَمَا اسْتَمْتَعَ الَّذِیْنَ مِنْ قَبْلِكُمْ بِخَلَاقِهِمْ وَخُضْتُمْ كَالَّذِیْ خَاضُوْا ؕ اُولٰٓىِٕكَ حَبِطَتْ اَعْمَالُهُمْ فِی الدُّنْیَا وَالْاٰخِرَةِ ۚ وَاُولٰٓىِٕكَ هُمُ الْخٰسِرُوْنَ ۟ أَلَمۡ يَأۡتِهِمۡ نَبَأُ ٱلَّذِينَ مِن قَبۡلِهِمۡ قَوۡمِ نُوحٖ وَعَادٖ وَثَمُودَ وَقَوۡمِ إِبۡرَٰهِيمَ وَأَصۡحَٰبِ مَدۡيَنَ وَٱلۡمُؤۡتَفِكَٰتِۚ أَتَتۡهُمۡ رُسُلُهُم بِٱلۡبَيِّنَٰتِۖ فَمَا كَانَ ٱللَّهُ لِيَظۡلِمَهُمۡ وَلَٰكِن كَانُوٓاْ أَنفُسَهُمۡ يَظۡلِمُونَ ۝٧٠ أَلَمۡ یَأۡتِهِمۡ نَبَأُ ٱلَّذِینَ مِن قَبۡلِهِمۡ قَوۡمِ نُوحࣲ وَعَادࣲ وَثَمُودَ وَقَوۡمِ إِبۡرَٰهِیمَ وَأَصۡحَٰبِ مَدۡیَنَ وَٱلۡمُؤۡتَفِكَٰتِۚ أَتَتۡهُمۡ رُسُلُهُم بِٱلۡبَیِّنَٰتِۖ فَمَا كَانَ ٱللَّهُ لِیَظۡلِمَهُمۡ وَلَٰكِن كَانُوۤا۟ أَنفُسَهُمۡ یَظۡلِمُونَ ۝٧٠ اَلَمْ یَاْتِهِمْ نَبَاُ الَّذِیْنَ مِنْ قَبْلِهِمْ قَوْمِ نُوْحٍ وَّعَادٍ وَّثَمُوْدَ ۙ۬ وَقَوْمِ اِبْرٰهِیْمَ وَاَصْحٰبِ مَدْیَنَ وَالْمُؤْتَفِكٰتِ ؕ اَتَتْهُمْ رُسُلُهُمْ بِالْبَیِّنٰتِ ۚ فَمَا كَانَ اللّٰهُ لِیَظْلِمَهُمْ وَلٰكِنْ كَانُوْۤا اَنْفُسَهُمْ یَظْلِمُوْنَ ۟ وَٱلۡمُؤۡمِنُونَ وَٱلۡمُؤۡمِنَٰتُ بَعۡضُهُمۡ أَوۡلِيَآءُ بَعۡضٖۚ يَأۡمُرُونَ بِٱلۡمَعۡرُوفِ وَيَنۡهَوۡنَ عَنِ ٱلۡمُنكَرِ وَيُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤۡتُونَ ٱلزَّكَوٰةَ وَيُطِيعُونَ ٱللَّهَ وَرَسُولَهُۥٓۚ أُوْلَٰٓئِكَ سَيَرۡحَمُهُمُ ٱللَّهُۗ إِنَّ ٱللَّهَ عَزِيزٌ حَكِيمٞ ۝٧١ وَٱلۡمُؤۡمِنُونَ وَٱلۡمُؤۡمِنَٰتُ بَعۡضُهُمۡ أَوۡلِیَاۤءُ بَعۡضࣲۚ یَأۡمُرُونَ بِٱلۡمَعۡرُوفِ وَیَنۡهَوۡنَ عَنِ ٱلۡمُنكَرِ وَیُقِیمُونَ ٱلصَّلَوٰةَ وَیُؤۡتُونَ ٱلزَّكَوٰةَ وَیُطِیعُونَ ٱللَّهَ وَرَسُولَهُۥۤۚ أُو۟لَٰۤئِكَ سَیَرۡحَمُهُمُ ٱللَّهُۗ إِنَّ ٱللَّهَ عَزِیزٌ حَكِیمࣱ ۝٧١ وَالْمُؤْمِنُوْنَ وَالْمُؤْمِنٰتُ بَعْضُهُمْ اَوْلِیَآءُ بَعْضٍ ۘ یَاْمُرُوْنَ بِالْمَعْرُوْفِ وَیَنْهَوْنَ عَنِ الْمُنْكَرِ وَیُقِیْمُوْنَ الصَّلٰوةَ وَیُؤْتُوْنَ الزَّكٰوةَ وَیُطِیْعُوْنَ اللّٰهَ وَرَسُوْلَهٗ ؕ اُولٰٓىِٕكَ سَیَرْحَمُهُمُ اللّٰهُ ؕ اِنَّ اللّٰهَ عَزِیْزٌ حَكِیْمٌ ۟ وَعَدَ ٱللَّهُ ٱلۡمُؤۡمِنِينَ وَٱلۡمُؤۡمِنَٰتِ جَنَّٰتٖ تَجۡرِي مِن تَحۡتِهَا ٱلۡأَنۡهَٰرُ خَٰلِدِينَ فِيهَا وَمَسَٰكِنَ طَيِّبَةٗ فِي جَنَّٰتِ عَدۡنٖۚ وَرِضۡوَٰنٞ مِّنَ ٱللَّهِ أَكۡبَرُۚ ذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ ۝٧٢ وَعَدَ ٱللَّهُ ٱلۡمُؤۡمِنِینَ وَٱلۡمُؤۡمِنَٰتِ جَنَّٰتࣲ تَجۡرِی مِن تَحۡتِهَا ٱلۡأَنۡهَٰرُ خَٰلِدِینَ فِیهَا وَمَسَٰكِنَ طَیِّبَةࣰ فِی جَنَّٰتِ عَدۡنࣲۚ وَرِضۡوَٰنࣱ مِّنَ ٱللَّهِ أَكۡبَرُۚ ذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِیمُ ۝٧٢ وَعَدَ اللّٰهُ الْمُؤْمِنِیْنَ وَالْمُؤْمِنٰتِ جَنّٰتٍ تَجْرِیْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِیْنَ فِیْهَا وَمَسٰكِنَ طَیِّبَةً فِیْ جَنّٰتِ عَدْنٍ ؕ وَرِضْوَانٌ مِّنَ اللّٰهِ اَكْبَرُ ؕ ذٰلِكَ هُوَ الْفَوْزُ الْعَظِیْمُ ۟۠ يَٰٓأَيُّهَا ٱلنَّبِيُّ جَٰهِدِ ٱلۡكُفَّارَ وَٱلۡمُنَٰفِقِينَ وَٱغۡلُظۡ عَلَيۡهِمۡۚ وَمَأۡوَىٰهُمۡ جَهَنَّمُۖ وَبِئۡسَ ٱلۡمَصِيرُ ۝٧٣ یَٰۤأَیُّهَا ٱلنَّبِیُّ جَٰهِدِ ٱلۡكُفَّارَ وَٱلۡمُنَٰفِقِینَ وَٱغۡلُظۡ عَلَیۡهِمۡۚ وَمَأۡوَىٰهُمۡ جَهَنَّمُۖ وَبِئۡسَ ٱلۡمَصِیرُ ۝٧٣ یٰۤاَیُّهَا النَّبِیُّ جَاهِدِ الْكُفَّارَ وَالْمُنٰفِقِیْنَ وَاغْلُظْ عَلَیْهِمْ ؕ وَمَاْوٰىهُمْ جَهَنَّمُ ؕ وَبِئْسَ الْمَصِیْرُ ۟ يَحۡلِفُونَ بِٱللَّهِ مَا قَالُواْ وَلَقَدۡ قَالُواْ كَلِمَةَ ٱلۡكُفۡرِ وَكَفَرُواْ بَعۡدَ إِسۡلَٰمِهِمۡ وَهَمُّواْ بِمَا لَمۡ يَنَالُواْۚ وَمَا نَقَمُوٓاْ إِلَّآ أَنۡ أَغۡنَىٰهُمُ ٱللَّهُ وَرَسُولُهُۥ مِن فَضۡلِهِۦۚ فَإِن يَتُوبُواْ يَكُ خَيۡرٗا لَّهُمۡۖ وَإِن يَتَوَلَّوۡاْ يُعَذِّبۡهُمُ ٱللَّهُ عَذَابًا أَلِيمٗا فِي ٱلدُّنۡيَا وَٱلۡأٓخِرَةِۚ وَمَا لَهُمۡ فِي ٱلۡأَرۡضِ مِن وَلِيّٖ وَلَا نَصِيرٖ ۝٧٤ یَحۡلِفُونَ بِٱللَّهِ مَا قَالُوا۟ وَلَقَدۡ قَالُوا۟ كَلِمَةَ ٱلۡكُفۡرِ وَكَفَرُوا۟ بَعۡدَ إِسۡلَٰمِهِمۡ وَهَمُّوا۟ بِمَا لَمۡ یَنَالُوا۟ۚ وَمَا نَقَمُوۤا۟ إِلَّاۤ أَنۡ أَغۡنَىٰهُمُ ٱللَّهُ وَرَسُولُهُۥ مِن فَضۡلِهِۦۚ فَإِن یَتُوبُوا۟ یَكُ خَیۡرࣰا لَّهُمۡۖ وَإِن یَتَوَلَّوۡا۟ یُعَذِّبۡهُمُ ٱللَّهُ عَذَابًا أَلِیمࣰا فِی ٱلدُّنۡیَا وَٱلۡءَاخِرَةِۚ وَمَا لَهُمۡ فِی ٱلۡأَرۡضِ مِن وَلِیࣲّ وَلَا نَصِیرࣲ ۝٧٤ یَحْلِفُوْنَ بِاللّٰهِ مَا قَالُوْا ؕ وَلَقَدْ قَالُوْا كَلِمَةَ الْكُفْرِ وَكَفَرُوْا بَعْدَ اِسْلَامِهِمْ وَهَمُّوْا بِمَا لَمْ یَنَالُوْا ۚ وَمَا نَقَمُوْۤا اِلَّاۤ اَنْ اَغْنٰىهُمُ اللّٰهُ وَرَسُوْلُهٗ مِنْ فَضْلِهٖ ۚ فَاِنْ یَّتُوْبُوْا یَكُ خَیْرًا لَّهُمْ ۚ وَاِنْ یَّتَوَلَّوْا یُعَذِّبْهُمُ اللّٰهُ عَذَابًا اَلِیْمًا ۙ فِی الدُّنْیَا وَالْاٰخِرَةِ ۚ وَمَا لَهُمْ فِی الْاَرْضِ مِنْ وَّلِیٍّ وَّلَا نَصِیْرٍ ۟ وَمِنۡهُم مَّنۡ عَٰهَدَ ٱللَّهَ لَئِنۡ ءَاتَىٰنَا مِن فَضۡلِهِۦ لَنَصَّدَّقَنَّ وَلَنَكُونَنَّ مِنَ ٱلصَّٰلِحِينَ ۝٧٥ وَمِنۡهُم مَّنۡ عَٰهَدَ ٱللَّهَ لَئِنۡ ءَاتَىٰنَا مِن فَضۡلِهِۦ لَنَصَّدَّقَنَّ وَلَنَكُونَنَّ مِنَ ٱلصَّٰلِحِینَ ۝٧٥ وَمِنْهُمْ مَّنْ عٰهَدَ اللّٰهَ لَىِٕنْ اٰتٰىنَا مِنْ فَضْلِهٖ لَنَصَّدَّقَنَّ وَلَنَكُوْنَنَّ مِنَ الصّٰلِحِیْنَ ۟ فَلَمَّآ ءَاتَىٰهُم مِّن فَضۡلِهِۦ بَخِلُواْ بِهِۦ وَتَوَلَّواْ وَّهُم مُّعۡرِضُونَ ۝٧٦ فَلَمَّاۤ ءَاتَىٰهُم مِّن فَضۡلِهِۦ بَخِلُوا۟ بِهِۦ وَتَوَلَّوا۟ وَّهُم مُّعۡرِضُونَ ۝٧٦ فَلَمَّاۤ اٰتٰىهُمْ مِّنْ فَضْلِهٖ بَخِلُوْا بِهٖ وَتَوَلَّوْا وَّهُمْ مُّعْرِضُوْنَ ۟ فَأَعۡقَبَهُمۡ نِفَاقٗا فِي قُلُوبِهِمۡ إِلَىٰ يَوۡمِ يَلۡقَوۡنَهُۥ بِمَآ أَخۡلَفُواْ ٱللَّهَ مَا وَعَدُوهُ وَبِمَا كَانُواْ يَكۡذِبُونَ ۝٧٧ فَأَعۡقَبَهُمۡ نِفَاقࣰا فِی قُلُوبِهِمۡ إِلَىٰ یَوۡمِ یَلۡقَوۡنَهُۥ بِمَاۤ أَخۡلَفُوا۟ ٱللَّهَ مَا وَعَدُوهُ وَبِمَا كَانُوا۟ یَكۡذِبُونَ ۝٧٧ فَاَعْقَبَهُمْ نِفَاقًا فِیْ قُلُوْبِهِمْ اِلٰی یَوْمِ یَلْقَوْنَهٗ بِمَاۤ اَخْلَفُوا اللّٰهَ مَا وَعَدُوْهُ وَبِمَا كَانُوْا یَكْذِبُوْنَ ۟ أَلَمۡ يَعۡلَمُوٓاْ أَنَّ ٱللَّهَ يَعۡلَمُ سِرَّهُمۡ وَنَجۡوَىٰهُمۡ وَأَنَّ ٱللَّهَ عَلَّٰمُ ٱلۡغُيُوبِ ۝٧٨ أَلَمۡ یَعۡلَمُوۤا۟ أَنَّ ٱللَّهَ یَلْمِزُوْنَ الْمُطَّوِّعِیْنَ مِنَ الْمُؤْمِنِیْنَ فِی الصَّدَقٰتِ وَالَّذِیْنَ لَا یَجِدُوْنَ اِلَّا جُهْدَهُمْ فَیَسْخَرُوْنَ مِنْهُمْ ؕ سَخِرَ اللّٰهُ مِنْهُمْ ؗ وَلَهُمْ عَذَابٌ اَلِیْمٌ ۟ ٱلۡقَوۡمَ ٱلۡفَٰسِقِينَ ۝٨٠ ٱسۡتَغۡفِرۡ لَهُمۡ أَوۡ لَا تَسۡتَغۡفِرۡ لَهُمۡ إِن تَسۡتَغۡفِرۡ لَهُمۡ سَبۡعِینَ مَرَّةࣰ فَلَن یَغۡفِرَ ٱللَّهُ لَهُمۡۚ ذَٰلِكَ بِأَنَّهُمۡ كَفَرُوا۟ بِٱللَّهِ وَرَسُولِهِۦۗ وَٱللَّهُ لَا یَهۡدِی ٱلۡقَوۡمَ ٱلۡفَٰسِقِینَ ۝٨٠ اِسْتَغْفِرْ لَهُمْ اَوْ لَا تَسْتَغْفِرْ لَهُمْ ؕ اِنْ تَسْتَغْفِرْ لَهُمْ سَبْعِیْنَ مَرَّةً فَلَنْ یَّغْفِرَ اللّٰهُ لَهُمْ ؕ ذٰلِكَ بِاَنَّهُمْ كَفَرُوْا بِاللّٰهِ وَرَسُوْلِهٖ ؕ وَاللّٰهُ لَا یَهْدِی الْقَوْمَ الْفٰسِقِیْنَ ۟۠ فَرِحَ ٱلۡمُخَلَّفُونَ بِمَقۡعَدِهِمۡ خِلَٰفَ رَسُولِ ٱللَّهِ وَكَرِهُوٓاْ أَن يُجَٰهِدُواْ خِلَٰفَ رَسُولِ ٱللَّهِ وَكَرِهُوۤا۟ أَن یُجَٰهِدُوا۟ بِأَمۡوَٰلِهِمۡ وَأَنفُسِهِمۡ فِی سَبِیلِ ٱللَّهِ وَقَالُوا۟ لَا تَنفِرُوا۟ فِی ٱلۡحَرِّۗ قُلۡ نَارُ جَهَنَّمَ أَشَدُّ حَرࣰّاۚ لَّوۡ كَانُوا۟ یَفۡقَهُونَ ۝٨١ فَرِحَ الْمُخَلَّفُوْنَ بِمَقْعَدِهِمْ خِلٰفَ رَسُوْلِ اللّٰهِ وَكَرِهُوْۤا اَنْ یُّجَاهِدُوْا بِاَمْوَالِهِمْ وَاَنْفُسِهِمْ فِیْ سَبِیْلِ اللّٰهِ وَقَالُوْا لَا تَنْفِرُوْا فِی الْحَرِّ ؕ قُلْ نَارُ جَهَنَّمَ اَشَدُّ حَرًّا ؕ لَوْ كَانُوْا یَفْقَهُوْنَ ۟ فَلۡيَضۡحَكُواْ قَلِيلٗا وَلۡيَبۡكُواْ كَثِيرٗا جَزَآءَۢ بِمَا كَانُواْ يَكۡسِبُونَ ۝٨٢ فَلۡیَضۡحَكُوا۟ قَلِیلࣰا مِّنۡهُمۡ فَٱسۡتَـٔۡذَنُوكَ لِلۡخُرُوجِ فَقُل لَّن تَخۡرُجُواْ مَعِيَ أَبَدٗا وَلَن تُقَٰتِلُواْ مَعِيَ عَدُوًّاۖ إِنَّكُمۡ رَضِيتُم بِٱلۡقُعُودِ أَوَّلَ مَرَّةٖ فَٱقۡعُدُواْ مَعَ ٱلۡخَٰلِفِينَ ۝٨٣ مَرَّةࣲ فَٱقۡعُدُوا۟ مَعَ ٱلۡخَٰلِفِینَ ۝٨٣ فَاِنْ رَّجَعَكَ اللّٰهُ اِلٰی طَآىِٕفَةٍ مِّنْهُمْ فَاسْتَاْذَنُوْكَ لِلْخُرُوْجِ فَقُلْ لَّنْ تَخْرُجُوْا مَعِیَ اَبَدًا وَّلَنْ تُقَاتِلُوْا مَعِیَ عَدُوًّا ؕ اِنَّكُمْ رَضِیْتُمْ بِالْقُعُوْدِ اَوَّلَ مَرَّةٍ فَاقْعُدُوْا مَعَ الْخٰلِفِیْنَ ۟ وَلَا تُصَلِّ عَلَىٰٓ أَحَدٖ مِّنۡهُم مَّاتَ أَبَدٗا وَلَا تَقُمۡ عَلَىٰ قَبۡرِهِۦٓۖ إِنَّهُمۡ وَمَاتُوا۟ وَهُمۡ فَٰسِقُونَ ۝٨٤ وَلَا تُصَلِّ عَلٰۤی اَحَدٍ مِّنْهُمْ مَّاتَ اَبَدًا وَّلَا تَقُمْ عَلٰی قَبْرِهٖ ؕ اِنَّهُمْ كَفَرُوْا بِاللّٰهِ وَرَسُوْلِهٖ وَمَاتُوْا وَهُمْ فٰسِقُوْنَ ۟ وَلَا ٱلَّذِينَ إِذَا مَآ أَتَوۡكَ لِتَحۡمِلَهُمۡ قُلۡتَ لَآ أَجِدُ مَآ أَحۡمِلُكُمۡ عَلَيۡهِ تَوَلَّواْ وَّأَعۡيُنُهُمۡ تَفِيضُ مِنَ ٱلدَّمۡعِ حَزَنًا أَلَّا يَجِدُواْ مَا يُنفِقُونَ ۝٩٢ وَلَا عَلَى ٱلَّذِینَ إِذَا مَاۤ أَتَوۡكَ لِتَحۡمِلَهُمۡ قُلۡتَ لَاۤ أَجِدُ مَاۤ أَحۡمِلُكُمۡ عَلَیۡهِ تَوَلَّوا۟ وَّأَعۡیُنُهُمۡ تَفِیضُ مِنَ ٱلدَّمۡعِ حَزَنًا أَلَّا یَجِدُوا۟ مَا یُنفِقُونَ ۝٩٢ وَّلَا عَلَی الَّذِیْنَ اِذَا مَاۤ اَتَوْكَ لِتَحْمِلَهُمْ قُلْتَ لَاۤ اَجِدُ مَاۤ اَحْمِلُكُمْ عَلَیْهِ ۪ تَوَلَّوْا وَّاَعْیُنُهُمْ تَفِیْضُ مِنَ الدَّمْعِ حَزَنًا اَلَّا یَجِدُوْا مَا یُنْفِقُوْنَ ۟ؕ إِنَّمَا ٱلسَّبِيلُ عَلَى ٱلَّذِينَ يَسۡتَـٔۡذِنُونَكَ وَهُمۡ أَغۡنِيَآءُۚ رَضُواْ بِأَن يَكُونُواْ مَعَ ٱلۡخَوَالِفِ وَطَبَعَ ٱللَّهُ عَلَىٰ قُلُوبِهِمۡ فَهُمۡ لَا يَعۡلَمُونَ ۝٩٣ إِنَّمَا ٱلسَّبِیلُ عَلَى ٱلَّذِینَ یَسۡتَءۡذِنُونَكَ وَهُمۡ أَغۡنِیَاۤءُۚ رَضُوا۟ بِأَن یَكُونُوا۟ مَعَ ٱلۡخَوَالِفِ وَطَبَعَ ٱللَّهُ عَلَىٰ قُلُوبِهِمۡ فَهُمۡ لَا یَعۡلَمُونَ ۝٩٣ اِنَّمَا السَّبِیْلُ عَلَی الَّذِیْنَ یَسْتَاْذِنُوْنَكَ وَهُمْ اَغْنِیَآءُ ۚ رَضُوْا بِاَنْ یَّكُوْنُوْا مَعَ الْخَوَالِفِ ۙ وَطَبَعَ اللّٰهُ عَلٰی قُلُوْبِهِمْ فَهُمْ لَا یَعْلَمُوْنَ ۟ يَعۡتَذِرُونَ إِلَيۡكُمۡ إِذَا رَجَعۡتُمۡ إِلَيۡهِمۡۚ قُل لَّا تَعۡتَذِرُواْ لَن نُّؤۡمِنَ لَكُمۡ قَدۡ نَبَّأَنَا ٱللَّهُ مِنۡ أَخۡبَارِكُمۡۚ وَسَيَرَى ٱللَّهُ عَمَلَكُمۡ وَرَسُولُهُۥ ثُمَّ تُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلۡغَيۡبِ وَٱلشَّهَٰدَةِ فَيُنَبِّئُكُم بِمَا كُنتُمۡ تَعۡمَلُونَ ۝٩٤ یَعۡتَذِرُونَ إِلَیۡكُمۡ إِذَا رَجَعۡتُمۡ إِلَیۡهِمۡۚ قُل لَّا تَعۡتَذِرُوا۟ لَن نُّؤۡمِنَ لَكُمۡ قَدۡ نَبَّأَنَا ٱللَّهُ مِنۡ أَخۡبَارِكُمۡۚ وَسَیَرَى ٱللَّهُ عَمَلَكُمۡ وَرَسُولُهُۥ ثُمَّ تُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلۡغَیۡبِ وَٱلشَّهَٰدَةِ فَیُنَبِّئُكُم بِمَا كُنتُمۡ تَعۡمَلُونَ ۝٩٤ یَعْتَذِرُوْنَ اِلَیْكُمْ اِذَا رَجَعْتُمْ اِلَیْهِمْ ؕ قُلْ لَّا تَعْتَذِرُوْا لَنْ نُّؤْمِنَ لَكُمْ قَدْ نَبَّاَنَا اللّٰهُ مِنْ اَخْبَارِكُمْ ؕ وَسَیَرَی اللّٰهُ عَمَلَكُمْ وَرَسُوْلُهٗ ثُمَّ تُرَدُّوْنَ اِلٰی عٰلِمِ الْغَیْبِ وَالشَّهَادَةِ فَیُنَبِّئُكُمْ بِمَا كُنْتُمْ تَعْمَلُوْنَ ۟ سَيَحۡلِفُونَ بِٱللَّهِ لَكُمۡ إِذَا ٱنقَلَبۡتُمۡ إِلَيۡهِمۡ لِتُعۡرِضُواْ عَنۡهُمۡۖ فَأَعۡرِضُواْ عَنۡهُمۡۖ إِنَّهُمۡ رِجۡسٞۖ وَمَأۡوَىٰهُمۡ جَهَنَّمُ جَزَآءَۢ بِمَا كَانُواْ يَكۡسِبُونَ ۝٩٥ سَیَحۡلِفُونَ بِٱللَّهِ لَكُمۡ إِذَا ٱنقَلَبۡتُمۡ إِلَیۡهِمۡ لِتُعۡرِضُوا۟ عَنۡهُمۡۖ فَأَعۡرِضُوا۟ عَنۡهُمۡۖ إِنَّهُمۡ رِجۡسࣱۖ وَمَأۡوَىٰهُمۡ جَهَنَّمُ جَزَاۤءَۢ بِمَا كَانُوا۟ یَكۡسِبُونَ ۝٩٥ سَیَحْلِفُوْنَ بِاللّٰهِ لَكُمْ اِذَا انْقَلَبْتُمْ اِلَیْهِمْ لِتُعْرِضُوْا عَنْهُمْ ؕ فَاَعْرِضُوْا عَنْهُمْ ؕ اِنَّهُمْ رِجْسٌ ؗ وَّمَاْوٰىهُمْ جَهَنَّمُ ۚ جَزَآءً بِمَا كَانُوْا یَكْسِبُوْنَ ۟ يَحۡلِفُونَ لَكُمۡ لِتَرۡضَوۡاْ عَنۡهُمۡۖ فَإِن تَرۡضَوۡاْ عَنۡهُمۡ فَإِنَّ ٱللَّهَ لَا يَرۡضَىٰ عَنِ ٱلۡقَوۡمِ ٱلۡفَٰسِقِينَ ۝٩٦ یَحۡلِفُونَ لَكُمۡ لِتَرۡضَوۡا۟ عَنۡهُمۡۖ فَإِن تَرۡضَوۡا۟ عَنۡهُمۡ فَإِنَّ ٱللَّهَ لَا یَرۡضَىٰ عَنِ ٱلۡقَوۡمِ ٱلۡفَٰسِقِینَ ۝٩٦ یَحْلِفُوْنَ لَكُمْ لِتَرْضَوْا عَنْهُمْ ۚ فَاِنْ تَرْضَوْا عَنْهُمْ فَاِنَّ اللّٰهَ لَا یَرْضٰی عَنِ الْقَوْمِ الْفٰسِقِیْنَ ۟ ٱلۡأَعۡرَابُ أَشَدُّ كُفۡرٗا وَنِفَاقٗا وَأَجۡدَرُ أَلَّا يَعۡلَمُواْ حُدُودَ مَآ أَنزَلَ ٱللَّهُ عَلَىٰ رَسُولِهِۦۗ وَٱللَّهُ عَلِيمٌ حَكِيمٞ ۝٩٧ ٱلۡأَعۡرَابُ أَشَدُّ كُفۡرࣰا وَنِفَاقࣰا وَأَجۡدَرُ أَلَّا یَعۡلَمُوا۟ حُدُودَ مَاۤ أَنزَلَ ٱللَّهُ عَلَىٰ رَسُولِهِۦۗ وَٱللَّهُ عَلِیمٌ حَكِیمࣱ ۝٩٧ اَلْاَعْرَابُ اَشَدُّ كُفْرًا وَّنِفَاقًا وَّاَجْدَرُ اَلَّا یَعْلَمُوْا حُدُوْدَ مَاۤ اَنْزَلَ اللّٰهُ عَلٰی رَسُوْلِهٖ ؕ وَاللّٰهُ عَلِیْمٌ حَكِیْمٌ ۟ وَمِنَ ٱلۡأَعۡرَابِ مَن يَتَّخِذُ مَا يُنفِقُ مَغۡرَمٗا وَيَتَرَبَّصُ بِكُمُ ٱلدَّوَآئِرَۚ عَلَيۡهِمۡ دَآئِرَةُ ٱلسَّوۡءِۗ وَٱللَّهُ سَمِيعٌ عَلِيمٞ ۝٩٨ وَمِنَ ٱلۡأَعۡرَابِ مَن یَتَّخِذُ مَا یُنفِقُ مَغۡرَمࣰا وَیَتَرَبَّصُ بِكُمُ ٱلدَّوَاۤئِرَۚ عَلَیۡهِمۡ دَاۤئِرَةُ ٱلسَّوۡءِۗ وَٱللَّهُ سَمِیعٌ عَلِیمࣱ ۝٩٨ وَمِنَ الْاَعْرَابِ مَنْ یَّتَّخِذُ مَا یُنْفِقُ مَغْرَمًا وَّیَتَرَبَّصُ بِكُمُ الدَّوَآىِٕرَ ؕ عَلَیْهِمْ دَآىِٕرَةُ السَّوْءِ ؕ وَاللّٰهُ سَمِیْعٌ عَلِیْمٌ ۟ وَمِنَ ٱلۡأَعۡرَابِ مَن يُؤۡمِنُ بِٱللَّهِ وَٱلۡيَوۡمِ ٱلۡأٓخِرِ وَيَتَّخِذُ مَا يُنفِقُ قُرُبَٰتٍ عِندَ ٱللَّهِ وَصَلَوَٰتِ ٱلرَّسُولِۚ أَلَآ إِنَّهَا قُرۡبَةٞ لَّهُمۡۚ سَيُدۡخِلُهُمُ ٱللَّهُ فِي رَحۡمَتِهِۦٓۚ إِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٞ ۝٩٩ وَمِنَ ٱلۡأَعۡرَابِ مَن یُؤۡمِنُ بِٱللَّهِ وَٱلۡیَوۡمِ ٱلۡءَاخِرِ وَیَتَّخِذُ مَا یُنفِقُ قُرُبَٰتٍ عِندَ ٱللَّهِ وَصَلَوَٰتِ ٱلرَّسُولِۚ أَلَاۤ إِنَّهَا قُرۡبَةࣱ لَّهُمۡۚ سَیُدۡخِلُهُمُ ٱللَّهُ فِی رَحۡمَتِهِۦۤۚ إِنَّ ٱللَّهَ غَفُورࣱ رَّحِیمࣱ ۝٩٩ وَمِنَ الْاَعْرَابِ مَنْ یُّؤْمِنُ بِاللّٰهِ وَالْیَوْمِ الْاٰخِرِ وَیَتَّخِذُ مَا یُنْفِقُ قُرُبٰتٍ عِنْدَ اللّٰهِ وَصَلَوٰتِ الرَّسُوْلِ ؕ اَلَاۤ اِنَّهَا قُرْبَةٌ لَّهُمْ ؕ سَیُدْخِلُهُمُ اللّٰهُ فِیْ رَحْمَتِهٖ ؕ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِیْمٌ ۟۠ وَٱلسَّٰبِقُونَ ٱلۡأَوَّلُونَ مِنَ ٱلۡمُهَٰجِرِينَ وَٱلۡأَنصَارِ وَٱلَّذِينَ ٱتَّبَعُوهُم بِإِحۡسَٰنٖ رَّضِيَ ٱللَّهُ عَنۡهُمۡ وَرَضُواْ عَنۡهُ وَأَعَدَّ لَهُمۡ جَنَّٰتٖ تَجۡرِي تَحۡتَهَا ٱلۡأَنۡهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدٗاۚ ذَٰلِكَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ ۝١٠٠ وَٱلسَّٰبِقُونَ ٱلۡأَوَّلُونَ مِنَ ٱلۡمُهَٰجِرِینَ وَٱلۡأَنصَارِ وَٱلَّذِینَ ٱتَّبَعُوهُم بِإِحۡسَٰنࣲ رَّضِیَ ٱللَّهُ عَنۡهُمۡ وَرَضُوا۟ عَنۡهُ وَأَعَدَّ لَهُمۡ جَنَّٰتࣲ تَجۡرِی تَحۡتَهَا ٱلۡأَنۡهَٰرُ خَٰلِدِینَ فِیهَاۤ أَبَدࣰاۚ ذَٰلِكَ ٱلۡفَوۡزُ ٱلۡعَظِیمُ ۝١٠٠ وَالسّٰبِقُوْنَ الْاَوَّلُوْنَ مِنَ الْمُهٰجِرِیْنَ وَالْاَنْصَارِ وَالَّذِیْنَ اتَّبَعُوْهُمْ بِاِحْسَانٍ ۙ رَّضِیَ اللّٰهُ عَنْهُمْ وَرَضُوْا عَنْهُ وَاَعَدَّ لَهُمْ جَنّٰتٍ تَجْرِیْ تَحْتَهَا الْاَنْهٰرُ خٰلِدِیْنَ فِیْهَاۤ اَبَدًا ؕ ذٰلِكَ الْفَوْزُ الْعَظِیْمُ ۟ وَمِمَّنۡ حَوۡلَكُم مِّنَ ٱلۡأَعۡرَابِ مُنَٰفِقُونَۖ وَمِنۡ أَهۡلِ ٱلۡمَدِينَةِ مَرَدُواْ عَلَى ٱلنِّفَاقِ لَا تَعۡلَمُهُمۡۖ نَحۡنُ نَعۡلَمُهُمۡۚ سَنُعَذِّبُهُم مَّرَّتَيۡنِ ثُمَّ يُرَدُّونَ إِلَىٰ عَذَابٍ عَظِيمٖ ۝١٠١ وَمِمَّنۡ حَوۡلَكُم مِّنَ ٱلۡأَعۡرَابِ مُنَٰفِقُونَۖ وَمِنۡ أَهۡلِ ٱلۡمَدِینَةِ مَرَدُوا۟ عَلَى ٱلنِّفَاقِ لَا تَعۡلَمُهُمۡۖ نَحۡنُ نَعۡلَمُهُمۡۚ سَنُعَذِّبُهُم مَّرَّتَیۡنِ ثُمَّ یُرَدُّونَ إِلَىٰ عَذَابٍ عَظِیمࣲ ۝١٠١ وَمِمَّنْ حَوْلَكُمْ مِّنَ الْاَعْرَابِ مُنٰفِقُوْنَ ۛؕ وَمِنْ اَهْلِ الْمَدِیْنَةِ ؔۛ۫ مَرَدُوْا عَلَی النِّفَاقِ ۫ لَا تَعْلَمُهُمْ ؕ نَحْنُ نَعْلَمُهُمْ ؕ سَنُعَذِّبُهُمْ مَّرَّتَیْنِ ثُمَّ یُرَدُّوْنَ اِلٰی عَذَابٍ عَظِیْمٍ ۟ۚ وَءَاخَرُونَ ٱعۡتَرَفُواْ بِذُنُوبِهِمۡ خَلَطُواْ عَمَلٗا صَٰلِحٗا وَءَاخَرَ سَيِّئًا عَسَى ٱللَّهُ أَن يَتُوبَ عَلَيۡهِمۡۚ إِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٌ ۝١٠٢ وَءَاخَرُونَ ٱعۡتَرَفُوا۟ بِذُنُوبِهِمۡ خَلَطُوا۟ عَمَلࣰا صَٰلِحࣰا وَءَاخَرَ سَیِّئًا عَسَى ٱللَّهُ أَن یَتُوبَ عَلَیۡهِمۡۚ إِنَّ ٱللَّهَ غَفُورࣱ رَّحِیمٌ ۝١٠٢ وَاٰخَرُوْنَ اعْتَرَفُوْا بِذُنُوْبِهِمْ خَلَطُوْا عَمَلًا صَالِحًا وَّاٰخَرَ سَیِّئًا ؕ عَسَی اللّٰهُ اَنْ یَّتُوْبَ عَلَیْهِمْ ؕ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِیْمٌ ۟ خُذۡ مِنۡ أَمۡوَٰلِهِمۡ صَدَقَةٗ تُطَهِّرُهُمۡ وَتُزَكِّيهِم بِهَا وَصَلِّ عَلَيۡهِمۡۖ إِنَّ صَلَوٰتَكَ سَكَنٞ لَّهُمۡۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌ ۝١٠٣ خُذۡ مِنۡ أَمۡوَٰلِهِمۡ صَدَقَةࣰ تُطَهِّرُهُمۡ وَتُزَكِّیهِم بِهَا وَصَلِّ عَلَیۡهِمۡۖ إِنَّ صَلَوٰتَكَ سَكَنࣱ لَّهُمۡۗ وَٱللَّهُ سَمِیعٌ عَلِیمٌ ۝١٠٣ خُذْ مِنْ اَمْوَالِهِمْ صَدَقَةً تُطَهِّرُهُمْ وَتُزَكِّیْهِمْ بِهَا وَصَلِّ عَلَیْهِمْ ؕ اِنَّ صَلٰوتَكَ سَكَنٌ لَّهُمْ ؕ وَاللّٰهُ سَمِیْعٌ عَلِیْمٌ ۟ أَلَمۡ يَعۡلَمُوٓاْ أَنَّ ٱللَّهَ هُوَ يَقۡبَلُ ٱلتَّوۡبَةَ عَنۡ عِبَادِهِۦ وَيَأۡخُذُ ٱلصَّدَقَٰتِ وَأَنَّ ٱللَّهَ هُوَ ٱلتَّوَّابُ ٱلرَّحِيمُ ۝١٠٤ أَلَمۡ یَعۡلَمُوۤا۟ أَنَّ ٱللَّهَ هُوَ یَقۡبَلُ ٱلتَّوۡبَةَ عَنۡ عِبَادِهِۦ وَیَأۡخُذُ ٱلصَّدَقَٰتِ وَأَنَّ ٱللَّهَ هُوَ ٱلتَّوَّابُ ٱلرَّحِیمُ ۝١٠٤ اَلَمْ یَعْلَمُوْۤا اَنَّ اللّٰهَ هُوَ یَقْبَلُ التَّوْبَةَ عَنْ عِبَادِهٖ وَیَاْخُذُ الصَّدَقٰتِ وَاَنَّ اللّٰهَ هُوَ التَّوَّابُ الرَّحِیْمُ ۟ وَقُلِ ٱعۡمَلُواْ فَسَيَرَى ٱللَّهُ عَمَلَكُمۡ وَرَسُولُهُۥ وَٱلۡمُؤۡمِنُونَۖ وَسَتُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلۡغَيۡبِ وَٱلشَّهَٰدَةِ فَيُنَبِّئُكُم بِمَا كُنتُمۡ تَعۡمَلُونَ ۝١٠٥ وَقُلِ ٱعۡمَلُوا۟ فَسَیَرَى ٱللَّهُ عَمَلَكُمۡ وَرَسُولُهُۥ وَٱلۡمُؤۡمِنُونَۖ وَسَتُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلۡغَیۡبِ وَٱلشَّهَٰدَةِ فَیُنَبِّئُكُم بِمَا كُنتُمۡ تَعۡمَلُونَ ۝١٠٥ وَقُلِ اعْمَلُوْا فَسَیَرَی اللّٰهُ عَمَلَكُمْ وَرَسُوْلُهٗ وَالْمُؤْمِنُوْنَ ؕ وَسَتُرَدُّوْنَ اِلٰی عٰلِمِ الْغَیْبِ وَالشَّهَادَةِ فَیُنَبِّئُكُمْ بِمَا كُنْتُمْ تَعْمَلُوْنَ ۟ۚ وَءَاخَرُونَ مُرۡجَوۡنَ لِأَمۡرِ ٱللَّهِ إِمَّا يُعَذِّبُهُمۡ وَإِمَّا يَتُوبُ عَلَيۡهِمۡۗ وَٱللَّهُ عَلِيمٌ حَكِيمٞ ۝١٠٦ وَءَاخَرُونَ مُرۡجَوۡنَ لِأَمۡرِ ٱللَّهِ إِمَّا یُعَذِّبُهُمۡ وَإِمَّا یَتُوبُ عَلَیۡهِمۡۗ وَٱللَّهُ عَلِیمٌ حَكِیمࣱ ۝١٠٦ وَاٰخَرُوْنَ مُرْجَوْنَ لِاَمْرِ اللّٰهِ اِمَّا یُعَذِّبُهُمْ وَاِمَّا یَتُوْبُ عَلَیْهِمْ ؕ وَاللّٰهُ عَلِیْمٌ حَكِیْمٌ ۟ وَٱلَّذِينَ ٱتَّخَذُواْ مَسۡجِدٗا ضِرَارٗا وَكُفۡرٗا وَتَفۡرِيقَۢا بَيۡنَ ٱلۡمُؤۡمِنِينَ وَإِرۡصَادٗا لِّمَنۡ حَارَبَ ٱللَّهَ وَرَسُولَهُۥ مِن قَبۡلُۚ وَلَيَحۡلِفُنَّ إِنۡ أَرَدۡنَآ إِلَّا ٱلۡحُسۡنَىٰۖ وَٱللَّهُ يَشۡهَدُ إِنَّهُمۡ لَكَٰذِبُونَ ۝١٠٧ وَٱلَّذِینَ ٱتَّخَذُوا۟ مَسۡجِدࣰا ضِرَارࣰا وَكُفۡرࣰا وَتَفۡرِیقَۢا بَیۡنَ ٱلۡمُؤۡمِنِینَ وَإِرۡصَادࣰا لِّمَنۡ حَارَبَ ٱللَّهَ وَرَسُولَهُۥ مِن قَبۡلُۚ وَلَیَحۡلِفُنَّ إِنۡ أَرَدۡنَاۤ إِلَّا ٱلۡحُسۡنَىٰۖ وَٱللَّهُ یَشۡهَدُ إِنَّهُمۡ لَكَٰذِبُونَ ۝١٠٧ وَالَّذِیْنَ اتَّخَذُوْا مَسْجِدًا ضِرَارًا وَّكُفْرًا وَّتَفْرِیْقًا بَیْنَ الْمُؤْمِنِیْنَ وَاِرْصَادًا لِّمَنْ حَارَبَ اللّٰهَ وَرَسُوْلَهٗ مِنْ قَبْلُ ؕ وَلَیَحْلِفُنَّ اِنْ اَرَدْنَاۤ اِلَّا الْحُسْنٰی ؕ وَاللّٰهُ یَشْهَدُ اِنَّهُمْ لَكٰذِبُوْنَ ۟ لَا تَقُمۡ فِيهِ أَبَدٗاۚ لَّمَسۡجِدٌ أُسِّسَ عَلَى ٱلتَّقۡوَىٰ مِنۡ أَوَّلِ يَوۡمٍ أَحَقُّ أَن تَقُومَ فِيهِۚ فِيهِ رِجَالٞ يُحِبُّونَ أَن يَتَطَهَّرُواْۚ وَٱللَّهُ يُحِبُّ ٱلۡمُطَّهِّرِينَ ۝١٠٨ لَا تَقُمۡ فِیهِ أَبَدࣰاۚ لَّمَسۡجِدٌ أُسِّسَ عَلَى ٱلتَّقۡوَىٰ مِنۡ أَوَّلِ یَوۡمٍ أَحَقُّ أَن تَقُومَ فِیهِۚ فِیهِ رِجَالࣱ یُحِبُّونَ أَن یَتَطَهَّرُوا۟ۚ وَٱللَّهُ یُحِبُّ ٱلۡمُطَّهِّرِینَ ۝١٠٨ لَا تَقُمْ فِیْهِ اَبَدًا ؕ لَمَسْجِدٌ اُسِّسَ عَلَی التَّقْوٰی مِنْ اَوَّلِ یَوْمٍ اَحَقُّ اَنْ تَقُوْمَ فِیْهِ ؕ فِیْهِ رِجَالٌ یُّحِبُّوْنَ اَنْ یَّتَطَهَّرُوْا ؕ وَاللّٰهُ یُحِبُّ الْمُطَّهِّرِیْنَ ۟ أَفَمَنۡ أَسَّسَ بُنۡيَٰنَهُۥ عَلَىٰ تَقۡوَىٰ مِنَ ٱللَّهِ وَرِضۡوَٰنٍ خَيۡرٌ أَم مَّنۡ أَسَّسَ بُنۡيَٰنَهُۥ عَلَىٰ شَفَا جُرُفٍ هَارٖ فَٱنۡهَارَ بِهِۦ فِي نَارِ جَهَنَّمَۗ وَٱللَّهُ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلظَّٰلِمِينَ ۝١٠٩ أَفَمَنۡ أَسَّسَ بُنۡیَٰنَهُۥ عَلَىٰ تَقۡوَىٰ مِنَ ٱللَّهِ وَرِضۡوَٰنٍ خَیۡرٌ أَم مَّنۡ أَسَّسَ بُنۡیَٰنَهُۥ عَلَىٰ شَفَا جُرُفٍ هَارࣲ فَٱنۡهَارَ بِهِۦ فِی نَارِ جَهَنَّمَۗ وَٱللَّهُ لَا یَهۡدِی ٱلۡقَوۡمَ ٱلظَّٰلِمِینَ ۝١٠٩ اَفَمَنْ اَسَّسَ بُنْیَانَهٗ عَلٰی تَقْوٰی مِنَ اللّٰهِ وَرِضْوَانٍ خَیْرٌ اَمْ مَّنْ اَسَّسَ بُنْیَانَهٗ عَلٰی شَفَا جُرُفٍ هَارٍ فَانْهَارَ بِهٖ فِیْ نَارِ جَهَنَّمَ ؕ وَاللّٰهُ لَا یَهْدِی الْقَوْمَ الظّٰلِمِیْنَ قُلُوبِهِمۡ إِلَّاۤ أَن تَقَطَّعَ قُلُوبُهُمۡۗ وَٱللَّهُ عَلِیمٌ حَكِیمٌ ۝١١٠ لَا یَزَالُ بُنْیَانُهُمُ الَّذِیْ بَنَوْا رِیْبَةً فِیْ قُلُوْبِهِمْ اِلَّاۤ اَنْ تَقَطَّعَ قُلُوْبُهُمْ ؕ وَاللّٰهُ عَلِیْمٌ حَكِیْمٌ ۟۠ إِنَّ ٱللَّهَ ٱشۡتَرَىٰ مِنَ ٱلۡمُؤۡمِنِينَ أَنفُسَهُمۡ وَأَمۡوَٰلَهُم بِأَنَّ لَهُمُ ٱلۡجَنَّةَۚ يُقَٰتِلُونَ فِي سَبِيلِ ٱللَّهِ فَيَقۡتُلُونَ وَيُقۡتَلُونَۖ وَعۡدًا عَلَيۡهِ حَقّٗا فِي ٱلتَّوۡرَىٰةِ وَٱلۡإِنجِيلِ وَٱلۡقُرۡءَانِۚ وَمَنۡ أَوۡفَىٰ بِعَهۡدِهِۦ مِنَ ٱللَّهِۚ فَٱسۡتَبۡشِرُواْ بِبَيۡعِكُمُ ٱلَّذِي بَايَعۡتُم بِهِۦۚ وَذَٰلِكَ هُوَ ٱلۡفَوۡزُ ٱلۡعَظِيمُ ۝١١١ إِنَّ ٱلۡحَٰمِدُونَ ٱلسَّٰۤئِحُونَ ٱلرَّٰكِعُونَ ٱلسَّٰجِدُونَ ٱلۡءَامِرُونَ بِٱلۡمَعۡرُوفِ وَٱلنَّاهُونَ عَنِ ٱلۡمُنكَرِ وَٱلۡحَٰفِظُونَ لِحُدُودِ ٱللَّهِۗ وَبَشِّرِ ٱلۡمُؤۡمِنِینَ ۝١١٢ اَلتَّآىِٕبُوْنَ الْعٰبِدُوْنَ الْحٰمِدُوْنَ السَّآىِٕحُوْنَ الرّٰكِعُوْنَ السّٰجِدُوْنَ الْاٰمِرُوْنَ بِالْمَعْرُوْفِ وَالنَّاهُوْنَ عَنِ الْمُنْكَرِ وَالْحٰفِظُوْنَ لِحُدُوْدِ اللّٰهِ ؕ وَبَشِّرِ الْمُؤْمِنِیْنَ ۟ مَا كَانَ لِلنَّبِيِّ وَٱلَّذِينَ ءَامَنُوٓاْ أَن يَسۡتَغۡفِرُواْ لِلۡمُشۡرِكِينَ وَلَوۡ كَانُوٓاْ أُوْلِي قُرۡبَىٰ مِنۢ بَعۡدِ مَا تَبَيَّنَ لَهُمۡ أَنَّهُمۡ أَصۡحَٰبُ ٱلۡجَحِيمِ ۝١١٣ مَا كَانَ لِلنَّبِیِّ وَٱلَّذِینَ ءَامَنُوۤا۟ أَن یَسۡتَغۡفِرُوا۟ لِلۡمُشۡرِكِینَ وَلَوۡ كَانُوۤا۟ أُو۟لِی قُرۡبَىٰ مِنۢ بَعۡدِ مَا تَبَیَّنَ لَهُمۡ أَنَّهُمۡ أَصۡحَٰبُ ٱلۡجَحِیمِ ۝١١٣ مَا كَانَ لِلنَّبِیِّ وَالَّذِیْنَ اٰمَنُوْۤا اَنْ یَّسْتَغْفِرُوْا لِلْمُشْرِكِیْنَ وَلَوْ كَانُوْۤا اُولِیْ قُرْبٰی مِنْ بَعْدِ مَا تَبَیَّنَ لَهُمْ اَنَّهُمْ اَصْحٰبُ الْجَحِیْمِ ۟ وَمَا كَانَ ٱسۡتِغۡفَارُ إِبۡرَٰهِيمَ لِأَبِيهِ إِلَّا عَن مَّوۡعِدَةٖ وَعَدَهَآ إِيَّاهُ فَلَمَّا تَبَيَّنَ لَهُۥٓ أَنَّهُۥ عَدُوّٞ لِّلَّهِ تَبَرَّأَ مِنۡهُۚ إِنَّ إِبۡرَٰهِيمَ لَأَوَّٰهٌ حَلِيمٞ ۝١١٤ وَمَا كَانَ ٱسۡتِغۡفَارُ إِبۡرَٰهِیمَ لِأَبِیهِ إِلَّا عَن مَّوۡعِدَةࣲ وَعَدَهَاۤ إِیَّاهُ فَلَمَّا تَبَیَّنَ لَهُۥۤ أَنَّهُۥ عَدُوࣱّ لِّلَّهِ تَبَرَّأَ مِنۡهُۚ إِنَّ إِبۡرَٰهِیمَ لَأَوَّٰهٌ حَلِیمࣱ ۝١١٤ وَمَا كَانَ اسْتِغْفَارُ اِبْرٰهِیْمَ لِاَبِیْهِ اِلَّا عَنْ مَّوْعِدَةٍ وَّعَدَهَاۤ اِیَّاهُ ۚ فَلَمَّا تَبَیَّنَ لَهٗۤ اَنَّهٗ عَدُوٌّ لِّلّٰهِ تَبَرَّاَ مِنْهُ ؕ اِنَّ اِبْرٰهِیْمَ لَاَوَّاهٌ حَلِیْمٌ ۟ وَمَا كَانَ ٱللَّهُ لِيُضِلَّ قَوۡمَۢا بَعۡدَ إِذۡ هَدَىٰهُمۡ حَتَّىٰ يُبَيِّنَ لَهُم مَّا يَتَّقُونَۚ إِنَّ ٱللَّهَ بِكُلِّ شَيۡءٍ عَلِيمٌ ۝١١٥ وَمَا كَانَ ٱللَّهُ لِیُضِلَّ قَوۡمَۢا بَعۡدَ إِذۡ هَدَىٰهُمۡ حَتَّىٰ یُبَیِّنَ لَهُم مَّا یَتَّقُونَۚ إِنَّ ٱللَّهَ بِكُلِّ شَیۡءٍ عَلِیمٌ ۝١١٥ وَمَا كَانَ اللّٰهُ لِیُضِلَّ قَوْمًا بَعْدَ اِذْ هَدٰىهُمْ حَتّٰی یُبَیِّنَ لَهُمْ مَّا یَتَّقُوْنَ ؕ اِنَّ اللّٰهَ بِكُلِّ شَیْءٍ عَلِیْمٌ ۟ إِنَّ ٱللَّهَ لَهُۥ مُلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۖ يُحۡيِۦ وَيُمِيتُۚ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِيّٖ وَلَا نَصِيرٖ ۝١١٦ إِنَّ ٱللَّهَ لَهُۥ مُلۡكُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۖ یُحۡیِۦ وَیُمِیتُۚ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِیࣲّ وَلَا نَصِیرࣲ ۝١١٦ اِنَّ اللّٰهَ لَهٗ مُلْكُ السَّمٰوٰتِ وَالْاَرْضِ ؕ یُحْیٖ وَیُمِیْتُ ؕ وَمَا لَكُمْ مِّنْ دُوْنِ اللّٰهِ مِنْ وَّلِیٍّ وَّلَا نَصِیْرٍ ۟ لَّقَد تَّابَ ٱللَّهُ عَلَى ٱلنَّبِيِّ وَٱلۡمُهَٰجِرِينَ وَٱلۡأَنصَارِ ٱلَّذِينَ ٱتَّبَعُوهُ فِي سَاعَةِ ٱلۡعُسۡرَةِ مِنۢ بَعۡدِ مَا كَادَ يَزِيغُ قُلُوبُ فَرِيقٖ مِّنۡهُمۡ ثُمَّ تَابَ عَلَيۡهِمۡۚ إِنَّهُۥ بِهِمۡ رَءُوفٞ رَّحِيمٞ ۝١١٧ لَّقَد تَّابَ ٱللَّهُ عَلَى ٱلنَّبِیِّ وَٱلۡمُهَٰجِرِینَ وَٱلۡأَنصَارِ ٱلَّذِینَ ٱتَّبَعُوهُ فِی سَاعَةِ ٱلۡعُسۡرَةِ مِنۢ بَعۡدِ مَا كَادَ یَزِیغُ قُلُوبُ فَرِیقࣲ مِّنۡهُمۡ ثُمَّ تَابَ عَلَیۡهِمۡۚ إِنَّهُۥ بِهِمۡ رَءُوفࣱ رَّحِیمࣱ ۝١١٧ لَقَدْ تَّابَ اللّٰهُ عَلَی النَّبِیِّ وَالْمُهٰجِرِیْنَ وَالْاَنْصَارِ الَّذِیْنَ اتَّبَعُوْهُ فِیْ سَاعَةِ الْعُسْرَةِ مِنْ بَعْدِ مَا كَادَ یَزِیْغُ قُلُوْبُ فَرِیْقٍ مِّنْهُمْ ثُمَّ تَابَ عَلَیْهِمْ ؕ اِنَّهٗ بِهِمْ رَءُوْفٌ رَّحِیْمٌ ۟ۙ وَعَلَى ٱلثَّلَٰثَةِ ٱلَّذِينَ خُلِّفُواْ حَتَّىٰٓ إِذَا ضَاقَتۡ عَلَيۡهِمُ ٱلۡأَرۡضُ بِمَا رَحُبَتۡ وَضَاقَتۡ عَلَيۡهِمۡ أَنفُسُهُمۡ وَظَنُّوٓاْ أَن لَّا مَلۡجَأَ مِنَ ٱللَّهِ إِلَّآ إِلَيۡهِ ثُمَّ تَابَ عَلَيۡهِمۡ لِيَتُوبُوٓاْۚ إِنَّ ٱللَّهَ هُوَ ٱلتَّوَّابُ ٱلرَّحِيمُ ۝١١٨ وَعَلَى ٱلثَّلَٰثَةِ ٱلَّذِینَ خُلِّفُوا۟ حَتَّىٰۤ إِذَا ضَاقَتۡ عَلَیۡهِمُ ٱلۡأَرۡضُ بِمَا رَحُبَتۡ وَضَاقَتۡ عَلَیۡهِمۡ أَنفُسُهُمۡ وَظَنُّوۤا۟ أَن لَّا مَلۡجَأَ مِنَ ٱللَّهِ إِلَّاۤ إِلَیۡهِ ثُمَّ تَابَ عَلَیۡهِمۡ لِیَتُوبُوۤا۟ۚ إِنَّ ٱللَّهَ هُوَ ٱلتَّوَّابُ ٱلرَّحِیمُ ۝١١٨ وَّعَلَی الثَّلٰثَةِ الَّذِیْنَ خُلِّفُوْا ؕ حَتّٰۤی اِذَا ضَاقَتْ عَلَیْهِمُ الْاَرْضُ بِمَا رَحُبَتْ وَضَاقَتْ عَلَیْهِمْ اَنْفُسُهُمْ وَظَنُّوْۤا اَنْ لَّا مَلْجَاَ مِنَ اللّٰهِ اِلَّاۤ اِلَیْهِ ؕ ثُمَّ تَابَ عَلَیْهِمْ لِیَتُوْبُوْا ؕ اِنَّ اللّٰهَ هُوَ التَّوَّابُ الرَّحِیْمُ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ ٱتَّقُواْ ٱللَّهَ وَكُونُواْ مَعَ ٱلصَّٰدِقِينَ ۝١١٩ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَكُونُوا۟ مَعَ ٱلصَّٰدِقِینَ ۝١١٩ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوا اتَّقُوا اللّٰهَ وَكُوْنُوْا مَعَ الصّٰدِقِیْنَ ۟ مَا كَانَ لِأَهۡلِ ٱلۡمَدِينَةِ وَمَنۡ حَوۡلَهُم مِّنَ ٱلۡأَعۡرَابِ أَن يَتَخَلَّفُواْ عَن رَّسُولِ ٱللَّهِ وَلَا يَرۡغَبُواْ بِأَنفُسِهِمۡ عَن نَّفۡسِهِۦۚ ذَٰلِكَ بِأَنَّهُمۡ لَا يُصِيبُهُمۡ ظَمَأٞ وَلَا نَصَبٞ وَلَا مَخۡمَصَةٞ فِي سَبِيلِ ٱللَّهِ وَلَا يَطَـُٔونَ مَوۡطِئٗا يَغِيظُ ٱلۡكُفَّارَ وَلَا يَنَالُونَ مِنۡ عَدُوّٖ نَّيۡلًا إِلَّا كُتِبَ لَهُم بِهِۦ عَمَلٞ صَٰلِحٌۚ إِنَّ ٱللَّهَ لَا يُضِيعُ أَجۡرَ ٱلۡمُحۡسِنِينَ ۝١٢٠ مَا كَانَ لِأَهۡلِ ٱلۡمَدِینَةِ وَمَنۡ حَوۡلَهُم مِّنَ ٱلۡأَعۡرَابِ أَن یَتَخَلَّفُوا۟ عَن رَّسُولِ ٱللَّهِ وَلَا یَرۡغَبُوا۟ بِأَنفُسِهِمۡ عَن نَّفۡسِهِۦۚ ذَٰلِكَ بِأَنَّهُمۡ لَا یُصِیبُهُمۡ ظَمَأࣱ وَلَا نَصَبࣱ وَلَا مَخۡمَصَةࣱ فِی سَبِیلِ ٱللَّهِ وَلَا یَطَءُونَ مَوۡطِئࣰا یَغِیظُ ٱلۡكُفَّارَ وَلَا یَنَالُونَ مِنۡ عَدُوࣲّ نَّیۡلًا إِلَّا كُتِبَ لَهُم بِهِۦ عَمَلࣱ صَٰلِحٌۚ إِنَّ ٱللَّهَ لَا یُضِیعُ أَجۡرَ ٱلۡمُحۡسِنِینَ ۝١٢٠ مَا كَانَ لِاَهْلِ الْمَدِیْنَةِ وَمَنْ حَوْلَهُمْ مِّنَ الْاَعْرَابِ اَنْ یَّتَخَلَّفُوْا عَنْ رَّسُوْلِ اللّٰهِ وَلَا یَرْغَبُوْا بِاَنْفُسِهِمْ عَنْ نَّفْسِهٖ ؕ ذٰلِكَ بِاَنَّهُمْ لَا یُصِیْبُهُمْ ظَمَاٌ وَّلَا نَصَبٌ وَّلَا مَخْمَصَةٌ فِیْ سَبِیْلِ اللّٰهِ وَلَا یَطَـُٔوْنَ مَوْطِئًا یَّغِیْظُ الْكُفَّارَ وَلَا یَنَالُوْنَ مِنْ عَدُوٍّ نَّیْلًا اِلَّا كُتِبَ لَهُمْ بِهٖ عَمَلٌ صَالِحٌ ؕ اِنَّ اللّٰهَ لَا یُضِیْعُ اَجْرَ الْمُحْسِنِیْنَ ۟ۙ وَلَا يُنفِقُونَ نَفَقَةٗ صَغِيرَةٗ وَلَا كَبِيرَةٗ وَلَا يَقۡطَعُونَ وَادِيًا إِلَّا كُتِبَ لَهُمۡ لِيَجۡزِيَهُمُ ٱللَّهُ أَحۡسَنَ مَا كَانُواْ يَعۡمَلُونَ ۝١٢١ وَلَا یُنفِقُونَ نَفَقَةࣰ صَغِیرَةࣰ وَلَا كَبِیرَةࣰ وَلَا یَقۡطَعُونَ وَادِیًا إِلَّا كُتِبَ لَهُمۡ لِیَجۡزِیَهُمُ ٱللَّهُ أَحۡسَنَ مَا كَانُوا۟ یَعۡمَلُونَ ۝١٢١ وَلَا یُنْفِقُوْنَ نَفَقَةً صَغِیْرَةً وَّلَا كَبِیْرَةً وَّلَا یَقْطَعُوْنَ وَادِیًا اِلَّا كُتِبَ لَهُمْ لِیَجْزِیَهُمُ اللّٰهُ اَحْسَنَ مَا كَانُوْا یَعْمَلُوْنَ ۟ وَمَا كَانَ ٱلۡمُؤۡمِنُونَ لِيَنفِرُواْ كَآفَّةٗۚ فَلَوۡلَا نَفَرَ مِن كُلِّ فِرۡقَةٖ مِّنۡهُمۡ طَآئِفَةٞ لِّيَتَفَقَّهُواْ فِي ٱلدِّينِ وَلِيُنذِرُواْ قَوۡمَهُمۡ إِذَا رَجَعُوٓاْ إِلَيۡهِمۡ لَعَلَّهُمۡ يَحۡذَرُونَ ۝١٢٢ وَمَا كَانَ ٱلۡمُؤۡمِنُونَ لِیَنفِرُوا۟ كَاۤفَّةࣰۚ فَلَوۡلَا نَفَرَ مِن كُلِّ فِرۡقَةࣲ مِّنۡهُمۡ طَاۤئِفَةࣱ لِّیَتَفَقَّهُوا۟ فِی ٱلدِّینِ وَلِیُنذِرُوا۟ قَوۡمَهُمۡ إِذَا رَجَعُوۤا۟ إِلَیۡهِمۡ لَعَلَّهُمۡ یَحۡذَرُونَ ۝١٢٢ وَمَا كَانَ الْمُؤْمِنُوْنَ لِیَنْفِرُوْا كَآفَّةً ؕ فَلَوْلَا نَفَرَ مِنْ كُلِّ فِرْقَةٍ مِّنْهُمْ طَآىِٕفَةٌ لِّیَتَفَقَّهُوْا فِی الدِّیْنِ وَلِیُنْذِرُوْا قَوْمَهُمْ اِذَا رَجَعُوْۤا اِلَیْهِمْ لَعَلَّهُمْ یَحْذَرُوْنَ ۟۠ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ قَٰتِلُواْ ٱلَّذِينَ يَلُونَكُم مِّنَ ٱلۡكُفَّارِ وَلۡيَجِدُواْ فِيكُمۡ غِلۡظَةٗۚ وَٱعۡلَمُوٓاْ أَنَّ ٱللَّهَ مَعَ ٱلۡمُتَّقِينَ ۝١٢٣ یَٰۤأَیُّهَا ٱلَّذِینَ ءَامَنُوا۟ قَٰتِلُوا۟ ٱلَّذِینَ یَلُونَكُم مِّنَ ٱلۡكُفَّارِ وَلۡیَجِدُوا۟ فِیكُمۡ غِلۡظَةࣰۚ وَٱعۡلَمُوۤا۟ أَنَّ ٱللَّهَ مَعَ ٱلۡمُتَّقِینَ ۝١٢٣ یٰۤاَیُّهَا الَّذِیْنَ اٰمَنُوْا قَاتِلُوا الَّذِیْنَ یَلُوْنَكُمْ مِّنَ الْكُفَّارِ وَلْیَجِدُوْا فِیْكُمْ غِلْظَةً ؕ وَاعْلَمُوْۤا اَنَّ اللّٰهَ مَعَ الْمُتَّقِیْنَ ۟ وَإِذَا مَآ أُنزِلَتۡ سُورَةٞ فَمِنۡهُم مَّن يَقُولُ أَيُّكُمۡ زَادَتۡهُ هَٰذِهِۦٓ إِيمَٰنٗاۚ فَأَمَّا ٱلَّذِينَ ءَامَنُواْ فَزَادَتۡهُمۡ إِيمَٰنٗا وَهُمۡ يَسۡتَبۡشِرُونَ ۝١٢٤ وَإِذَا مَاۤ أُنزِلَتۡ سُورَةࣱ فَمِنۡهُم مَّن یَقُولُ أَیُّكُمۡ زَادَتۡهُ هَٰذِهِۦۤ إِیمَٰنࣰاۚ فَأَمَّا ٱلَّذِینَ ءَامَنُوا۟ فَزَادَتۡهُمۡ إِیمَٰنࣰا وَهُمۡ یَسۡتَبۡشِرُونَ ۝١٢٤ وَاِذَا مَاۤ اُنْزِلَتْ سُوْرَةٌ فَمِنْهُمْ مَّنْ یَّقُوْلُ اَیُّكُمْ زَادَتْهُ هٰذِهٖۤ اِیْمَانًا ۚ فَاَمَّا الَّذِیْنَ اٰمَنُوْا فَزَادَتْهُمْ اِیْمَانًا وَّهُمْ یَسْتَبْشِرُوْنَ ۟ وَأَمَّا ٱلَّذِينَ فِي قُلُوبِهِم مَّرَضٞ فَزَادَتۡهُمۡ رِجۡسًا إِلَىٰ رِجۡسِهِمۡ وَمَاتُواْ وَهُمۡ كَٰفِرُونَ ۝١٢٥ وَأَمَّا ٱلَّذِینَ فِی قُلُوبِهِم مَّرَضࣱ فَزَادَتۡهُمۡ رِجۡسًا إِلَىٰ رِجۡسِهِمۡ وَمَاتُوا۟ وَهُمۡ كَٰفِرُونَ ۝١٢٥ وَاَمَّا الَّذِیْنَ فِیْ قُلُوْبِهِمْ مَّرَضٌ فَزَادَتْهُمْ رِجْسًا اِلٰی رِجْسِهِمْ وَمَاتُوْا وَهُمْ كٰفِرُوْنَ ۟ أَوَلَا يَرَوۡنَ أَنَّهُمۡ يُفۡتَنُونَ فِي كُلِّ عَامٖ مَّرَّةً أَوۡ مَرَّتَيۡنِ ثُمَّ لَا يَتُوبُونَ وَلَا هُمۡ يَذَّكَّرُونَ ۝١٢٦ أَوَلَا یَرَوۡنَ أَنَّهُمۡ یُفۡتَنُونَ فِی كُلِّ عَامࣲ مَّرَّةً أَوۡ مَرَّتَیۡنِ ثُمَّ لَا یَتُوبُونَ وَلَا هُمۡ یَذَّكَّرُونَ ۝١٢٦ اَوَلَا یَرَوْنَ اَنَّهُمْ یُفْتَنُوْنَ فِیْ كُلِّ عَامٍ مَّرَّةً اَوْ مَرَّتَیْنِ ثُمَّ لَا یَتُوْبُوْنَ وَلَا هُمْ یَذَّكَّرُوْنَ ۟ وَإِذَا مَآ أُنزِلَتۡ سُورَةٞ نَّظَرَ بَعۡضُهُمۡ إِلَىٰ بَعۡضٍ هَلۡ يَرَىٰكُم مِّنۡ أَحَدٖ ثُمَّ ٱنصَرَفُواْۚ صَرَفَ ٱللَّهُ قُلُوبَهُم بِأَنَّهُمۡ قَوۡمٞ لَّا يَفۡقَهُونَ ۝١٢٧ وَإِذَا مَاۤ أُنزِلَتۡ سُورَةࣱ نَّظَرَ بَعۡضُهُمۡ إِلَىٰ بَعۡضٍ هَلۡ یَرَىٰكُم مِّنۡ أَحَدࣲ ثُمَّ ٱنصَرَفُوا۟ۚ صَرَفَ ٱللَّهُ قُلُوبَهُم بِأَنَّهُمۡ قَوۡمࣱ لَّا یَفۡقَهُونَ ۝١٢٧ وَاِذَا مَاۤ اُنْزِلَتْ سُوْرَةٌ نَّظَرَ بَعْضُهُمْ اِلٰی بَعْضٍ ؕ هَلْ یَرٰىكُمْ مِّنْ اَحَدٍ ثُمَّ انْصَرَفُوْا ؕ صَرَفَ اللّٰهُ قُلُوْبَهُمْ بِاَنَّهُمْ قَوْمٌ لَّا یَفْقَهُوْنَ ۟ لَقَدۡ جَآءَكُمۡ رَسُولٞ مِّنۡ أَنفُسِكُمۡ عَزِيزٌ عَلَيۡهِ مَا عَنِتُّمۡ حَرِيصٌ عَلَيۡكُم بِٱلۡمُؤۡمِنِينَ رَءُوفٞ رَّحِيمٞ ۝١٢٨ لَقَدۡ جَاۤءَكُمۡ رَسُولࣱ مِّنۡ أَنفُسِكُمۡ عَزِیزٌ عَلَیۡهِ مَا عَنِتُّمۡ حَرِیصٌ عَلَیۡكُم بِٱلۡمُؤۡمِنِینَ رَءُوفࣱ رَّحِیمࣱ ۝١٢٨ لَقَدْ جَآءَكُمْ رَسُوْلٌ مِّنْ اَنْفُسِكُمْ عَزِیْزٌ عَلَیْهِ مَا عَنِتُّمْ حَرِیْصٌ عَلَیْكُمْ بِالْمُؤْمِنِیْنَ رَءُوْفٌ رَّحِیْمٌ ۟ فَإِن تَوَلَّوۡاْ فَقُلۡ حَسۡبِيَ ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَۖ عَلَيۡهِ تَوَكَّلۡتُۖ وَهُوَ رَبُّ ٱلۡعَرۡشِ ٱلۡعَظِيمِ ۝١٢٩ فَإِن تَوَلَّوۡا۟ فَقُلۡ حَسۡبِیَ ٱللَّهُ لَاۤ إِلَٰهَ إِلَّا هُوَۖ عَلَیۡهِ تَوَكَّلۡتُۖ وَهُوَ رَبُّ ٱلۡعَرۡشِ ٱلۡعَظِیمِ ۝١٢٩ فَاِنْ تَوَلَّوْا فَقُلْ حَسْبِیَ اللّٰهُ ۖۗؗ لَاۤ اِلٰهَ اِلَّا هُوَ ؕ عَلَیْهِ تَوَكَّلْتُ وَهُوَ رَبُّ الْعَرْشِ الْعَظِیْمِ ۟۠
